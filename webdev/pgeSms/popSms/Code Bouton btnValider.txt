// **** Clic sur btnValider (onclick navigateur) ( dispMessage ) ****

cpNavSms.NavCompteurSms(saiPopMessage,_sMessageFinalisé,Faux,_nTailleMessage,_mPrixVenteUnitaireHT,_nCreditMessage,_mTarifUnitaire,Vrai,_bEstCliquezIci)
MixGestionInterface(saiPopMessage)

// **** Clic sur btnValider ( dispMessage ) (serveur) ****

mCoutTotal est un monétaire
nTailleMessage	est un entier
sMessage		est une chaîne
nCreditMessage	est un entier
mPrixMessage	est un monétaire

nTailleMessage	= _nTailleMessage
sMessage		= _sMessageFinalisé
nCreditMessage	= _nCreditMessage
mPrixMessage	= _mPrixVenteUnitaireHT

// On recalcule côté serveur le prix du modèle
CompteurSms(saiPopMessage,_sMessageFinalisé,Faux,_nTailleMessage,_mPrixVenteUnitaireHT,_nCreditMessage,_mTarifUnitaire,Vrai,_bEstCliquezIci)

// On contrôle les valeurs navigateur au valeurs serveur
si nTailleMessage <> _nTailleMessage _ou_ sMessage <> _sMessageFinalisé _ou_ nCreditMessage <> _nCreditMessage _ou_ mPrixMessage <> _mPrixVenteUnitaireHT alors
	Trace("Taille Navigateur : ", nTailleMessage, "Taille serveur ", _nTailleMessage)
	Trace("Message Navigateur : ", sMessage, "Message serveur ", _sMessageFinalisé)
	Trace("Crédit Navigateur : ", nCreditMessage, "Crédit serveur ", _nCreditMessage)
	Trace("Prix Navigateur : ", mPrixMessage, "Prix serveur ", _mPrixVenteUnitaireHT)
	
	PopupAfficheMessage("Problème de concordance entre navigateur et serveur","Erreur",__OPERATION_SMS__,__TOAST_INFORMATION__)
	retour
fin

si SaisieMessageOK() alors
	saiMessage					= _sMessageFinalisé
	btnMessageSaisir.Libellé	= (SansEspace(saiMessage) <> "") ? "Modifier le message" sinon "Saisir le message"
	__stOperation.stOperationSms.Message = _sMessageFinalisé
	mCoutTotal					= CalculeCoutOperationSms(__stOperation)
	libTrfMontantHT				= NumériqueVersChaîne(mCoutTotal, ",2fS") +" €"
	cpDeviseur.DevisSms(__stOperation)
	MixGestionInterface(saiMessage)
	TunnelAfficheInformationsBandeauDroit()
	PopupFerme(popSms)
fin
