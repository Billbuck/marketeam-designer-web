procédure PréparationRequeteImportXls(stBaseDeDonnées est une structWsBaseDeDonnées, pFichierTemporaire est une chaîne)

nModulo 			est un entier
nColonne			est un entier
sInsert				est une chaîne ANSI
sSexeProfil			est une chaîne ANSI
sValues				est une chaîne ANSI
sEncodage 			est une chaîne
sContenu			est une chaîne ANSI
sColonne 			est une chaîne ANSI
sValeur				est une chaîne ANSI
tabValeur			est un tableau de chaînes ANSI
stImportChamp		est une structWsBddChamp
stTestLigne   		est une StructWsBddTestLigne
xDocument 			est un xlsDocument
tabRequete			est un tableau de chaînes ANSI
taaPrenom 			est un tableau associatif de chaînes ANSI
taaCivilite 		est un tableau associatif de chaînes ANSI

taaPrenom 		= ChargeTablePrenoms()
taaCivilite		= ChargeTableCivilites()

sInsert = "IdBase"
pour tout stImportChamp de stBaseDeDonnées.tabImportChamp sur Ordre
	sInsert += "," + stImportChamp.Champ
	si stImportChamp.Champ = __CHAMP_PRENOM__ alors
		sSexeProfil = ",Sexe,Profil"
	fin
fin
sInsert += sSexeProfil

xDocument = xlsOuvre(fRepDonnées() + ["\"] + pFichierTemporaire)

pour nLigne = 2 _à_ xDocument..NombreLigne
	nModulo++
	VariableRAZ(stTestLigne)
	tabValeur.SupprimeTout()

	pour nColonne = 1 _à_ xDocument..NombreColonne
		tabValeur.Ajoute(xDocument[nLigne,nColonne]..Valeur)
	fin

	sValeur = "ValIdBase"

	pour tout stImportChamp de stBaseDeDonnées.tabImportChamp sur ordre

		sContenu = ""

		pour tout nColonne de stImportChamp.tabColonne
			selon stImportChamp.Extrait
				cas 0
					si stImportChamp.Code = __CHAMP_CODE_CONTACT__ alors // Si c'est la colonne Contact, on transforme la civilité longue en courte
						si stBaseDeDonnées.stMapping.taaType[nColonne] = __CHAMP_CODE_CIVILITE__ alors
							sContenu += [" "] + CiviliteLongueVersCourte(tabValeur[nColonne])
						sinon
							sContenu += [" "] + tabValeur[nColonne]
						fin

					sinon
						sContenu += [" "] + tabValeur[nColonne]
					fin

				cas 1 // Pour récupérer le Code Postal
					sContenu = tabValeur[nColonne].ExtraitChaîne(1, " ")

				cas 2 // Pour récupérer la Ville
					sContenu = tabValeur[nColonne][[tabValeur[nColonne].Position(" ")+1 à]]	
			fin
		fin

		sColonne = TraiteValeur(sContenu, stImportChamp.Type, stImportChamp.Code, sEncodage)

		si stImportChamp.Champ = __CHAMP_PRENOM__ alors
			sSexeProfil = AnalysePrenom(sColonne, taaPrenom)
		fin

		AnalyseLigne(stTestLigne, sColonne , stImportChamp.Code)

		sValeur	+= [","] + sColonne
	fin

	si ComptabiliseLigne(stBaseDeDonnées, stTestLigne) alors
		sValeur += sSexeProfil
		sValues += [","] + "(" + sValeur + ")"
	fin

	si modulo(nModulo,100) = 0 alors
		tabRequete.Ajoute([
			START TRANSACTION;
			INSERT INTO clt_base_ligne ([%sInsert%])
			VALUES
			[%sValues%];
			COMMIT;
		])
		sValues = ""
		nModulo = 0
	fin
fin

si nModulo > 0 alors
	tabRequete.Ajoute([
		START TRANSACTION;
		INSERT INTO clt_base_ligne ([%sInsert%])
		VALUES
		[%sValues%];
		COMMIT;
	])
fin

renvoyer tabRequete

fin :
xDocument.Ferme()


	procédure interne CiviliteLongueVersCourte(pCiviliteLongue)

		sCiviliteCourte est une chaîne ANSI

		sCiviliteCourte = taaCivilite[Majuscule(SansAccent(pCiviliteLongue))]

		si sCiviliteCourte = "" alors
			renvoyer pCiviliteLongue
		sinon
			renvoyer sCiviliteCourte
		fin

	fin
