procédure BddImportation(stRequeteBddImportation est une structWsRequeteBddImportation)

sFichierTemporaire				est une chaîne ANSI
stBaseDeDonnées					est une structWsBaseDeDonnées
jsonStructImport				est une chaîne ANSI
stInformations					est une structInformations
jsonStructInformations			est une chaîne ANSI
stImportationReponse			est une structWsReponseBddImportation

si stRequeteBddImportation.FichierNom.SansEspace() = "" alors
	GénéreJsonErreur("Nom de fichier","FichierNom absent")
	renvoyer Faux
fin

si stRequeteBddImportation.FichierContenu.Vide = Vrai alors
	GénéreJsonErreur("FichierContenu","vide")
	renvoyer Faux
fin

selon "." + stRequeteBddImportation.FichierFormat.Minuscule()
	cas __CSV__, __TXT__ 	: sFichierTemporaire	= "Test"+ GénèreMotDePasse(8) + __CSV__
	cas __XLS__, __XLSX__	: sFichierTemporaire	= "Test"+ GénèreMotDePasse(8) + __XLSX__
	autre cas				: GénéreJsonErreur("Fichier","Format de fichier non pris en charge") ; renvoyer Faux
fin

stBaseDeDonnées.TypeFichier	= "." + stRequeteBddImportation.FichierFormat.Minuscule()
stBaseDeDonnées.stMapping		= stRequeteBddImportation.stMapping

//si pas fSauveBuffer(fRepDonnées() +["\"] + sFichierTemporaire, Décode(stRequeteBddImportation.FichierContenu,encodeBASE64SansRC)) ALORS
si pas fSauveBuffer(fRepDonnées() +["\"] + sFichierTemporaire, stRequeteBddImportation.FichierContenu) alors
	GénéreJsonErreur("Impossible de copier le fichier temporaire.",500)
	renvoyer Faux
fin

// Détection d'encodage 09/07/2026 : UNE fois pour tout le fichier (fini la
// devinette valeur par valeur qui produisait du « BÃ¢timent »). Stocké dans
// la structure -> sérialisé dans wbs_base.Json -> réutilisé par BddTraitement.
si stBaseDeDonnées.TypeFichier dans (__CSV__, __TXT__) alors
	stBaseDeDonnées.Encodage = DétecteEncodageFichier(fRepDonnées() +["\"] + sFichierTemporaire)
fin

stBaseDeDonnées.taaColonne 	= RécupérerEntêtesFichier(fRepDonnées() +["\"] + sFichierTemporaire, stBaseDeDonnées.TypeFichier, stBaseDeDonnées.Encodage)

si stBaseDeDonnées.stMapping.tabStandard.Occurrence = 0 _et_ stBaseDeDonnées.stMapping.tabOptionnel.Occurrence = 0 alors
	stBaseDeDonnées.stMapping = RubriqueAutomatique(stBaseDeDonnées.taaColonne)
	// Affinage 09/07/2026 : le type des optionnels restés Texte est inféré
	// depuis le CONTENU des 50 premières lignes (Date/Email/Entier/Décimal)
	AffineTypesOptionnels(stBaseDeDonnées.stMapping, stBaseDeDonnées.taaColonne, RécupérerEchantillonFichier(fRepDonnées() +["\"] + sFichierTemporaire, stBaseDeDonnées.TypeFichier))
sinon
	stInformations = VérifierColonnesMapping(stBaseDeDonnées)
	si stInformations.tabColonnesManquantes.Occurrence > 0 _ou_ stInformations.tabColonnesExploitées.Occurrence = 0
		Sérialise(stInformations, jsonStructInformations,psdJSON)
		//	fAjouteTexte(fRepDonnées() + "\Importation.txt",RC + "Informations" + RC + bufInformations)
		GénéreJsonErreur("Erreur d'importation",jsonStructInformations)
		renvoyer Faux
	fin
fin


stBaseDeDonnées.tabImportChamp = IdentificationDesColonnes(stBaseDeDonnées)

bEmail 		est un booléen
bPortable 	est un booléen
bTel 		est un booléen
nAdresse 	est un entier

pour tout stBddChamp de stBaseDeDonnées.tabImportChamp
	selon stBddChamp.Code
		cas __CHAMP_CODE_EMAIL__ 		: bEmail = Vrai
		cas __CHAMP_CODE_PORTABLE__		: bPortable	= Vrai
		cas __CHAMP_CODE_TEL__			: bTel = Vrai
		cas __CHAMP_CODE_NUMERO_VOIE__	: nAdresse += 1
		cas __CHAMP_CODE_NOM_VOIE__		: nAdresse += 1
		cas __CHAMP_CODE_ADRESSE1__		: nAdresse += 2
		cas __CHAMP_CODE_CODE_POSTAL__	: nAdresse += 1
		cas __CHAMP_CODE_VILLE__		: nAdresse += 1
		cas __CHAMP_CODE_CDP_VILLE__	: nAdresse += 2
		autre cas
	fin
fin

si bEmail = Faux _et_ bPortable = Faux _et_ bTel = Faux  _et_ nAdresse < 4 alors
	GénéreJsonErreur("Aucun enregistrement exploitable","Aucune colonne Portable, Tel, Email ou Adresse")
	renvoyer Faux
