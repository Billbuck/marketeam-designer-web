procédure AnalysePrenom(local sPrenom, taaPrenom tableau associatif de chaînes ANSI)

sProfil	est une chaîne ANSI
sSexe	est une chaîne ANSI
sRetour	est une chaîne ANSI


si sPrenom <> "NULL" alors
	sPrenom = SansAccent(sPrenom)
	sPrenom = Majuscule(sPrenom)
	sPrenom = Remplace(sPrenom,"'","")
	sPrenom = Remplace(sPrenom,"-"," ")

	si pas taaPrenom[sPrenom]..Vide alors
		sProfil	= taaPrenom[sPrenom][[1]]
		sSexe	= taaPrenom[sPrenom][[2]]
	sinon
		sProfil	= __TEL_PROFIL_NON_DETERMINE__
		sSexe	= __TEL_PROFIL_NON_DETERMINE__
	fin
sinon
	sProfil	= __TEL_PROFIL_NON_DETERMINE__
	sSexe	= __TEL_PROFIL_NON_DETERMINE__
fin

sRetour = ",'" + sProfil + "','" + sSexe + "'"

renvoyer sRetour
