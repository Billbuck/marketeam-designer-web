// Résumé : Infère le type (__CHAMP_TYPE_*) d'une colonne à partir de ses
//          VALEURS (échantillon des premières lignes du fichier), et non plus
//          seulement de son en-tête. Détection volontairement CONSERVATRICE :
//          en cas de doute, on renvoie Texte (le type reste modifiable par
//          l'utilisateur dans l'écran de mapping).
//
// Créée le 09/07/2026 (chantier refonte RubriqueAutomatique, point 3).
//
// Règles :
//	- Au moins 3 valeurs non vides, sinon Texte (échantillon trop faible).
//	- Un type n'est retenu que si >= 90 % des valeurs non vides le respectent.
//	- Types détectés : Date, Email, Entier, Décimal. Les types « risqués »
//	  (téléphone, code postal...) ne sont PAS inférés : trop de faux positifs,
//	  et les colonnes concernées sont normalement reconnues par leur en-tête.
//	- Un nombre à zéro de tête (« 0612... », « 09170 ») n'est PAS un entier :
//	  c'est un code (téléphone, code postal...) qui perdrait son zéro.
//
// Paramètres :
//	pValeurs : valeurs de la colonne séparées par RC (cf. RécupérerEchantillonFichier)
// Valeur de retour :
// 	chaîne : constante __CHAMP_TYPE_*  (TXT par défaut)
//
PROCÉDURE InféreTypeColonne(pValeurs est une chaîne ANSI)

nNonVide		est un entier
nDate			est un entier
nEmail		est un entier
nEntier		est un entier
nDécimal		est un entier
sValeur		est une chaîne

POUR TOUTE CHAÎNE sVal DE pValeurs SÉPARÉE PAR RC
	sValeur = sVal.SansEspace()
	SI sValeur = "" ALORS CONTINUER

	nNonVide++
	SELON Vrai
		CAS EstDate(sValeur)	: nDate++
		CAS EstEmail(sValeur)	: nEmail++
		CAS EstEntier(sValeur)	: nEntier++
		CAS EstDécimal(sValeur)	: nDécimal++
	FIN
FIN

SI nNonVide < 3 ALORS RENVOYER __CHAMP_TYPE_TEXTE__

// Seuil : au moins 90 % des valeurs non vides (comparaison en entiers)
SELON Vrai
	CAS nDate * 10 >= nNonVide * 9		: RENVOYER __CHAMP_TYPE_DATE__
	CAS nEmail * 10 >= nNonVide * 9		: RENVOYER __CHAMP_TYPE_EMAIL__
	CAS nEntier * 10 >= nNonVide * 9	: RENVOYER __CHAMP_TYPE_ENTIER__
	// Une colonne numérique mixte (entiers + décimaux) est décimale
	CAS (nDécimal + nEntier) * 10 >= nNonVide * 9 _ET_ nDécimal > 0	: RENVOYER __CHAMP_TYPE_DECIMAL__
	AUTRE CAS							: RENVOYER __CHAMP_TYPE_TEXTE__
FIN

// Même logique de normalisation que CleanDate (TraiteValeur) :
	// chiffres uniquement, 6 ou 8 caractères, date calendaire valide
	PROCÉDURE INTERNE EstDate(pVal est une chaîne) : booléen

		sEntrée	est une chaîne		= pVal.Remplace([" ","-","/","\","_",":",".",","], "")
		dDate		est une Date

		SI sEntrée.CaractOccurrenceType(ctNumérique) <> sEntrée.Taille() ALORS RENVOYER Faux

		SI sEntrée.Taille() = 6 ALORS
			sEntrée = sEntrée[[À 4]] + "20" + sEntrée[[5 À]]
		FIN
		SI sEntrée.Taille() <> 8 ALORS RENVOYER Faux

		SI Val(sEntrée.Droite(4)) >= 1900 _ET_ Val(sEntrée.Droite(4)) <= 2100 ALORS
			dDate = sEntrée.VersDate("JJMMAAAA")
		SINON
			dDate = sEntrée.VersDate("AAAAMMJJ")
		FIN

		RENVOYER DateValide(dDate)
	FIN


	PROCÉDURE INTERNE EstEmail(pVal est une chaîne) : booléen

		SI PAS pVal.Contient("@") ALORS RENVOYER Faux
		RENVOYER EmailVérifieAdresse(pVal.Minuscule(), emailValideSyntaxeStrict) = emailAdresseValide
	FIN


	PROCÉDURE INTERNE EstEntier(pVal est une chaîne) : booléen

		SI pVal.CaractOccurrenceType(ctNumérique) <> pVal.Taille() ALORS RENVOYER Faux
		// Zéro de tête = code (téléphone, code postal...), pas un nombre
		SI pVal.Taille() > 1 _ET_ pVal[[1]] = "0" ALORS RENVOYER Faux
		RENVOYER Vrai
	FIN


	PROCÉDURE INTERNE EstDécimal(pVal est une chaîne) : booléen

		sNombre est une chaîne = pVal.Remplace(",", ".")

		SI sNombre.Occurrence(".") <> 1 ALORS RENVOYER Faux

		sNombre = sNombre.Remplace(".", "")
		SI sNombre = "" ALORS RENVOYER Faux

		RENVOYER sNombre.CaractOccurrenceType(ctNumérique) = sNombre.Taille()
	FIN
