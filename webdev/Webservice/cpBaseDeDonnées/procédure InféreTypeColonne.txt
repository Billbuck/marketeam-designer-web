// Résumé : Transforme le mapping (standard + optionnel) en tableau de champs
//          d'import (structWsBddChamp) avec code, champ SQL, libellé et type.
//
// Refonte 09/07/2026 : les ~20 blocs « cas » identiques (copier/coller) sont
// remplacés par une TABLE DE DÉFINITIONS (code -> champ / type / extrait).
// Ajouter un champ standard = ajouter UNE ligne AjouteDef.
// Seul le cas composite CDP_VILLE (code postal + ville dans la même colonne,
// qui génère DEUX champs avec extraction) reste traité à part.
//
// Paramètres :
//	stBaseDeDonnées (structWsBaseDeDonnées) : base en cours d'import (stMapping + taaColonne)
// Valeur de retour :
// 	tableau (structWsBddChamp) : // 	tableau de structWsBddChamp
//
procédure IdentificationDesColonnes(local stBaseDeDonnées est une structWsBaseDeDonnées)

// La structure structDefChampStandard est déclarée dans le code de
// DÉCLARATION de la collection cpBaseDeDonnées (déplacée le 09/07/2026).

nInd				est un entier
nOrdre			est un entier
stImportChamp	est une structWsBddChamp
tabImportChamp	est un tableau de structWsBddChamp
taaDef			est un tableau associatif de structDefChampStandard

// ------------------------------------------------------------------
// TABLE DES DÉFINITIONS : code standard -> (champ SQL, type, extrait)
// ------------------------------------------------------------------
AjouteDef(__CHAMP_CODE_REFERENCE__,		__CHAMP_REFERENCE__,	__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_SOCIETE__,		__CHAMP_SOCIETE__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_ENSEIGNE__,		__CHAMP_ENSEIGNE__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_CIVILITE__,		__CHAMP_CIVILITE__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_NOM__,			__CHAMP_NOM__,			__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_PRENOM__,		__CHAMP_PRENOM__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_CONTACT__,		__CHAMP_CONTACT__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_TEL__,			__CHAMP_TEL__,			__CHAMP_TYPE_TELEPHONE__,	0)
AjouteDef(__CHAMP_CODE_PORTABLE__,		__CHAMP_PORTABLE__,		__CHAMP_TYPE_PORTABLE__,	0)
AjouteDef(__CHAMP_CODE_EMAIL__,			__CHAMP_EMAIL__,		__CHAMP_TYPE_EMAIL__,		0)
AjouteDef(__CHAMP_CODE_ADRESSE1__,		__CHAMP_ADRESSE1__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_ADRESSE2__,		__CHAMP_ADRESSE2__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_ADRESSE3__,		__CHAMP_ADRESSE3__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_ADRESSE4__,		__CHAMP_ADRESSE4__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_CODE_POSTAL__,	__CHAMP_CODE_POSTAL__,	__CHAMP_TYPE_CODEPOSTAL__,	0)
AjouteDef(__CHAMP_CODE_VILLE__,			__CHAMP_VILLE__,		__CHAMP_TYPE_TEXTE__,		0)
AjouteDef(__CHAMP_CODE_PAYS__,			__CHAMP_PAYS__,			__CHAMP_TYPE_TEXTE__,		2)
AjouteDef(__CHAMP_CODE_ALLIAGE__,		__CHAMP_ALLIAGE__,		__CHAMP_TYPE_ALLIAGE__,	0)

// ------------------------------------------------------------------
// Champs STANDARD
// ------------------------------------------------------------------
pour tout stImportStandard de stBaseDeDonnées.stMapping.tabStandard

	si stImportStandard.Type = __CHAMP_CODE_CDP_VILLE__ alors
		// Cas composite : la même colonne alimente DEUX champs par extraction
		AjouteChamp(stImportStandard, __CHAMP_CODE_CODE_POSTAL__, __CHAMP_CODE_POSTAL__, __CHAMP_TYPE_CODEPOSTAL__, 1)
		AjouteChamp(stImportStandard, __CHAMP_CODE_VILLE__, __CHAMP_VILLE__, __CHAMP_TYPE_TEXTE__, 2)

	sinon si taaDef[stImportStandard.Type]..Vide = Faux alors
		AjouteChamp(stImportStandard, stImportStandard.Type, taaDef[stImportStandard.Type].Champ, taaDef[stImportStandard.Type].Type, taaDef[stImportStandard.Type].Extrait)
	fin
	// code inconnu -> ignoré (comme l'ancien « autre cas »)
fin

// ------------------------------------------------------------------
// Champs OPTIONNELS
// ------------------------------------------------------------------
pour tout stImportOptionnel de stBaseDeDonnées.stMapping.tabOptionnel
	nInd++
	VariableRAZ(stImportChamp)
	stImportChamp.Code			= ""
	stImportChamp.Champ			= "Champ" + nInd
	stImportChamp.Libelle		= stImportOptionnel.Libelle
	stImportChamp.Ordre			= ++nOrdre
	stImportChamp.Type			= stImportOptionnel.Type
	stImportChamp.Extrait		= 0
	stImportChamp.tabColonne	= [stBaseDeDonnées.taaColonne[stImportOptionnel.Colonne]]
	tabImportChamp.Ajoute(stImportChamp)
fin

renvoyer tabImportChamp

	procédure interne AjouteDef(pCode est une chaîne, pChamp est une chaîne, pType est une chaîne, pExtrait est un entier)

		stDef est une structDefChampStandard
		stDef.Champ		= pChamp
		stDef.Type		= pType
		stDef.Extrait	= pExtrait
		taaDef[pCode]	= stDef
	fin


	procédure interne AjouteChamp(pImportStandard est une structWsBddStandard, pCode est une chaîne ANSI, pChamp est une chaîne, pType est une chaîne, pExtrait est un entier)

		VariableRAZ(stImportChamp)
		stImportChamp.Code			= pCode
		stImportChamp.Champ			= pChamp
		stImportChamp.Libelle		= LibelleTypeChamp(pChamp)
		stImportChamp.Ordre			= ++nOrdre
		stImportChamp.Type			= pType
		stImportChamp.Extrait		= pExtrait
		pour tout sColonne de pImportStandard.Colonne
			stImportChamp.tabColonne.Ajoute(stBaseDeDonnées.taaColonne[sColonne])
		fin
		tabImportChamp.Ajoute(stImportChamp)
	fin
