// Résumé : Feuille Excel du détail de consommation des SMS de notification.
// Syntaxe :
//[ <Résultat> = ] ChargeFeuilleExcelConsoSms (<pIdMere>, <pFiltre>)
//
// Paramètres :
//	pIdMere : IdClient de la maison mère (client connecté)
//	pFiltre : filtres de la page "Année;Mois;IdSociété;IdContact" (0 = tous)
// Valeur de retour :
// 	structRapportExcelFeuille
//
procédure ChargeFeuilleExcelConsoSms(pIdMere, pFiltre)

nLigne					est un entier
sTexteRequete			est une chaîne
sdReqConsoSms			est une Source de Données
nColonne				est un entier
stRapportExcelFeuille	est une structRapportExcelFeuille
sWhere					est une chaîne
sAnnee					est une chaîne
nMois					est un entier
nIdSociete				est un entier
nIdContactFiltre		est un entier
nTotalSms				est un entier
mTotalHT				est un monétaire
mTotalTTC				est un monétaire
mTotalResteAPayer		est un monétaire


sAnnee				= pFiltre.ExtraitChaîne(1,";")
nMois				= pFiltre.ExtraitChaîne(2,";")
nIdSociete			= pFiltre.ExtraitChaîne(3,";")
nIdContactFiltre	= pFiltre.ExtraitChaîne(4,";")

// Mêmes filtres que la page pgeConsoNotification
si nMois > 0 alors
	sWhere = "AND MONTH(ope_sms_notification.DateHeureEnvoi) = " + nMois
fin
si nIdSociete > 0 alors
	sWhere += [RC] + "AND ope_sms_notification.IdClient = " + nIdSociete
fin
si nIdContactFiltre > 0 alors
	sWhere += [RC] + "AND ope_sms_notification.IdContact = " + nIdContactFiltre
fin

// Colonnes de la feuille
nColonne = 14

stRapportExcelFeuille.taaEntête["Client"]			= 2
stRapportExcelFeuille.tabColonne.Ajoute("Société")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_GAUCHE__)
stRapportExcelFeuille.tabColonne.Ajoute("Contact")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_GAUCHE__)

stRapportExcelFeuille.taaEntête["Diffusion"]		= 5
stRapportExcelFeuille.tabColonne.Ajoute("Jour")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)
stRapportExcelFeuille.tabColonne.Ajoute("Date")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)
stRapportExcelFeuille.tabColonne.Ajoute("Heure")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)
stRapportExcelFeuille.tabColonne.Ajoute("Portable")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)
stRapportExcelFeuille.tabColonne.Ajoute("Émetteur")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)

stRapportExcelFeuille.taaEntête["Message"]			= 2
stRapportExcelFeuille.tabColonne.Ajoute("Message")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_GAUCHE__)
stRapportExcelFeuille.tabColonne.Ajoute("Nbr SMS")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)

stRapportExcelFeuille.taaEntête["Facturation"]		= 5
stRapportExcelFeuille.tabColonne.Ajoute("Montant HT")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_DROITE__)
stRapportExcelFeuille.tabColonne.Ajoute("TVA")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)
stRapportExcelFeuille.tabColonne.Ajoute("Montant TTC")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_DROITE__)
stRapportExcelFeuille.tabColonne.Ajoute("Payé")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)
stRapportExcelFeuille.tabColonne.Ajoute("Reste à payer")
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_DROITE__)


sTexteRequete =
[
	SELECT
		IFNULL(clt_client.Societe,'') Societe,
		IFNULL(CONCAT_WS(' ', clt_contact.Prenom, clt_contact.Nom),'') Contact,
		IFNULL(DATE_FORMAT(ope_sms_notification.DateHeureEnvoi, '%Y%m%d'),'') DateBrute,
		IFNULL(DATE_FORMAT(ope_sms_notification.DateHeureEnvoi, '%d-%m-%Y'),'') DateEnvoi,
		IFNULL(DATE_FORMAT(ope_sms_notification.DateHeureEnvoi, '%H:%i:%s'),'') HeureEnvoi,
		IFNULL(ope_sms_notification.Portable,'') Portable,
		IFNULL(ope_sms_notification.LeFrom,'') Emetteur,
		IFNULL(ope_sms_notification.Message,'') Message,
		IFNULL(ope_sms_notification.NbrSmsFacture,0) NbrSms,
		IFNULL(ope_sms_notification.MontantHT,0) MontantHT,
		IFNULL(ope_sms_notification.TauxTVA,0) TauxTVA,
		IFNULL(ope_sms_notification.MontantTTC,0) MontantTTC,
		ope_sms_notification.EstPaye,
		CASE WHEN ope_sms_notification.EstPaye = 0 THEN IFNULL(ope_sms_notification.MontantTTC,0) ELSE 0 END ResteAPayer
	FROM
		ope_sms_notification
		JOIN clt_client ON clt_client.IdClient = ope_sms_notification.IdClient
		JOIN clt_mere ON clt_mere.IdClient = ope_sms_notification.IdClient
		LEFT JOIN clt_contact ON clt_contact.IdContact = ope_sms_notification.IdContact
	WHERE
		clt_mere.IdMere = [%pIdMere%]
		AND YEAR(ope_sms_notification.DateHeureEnvoi) = '[%sAnnee%]'
		[%sWhere%]
	ORDER BY
		ope_sms_notification.DateHeureEnvoi
]

si pas sdReqConsoSms.ExécuteRequêteSQL(ope_sms_notification..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	renvoyer stRapportExcelFeuille
fin

pour tout sdReqConsoSms
	nLigne = stRapportExcelFeuille.tabExcel.AjouteLigne()
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(sdReqConsoSms.Societe)
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(sdReqConsoSms.Contact)
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(DateVersJourEnLettre(sdReqConsoSms.DateBrute))
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(sdReqConsoSms.DateEnvoi)
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(sdReqConsoSms.HeureEnvoi)
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(cpFonction.Telephone(sdReqConsoSms.Portable,1,"."))
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(sdReqConsoSms.Emetteur)
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(sdReqConsoSms.Message)
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(NumériqueVersChaîne(sdReqConsoSms.NbrSms, "dS"))
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(NumériqueVersChaîne(sdReqConsoSms.MontantHT, ",3fS") + " €")
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(NumériqueVersChaîne(sdReqConsoSms.TauxTVA, ",2fS") + " %")
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(NumériqueVersChaîne(sdReqConsoSms.MontantTTC, ",3fS") + " €")
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(sdReqConsoSms.EstPaye = 1 ? "Oui" sinon "Non")
	stRapportExcelFeuille.tabExcel[nLigne].Ajoute(NumériqueVersChaîne(sdReqConsoSms.ResteAPayer, ",3fS") + " €")

	nTotalSms			+= sdReqConsoSms.NbrSms
	mTotalHT			+= sdReqConsoSms.MontantHT
	mTotalTTC			+= sdReqConsoSms.MontantTTC
	mTotalResteAPayer	+= sdReqConsoSms.ResteAPayer
fin

// Ligne Total en fin de feuille
si stRapportExcelFeuille.tabExcel.Occurrence > 0 alors
	nLigne = stRapportExcelFeuille.tabExcel.AjouteLigne()
	pour nInd = 1 à nColonne
		stRapportExcelFeuille.tabExcel[nLigne].Ajoute("")
	fin
	stRapportExcelFeuille.tabExcel[nLigne][1]	= "Total"
	stRapportExcelFeuille.tabExcel[nLigne][9]	= NumériqueVersChaîne(nTotalSms, "dS")
	stRapportExcelFeuille.tabExcel[nLigne][10]	= NumériqueVersChaîne(mTotalHT, ",3fS") + " €"
	stRapportExcelFeuille.tabExcel[nLigne][12]	= NumériqueVersChaîne(mTotalTTC, ",3fS") + " €"
	stRapportExcelFeuille.tabExcel[nLigne][14]	= NumériqueVersChaîne(mTotalResteAPayer, ",3fS") + " €"
fin

renvoyer stRapportExcelFeuille

fin:
sdReqConsoSms.AnnuleDéclaration()
