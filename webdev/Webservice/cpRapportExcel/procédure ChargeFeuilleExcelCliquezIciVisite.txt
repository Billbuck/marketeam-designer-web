// Résumé : <indiquez ici ce que fait la procédure>
// Syntaxe :
//[ <Résultat> = ] ChargeFeuilleExcelCliquezIciVisite (<pIdOperation> [, <pIdEmailEnvoi>])
//
// Paramètres :
//	pIdOperation : <indiquez ici le rôle de pIdOperation>
//	pIdEmailEnvoi (valeur par défaut=0) : < indiquez ici le rôle de pIdEmailEnvoi >
// Valeur de retour :
// 	Type indéterminé : <indiquez ici le rôle de la valeur de retour>
//
// Exemple :
// Indiquez ici un exemple d'utilisation.
//
//	stEmail (structEmail) : <indiquez ici le rôle de stEmail>
procédure ChargeFeuilleExcelCliquezIciVisite(pIdOperation, pDateHeureEmailFiltre = "")

nLigne					est un entier
sValeur 				est une chaîne
sTexteRequete			est une chaîne
sTexteAudience			est une chaîne
sSelect					est une chaîne
sdReqVisite				est une Source de Données
sListeField				est une chaîne
sChampOrderBy			est une chaîne
nColonne				est un entier
sWhere					est une chaîne
tabType					est un tableau de chaînes
stRapportExcelFeuille	est une structRapportExcelFeuille
taaColonneAnonymise 	est un tableau associatif de booléens

// On récupère la liste des offres (Interrupteurs, Combos, Champs de saisie et Optin) pour trier les résultats en sortie.
sTexteRequete = [
	SELECT
		GROUP_CONCAT("\"", COALESCE(ope_cliquezici_offre.Libelle, ope_cliquezici_offre.Offre), "\"") ListeField
	FROM
		ope_cliquezici_offre
		JOIN ope_cliquezici_bloc ON ope_cliquezici_bloc.IdCliquezIciBloc = ope_cliquezici_offre.IdCliquezIciBloc AND ope_cliquezici_bloc.`Type` IN ('[%__BLOC_OFFRE__%]' , '[%__BLOC_FORMULAIRE__%]' , '[%__BLOC_BOUTON__%]')
	WHERE
		ope_cliquezici_bloc.IdCliquezIci = [%pIdOperation%]
	ORDER BY
		FIELD(ope_cliquezici_bloc.`Type`, '[%__BLOC_OFFRE__%]' , '[%__BLOC_FORMULAIRE__%]' , '[%__BLOC_BOUTON__%]')
	
]

si pas sdReqVisite.ExécuteRequêteSQL(ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	renvoyer stRapportExcelFeuille
fin

si sdReqVisite.LitPremier() alors
	sListeField = sdReqVisite.ListeField
fin

// Si la liste des "libellés des offres" est vide : On trie le résultat par type (Interrupteurs, Combos, Champs de saisie et Optin)
si sListeField <> "" alors
	sChampOrderBy = [
		ORDER BY
			FIELD(ope_cliquezici_clic_selection.Libelle, [%sListeField%])
	]
sinon
	sChampOrderBy = [
		ORDER BY
			FIELD(ope_cliquezici_clic_selection.`Type`, '[%__FORMULAIRE_INTERRUPTEUR__%]', '[%__FORMULAIRE_COMBO__%]','[%__FORMULAIRE_SAISIE__%]','[%__FORMULAIRE_OPTIN__%]')
	]
fin


// Colonnes sélectionnées par le client pour le rapport

sTexteRequete	= [
	SELECT 
		dos_operation_rapport.Champ, 
		dos_operation_rapport.Libelle,
		dos_operation_rapport.Type
	FROM dos_operation_rapport
	WHERE dos_operation_rapport.IdOperation = [%pIdOperation%]
	ORDER BY dos_operation_rapport.Ordre
]

si pas sdReqVisite.ExécuteRequêteSQL(dos_operation_rapport..Connexion,hRequêteSansCorrection,sTexteRequete) alors
	renvoyer stRapportExcelFeuille
fin

pour tout sdReqVisite
	nColonne++
	taaColonneAnonymise[nColonne] = Vrai
	sSelect += ["," + RC] + "dos_base_ligne." + sdReqVisite.Champ + " AS 'Colonne" + nColonne + "'"
	stRapportExcelFeuille.taaEntête["Fiche client"]++
	stRapportExcelFeuille.tabColonne.Ajoute(sdReqVisite.Libelle)
	tabType.Ajoute(sdReqVisite.Type)
	stRapportExcelFeuille.tabAlignement.Ajoute(AlignementTypeChamp(sdReqVisite.Type))
fin

// On ajoute la colonne date & heure de visite
nColonne++
sSelect += ["," + RC]  + "DATE_FORMAT(ope_cliquezici_tracking." + ope_cliquezici_tracking.DateHeureClic..Nom + ", '%d-%m-%Y %H:%i:%s') AS 'Colonne" + nColonne + "'"
stRapportExcelFeuille.taaEntête["Visite"]++
stRapportExcelFeuille.tabColonne.Ajoute("Date & heure")
tabType.Ajoute(__CHAMP_TYPE_TEXTE__)
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)

// On ajoute la colonne Nombre de visite
nColonne++
sSelect += ["," + RC]  + "COUNT(ope_cliquezici_tracking.IdCliquezIci) AS 'Colonne" + nColonne + "'"
stRapportExcelFeuille.taaEntête["Visite"]++
stRapportExcelFeuille.tabColonne.Ajoute("Nombre")
tabType.Ajoute(__CHAMP_TYPE_TEXTE__)
stRapportExcelFeuille.tabAlignement.Ajoute(__ALIGNEMENT_CENTRE__)

si pDateHeureEmailFiltre = "" alors
	sWhere	= "ope_cliquezici_tracking.IdCliquezIci = " + pIdOperation
sinon
	sWhere	= "ope_cliquezici_tracking.DateHeureEmailFiltre = '" + pDateHeureEmailFiltre + "'"
fin


sTexteAudience =
[
	SELECT 
		dos_base.IdZecible,
		[%sSelect%]
	FROM 
		ope_cliquezici_tracking
		JOIN dos_base_ligne ON dos_base_ligne.IdBaseLigne = ope_cliquezici_tracking.IdBaseLigne
		JOIN dos_base ON dos_base.IdBase = dos_base_ligne.IdBase
	WHERE
		[%sWhere%]
	GROUP BY 
		dos_base_ligne.IdBaseLigne
	ORDER BY 
		ope_cliquezici_tracking.DateHeureClic DESC
]

si pas sdReqVisite.ExécuteRequêteSQL(ope_cliquezici_clic..Connexion, hRequêteSansCorrection, sTexteAudience) alors
	renvoyer stRapportExcelFeuille
fin

pour tout sdReqVisite
	nLigne = stRapportExcelFeuille.tabExcel.AjouteLigne()
	// La première colonne correspond à IdZecible
	pour nCol = 1 à nColonne
		sValeur = FormatValeur(tabType[nCol],{"sdReqVisite.Colonne"+nCol,indRubrique})
		// Si la base de données a été loué on anonymise les informations
		// On anonymise les visites uniquement car c'est pas Optin
		si sdReqVisite.IdZecible > 0 _et_ taaColonneAnonymise[nCol] = Vrai alors
			sValeur = cpFonction.Anonymise(sValeur)
		fin
		stRapportExcelFeuille.tabExcel[nLigne].Ajoute(sValeur)
	fin
fin

stRapportExcelFeuille.Titre 	= "Rapport de performance"
stRapportExcelFeuille.SousTitre = "CliquezIci Lead"

renvoyer stRapportExcelFeuille

fin:
sdReqVisite.AnnuleDéclaration()
