procédure CelluleExcel(local pLigne, local pColonne)

sLettre		est une chaîne
nReste		est un entier
sResultat	est une chaîne

tantque pColonne > 0
	pColonne--
	nReste		= modulo(pColonne,26)
	sLettre		= Caract(65 + nReste)
	sResultat	= sLettre + sResultat
	pColonne	= PartieEntière(pColonne / 26)
fin

sResultat = sResultat + pLigne

renvoyer sResultat
