//saiTest = "OFF"

si uplDocumentUpload..Occurrence > 0 alors
	sExtensionFichier = Droite(sFichierUpload,5)
	P = Position(sExtensionFichier,".",0) + 1
	sExtensionFichier = Majuscule(sExtensionFichier[[P	à]])
	si sExtensionFichier <> "PDF" alors
		UploadSupprime(uplDocumentUpload,1)
		PopupAfficheMessage(sFichierUpload +" : seuls les fichiers PDF sont pris en charge",*,__stOperation.TypeOperation,__TOAST_INFORMATION__)
	sinon
		AffichePopupProgression("Traitement en cours",_sTypeOperation)
		UploadLance(uplDocumentUpload)
	fin
