// Produites par cpDesigner.VerifieConcordanceBaseDesigner(stOperation).
//   - structDesignerAnomalieConcordance : une anomalie rattachée à un document.
//   - structDesignerRapportConcordance  : le rapport complet (par document).

structDesignerAnomalieConcordance est une Structure
	IndexDocument	est un entier		// Position dans tabStDocument (1..n)
	Designation		est une chaîne		// .Designation du document (titre de section)
	TypeProbleme	est une chaîne		// "ABSENT" / "ECART_CASSE" / "NON_RESOLVABLE" / "CHAMP_INCONNU" / "AMBIGU" / "AUCUNE_BASE"
	CleInseree		est une chaîne		// Clé telle qu'insérée (nettoyée)
	LibelleResolu	est une chaîne		// Libellé lisible (spécifique) ou nom technique (standard)
	Nature			est une chaîne		// "standard" / "specifique" / ""
	Detail			est une chaîne		// Message prêt à afficher pour cette anomalie
fin

structDesignerRapportConcordance est une Structure
	EstConcordant	est un booléen															// Vrai si 0 anomalie
	NbAnomalies		est un entier
	BaseChargee		est un booléen															// Vrai si au moins une base non supprimée
	tabAnomalies	est un tableau de structDesignerAnomalieConcordance		// déjà ordonné par document
fin

// =============================================================================
// Structures de l'état de RAPPROCHEMENT (chantier QR de rapprochement).
// Produites par cpDesigner.RecalculeEtatRapprochement(stOperation).
// CALCULÉES À LA DEMANDE — NON persistées (recalcul dynamique).
//
//   - structDesignerEtatRapprochementDocument : état d'UN document.
//   - structDesignerEtatRapprochement         : agrégat opération + tableau
//                                               par document.
//
// Correspondance avec le vocabulaire du cahier des charges :
//   EstPersonnalise  <-> bPersonnalise   QrPresent  <-> bQrPresent
//   QrRequis         <-> bQrRequis        EstConformeQr <-> bConformeQr
//   QrNecessaire     <-> bQrNecessaire    NbDocumentsPersonnalises <-> N
// =============================================================================

structDesignerEtatRapprochementDocument est une Structure
	IndexDocument		est un entier		// Position 1..n dans tabStDocument
	Designation			est une chaîne		// Libellé du document
	EstPersonnalise	est un booléen		// (A) pavé d'adresse OU (B) >=1 champ base
	QrPresent			est un booléen		// QR de rapprochement déjà présent (critère provisoire Phase 2)
	QrRequis				est un booléen		// EstPersonnalise ET QrNecessaire (agrégat)
	EstConformeQr		est un booléen		// QrRequis = QrPresent
fin

structDesignerEtatRapprochement est une Structure
	NbDocumentsPersonnalises	est un entier		// N
	QrNecessaire					est un booléen		// N >= 2
	EstToutConformeQr				est un booléen		// Vrai si tous les documents EstConformeQr
	tabDocuments					est un tableau de structDesignerEtatRapprochementDocument
fin
