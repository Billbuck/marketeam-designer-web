	procédure interne TypeValeur(pIdVar)

		selon Vrai
			cas Contient(pIdVar,"interrupteur") = Vrai
				renvoyer ".checked"
				//			renvoyer ".checked.toString()  : ""false"""

			cas Contient(pIdVar,"Optin") = Vrai _ou_ EstNumérique(pIdVar)
				renvoyer ""

			autre cas
				//			renvoyer [
				//			.value.replace(/"/g, '\\"')
				//			]
				renvoyer ".value"
		fin

	fin

	procédure interne TypeChamp(pIdVar)

		selon Vrai
			cas Contient(pIdVar,"interrupteur") = Vrai						: renvoyer __FORMULAIRE_INTERRUPTEUR__
			cas Contient(pIdVar,"Optin") = Vrai _ou_ EstNumérique(pIdVar)	: renvoyer __FORMULAIRE_OPTIN__
			cas Contient(pIdVar,"BlocCombo") = Vrai							: renvoyer __FORMULAIRE_COMBO__
			cas Contient(pIdVar,"input") = Vrai								: renvoyer __FORMULAIRE_SAISIE__
			autre cas														: STOP ; renvoyer "inconnu"
		fin

	fin
