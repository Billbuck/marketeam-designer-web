sScript = [
	// Liste des messages à afficher
	const messages = [[%sListeDesMessages%]];
	
	let currentIndex = 0; // Index du message actuel
	let isFirstMessage = true; // Alterne entre les deux éléments de message
	
	// Fonction pour animer l'apparition d'un message
	function animateMessage(element, text) {
	// Divise le texte en caractères et crée un span pour chacun
	element.innerHTML = text.split('').map(char => 
	`<span style="animation: dropIn 0.3s forwards;">${char}</span>`
	).join('');
	
	// Ajoute un délai à chaque caractère pour créer un effet cascade
	element.querySelectorAll('span').forEach((span, index) => {
	span.style.animationDelay = `${index * [%nVitesseDefilement%] }ms`;
	});
	}
	
	// Fonction pour faire disparaître un message
	function fadeOutMessage(element) {
	return new Promise((resolve) => {
	element.style.animation = 'fadeOut 0.5s forwards';
	setTimeout(resolve, 400); // Attendre que l'animation soit terminée
	});
	}
	
	// Fonction pour mettre à jour et animer le prochain message
	async function updateMessage() {
	// Détermine quel élément est actuel et lequel est le suivant
	const currentElement = document.getElementById(isFirstMessage ? 'message1' : 'message2');
	const nextElement = document.getElementById(isFirstMessage ? 'message2' : 'message1');
	
	// Fait disparaître le message actuel
	await fadeOutMessage(currentElement);
	
	// Prépare le prochain message
	currentIndex = (currentIndex + 1) % messages.length;
	nextElement.style.opacity = '0'; // Assure que le prochain élément est invisible
	animateMessage(nextElement, messages[currentIndex]);
	
	// Fait apparaître le nouveau message
	nextElement.style.animation = 'none';
	nextElement.offsetHeight; // Force un reflow
	nextElement.style.animation = null;
	nextElement.style.opacity = '1';
	
	// Prépare pour le prochain cycle
	isFirstMessage = !isFirstMessage;
	setTimeout(updateMessage, [%stProprieteTexteDefilant.Temporisation%]);
	}
	
	// Initialise l'animation dès que le DOM est chargé
	document.addEventListener('DOMContentLoaded', () => {
	const firstElement = document.getElementById('message1');
	animateMessage(firstElement, messages[0]);
	setTimeout(updateMessage, [%stProprieteTexteDefilant.Temporisation%]);
	});
]
