sCodeJS = [

	function adjustCardHeight(img) {
		const card = img.closest('.grid-item');
			if (card) {
				card.style.height = img.clientHeight + 'px'; // Ajuste la hauteur de la carte à celle de l'image
			}
	}
	
	function adjustCardHeights() {
	    const images = document.querySelectorAll('.grid-item img');
	    images.forEach(img => {
	        if (img.complete) {
	            adjustCardHeight(img);
	        } else {
	            img.onload = () => adjustCardHeight(img);
	        }
	    });
	}
	
	document.addEventListener("DOMContentLoaded", function() {
	    const NbrColonne = getComputedStyle(document.documentElement).getPropertyValue('--GrilleColonne');
	    const gridItems = document.querySelectorAll('.grid-item');
	    const numColonne = parseInt(NbrColonne);
	    gridItems.forEach((item, index) => {
	        const rowIndex = Math.floor(index / numColonne);
	        item.style.zIndex = gridItems.length - rowIndex;
	    });
	

	    const gridContainer = document.querySelector('.grid-container');
	    gridContainer.style.height = 'auto'; // Permet à container d'ajuster la hauteur basée sur le contenu

		adjustCardHeights(); // Ajuste les hauteurs initiales
	
	    // Ajuste la taille de la police dynamiquement
	    const adjustFontSize = () => {
	        const backContents = document.querySelectorAll('.back-content');
	        backContents.forEach(backContent => {
	            const gridItem = backContent.closest('.grid-item');
	            if (!gridItem) return; // Vérifie que l'élément gridItem existe
	
	            const buttonLink = backContent.querySelector('.bouton-lien');
	            if (!buttonLink) return; // Vérifie que l'élément bouton-lien existe
	
	            const linkHeight = buttonLink.offsetHeight;
	            const availableHeight = gridItem.offsetHeight - linkHeight - 40; // 40 for padding
	            const textElement = backContent.querySelector('.grille-block-title');
	            if (!textElement) return; // Vérifie que l'élément grille-block-title existe
	
	            const maxFontSize = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--initial-font-size'), 10) || [%stProprieteTexte.PoliceTailleL%];
	            let fontSize = maxFontSize;
	            textElement.style.fontSize = fontSize + 'px'; // Reset to max font size
	            while (textElement.scrollHeight > availableHeight && fontSize > 10) {
	                fontSize -= 1;
	                textElement.style.fontSize = fontSize + 'px';
	            }
	        });
	    };
	    
	    adjustFontSize();
	    
		window.addEventListener('resize', () => {
			adjustCardHeights(); // Ajuste les hauteurs lors du redimensionnement
			adjustFontSize();
		});
	});
]
