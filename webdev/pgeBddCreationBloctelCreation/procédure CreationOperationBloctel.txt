procédure CreationOperationBloctel(pIdBaseSource, pIdFiltreSource)

dhLimite est une DateHeure = DateHeureSys()

dhLimite.Minute += 15


VariableRAZ(_stOperation)

_stOperation.DateHeureCreation							= DateHeureSys()
_stOperation.DateEnvoiSouhaite							= DateSys()
_stOperation.DateHeureLimitePriseEnCharge				= dhLimite
_stOperation.DateHeureLimiteValidation					= dhLimite
_stOperation.Statut										= __Statut_Règlement_En_Attente__
_stOperation.UserCreation								= Partage.__vcUser.NomUser
_stOperation.IdClient									= Partage.__vcUser.IDClient
_stOperation.IdClientFacturation						= Partage.__vcUser.IDClient
_stOperation.IdAdresseFacturation						= Partage.AdresseDeFacturation(Partage.__vcUser.IDClient)
_stOperation.IdContact									= Partage.__vcUser.IDContact
_stOperation.IdValideur									= Partage.__vcUser.IDContact
_stOperation.TypeOperation								= __OPERATION_BLOCTEL__
_stOperation.TypeProjet									= __OPERATION_BLOCTEL__
_stOperation.NbrDestinataire							= (_nTotalPortable + _nTotalTelephone)
_stOperation.NomOperation								= "Traitement Bloctel fichier " + SansEspace(saiNom) + " - " + NumériqueVersChaîne((_nTotalPortable + _nTotalTelephone), "dS") + " numéros"
_stOperation.Designation								= "Traitement Bloctel fichier du " + DateVersChaîne(DateSys(),"Jjj JJ Mmm AAAA")
_stOperation.EstBrouillon								= Faux
_stOperation.EstModifiableParUnTiers					= Faux
_stOperation.EstBaseEnAttente							= Faux
_stOperation.EstDocumentEnAttente						= Faux
_stOperation.EstInjectionEnAttente						= Faux
_stOperation.EstChronodirect							= Faux
_stOperation.EstCliquezIci								= Faux
_stOperation.EstEnvoiImmediat							= Faux
_stOperation.EstVisibleParLeClient						= Vrai
_stOperation.EstVisibleParUnTiers						= Faux
_stOperation.DateHeureValidation						= DateHeureSys()
_stOperation.MontantBrutHT								= (_nTotalPortable + _nTotalTelephone) * _mPrixVenteUnitaireHT
_stOperation.MontantNetHT								= (_nTotalPortable + _nTotalTelephone) * _mPrixVenteUnitaireHT
_stOperation.MontantTVA									= ((_nTotalPortable + _nTotalTelephone) * _mPrixVenteUnitaireHT) * cpProjet.__moTauxTva / 100
_stOperation.MontantTTC									= ((_nTotalPortable + _nTotalTelephone) * _mPrixVenteUnitaireHT) + ((_nTotalPortable + _nTotalTelephone) * _mPrixVenteUnitaireHT) * cpProjet.__moTauxTva / 100
_stOperation.ResteAPayer								= ((_nTotalPortable + _nTotalTelephone) * _mPrixVenteUnitaireHT) + ((_nTotalPortable + _nTotalTelephone) * _mPrixVenteUnitaireHT) * cpProjet.__moTauxTva / 100

_stOperation.stOperationBloctel.IdBaseClient			= 0
_stOperation.stOperationBloctel.IdBaseSource			= pIdBaseSource
_stOperation.stOperationBloctel.IdFiltreSource			= pIdFiltreSource
_stOperation.stOperationBloctel.NomBaseClient			= SansEspace(saiNom)
_stOperation.stOperationBloctel.Statut					= __STATUT_BLOCTEL_EN_PREPARATION__
_stOperation.stOperationBloctel.NbrTelephoneEntree		= _nTotalTelephone
_stOperation.stOperationBloctel.NbrPortableEntree		= _nTotalPortable
_stOperation.stOperationBloctel.PrixUnitaireAchat		= _mPrixAchatUnitaireHT
_stOperation.stOperationBloctel.PrixUnitaireFacture		= _mPrixVenteUnitaireHT

renvoyer AjoutModificationOperationBloctel(_stOperation)
