procédure ImportImageDessus()

sRepertoireUpload	est une chaîne
sImageUpload		est une chaîne
sRepImgUpload		est une chaîne
imgUpload			est une WL.Image


sRepertoireUpload	= cpProjet.__sRepRacine + ["\"] + cpProjet.__sRepUpload
imgImageDessus			= ""
libRésolutionDessus		= ""


// On supprime l'ancien fichier
si _sDessusFichierGénéré <> "" alors
	si fFichierExiste(sRepertoireUpload +"\"+ _sDessusFichierGénéré) alors
		fSupprime(sRepertoireUpload +"\"+ _sDessusFichierGénéré)
	fin
fin


_sDessusFichierGénéré 	= GénèreMotDePasse(16)
sImageUpload			= UploadCopieFichier(uplImageDessus, sRepertoireUpload, _sDessusFichierGénéré)
_sDessusFichierGénéré	+= fExtraitChemin(sImageUpload,fExtension)
_sGrattageExtension		= Minuscule(fExtraitChemin(_sDessusFichierGénéré,fExtension))

sRepImgUpload		= sRepertoireUpload + "\" + _sDessusFichierGénéré

si Contient(_sGrattageExtension,"svg",MotComplet+SansCasse) alors
	imgImageDessus		= sRepImgUpload
	libRésolutionDessus	= ""
	_nImage1Hauteur		= 0
	_nImage1Largeur		= 0

sinon
	imgUpload			= sRepImgUpload
	_nImage1Hauteur		= imgUpload.Hauteur
	_nImage1Largeur		= imgUpload.Largeur

	si _bConversionAutomatique = Vrai _et_ (_nImage1Hauteur > _nImageUploadHauteurMax _ou_ _nImage1Largeur > _nImageUploadLargeurMax _ou_ _nImage1Largeur < _nImageUploadLargeurMin) alors
		si RedimentionneImage(sRepImgUpload,sRepertoireUpload) alors
			imgUpload			= sRepImgUpload
			_nImage1Hauteur		= imgUpload.Hauteur
			_nImage1Largeur		= imgUpload.Largeur
			PopupAfficheMessage("Conversion de l'image réussie","Conversion",__OPERATION_MARKETEAM__,__TOAST_INFORMATION__,2 s)
		sinon
			PopupAfficheMessage("Échec de conversion" + RC + sRepImgUpload,"Conversion",__OPERATION_MARKETEAM__,__TOAST_ANNULATION__,15 s)
			retour
		fin
	fin

	selon Vrai
		cas _nImage1Hauteur > _nImageUploadHauteurMax _et_ _nImage1Largeur > _nImageUploadLargeurMax
			PopupAfficheMessage("La largeur ("+ _nImage1Hauteur + " px) et la hauteur (" + _nImage1Hauteur + " px) de votre image sont hors norme" + RC + "Largeur Maximum : " + _nImageUploadLargeurMax + " px" +RC + "Hauteur Maximum : " + _nImageUploadHauteurMax + " px","Format d'image incorrect",__OPERATION_MARKETEAM__,__TOAST_ANNULATION__, 15 s)
			imgUpload			= ""
			saiNomImage			= ""
			libRésolutionDessus	= ""

		cas _nImage1Largeur < _nImageUploadLargeurMin
			PopupAfficheMessage("La largeur de votre image ("+ _nImage1Largeur + " px) est hors norme" + RC + "Largeur Minimum : " + _nImageUploadLargeurMin + " px","Format d'image incorrect",__OPERATION_MARKETEAM__,__TOAST_ANNULATION__, 15 s)
			imgUpload			= ""
			saiNomImage			= ""
			libRésolution		= ""

		cas _nImage1Largeur > _nImageUploadLargeurMax
			PopupAfficheMessage("La largeur de votre image ("+ _nImage1Largeur + " px) est hors norme" + RC + "Largeur Maximum : " + _nImageUploadLargeurMax + " px","Format d'image incorrect",__OPERATION_MARKETEAM__,__TOAST_ANNULATION__, 15 s)
			imgUpload			= ""
			saiNomImage			= ""
			libRésolutionDessus	= ""

		cas _nImage1Hauteur > _nImageUploadHauteurMax
			PopupAfficheMessage("La hauteur de votre image ("+ _nImage1Hauteur + " px) est hors norme" + RC + "Hauteur Maximum : " + _nImageUploadHauteurMax + " px","Format d'image incorrect",__OPERATION_MARKETEAM__,__TOAST_ANNULATION__, 15 s)
			imgUpload			= ""
			saiNomImage			= ""
			libRésolutionDessus	= ""


		autre cas
			imgImageDessus		= imgUpload
			libRésolutionDessus	= "Image dessus : " + NumériqueVersChaîne(_nImage1Largeur, "dS") +" px x " + NumériqueVersChaîne(_nImage1Hauteur, "dS") +" px"

	fin
fin
