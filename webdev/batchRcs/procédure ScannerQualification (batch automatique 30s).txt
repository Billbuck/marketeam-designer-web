
// Paramètres d'automatisme de la procédure :

// - Type d'automatisme : Timer/Procédure différée.
// - Nombre d'exécutions : infini.
// - Espacement entre les appels : 30 secondes (00:00:30:00 au format HH:MM:SS:CC).
// - Comportement lors des appels suivants de la procédure : ignorer le timer et exécuter immédiatement la procédure.
// - Mode de démarrage : démarrage automatique après le code d'initialisation de la fenêtre, avec un délai de 30 secondes (00:00:30:00).


// Résumé : Qualification RCS des bases clients (capability check Infobip, gratuit).
// Mode HYBRIDE selon le nombre de portables distincts (cahier du 23/09/2026) :
//
//	SYNCHRONE (n <= Rcs_Qualification_AutoMax) : paquets de
//	Rcs_Capacite_TailleLotSync (500) envoyés à /rcs/2/capability-check/query
//	par vagues de Rcs_Capacite_Parallelisme (4) tâches parallèles
//	(TacheQualifCapacite : réponse immédiate, écriture des EstRcs par
//	valeur, jauge NbrTraiteRcs). Plus de 10 % de paquets en échec → ERREUR.
//	Fin dans le même cycle : TRAITE + compteurs (mesures § 6 : 500 numéros
//	en 1,23 s, 4 paquets parallèles en 1,74 s, 50 000 en ~45 s).
//
//	ASYNCHRONE (n > AutoMax) : lots de Rcs_Capacite_TailleLot (10 000) vers
//	/rcs/2/capability-check/notify (EnvoieCapabilityCheck), résultats reçus
//	par la page pgeRcsCapacite qui tague les lignes et les compteurs.
//	La CONSOLIDATION est faite ici (seule autorité) : quand tous les lots
//	sont TERMINE, ou quand le dernier retour date de plus de
//	Rcs_Capacite_InactiviteMinutes : lignes restées NULL → invalide,
//	lots restants TERMINE, puis TRAITE + compteurs.
//
// Filet : base EN_COURS sans aucun lot depuis plus de
// Rcs_Capacite_InactiviteMinutes (échec avant le premier envoi, ou crash
// pendant un traitement synchrone) → ERREUR.
procédure ScannerQualification()

nTailleLotSync	est un entier
nTailleLotAsync	est un entier
nParallelisme	est un entier
nAutoMax			est un entier
nInactiviteMin	est un entier
sSender			est une chaîne
sNotifyUrl		est une chaîne
sCleWebhook		est une chaîne
nIdBase			est un entier
sNomBase			est une chaîne
sSociete			est une chaîne
nNbrPortable	est un entier
nNbrLot			est un entier
sTexteRequete	est une chaîne
sdReqBase		est une Source de Données
sdReqAgent		est une Source de Données
sdReqPortable	est une Source de Données
sdReqMaj			est une Source de Données
sdReqCompte		est une Source de Données
sdReqEcheance	est une Source de Données
tabNumeros		est un tableau de chaînes
bErreurBase		est un booléen
dhDebutBase		est une DateHeure
duBase			est une Durée


	// Normalisation d'un portable pour Infobip : sans espace ni +,
	// 06XXXXXXXX → 336XXXXXXXX (les bases sont déjà en 33X, ceinture)
	procédure interne NormalisePortableQualif(pNumeroCdb)
		sNumCdb est une chaîne = SansEspace(pNumeroCdb)
		sNumCdb = Remplace(sNumCdb, " ", "")
		sNumCdb = Remplace(sNumCdb, "+", "")
		si Gauche(sNumCdb, 1) = "0" _et_ Taille(sNumCdb) = 10 alors
			sNumCdb = "33" + Milieu(sNumCdb, 2)
		fin
		renvoyer sNumCdb
	fin


	// Échappement SQL minimal (apostrophes)
	procédure interne EchappeSqlQualif(pTexteCdb)
		renvoyer Remplace(pTexteCdb, "'", "''")
	fin


	// Changement de statut d'une base (sans compteurs)
	procédure interne MajStatutBaseQualif(pIdBaseCdb, pStatutCdb)
		sReqCdb	est une chaîne
		sdMajCdb	est une Source de Données

		sReqCdb = [
			UPDATE clt_base SET
			StatutRcs = '[%pStatutCdb%]',
			DateHeureModification = NOW(),
			UserModification = 'batchRcs'
			WHERE clt_base.IdBase = [%pIdBaseCdb%]
		]
		si pas sdMajCdb.ExécuteRequêteSQL(clt_base..Connexion, hRequêteSansCorrection, sReqCdb) alors
			Trace("ScannerQualification : erreur UPDATE statut base " + pIdBaseCdb, HErreurInfo(hErrComplet))
		fin
		sdMajCdb.AnnuleDéclaration()
	fin


	// Consolidation d'une base : TRAITE + compteurs RCS / SMS seul / invalides
	procédure interne ConsolideBaseQualif(pIdBaseCdb)
		sReqCdb	est une chaîne
		sdMajCdb	est une Source de Données

		sReqCdb = [
			UPDATE clt_base SET
			StatutRcs = '[%__BASE_RCS_TRAITE__%]',
			DateHeureRcs = NOW(),
			DateHeureModification = NOW(),
			UserModification = 'batchRcs',
			NbrRcs = (SELECT COUNT(*) FROM clt_base_ligne WHERE clt_base_ligne.IdBase = [%pIdBaseCdb%] AND clt_base_ligne.EstRcs = [%__LIGNE_RCS_OUI__%]),
			NbrSmsSeul = (SELECT COUNT(*) FROM clt_base_ligne WHERE clt_base_ligne.IdBase = [%pIdBaseCdb%] AND clt_base_ligne.EstRcs = [%__LIGNE_RCS_SMS__%]),
			NbrInvalide = (SELECT COUNT(*) FROM clt_base_ligne WHERE clt_base_ligne.IdBase = [%pIdBaseCdb%] AND clt_base_ligne.EstRcs = [%__LIGNE_RCS_INVALIDE__%])
			WHERE clt_base.IdBase = [%pIdBaseCdb%]
			AND clt_base.StatutRcs = '[%__BASE_RCS_EN_COURS__%]'
		]
		si pas sdMajCdb.ExécuteRequêteSQL(clt_base..Connexion, hRequêteSansCorrection, sReqCdb) alors
			Trace("ScannerQualification : erreur consolidation base " + pIdBaseCdb, HErreurInfo(hErrComplet))
		fin
		sdMajCdb.AnnuleDéclaration()
	fin


	// Envoi d'un lot ASYNCHRONE (notify) et enregistrement dans clt_base_rcs_lot.
	// Utilise sSender et sNotifyUrl de la procédure hôte.
	procédure interne EnvoieLotQualif(pIdBaseCdb, tabNumerosCdb est un tableau de chaînes)
		sReqCdb	est une chaîne
		sdLotCdb	est une Source de Données
		sBulkCdb	est une chaîne
		sErrCdb	est une chaîne

		si cpRcs.EnvoieCapabilityCheck(sSender, tabNumerosCdb, sNotifyUrl, sBulkCdb, sErrCdb) alors
			sReqCdb = [
				INSERT INTO clt_base_rcs_lot (IdBase, BulkId, NbrEnvoye, Statut)
				VALUES ([%pIdBaseCdb%], '[%EchappeSqlQualif(sBulkCdb)%]', [%tabNumerosCdb..Occurrence%], 'ENVOYE')
			]
			si pas sdLotCdb.ExécuteRequêteSQL(clt_base_rcs_lot..Connexion, hRequêteSansCorrection, sReqCdb) alors
				Trace("ScannerQualification : erreur INSERT lot base " + pIdBaseCdb, HErreurInfo(hErrComplet))
			fin
			sdLotCdb.AnnuleDéclaration()
			renvoyer Vrai
		fin

		// Échec HTTP : lot en erreur, message conservé (500 caractères max)
		sErrCdb = Gauche(sErrCdb, 500)
		sReqCdb = [
			INSERT INTO clt_base_rcs_lot (IdBase, NbrEnvoye, Statut, Erreur)
			VALUES ([%pIdBaseCdb%], [%tabNumerosCdb..Occurrence%], 'ERREUR', '[%EchappeSqlQualif(sErrCdb)%]')
		]
		si pas sdLotCdb.ExécuteRequêteSQL(clt_base_rcs_lot..Connexion, hRequêteSansCorrection, sReqCdb) alors
			Trace("ScannerQualification : erreur INSERT lot erreur base " + pIdBaseCdb, HErreurInfo(hErrComplet))
		fin
		sdLotCdb.AnnuleDéclaration()
		Trace("ScannerQualification : échec capability check asynchrone base " + pIdBaseCdb + " : " + sErrCdb)
		renvoyer Faux
	fin


	// Qualification SYNCHRONE d'une base : paquets de nTailleLotSync envoyés
	// par vagues de nParallelisme tâches parallèles (TacheQualifCapacite).
	// Renvoie Faux si plus de 10 % des paquets sont en échec.
	procédure interne QualifieBaseSync(pIdBaseSync)
		sReqSync				est une chaîne
		sdReqPortSync		est une Source de Données
		tabPaquetSync		est un tableau de chaînes
		sPaquetSync			est une chaîne
		nDansPaquetSync	est un entier
		tabTacheSync		est un tableau de TâchesParallèles
		nIndPaquetSync		est un entier
		nIndTacheSync		est un entier
		nEchecSync			est un entier
		nRetourSync			est un entier

		// Extraction des portables distincts, découpage en paquets (chaînes TAB)
		sReqSync = [
			SELECT DISTINCT clt_base_ligne.Portable
			FROM clt_base_ligne
			WHERE clt_base_ligne.IdBase = [%pIdBaseSync%]
			AND clt_base_ligne.Portable IS NOT NULL
			AND clt_base_ligne.Portable <> ''
		]
		si pas sdReqPortSync.ExécuteRequêteSQL(clt_base_ligne..Connexion, hRequêteSansCorrection, sReqSync) alors
			Trace("ScannerQualification : erreur extraction portables base " + pIdBaseSync, HErreurInfo(hErrComplet))
			renvoyer Faux
		fin
		pour tout sdReqPortSync
			si sPaquetSync <> "" alors sPaquetSync += TAB
			sPaquetSync += NormalisePortableQualif(sdReqPortSync.Portable)
			nDansPaquetSync++
			si nDansPaquetSync >= nTailleLotSync alors
				tabPaquetSync.Ajoute(sPaquetSync)
				sPaquetSync			= ""
				nDansPaquetSync	= 0
			fin
		fin
		sdReqPortSync.AnnuleDéclaration()
		si sPaquetSync <> "" alors tabPaquetSync.Ajoute(sPaquetSync)

		si tabPaquetSync..Occurrence = 0 alors renvoyer Vrai

		// Vagues de nParallelisme tâches, attente de la vague avant la suivante
		nIndPaquetSync = 1
		tantque nIndPaquetSync <= tabPaquetSync..Occurrence
			tabTacheSync.SupprimeTout()
			pour nIndTacheSync = 1 _à_ nParallelisme
				si nIndPaquetSync > tabPaquetSync..Occurrence alors sortir
				tabTacheSync.Ajoute(TâcheParallèleExécute(TacheQualifCapacite, (pIdBaseSync, sSender, tabPaquetSync[nIndPaquetSync])))
				nIndPaquetSync++
			fin
			TâcheParallèleAttendToutes(tabTacheSync)
			pour nIndTacheSync = 1 _à_ tabTacheSync..Occurrence
				nRetourSync = tabTacheSync[nIndTacheSync].ValeurRenvoyée
				si nRetourSync = 0 alors nEchecSync++
			fin
		fin

		si nEchecSync * 10 > tabPaquetSync..Occurrence alors
			Trace("ScannerQualification : base " + pIdBaseSync + " en erreur - " + nEchecSync + " paquets en échec sur " + tabPaquetSync..Occurrence)
			renvoyer Faux
		fin
		renvoyer Vrai
	fin


// ------------------------------------------------------------------
// 0) Paramètre d'inactivité (utile même sans nouvelle base)
// ------------------------------------------------------------------
nInactiviteMin = cpFonction.ValeurParametre("Rcs_Capacite_InactiviteMinutes", prm_parametre.ValNumerique..Nom)
si nInactiviteMin <= 0 alors nInactiviteMin = 10

// ------------------------------------------------------------------
// 1) Bases en attente de qualification
// ------------------------------------------------------------------
sTexteRequete = [
	SELECT
	clt_base.IdBase,
	IFNULL(clt_base.Nom, '') Nom,
	IFNULL(clt_client.Societe, '') Societe
	FROM
	clt_base
	LEFT JOIN clt_client ON clt_client.IdClient = clt_base.IdClient
	WHERE
	clt_base.StatutRcs = '[%__BASE_RCS_A_TRAITER__%]'
]

si pas sdReqBase.ExécuteRequêteSQL(clt_base..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Trace("Erreur requête ScannerQualification", sTexteRequete, HErreurInfo(hErrComplet))
	retour
fin

si sdReqBase.LitPremier(hSansRafraîchir) alors

	// Agent RCS actif (le premier) et paramètres, lus seulement s'il y a du travail
	sTexteRequete = [
		SELECT rcs_agent.SenderName
		FROM rcs_agent
		WHERE rcs_agent.EstActif = 1
		ORDER BY rcs_agent.IdRcsAgent
		LIMIT 1
	]
	si sdReqAgent.ExécuteRequêteSQL(rcs_agent..Connexion, hRequêteSansCorrection, sTexteRequete) alors
		si sdReqAgent.LitPremier() alors sSender = sdReqAgent.SenderName
	fin
	sdReqAgent.AnnuleDéclaration()

	nTailleLotSync = cpFonction.ValeurParametre("Rcs_Capacite_TailleLotSync", prm_parametre.ValNumerique..Nom)
	si nTailleLotSync <= 0 _ou_ nTailleLotSync > 500 alors nTailleLotSync = 500

	nParallelisme = cpFonction.ValeurParametre("Rcs_Capacite_Parallelisme", prm_parametre.ValNumerique..Nom)
	si nParallelisme <= 0 alors nParallelisme = 4

	nAutoMax = cpFonction.ValeurParametre("Rcs_Qualification_AutoMax", prm_parametre.ValNumerique..Nom)
	si nAutoMax <= 0 alors nAutoMax = 50000

	nTailleLotAsync = cpFonction.ValeurParametre("Rcs_Capacite_TailleLot", prm_parametre.ValNumerique..Nom)
	si nTailleLotAsync <= 0 _ou_ nTailleLotAsync > 10000 alors nTailleLotAsync = 10000

	sNotifyUrl	= cpFonction.ValeurParametre("Rcs_Capacite_Webhook_Url", prm_parametre.ValChaine..Nom)
	sCleWebhook	= cpFonction.ValeurParametre("Rcs_Webhook_Cle", prm_parametre.ValChaine..Nom)
	si sNotifyUrl = "Erreur" alors sNotifyUrl = ""
	si sCleWebhook = "Erreur" alors sCleWebhook = ""
	sNotifyUrl += sCleWebhook

	si sSender = "" alors
		Trace("ScannerQualification : aucun agent RCS actif, bases laissées en attente")
	sinon

		pour tout sdReqBase
			nIdBase		= sdReqBase.IdBase
			sNomBase		= sdReqBase.Nom
			sSociete		= sdReqBase.Societe
			bErreurBase	= Faux

			// (a) Nombre de portables distincts : aiguillage synchrone / asynchrone
			nNbrPortable = 0
			sTexteRequete = [
				SELECT COUNT(DISTINCT clt_base_ligne.Portable) NbrPortable
				FROM clt_base_ligne
				WHERE clt_base_ligne.IdBase = [%nIdBase%]
				AND clt_base_ligne.Portable IS NOT NULL
				AND clt_base_ligne.Portable <> ''
			]
			si sdReqCompte.ExécuteRequêteSQL(clt_base_ligne..Connexion, hRequêteSansCorrection, sTexteRequete) alors
				si sdReqCompte.LitPremier() alors nNbrPortable = sdReqCompte.NbrPortable
			fin
			sdReqCompte.AnnuleDéclaration()

			// (b) Prise en main : EN_COURS + compteurs de jauge
			sTexteRequete = [
				UPDATE clt_base SET
				StatutRcs = '[%__BASE_RCS_EN_COURS__%]',
				NbrPortableRcs = [%nNbrPortable%],
				NbrTraiteRcs = 0,
				DateHeureModification = NOW(),
				UserModification = 'batchRcs'
				WHERE clt_base.IdBase = [%nIdBase%]
			]
			si pas sdReqMaj.ExécuteRequêteSQL(clt_base..Connexion, hRequêteSansCorrection, sTexteRequete) alors
				Trace("ScannerQualification : erreur passage EN_COURS base " + nIdBase, HErreurInfo(hErrComplet))
			fin

			// (c) Requalification propre : résultats précédents effacés,
			// lots de la qualification précédente purgés
			sTexteRequete = "UPDATE clt_base_ligne SET EstRcs = NULL WHERE IdBase = " + nIdBase
			si pas sdReqMaj.ExécuteRequêteSQL(clt_base_ligne..Connexion, hRequêteSansCorrection, sTexteRequete) alors
				Trace("ScannerQualification : erreur remise à NULL EstRcs base " + nIdBase, HErreurInfo(hErrComplet))
			fin
			sTexteRequete = "DELETE FROM clt_base_rcs_lot WHERE IdBase = " + nIdBase
			si pas sdReqMaj.ExécuteRequêteSQL(clt_base_rcs_lot..Connexion, hRequêteSansCorrection, sTexteRequete) alors
				Trace("ScannerQualification : erreur purge des lots base " + nIdBase, HErreurInfo(hErrComplet))
			fin
			sdReqMaj.AnnuleDéclaration()

			// (d) Base sans portable : traitée à vide
			si nNbrPortable = 0 alors
				ConsolideBaseQualif(nIdBase)
				Trace("ScannerQualification : base " + nIdBase + " sans portable, traitée à vide")
				tblQualification.AjouteLigne(DateHeureSys(), sNomBase, sSociete, 0, 0, "", 0, 0, 0, "Aucun portable")
				continuer
			fin

			si nNbrPortable <= nAutoMax alors
				// ------------------------------------------------------
				// Mode SYNCHRONE : réponse immédiate, fin dans ce cycle
				// ------------------------------------------------------
				nNbrLot		= PartieEntière((nNbrPortable + nTailleLotSync - 1) / nTailleLotSync)
				dhDebutBase	= DateHeureSys()

				si QualifieBaseSync(nIdBase) alors
					ConsolideBaseQualif(nIdBase)
					duBase = DateHeureSys() - dhDebutBase

					// Compteurs consolidés pour l'affichage
					sTexteRequete = "SELECT IFNULL(NbrRcs,0) NbrRcs, IFNULL(NbrSmsSeul,0) NbrSmsSeul, IFNULL(NbrInvalide,0) NbrInvalide FROM clt_base WHERE IdBase = " + nIdBase
					si sdReqCompte.ExécuteRequêteSQL(clt_base..Connexion, hRequêteSansCorrection, sTexteRequete) alors
						si sdReqCompte.LitPremier() alors
							tblQualification.AjouteLigne(DateHeureSys(), sNomBase, sSociete, nNbrPortable, nNbrLot, PartieEntière(duBase..EnMillisecondes / 1000) + " s", sdReqCompte.NbrRcs, sdReqCompte.NbrSmsSeul, sdReqCompte.NbrInvalide, "Terminée")
						fin
					fin
					sdReqCompte.AnnuleDéclaration()
					Trace("ScannerQualification : base " + nIdBase + " qualifiée en " + PartieEntière(duBase..EnMillisecondes / 1000) + " s (" + nNbrPortable + " portables, " + nNbrLot + " paquets)")
				sinon
					MajStatutBaseQualif(nIdBase, __BASE_RCS_ERREUR__)
					tblQualification.AjouteLigne(DateHeureSys(), sNomBase, sSociete, nNbrPortable, nNbrLot, "", "", "", "", "Erreur (échecs > 10 %)")
				fin

			sinon
				// ------------------------------------------------------
				// Mode ASYNCHRONE : lots de 10 000, résultats par webhook
				// ------------------------------------------------------
				si sNotifyUrl = "" alors
					Trace("ScannerQualification : Rcs_Capacite_Webhook_Url manquant, base " + nIdBase + " en erreur")
					MajStatutBaseQualif(nIdBase, __BASE_RCS_ERREUR__)
					continuer
				fin

				nNbrLot = 0
				tabNumeros.SupprimeTout()

				si pas sdReqPortable.ExécuteRequêteSQL(clt_base_ligne..Connexion, hRequêteSansCorrection, ChaîneConstruit("SELECT DISTINCT clt_base_ligne.Portable FROM clt_base_ligne WHERE clt_base_ligne.IdBase = %1 AND clt_base_ligne.Portable IS NOT NULL AND clt_base_ligne.Portable <> ''", nIdBase)) alors
					Trace("ScannerQualification : erreur extraction portables base " + nIdBase, HErreurInfo(hErrComplet))
					MajStatutBaseQualif(nIdBase, __BASE_RCS_ERREUR__)
					continuer
				fin

				pour tout sdReqPortable
					tabNumeros.Ajoute(NormalisePortableQualif(sdReqPortable.Portable))
					si tabNumeros..Occurrence >= nTailleLotAsync alors
						si pas EnvoieLotQualif(nIdBase, tabNumeros) alors
							bErreurBase = Vrai
							sortir
						fin
						nNbrLot++
						tabNumeros.SupprimeTout()
					fin
				fin
				sdReqPortable.AnnuleDéclaration()

				si bErreurBase = Faux _et_ tabNumeros..Occurrence > 0 alors
					si EnvoieLotQualif(nIdBase, tabNumeros) alors
						nNbrLot++
						tabNumeros.SupprimeTout()
					sinon
						bErreurBase = Vrai
					fin
				fin

				si bErreurBase alors
					MajStatutBaseQualif(nIdBase, __BASE_RCS_ERREUR__)
					tblQualification.AjouteLigne(DateHeureSys(), sNomBase, sSociete, nNbrPortable, nNbrLot, "", "", "", "", "Erreur d'envoi")
				sinon
					tblQualification.AjouteLigne(DateHeureSys(), sNomBase, sSociete, nNbrPortable, nNbrLot, "", "", "", "", "Envoyé - attente retours")
				fin
			fin
		fin

	fin
fin
sdReqBase.AnnuleDéclaration()

// ------------------------------------------------------------------
// 2) Consolidation ASYNCHRONE (seule autorité) : tous les lots TERMINE,
//    ou aucun retour depuis Rcs_Capacite_InactiviteMinutes
// ------------------------------------------------------------------
sTexteRequete = [
	SELECT clt_base.IdBase, IFNULL(clt_base.Nom, '') Nom
	FROM clt_base
	WHERE clt_base.StatutRcs = '[%__BASE_RCS_EN_COURS__%]'
	AND EXISTS (SELECT 1 FROM clt_base_rcs_lot WHERE clt_base_rcs_lot.IdBase = clt_base.IdBase)
	AND (
		NOT EXISTS (SELECT 1 FROM clt_base_rcs_lot WHERE clt_base_rcs_lot.IdBase = clt_base.IdBase AND clt_base_rcs_lot.Statut = 'ENVOYE')
		OR (SELECT MAX(COALESCE(clt_base_rcs_lot.DateHeureDernierRetour, clt_base_rcs_lot.DateHeureEnvoi)) FROM clt_base_rcs_lot WHERE clt_base_rcs_lot.IdBase = clt_base.IdBase) < NOW() - INTERVAL [%nInactiviteMin%] MINUTE
	)
]

si sdReqEcheance.ExécuteRequêteSQL(clt_base..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	pour tout sdReqEcheance
		nIdBase = sdReqEcheance.IdBase

		// Numéros restés sans réponse → invalide (lignes avec portable seulement)
		sTexteRequete = [
			UPDATE clt_base_ligne SET EstRcs = [%__LIGNE_RCS_INVALIDE__%]
			WHERE clt_base_ligne.IdBase = [%nIdBase%]
			AND clt_base_ligne.EstRcs IS NULL
			AND clt_base_ligne.Portable IS NOT NULL
			AND clt_base_ligne.Portable <> ''
		]
		si pas sdReqMaj.ExécuteRequêteSQL(clt_base_ligne..Connexion, hRequêteSansCorrection, sTexteRequete) alors
			Trace("ScannerQualification : erreur invalidation des silencieux base " + nIdBase, HErreurInfo(hErrComplet))
		fin

		// Lots encore ouverts → TERMINE
		sTexteRequete = "UPDATE clt_base_rcs_lot SET Statut = 'TERMINE', DateHeureFin = NOW() WHERE IdBase = " + nIdBase + " AND Statut = 'ENVOYE'"
		si pas sdReqMaj.ExécuteRequêteSQL(clt_base_rcs_lot..Connexion, hRequêteSansCorrection, sTexteRequete) alors
			Trace("ScannerQualification : erreur clôture des lots base " + nIdBase, HErreurInfo(hErrComplet))
		fin
		sdReqMaj.AnnuleDéclaration()

		ConsolideBaseQualif(nIdBase)
		Trace("ScannerQualification : base " + nIdBase + " (" + sdReqEcheance.Nom + ") consolidée (asynchrone)")
	fin
fin
sdReqEcheance.AnnuleDéclaration()

// ------------------------------------------------------------------
// 3) Filet : base EN_COURS sans aucun lot depuis trop longtemps
//    (échec avant le premier envoi, ou crash pendant un synchrone)
// ------------------------------------------------------------------
sTexteRequete = [
	SELECT clt_base.IdBase, IFNULL(clt_base.Nom, '') Nom
	FROM clt_base
	WHERE clt_base.StatutRcs = '[%__BASE_RCS_EN_COURS__%]'
	AND NOT EXISTS (SELECT 1 FROM clt_base_rcs_lot WHERE clt_base_rcs_lot.IdBase = clt_base.IdBase)
	AND clt_base.DateHeureModification < NOW() - INTERVAL [%nInactiviteMin%] MINUTE
]

si sdReqEcheance.ExécuteRequêteSQL(clt_base..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	pour tout sdReqEcheance
		Trace("ScannerQualification : base " + sdReqEcheance.IdBase + " (" + sdReqEcheance.Nom + ") EN_COURS sans lot depuis " + nInactiviteMin + " min, passée en erreur")
		MajStatutBaseQualif(sdReqEcheance.IdBase, __BASE_RCS_ERREUR__)
	fin
fin
sdReqEcheance.AnnuleDéclaration()
