procédure CompteurSms(local pMessage, pMessageFinalisé,  bEstUnicode est un booléen, nTailleMessage est un entier, mPrixMessage est un monétaire, nCreditMessage est un entier, mPrixUnitaireSms est un monétaire, bEstStopSms est un booléen, bEstCliquezIci est un booléen)


sMessageBrut		est une chaîne
nTailleMessage		= 0
mPrixMessage		= 0
nCreditMessage		= 0
                	
sMessageBrut		= MessageSmsBrut(pMessage)
pMessageFinalisé	= MessageSmsFinalisé(sMessageBrut, bEstCliquezIci, bEstStopSms)

si sMessageBrut <> "" alors
	nTailleMessage 	= TailleSms(pMessageFinalisé, bEstUnicode)
	nCreditMessage	= CalculeTailleSms(nTailleMessage, bEstUnicode)
	mPrixMessage	= nCreditMessage * mPrixUnitaireSms
