	procédure interne AjouteEtape(pNumero est un entier, pLibelle est une chaîne, pOk est un booléen, pDetail est une chaîne)
		sRapport += "Étape " + pNumero + " — " + pLibelle + " "
		si pOk alors
			sRapport += "OK"
		sinon
			sRapport += "ÉCHEC"
		fin
		si pDetail <> "" alors sRapport += "  (" + pDetail + ")"
		sRapport += RC
	fin

	procédure interne FabriqueBoutonTest(pOrdre est un entier, pLibelle est une chaîne)
		stBoutonNeuf est une structOperationRcsBouton
		stBoutonNeuf.IdRcsBouton	= 0
		stBoutonNeuf.Ordre			= pOrdre
		stBoutonNeuf.TypeBouton		= "OPEN_URL"
		stBoutonNeuf.Libelle		= pLibelle
		renvoyer stBoutonNeuf
	fin

	procédure interne IndiceCarteParId(pStLu est une structOperation, pIdCarte est un entier)
		nIndCherche est un entier
		pour nIndCherche = 1 _à_ pStLu.stOperationRcs.tabCarte..Occurrence
			si pStLu.stOperationRcs.tabCarte[nIndCherche].IdRcsCarte = pIdCarte alors
				renvoyer nIndCherche
			fin
		fin
		renvoyer 0
	fin

	procédure interne IndiceCarteParTitre(pStLu est une structOperation, pTitre est une chaîne)
		nIndChercheTitre est un entier
		pour nIndChercheTitre = 1 _à_ pStLu.stOperationRcs.tabCarte..Occurrence
			si pStLu.stOperationRcs.tabCarte[nIndChercheTitre].Titre = pTitre alors
				renvoyer nIndChercheTitre
			fin
		fin
		renvoyer 0
	fin

	procédure interne DetailEcartRelecture(pStLu est une structOperation)
		sDetail est une chaîne = "attendu 1 carte « Carte test 1 » avec 2 boutons"
		sDetail += ", trouvé " + pStLu.stOperationRcs.tabCarte..Occurrence + " carte(s)"
		si pStLu.stOperationRcs.tabCarte..Occurrence >= 1 alors
			sDetail += ", titre « " + pStLu.stOperationRcs.tabCarte[1].Titre + " », "
			sDetail += pStLu.stOperationRcs.tabCarte[1].tabBouton..Occurrence + " bouton(s)"
		fin
		renvoyer sDetail
	fin

	procédure interne DetailEcartNumeros(pStLu est une structOperation, pIdCarte est un entier, pIdBoutonA est un entier, pIdBoutonB est un entier)
		sDetail est une chaîne = "attendu carte " + pIdCarte + " et boutons " + pIdBoutonA + " / " + pIdBoutonB
		si pStLu.stOperationRcs.tabCarte..Occurrence >= 1 alors
			sDetail += ", trouvé carte " + pStLu.stOperationRcs.tabCarte[1].IdRcsCarte
			si pStLu.stOperationRcs.tabCarte[1].tabBouton..Occurrence >= 1 alors
				sDetail += ", bouton " + pStLu.stOperationRcs.tabCarte[1].tabBouton[1].IdRcsBouton
			fin
		sinon
			sDetail += ", trouvé 0 carte"
		fin
		renvoyer sDetail
	fin
