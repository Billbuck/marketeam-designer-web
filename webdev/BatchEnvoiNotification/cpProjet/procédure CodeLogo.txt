procédure CodeLogo(pType)

sCode est une chaîne

selon pType
	cas __OPERATION_COURRIER__ , __OPERATION_RNVP__, __OPERATION_COURRIER_GESTION__				: sCode = __OPERATION_COURRIER__
	cas __OPERATION_TRACTEVA__, __OPERATION_SMARTEVA__ 											: sCode = "TTB"
	cas __OPERATION_SMS__, __OPERATION_SMS_NOTIFICATION__, __OPERATION_SMS_INTERACTIF__			: sCode = __OPERATION_SMS__
	cas __OPERATION_VMS__, __OPERATION_VMS_UNITAIRE__											: sCode = __OPERATION_VMS__
	cas __OPERATION_EMAIL__, __OPERATION_EMAIL_NOTIFICATION__, __OPERATION_EMAIL_INTERACTIF__	: sCode = __OPERATION_EMAIL__
	cas __OPERATION_TEL__ , __OPERATION_BLOCTEL__												: sCode = __OPERATION_TEL__
	cas __OPERATION_BASE_DE_DONNEES__															: sCode = __OPERATION_BASE_DE_DONNEES__
	autres cas																					: sCode = __OPERATION_MARKETEAM__
fin

renvoyer Minuscule(sCode)
