procédure IconePrintTypeOperation(local pTypeOperation, local pEstCliquezIci)

sCheminIcone	est une chaîne
sTypeOperation	est une chaîne

sTypeOperation = Partage.TypeProjet(pTypeOperation, pEstCliquezIci)

selon sTypeOperation
	cas __OPERATION_COURRIER__										: sCheminIcone = fRepExe() + "\Print\ltr-icone-print" + __PNG__
	cas __OPERATION_TRACTEVA__										: sCheminIcone = fRepExe() + "\Print\tra-icone-print" + __PNG__
	cas __OPERATION_SMARTEVA__										: sCheminIcone = fRepExe() + "\Print\sma-icone-print" + __PNG__
	cas __OPERATION_SMS__											: sCheminIcone = fRepExe() + "\Print\sms-icone-print" + __PNG__
	cas __OPERATION_SMS_INTERACTIF__								: sCheminIcone = fRepExe() + "\Print\smi-icone-print" + __PNG__
	cas __OPERATION_VMS__											: sCheminIcone = fRepExe() + "\Print\vms-icone-print" + __PNG__
	cas __OPERATION_TEL__											: sCheminIcone = fRepExe() + "\Print\tel-icone-print" + __PNG__
	cas __OPERATION_EMAIL__											: sCheminIcone = fRepExe() + "\Print\eml-icone-print" + __PNG__
	cas __OPERATION_RNVP__, __OPERATION_BLOCTEL__					: sCheminIcone = fRepExe() + "\Print\bdd-icone-print" + __PNG__
	cas __OPERATION_COURRIER_GESTION__								: sCheminIcone = fRepExe() + "\Print\gst-icone-print" + __PNG__
	cas __OPERATION_MARKETEAM__										: sCheminIcone = fRepExe() + "\Print\mkt-icone-print" + __PNG__
	autre cas					
fin

renvoyer sCheminIcone
