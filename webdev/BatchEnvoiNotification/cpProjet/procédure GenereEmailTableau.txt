// Résumé : <indiquez ici ce que fait la procédure>
// Syntaxe :
//[ <Résultat> = ] GenereEmailTableau (<stEmail> est structEmail)
//
// Paramètres :
//	stEmail (structEmail) : <indiquez ici le rôle de stEmail>
// Valeur de retour :
// 	chaîne Unicode : <indiquez ici le rôle de la valeur de retour>
//
// Exemple :
// <Indiquez ici un exemple d'utilisation>
//
procédure GenereEmailTableau(stEmail est une structEmail)

nLigne				est un entier
nColonne			est un entier
sEnteteDeColonne	est une chaîne
sEntete				est une chaîne
sTableauDesktop		est une chaîne
sTableauMobile		est une chaîne
sLigneDesktop		est une chaîne
sLigneMobile		est une chaîne
sLigneTableau		est une chaîne
sTableauComplet		est une chaîne



pour tout stEntete de stEmail.tabTableauEntête
	sEntete = [
		<td class="table-cell" align="center" valign="middle" style="font-size:12px; color:#FFFFFF; background-color:[%stEmail.Couleur.XL%]; font-weight:bold; text-align:center; vertical-align:middle; border:1px solid [%stEmail.Couleur.Tableau%]; height:30px; padding-left:5px; padding-right:5px;">[%stEntete.Libelle%]</td>
	]
	sEnteteDeColonne += [RC] + sEntete
fin


pour nLigne = 1 _à_ stEmail.tabTableauValeur.Occurrence
	sTableauDesktop = ""
	sTableauMobile	= ""
	pour nColonne = 1 _à_ stEmail.tabTableauValeur[nLigne].Occurrence

		sLigneDesktop = [
			<td class="table-cell desktop-only" align="[%stEmail.tabTableauEntête[nColonne].Alignement%]" valign="middle">[%stEmail.tabTableauValeur[nLigne][nColonne]%]</td>
		]

		sTableauDesktop += [RC] + sLigneDesktop

		si stEmail.tabTableauValeur[nLigne][nColonne] <> "" alors
			sLigneMobile = [
				<p class="table-cell" style="margin:5px 0;"><span class="label">[%stEmail.tabTableauEntête[nColonne].Libelle%] : </span><span class="value">[%stEmail.tabTableauValeur[nLigne][nColonne]%]</span></p>
			]
			sTableauMobile += [RC] + sLigneMobile
		fin
	fin

	sLigneTableau = [
		<tr class="table-row">
			<!-- Version Desktop -->
			[%sTableauDesktop%]
			<!-- Version Mobile -->
			<!--[if !mso]><!-->
			<td class="mobile-only" style="padding:0px !important; height:auto !important; border:none;">
				[%sTableauMobile%]
			</td>
			<!--<![endif]-->
		</tr>
	]

	sTableauComplet += RC + sLigneTableau

fin

sTableau est une chaîne = [
	<!-- Conteneur Tableau -->
	<table width="100%" cellpadding="0" cellspacing="0" border="0" align="center">
		<tr>
			<td align="center" style="padding-left: 15px; padding-right: 15px;">

				<!-- Espace vide en haut -->
				<table width="100%" cellpadding="0" cellspacing="0" border="0">
					<tr>
						<td height="15"></td>
					</tr>
				</table>

				<table class="container" width="100%" cellpadding="0" cellspacing="0" border="0" align="center">
					<tr>
						<td class="content" style="padding-bottom: 15px; padding-top: 15px;">

							<!-- Tableau des données -->
							<table class="table-container" width="100%" cellpadding="0" cellspacing="0" border="0">

								<!-- En-tête du tableau -->
								<tr class="table-header desktop-only">
									[%sEnteteDeColonne%]
								</tr>

								<!-- Lignes de données -->
								[%sTableauComplet%]
							</table>
						</td>
					</tr>
				</table>

				<!-- Espace vide en bas -->
				<table width="100%" cellpadding="0" cellspacing="0" border="0">
					<tr>
						<td height="15"></td>
					</tr>
				</table>

			</td>
		</tr>
	</table>
]

renvoyer sTableau
