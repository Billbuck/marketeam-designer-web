procédure OptionModifiee(IndiceOptionModifiee)

// Ordre des options aligné sur les libellés posés par InitOngletRetours :
// 1 Intentions, 2 Interactions, 3 Redirections, 4 Réponses, 5 Leads.
// Leads par défaut (selFiltre = 5 à l'initialisation de la page).
selon IndiceOptionModifiee
	cas 1 : _sFiltreRetourFamille = __RETOUR_FAMILLE_INTENTION__
	cas 2 : _sFiltreRetourFamille = __RETOUR_FAMILLE_INTERACTION__
	cas 3 : _sFiltreRetourFamille = __RETOUR_FAMILLE_REDIRECTION__
	cas 4 : _sFiltreRetourFamille = __RETOUR_FAMILLE_REPONSE__
	cas 5 : _sFiltreRetourFamille = __RETOUR_FAMILLE_LEAD__
fin

InitOngletRetours()
