procédure OptionModifiee(IndiceOptionModifiee)

// Ordre des options aligné sur les libellés posés par InitOngletRetours
// (décision du 29/09) : 1 Leads, 2 Intentions, 3 Interactions,
// 4 Redirections, 5 Réponses. Leads par défaut (option 1).
selon IndiceOptionModifiee
	cas 1 : _sFiltreRetourFamille = __RETOUR_FAMILLE_LEAD__
	cas 2 : _sFiltreRetourFamille = __RETOUR_FAMILLE_INTENTION__
	cas 3 : _sFiltreRetourFamille = __RETOUR_FAMILLE_INTERACTION__
	cas 4 : _sFiltreRetourFamille = __RETOUR_FAMILLE_REDIRECTION__
	cas 5 : _sFiltreRetourFamille = __RETOUR_FAMILLE_REPONSE__
fin

InitOngletRetours()
