// Résumé : Onglet Retours (remplace InitOngletLead du SMS).
// Un seul tableau pour tous les retours de l'opération (vue_rcs_retour) :
// leads, intentions de contact, interactions, redirections et réponses.
// Décisions du 29/09 : les réponses sont affichées dès la V1 (le texte de
// la réponse est dans la colonne Détail) ; les redirections sont une
// famille à part (REDIRECTION), plus une action de la famille INTERACTION.
// Sélecteur selFiltre en en-tête, 5 options dans l'ordre des familles :
// 1 Leads, 2 Intentions, 3 Interactions, 4 Redirections, 5 Réponses —
// Leads par défaut. OptionModifiee pose _sFiltreRetourFamille et rappelle
// cette procédure : filtre côté serveur, une seule condition sur la famille.
// La zone réutilise znrLead du SMS avec trois attributs ajoutés :
// attRetourAction, attRetourDetail, attRetourCanal.
procédure InitOngletRetours()

nInd					est un entier
sTexteRequete		est une chaîne
sdReqRetour			est une Source de Données
sCondition			est une chaîne
nNbrLead				est un entier
nNbrIntention		est un entier
nNbrInteraction	est un entier
nNbrRedirection	est un entier
nNbrReponse			est un entier


// ------------------------------------------------------------------
// Pastilles : comptage par famille (réponses comprises)
// ------------------------------------------------------------------
sTexteRequete =
[
	SELECT
	vue_rcs_retour.Famille,
	COUNT(*) Total
	FROM vue_rcs_retour
	WHERE vue_rcs_retour.IdOperation = [%_pIdOperation%]
	GROUP BY vue_rcs_retour.Famille
]

si pas sdReqRetour.ExécuteRequêteSQL(clt_client..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

pour tout sdReqRetour
	selon sdReqRetour.Famille
		cas __RETOUR_FAMILLE_LEAD__			: nNbrLead			= sdReqRetour.Total
		cas __RETOUR_FAMILLE_INTENTION__		: nNbrIntention	= sdReqRetour.Total
		cas __RETOUR_FAMILLE_INTERACTION__	: nNbrInteraction	= sdReqRetour.Total
		cas __RETOUR_FAMILLE_REDIRECTION__	: nNbrRedirection	= sdReqRetour.Total
		cas __RETOUR_FAMILLE_REPONSE__		: nNbrReponse		= sdReqRetour.Total
	fin
fin

selFiltre[1].Libellé	= "Leads (" + NumériqueVersChaîne(nNbrLead, "dS") + ")"
selFiltre[2].Libellé	= "Intentions (" + NumériqueVersChaîne(nNbrIntention, "dS") + ")"
selFiltre[3].Libellé	= "Interactions (" + NumériqueVersChaîne(nNbrInteraction, "dS") + ")"
selFiltre[4].Libellé	= "Redirections (" + NumériqueVersChaîne(nNbrRedirection, "dS") + ")"
selFiltre[5].Libellé	= "Réponses (" + NumériqueVersChaîne(nNbrReponse, "dS") + ")"

// ------------------------------------------------------------------
// Filtre de la famille active (posée par OptionModifiee)
// ------------------------------------------------------------------
sCondition = "vue_rcs_retour.Famille = '" + _sFiltreRetourFamille + "'"

// ------------------------------------------------------------------
// Tableau des retours de la famille filtrée
// ------------------------------------------------------------------
sTexteRequete =
[
	SELECT
	l.contact,
	l.Civilite,
	l.Prenom,
	l.Nom,
	l.Portable,
	vue_rcs_retour.Action,
	vue_rcs_retour.Libelle,
	vue_rcs_retour.Detail,
	vue_rcs_retour.DateHeure,
	vue_rcs_retour.Canal
	FROM
	vue_rcs_retour
	JOIN dos_base_ligne l ON l.IdBaseLigne = vue_rcs_retour.IdBaseLigne
	WHERE
	vue_rcs_retour.IdOperation = [%_pIdOperation%]
	AND [%sCondition%]
	ORDER BY 
	vue_rcs_retour.DateHeure DESC
]

si pas sdReqRetour.ExécuteRequêteSQL(clt_client..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Trace("Erreur requête","Date",DateVersChaîne(DateSys(),"Jjjj JJ mmm"),"Heure",HeureVersChaîne(HeureSys(),"HH:MM:SS"),"Erreur",HErreurInfo(),"Requête",sTexteRequete)
	retour
fin

znrLead.SupprimeTout()

pour tout sdReqRetour
	nInd = znrLead.AjouteLigne()
	si sdReqRetour.Nom = "" _et_ sdReqRetour.Prenom = ""
		znrLead[nInd].attLeadContact 		= NomPropre(sdReqRetour.Contact)
	sinon
		znrLead[nInd].attLeadContact 		= CiviliteCourte(sdReqRetour.Civilite) + [" "] + Majuscule(sdReqRetour.Nom) + [" "] + NomPropre(sdReqRetour.Prenom)
	fin
	znrLead[nInd].attRetourAction			= Partage.LibelleRetourRcsAction(sdReqRetour.Action)
	znrLead[nInd].attLeadChoix				= sdReqRetour.Libelle
	znrLead[nInd].attRetourDetail			= Remplace(sdReqRetour.Detail,RC," + ")
	znrLead[nInd].attLeadDate				= DateVersChaîne(sdReqRetour.DateHeure..PartieDate,  "JJ/MM/AA")
	znrLead[nInd].attLeadHeure				= HeureVersChaîne(sdReqRetour.DateHeure..PartieHeure, "HH:MM")
	znrLead[nInd].attLeadPortable			= Telephone(sdReqRetour.Portable ,1,".")
	znrLead[nInd].attRetourCanal			= sdReqRetour.Canal
fin

fin:
sdReqRetour.AnnuleDéclaration()
