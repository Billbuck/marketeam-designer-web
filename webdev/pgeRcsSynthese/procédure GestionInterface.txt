// Résumé : Visibilité des onglets et initialisation de chaque onglet visible.
// Différences avec pgeSmsSynthese :
//	- Performance visible dès la diffusion pour tout RCS (pas seulement
//	  avec une landing) ;
//	- l'onglet leads est renommé retours, visible dès la diffusion
//	  (pas de condition sur le nombre de leads : les interactions et
//	  intentions y figurent aussi) ;
//	- InitOngletRetours remplace InitOngletLead.
procédure GestionInterface()

si _stSynthèse.IdOperation = _pIdOperation

	onglet.reglement.Visible		= (_stSynthèse.Statut dans (__Statut_Règlement_En_Attente__, __Statut_Solde_En_Attente__, __Statut_Validation_En_Attente__) _et_ _stSynthèse.ResteAPayer > 0 _et_ _stSynthèse.IdValideur = Partage.__vcUser.IDContact)
	btnOperationPayer.Visible		= (_stSynthèse.Statut dans (__Statut_Règlement_En_Attente__, __Statut_Solde_En_Attente__, __Statut_Validation_En_Attente__) _et_ _stSynthèse.ResteAPayer > 0 _et_ _stSynthèse.IdValideur = Partage.__vcUser.IDContact)
	btnOperationValider.Visible		= (_stSynthèse.Statut = __Statut_Validation_En_Attente__ _et_ _stSynthèse.IdValideur = Partage.__vcUser.IDContact)
	btnOperationRefuser.Visible		= (_stSynthèse.Statut dans (__Statut_Validation_En_Attente__, __Statut_Règlement_En_Attente__) _et_ _stSynthèse.IdValideur = Partage.__vcUser.IDContact _et_ _stSynthèse.IdContact <> Partage.__vcUser.IDContact)
	onglet.rapport.Visible			= (_stSynthèse.Statut > __Statut_Prise_En_Charge__)
	onglet.performance.Visible		= (_stSynthèse.Statut > __Statut_Prise_En_Charge__)
	onglet.retours.Visible			= (_stSynthèse.Statut > __Statut_Prise_En_Charge__)
	btnReprogrammer.Visible			= (_stSynthèse.Statut <= __Statut_Prise_En_Charge_En_Attente__)

	onglet							= {"onglet." + _pOnglet,indChamp}

	InitOngletSynthèse()
	InitOngletInformations()
	InitOngletStatut()
	InitOngletDevis()

	si onglet.rapport.Visible			alors InitOngletRapport()
	si onglet.performance.Visible		alors InitOngletPerformance()
	si onglet.retours.Visible			alors InitOngletRetours()
	si onglet.reglement.Visible			alors InitOngletReglement()

	libTitre = onglet[onglet]..Libellé
fin
