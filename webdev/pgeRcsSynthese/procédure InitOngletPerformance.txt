// Résumé : Onglet Performance d'une opération RCS.
// Bloc Synthèse : Envoyés, Délivrés (dont SMS de repli), Lus, Interactions,
// Intentions de contact, Réponses (nombre seul), Leads — taux (29/09) :
// lus / délivrés, puis interactions, intentions et leads SUR LES DÉLIVRÉS.
// Champs RENOMMÉS depuis le SMS : Visite → Lu, Lien → Interaction.
// Champs NOUVEAUX : Intention (total / taux / nouveau), Réponse (total /
// nouveau), libPerformanceReçuRepli, tableaux tblInteractionBouton
// (famille INTERACTION) et tblIntention (famille INTENTION) — colonnes
// Catégorie / Bouton / Destinataires uniques pour les deux.
// Tableau des leads (29/09) : Réactions des boutons RCS pour toutes les
// opérations, puis offres de la landing pour les opérations interactives.
// Bloc redirections : landing uniquement (EstCliquezIci).
procédure InitOngletPerformance()

nCategorie				est un entier
sNomCookie				est une chaîne
sCookie					est une chaîne
vJsonCookie				est un Variant
bCookieExiste			est un booléen
sTexteRequete			est une chaîne
sdReqPerformance		est une Source de Données
sListeField				est une chaîne
sOrderBy					est une chaîne

btnTéléchargerPerformance.Visible		= _stSynthèse.TypeCliquezici dans (__ICI_URL_TRACKING__, __ICI_WEB__) _et_ (_stSynthèse.NbrAudienceUnique + _stSynthèse.NbrInteractionUnique + _stSynthèse.NbrLeadUnique) > 0
btnBddPerformance.Visible					= _stSynthèse.TypeCliquezici dans (__ICI_URL_TRACKING__, __ICI_WEB__)

//-------------------------------------------------------------------------------------------------------------------------------------------

sNomCookie										= "Performance|" + Partage.__vcUser.IDContact + "|" + _pIdOperation
sCookie											= CookieLit(sNomCookie)
si sCookie <> "" alors
	vJsonCookie		= JSONVersVariant(sCookie)
	bCookieExiste	= Vrai
sinon
	// Le cookie n'existait pas
	vJsonCookie.Envoyé		= 0
	vJsonCookie.Reçu			= 0
	vJsonCookie.Lu				= 0
	vJsonCookie.Interaction	= 0
	vJsonCookie.Intention	= 0
	vJsonCookie.Reponse		= 0
	vJsonCookie.Lead			= 0
fin

soit nNouveauEnvoyé			= Max(0,_stSynthèse.NbrDiffusion 				- Val(vJsonCookie.Envoyé))
soit nNouveauReçu			= Max(0,_stSynthèse.NbrDelivre 					- Val(vJsonCookie.Reçu))
soit nNouveauLu				= Max(0,_stSynthèse.NbrLu						- Val(vJsonCookie.Lu))
soit nNouveauInteraction	= Max(0,_stSynthèse.NbrInteractionUnique		- Val(vJsonCookie.Interaction))
soit nNouveauIntention		= Max(0,_stSynthèse.NbrIntentionContactUnique	- Val(vJsonCookie.Intention))
soit nNouveauReponse		= Max(0,_stSynthèse.NbrReponse					- Val(vJsonCookie.Reponse))
soit nNouveauLead			= Max(0,_stSynthèse.NbrLeadUnique				- Val(vJsonCookie.Lead))


libPerformanceNouveauEnvoye					= "+" + NumériqueVersChaîne(nNouveauEnvoyé, "dS")
libPerformanceNouveauEnvoye.Visible			= (nNouveauEnvoyé > 0)

libPerformanceNouveauReçu						= "+" + NumériqueVersChaîne(nNouveauReçu, "dS")
libPerformanceNouveauReçu.Visible			= (nNouveauReçu > 0)

libPerformanceNouveauLu							= "+" + NumériqueVersChaîne(nNouveauLu, "dS")
libPerformanceNouveauLu.Visible				= (nNouveauLu > 0)

libPerformanceNouveauInteraction				= "+" + NumériqueVersChaîne(nNouveauInteraction, "dS")
libPerformanceNouveauInteraction.Visible	= (nNouveauInteraction > 0)

libPerformanceNouveauIntention				= "+" + NumériqueVersChaîne(nNouveauIntention, "dS")
libPerformanceNouveauIntention.Visible		= (nNouveauIntention > 0)

libPerformanceNouveauReponse					= "+" + NumériqueVersChaîne(nNouveauReponse, "dS")
libPerformanceNouveauReponse.Visible		= (nNouveauReponse > 0)

libPerformanceNouveauLead						= "+" + NumériqueVersChaîne(nNouveauLead, "dS")
libPerformanceNouveauLead.Visible			= (nNouveauLead > 0)


// On enregistre le Cookie
vJsonCookie.Envoyé								= _stSynthèse.NbrDiffusion
vJsonCookie.Reçu									= _stSynthèse.NbrDelivre
vJsonCookie.Lu										= _stSynthèse.NbrLu
vJsonCookie.Interaction							= _stSynthèse.NbrInteractionUnique
vJsonCookie.Intention							= _stSynthèse.NbrIntentionContactUnique
vJsonCookie.Reponse								= _stSynthèse.NbrReponse
vJsonCookie.Lead									= _stSynthèse.NbrLeadUnique

CookieEcrit(sNomCookie, VariantVersJSON(vJsonCookie))


//-----------------------------------------------------------------------


libPerformanceTotalEnvoye				= NumériqueVersChaîne(_stSynthèse.NbrDiffusion, "dS")
libPerformanceTotalReçu					= NumériqueVersChaîne(_stSynthèse.NbrDelivre, "dS")
libPerformanceTotalLu					= NumériqueVersChaîne(_stSynthèse.NbrLu, "dS")
libPerformanceTotalInteraction		= NumériqueVersChaîne(_stSynthèse.NbrInteractionUnique, "dS")
libPerformanceTotalIntention			= NumériqueVersChaîne(_stSynthèse.NbrIntentionContactUnique, "dS")
libPerformanceTotalReponse				= NumériqueVersChaîne(_stSynthèse.NbrReponse, "dS")
libPerformanceTotalLead					= NumériqueVersChaîne(_stSynthèse.NbrLeadUnique, "dS")

//// dont n par SMS de repli, sous Délivrés
//si _stSynthèse.NbrRepliSms > 0 alors
//	libPerformanceReçuRepli				= "dont " + NumériqueVersChaîne(_stSynthèse.NbrRepliSms, "dS") + " par SMS de repli"
//	libPerformanceReçuRepli.Visible	= Vrai
//sinon
//	libPerformanceReçuRepli.Visible	= Faux
//fin

libPerformanceTauxEnvoye				= (_stSynthèse.NbrDestinataire > 0)	? Arrondi(_stSynthèse.NbrDiffusion					/ _stSynthèse.NbrDestinataire * 100,0)	+ " %"	sinon "0 %"
libPerformanceTauxReçu					= (_stSynthèse.NbrDiffusion > 0)	? Arrondi(_stSynthèse.NbrDelivre					/ _stSynthèse.NbrDiffusion * 100,0) 	+ " %"	sinon "0 %"
libPerformanceTauxLu						= (_stSynthèse.NbrDelivre > 0)		? Arrondi(_stSynthèse.NbrLu							/ _stSynthèse.NbrDelivre * 100,0) 		+ " %"	sinon "0 %"
// Taux des cartes Interactions / Intentions / Leads calculés sur les DÉLIVRÉS
// (correction du 29/09) : le taux de lecture est un plancher (accusés de lecture
// bloqués par certains réglages), les délivrés sont la base fiable.
libPerformanceTauxInteraction			= (_stSynthèse.NbrDelivre > 0)		? Arrondi(_stSynthèse.NbrInteractionUnique			/ _stSynthèse.NbrDelivre * 100,0) 		+ " %"	sinon "0 %"
libPerformanceTauxIntention			= (_stSynthèse.NbrDelivre > 0)		? Arrondi(_stSynthèse.NbrIntentionContactUnique		/ _stSynthèse.NbrDelivre * 100,0) 		+ " %"	sinon "0 %"
libPerformanceTauxLead					= (_stSynthèse.NbrDelivre > 0)		? Arrondi(_stSynthèse.NbrLeadUnique					/ _stSynthèse.NbrDelivre * 100,0)	 	+ " %"	sinon "0 %"


// **************************** TABLEAUX INTERACTIONS ET INTENTIONS PAR BOUTON *****************************

// Une seule mesure par bouton : les destinataires uniques (correction du 29/09 —
// la colonne Clics a été retirée). Une seule requête pour les deux familles :
// INTERACTION alimente tblInteractionBouton, INTENTION alimente tblIntention
// (placée entre Interactions et Redirections). Colonnes des deux tables :
// Catégorie, Bouton, Destinataires uniques.
sTexteRequete								=
[
	SELECT
	vue_rcs_retour.Famille,
	vue_rcs_retour.Action,
	vue_rcs_retour.Libelle,
	COUNT(DISTINCT vue_rcs_retour.IdBaseLigne) NbrUnique
	FROM vue_rcs_retour
	WHERE vue_rcs_retour.IdOperation = [%_pIdOperation%]
	AND vue_rcs_retour.Famille IN ('[%__RETOUR_FAMILLE_INTERACTION__%]','[%__RETOUR_FAMILLE_INTENTION__%]')
	GROUP BY vue_rcs_retour.Famille, vue_rcs_retour.Action, vue_rcs_retour.Libelle
	ORDER BY NbrUnique DESC
]

si pas HExécuteRequêteSQL(sdReqPerformance, ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

tblInteractionBouton.SupprimeTout()
tblIntention.SupprimeTout()

pour tout sdReqPerformance
	si sdReqPerformance.Famille = __RETOUR_FAMILLE_INTENTION__ alors
		tblIntention.AjouteLigne(Partage.LibelleRetourRcsAction(sdReqPerformance.Action), sdReqPerformance.Libelle, sdReqPerformance.NbrUnique)
	sinon
		tblInteractionBouton.AjouteLigne(Partage.LibelleRetourRcsAction(sdReqPerformance.Action), sdReqPerformance.Libelle, sdReqPerformance.NbrUnique)
	fin
fin

tblInteractionBouton.Visible	= (tblInteractionBouton.Occurrence > 0)
tblIntention.Visible				= (tblIntention.Occurrence > 0)


// **************************** TABLEAU DES LEADS — RÉACTIONS *****************************
// Tous les leads (correction du 29/09) : d'abord les Réactions des boutons RCS
// (famille LEAD, action REACTION, destinataires uniques par libellé) — pour
// toutes les opérations, avant le « retour » du bloc landing. Les offres de la
// landing s'ajoutent à la suite plus bas.

tblLead.SupprimeTout()
nCategorie	= 0

sTexteRequete								=
[
	SELECT
	vue_rcs_retour.Libelle,
	COUNT(DISTINCT vue_rcs_retour.IdBaseLigne) Total
	FROM vue_rcs_retour
	WHERE vue_rcs_retour.IdOperation = [%_pIdOperation%]
	AND vue_rcs_retour.Famille = '[%__RETOUR_FAMILLE_LEAD__%]'
	AND vue_rcs_retour.Action = '[%__RCS_ACTION_REACTION__%]'
	GROUP BY vue_rcs_retour.Libelle
	ORDER BY Total DESC
]

si pas HExécuteRequêteSQL(sdReqPerformance, ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

pour tout sdReqPerformance
	nCategorie++
	tblLead.AjouteLigne(sdReqPerformance.Libelle, sdReqPerformance.Total)
fin


// **************************** BLOC RCS INTERACTIF (landing) *****************************

si _stSynthèse.EstCliquezIci = Faux alors
	// dispStatistique[4].Visible	= Faux
	// dispStatistique[5].Visible	= Faux
	retour
fin



sTexteRequete								=
[
	SELECT TypeLien, Libelle, Url, COUNT(DISTINCT IdBaseLigne) Total FROM ope_cliquezici_lien WHERE IdCliquezIci = [%_pIdOperation%] GROUP BY TypeLien, Libelle, Url
]

si pas HExécuteRequêteSQL(sdReqPerformance, ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

tblRedirection.SupprimeTout()
nCategorie	= 0
pour tout sdReqPerformance
	nCategorie++
	tblRedirection.AjouteLigne()
	tblRedirection[nCategorie].colTypeLien	= Partage.LibelleCliquezIciTracking(sdReqPerformance.TypeLien)
	tblRedirection[nCategorie].colLien		= sdReqPerformance.Libelle
	tblRedirection[nCategorie].colUrl		= Remplace(sdReqPerformance.Url,"#","")
	tblRedirection[nCategorie].colTotal		= sdReqPerformance.Total
fin


sTexteRequete = [
	SELECT
	GROUP_CONCAT("\"", COALESCE(ope_cliquezici_offre.Libelle, ope_cliquezici_offre.Offre), "\"") ListeField
	FROM
	ope_cliquezici_offre
	JOIN ope_cliquezici_bloc ON ope_cliquezici_bloc.IdCliquezIciBloc = ope_cliquezici_offre.IdCliquezIciBloc AND ope_cliquezici_bloc.`Type` IN ('[%__BLOC_OFFRE__%]' ,'[%__BLOC_FORMULAIRE__%]' , '[%__BLOC_BOUTON__%]')
	WHERE
	ope_cliquezici_bloc.IdCliquezIci = [%_pIdOperation%]
	ORDER BY
	FIELD(ope_cliquezici_bloc.`Type`, '[%__BLOC_OFFRE__%]' , '[%__BLOC_FORMULAIRE__%]' , '[%__BLOC_BOUTON__%]')
]

si pas HExécuteRequêteSQL(sdReqPerformance, ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

si sdReqPerformance.LitPremier() alors
	sListeField = sdReqPerformance.ListeField
fin

si sListeField <> "" alors
	sOrderBy = [
		ORDER BY 
		FIELD(ope_cliquezici_clic_selection.Libelle, [%sListeField%]),
		Total DESC
	]
sinon
	sOrderBy = [
		ORDER BY 
		FIELD(ope_cliquezici_clic_selection.`Type`, '[%__FORMULAIRE_INTERRUPTEUR__%]', '[%__FORMULAIRE_COMBO__%]','[%__FORMULAIRE_SAISIE__%]','[%__FORMULAIRE_OPTIN__%]'),
		Total DESC
	]
fin


sTexteRequete =
[
	SELECT
	ope_cliquezici_clic_selection.Libelle,
	ope_cliquezici_clic_selection.Valeur,
	COUNT(ope_cliquezici_clic_selection.IdCliquezIciSelection) AS Total
	FROM 
	ope_cliquezici_clic
	JOIN 
	ope_cliquezici_clic_selection ON ope_cliquezici_clic_selection.IdCliquezIciClic = ope_cliquezici_clic.IdCliquezIciClic
	JOIN (
	SELECT 
	IdBaseLigne, 
	MAX(IdCliquezIciClic) AS MaxIdCliquezIciClic
	FROM 
	ope_cliquezici_clic
	WHERE 
	IdCliquezIci = [%_pIdOperation%]
	GROUP BY 
	IdBaseLigne
	) AS dernierClic ON dernierClic.MaxIdCliquezIciClic = ope_cliquezici_clic.IdCliquezIciClic
	WHERE
	ope_cliquezici_clic.IdCliquezIci = [%_pIdOperation%]
	GROUP BY 
	ope_cliquezici_clic_selection.Libelle,
	ope_cliquezici_clic_selection.Valeur
	[%sOrderBy%]
	
]

si pas HExécuteRequêteSQL(sdReqPerformance, ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

// Les offres s'ajoutent à la suite des Réactions (pas de remise à zéro)
pour tout sdReqPerformance
	nCategorie++
	tblLead.AjouteLigne(Majuscule(SansEspace(Remplace(sdReqPerformance.Libelle,":",""))) + [" : "] + sdReqPerformance.Valeur, sdReqPerformance.Total)
fin

fin:
HAnnuleDéclaration(sdReqPerformance)
