// (5) Code JavaScript : pointer events + resize + pinceau adaptatif
sCodeJS =
[
	$(document).ready(function() {
	
	// Sélecteurs
	const canvas  = document.getElementById("scratch-canvas");
	const ctx     = canvas.getContext("2d");
	const container = document.getElementById("scratch-container");
	const imgBelow = document.getElementById("imgBelow");
	
	const imgAbove = new Image();
	imgAbove.crossOrigin = "anonymous";
	imgAbove.src = "[%sUrlImageDessus%]"; // image du dessus
	
	// Paramètres
	let isDrawing       = false;
	let revealThreshold = 60; // % de grattage pour tout révéler
	// Valeur "de base" pour le pinceau quand la largeur conteneur ~ pLargeurL
	let baseBrushRadius = 40; 
	let referenceWidth  = [%pLargeurL%]; 
	
	// Dès que l'image du dessus est chargée, on initialise
	imgAbove.onload = function() {
	initScratch();
	};
	
	// Quand la fenêtre est redimensionnée => réinitialiser le grattage
	// (Evite le décalage du curseur)
	$(window).on("resize", function() {
	initScratch();
	});
	
	// Fonction init : recalcule la taille du canvas, redessine l'image
	function initScratch() {
	let width  = container.offsetWidth;
	let height = container.offsetHeight;
	
	canvas.width  = width;
	canvas.height = height;
	
	// Calcul adaptatif du pinceau
	let scale = width / referenceWidth;
	let brushRadius = baseBrushRadius * scale;
	if (brushRadius < 5) brushRadius = 5; // minimum
	
	// Dessiner l'image du dessus en "source-over"
	ctx.globalCompositeOperation = "source-over";
	ctx.drawImage(imgAbove, 0, 0, width, height);
	
	// Re-attacher les pointer events
	canvas.removeEventListener("pointerdown", onPointerDown);
	canvas.removeEventListener("pointermove", onPointerMove);
	canvas.removeEventListener("pointerup",   onPointerUp);
	canvas.removeEventListener("pointerleave",onPointerUp);
	
	// On stocke le brushRadius dans une variable globale
	window._brushRadius = brushRadius;
	
	// Attacher pointer events
	canvas.addEventListener("pointerdown", onPointerDown);
	canvas.addEventListener("pointermove", onPointerMove);
	canvas.addEventListener("pointerup",   onPointerUp);
	canvas.addEventListener("pointerleave",onPointerUp);
	
	isDrawing = false;
	}
	
	// Pointer events handlers
	function onPointerDown(e) {
	e.preventDefault();
	isDrawing = true;
	}
	function onPointerUp(e) {
	e.preventDefault();
	isDrawing = false;
	}
	function onPointerMove(e) {
	if(!isDrawing) return;
	e.preventDefault();
	
	let rect = canvas.getBoundingClientRect();
	let x = e.clientX - rect.left;
	let y = e.clientY - rect.top;
	
	let brushRadius = window._brushRadius || 40;
	ctx.globalCompositeOperation = "destination-out";
	ctx.beginPath();
	ctx.arc(x, y, brushRadius, 0, 2 * Math.PI);
	ctx.fill();
	
	let percent = getScratchedPercent();
	// console.log("Pourcentage gratté =", percent.toFixed(2));
	if(percent >= revealThreshold) {
	// Tout effacer => image dessous visible
	ctx.clearRect(0, 0, canvas.width, canvas.height);
	// Désactiver le grattage
	canvas.removeEventListener("pointermove", onPointerMove);
	// alert("Félicitations, plus de " + revealThreshold + "% a été gratté !");
	}
	}
	
	// Calcul du pourcentage gratté (alpha=0 => transparent)
	function getScratchedPercent() {
	try {
	let imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
	let pixels = imageData.data;
	let total = pixels.length / 4;
	let countTransparent = 0;
	
	for(let i = 3; i < pixels.length; i += 4) {
	if(pixels[i] === 0) {
	countTransparent++;
	}
	}
	return (countTransparent / total) * 100;
	} catch(e) {
	console.error("Lecture impossible (CORS?)", e);
	return 0;
	}
	}
	
	});
]

TableauAjoute(_tabCodeJS, sCodeJS)
renvoyer sCodeHTML

// PROCÉDURE INTERNE : CalculLargeurHauteur
	// ----------------------------------------------------------------
	procédure interne CalculLargeurHauteur(pLargeurPage)

		nHauteurCalc	est un entier
		nHauteurMax		est un entier	= 100

		si nLargeurImg = 0 alors nLargeurImg = 1
		si nHauteurImg = 0 alors nHauteurImg = 1

		si sFormat = "SVG" alors
			// Cas particulier : SVG => Hauteur en px, largeur en auto
			sHauteurCalc	= (pLargeurPage * nHauteurMax / 100) + "px"
			sLargeurCalc	= "auto"
		sinon
			// Calcul standard
			nHauteurCalc	= Min(nHauteurImg, pLargeurPage * nHauteurMax / 100, pLargeurPage * nHauteurImg / nLargeurImg)
			sHauteurCalc	= nHauteurCalc + "px"
			sLargeurCalc	= Min(nLargeurImg, (nLargeurImg / nHauteurImg) * nHauteurCalc) + "px"
		fin

		renvoyer (sLargeurCalc, sHauteurCalc)
	fin
