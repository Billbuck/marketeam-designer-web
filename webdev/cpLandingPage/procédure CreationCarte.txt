sCodeJS += RC + [
    function placeAllMarkers() {
        var pending = listMarkers[%pNumBloc%].length;
        if (pending === 0) {
            return;
        }

        listMarkers[%pNumBloc%].forEach(function(mObj) {
            if (mObj.lat !== null && mObj.lng !== null) {
                createMarker(mObj.lat, mObj.lng, mObj.title);
                doneOne();
            } else if (mObj.address) {
                geocoder[%pNumBloc%].geocode({ address: mObj.address }, function(results, status) {
                    if (status === "OK" && results[0]) {
                        var loc = results[0].geometry.location;
                        createMarker(loc.lat(), loc.lng(), mObj.title);
                    }
                    doneOne();
                });
            } else {
                // ni coords ni address => on ignore
                doneOne();
            }
        });

        function doneOne() {
            pending--;
            if (pending === 0) {
                // Quand tous sont traités, fitBounds si markers > 0
                if (markers[%pNumBloc%].length > 0) {

					if (markers[%pNumBloc%].length === 1) {
                        // Si un seul marqueur, définir le centre et le zoom souhaité
                        mapInstance[%pNumBloc%].setCenter(markers[%pNumBloc%][0].getPosition());
                        mapInstance[%pNumBloc%].setZoom([%stProprieteCarte.Zoom%]);
                    } else {
                        // Si plusieurs marqueurs, ajuster les limites pour les inclure tous
                        mapInstance[%pNumBloc%].fitBounds(bounds[%pNumBloc%]);
                    }
                }
            }
        }


        function createMarker(lat, lng, title) {
            var pos = { lat: lat, lng: lng };
            var mk = new google.maps.Marker({
                map: mapInstance[%pNumBloc%],
                position: pos,
                title: title || ""
            });
            markers[%pNumBloc%].push(mk);
            bounds[%pNumBloc%].extend(pos);
        }
    }

	// Appel de la fonction
	placeAllMarkers();

    // Sur resize => resize + fitBounds
    $(window).resize(function(){
        google.maps.event.trigger(mapInstance[%pNumBloc%], "resize");
        if (markers[%pNumBloc%].length > 0) {
            mapInstance[%pNumBloc%].fitBounds(bounds[%pNumBloc%]);
        }
    });

	});
]
