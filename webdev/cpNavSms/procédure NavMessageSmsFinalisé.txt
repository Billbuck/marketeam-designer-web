procédure NavMessageSmsFinalisé(local pMessage, pEstCliquezIci, pEstStopSms)

pMessage = Remplace(pMessage,RC,Caract(10))

si pEstCliquezIci alors
	selon Vrai 
		cas Contient(pMessage,"[" + cpSms.__sLibelleCliquezIci +"]",MotComplet) : pMessage = Remplace(pMessage,"[" + cpSms.__sLibelleCliquezIci +"]", cpSms.__sUrlCliquezIci)
		cas Contient(pMessage,cpSms.__sUrlCliquezIci) // On ne fait rien
		autre cas : pMessage += Caract(10) + cpSms.__sUrlCliquezIci
	fin
sinon
	pMessage	= Remplace(pMessage, Caract(10) + cpSms.__sUrlCliquezIci,"")
	pMessage	= Remplace(pMessage, " " + cpSms.__sUrlCliquezIci,"")
	pMessage	= Remplace(pMessage, cpSms.__sUrlCliquezIci,"")
	
	pMessage	= Remplace(pMessage, Caract(10) + "[" + cpSms.__sLibelleCliquezIci +"]","")
	pMessage	= Remplace(pMessage, " " + "[" + cpSms.__sLibelleCliquezIci +"]","")
	pMessage	= Remplace(pMessage, "[" + cpSms.__sLibelleCliquezIci +"]","")
fin

si pEstStopSms alors
	si pas Contient(pMessage,__STOP_SMS__,SansCasse) alors
		pMessage += Caract(10) + __STOP_SMS__
	fin
sinon
	si Contient(pMessage,__STOP_SMS__) alors
		pMessage	= Remplace(pMessage, Caract(10) + __STOP_SMS__,"")
		pMessage	= Remplace(pMessage, " " + __STOP_SMS__,"")
		pMessage	= Remplace(pMessage, __STOP_SMS__,"")
	fin
fin

renvoyer pMessage
