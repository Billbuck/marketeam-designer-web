procédure NavCalculeTailleSms(pTaille, pUnicode = Faux)

nMax est un entier
nLimit est un entier

nMax = (pUnicode = Faux) ? 160 sinon 70
nLimit = (pUnicode = Faux) ? 153 sinon 67

	renvoyer 1
