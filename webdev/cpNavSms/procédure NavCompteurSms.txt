procédure NavCompteurSms(local pMessage, pMessageFinalisé, bEstUnicode est un booléen, nTailleMessage est un entier, mPrixMessage est un numérique, nCreditMessage est un entier, mPrixUnitaireSms est un numérique, bEstStopSms est un booléen, bEstCliquezIci est un booléen)

sMessageBrut		est une chaîne
nTailleMessage		= 0
mPrixMessage		= 0
nCreditMessage		= 0

sMessageBrut		= NavMessageSmsBrut(pMessage)
pMessageFinalisé	= NavMessageSmsFinalisé(sMessageBrut,bEstCliquezIci,bEstStopSms)


si sMessageBrut <> "" alors
	nTailleMessage 	= NavTailleSms(pMessageFinalisé, bEstUnicode)
	nCreditMessage	= NavCalculeTailleSms(nTailleMessage, bEstUnicode)
	mPrixMessage	= nCreditMessage * mPrixUnitaireSms
fin
