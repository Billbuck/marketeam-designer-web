// pstBouton : bouton en cours d'édition. S'il est fourni, le sélecteur
// de mode d'ouverture est replacé sur la valeur enregistrée.
// popRcsBouton doit appeler RemplitFormulaireBouton(stParam, stBouton).
procédure RemplitFormulaireBouton(pstParam est une structRcsBoutonParametres, local pstBouton est une structOperationRcsBouton = Null)

saiBoutonTelephone			= pstParam.Telephone
saiBoutonMessage				= pstParam.Message
saiBoutonEmail					= pstParam.Email
saiBoutonEmailObjet			= pstParam.EmailObjet
saiBoutonEmailMessage		= pstParam.EmailMessage
saiBoutonConfirmation		= pstParam.Confirmation
saiBoutonUrl					= pstParam.Url
saiBoutonLatitude				= pstParam.Latitude
saiBoutonLongitude			= pstParam.Longitude
saiBoutonLieu					= pstParam.Lieu
saiBoutonAdresse				= pstParam.Adresse
saiBoutonAgendaTitre			= pstParam.AgendaTitre
saiBoutonAgendaDate			= pstParam.AgendaDate
saiBoutonAgendaHeureDebut	= pstParam.AgendaHeureDebut
saiBoutonAgendaHeureFin		= pstParam.AgendaHeureFin
saiBoutonAgendaLieu			= pstParam.AgendaLieu
saiBoutonAgendaDescription	= pstParam.AgendaDescription
saiBoutonUrlAppStore			= pstParam.UrlAppStore
saiBoutonUrlPlayStore		= pstParam.UrlPlayStore
saiBoutonCompte				= pstParam.Compte

si pstBouton <> Null alors
	PositionneOuverture()
fin


procédure interne PositionneOuverture()

	selBoutonOuverture = 1

	si pstBouton.UrlApplication = __RCS_URL_WEBVIEW__ alors
		selon pstBouton.UrlWebviewMode
			cas "FULL"
				selBoutonOuverture = 2
			cas "TALL"
				selBoutonOuverture = 3
			cas __RCS_WEBVIEW_HALF__
				selBoutonOuverture = 4
		fin
	fin
fin
