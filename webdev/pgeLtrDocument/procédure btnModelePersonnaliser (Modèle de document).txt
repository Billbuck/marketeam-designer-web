// Code Ajax serveur :
// ═══════════════════════════════════════════════════════════════════════════

jsDesigner	est un JSON
jsWrapper	est un JSON

si _stDocument.NomFichierTemp = "" alors
	retour
fin

// ─── Construction de l'échantillon d'aperçu ──────────────────────────────
tabEchantillon est un tableau de StructBaseLigne
tabEchantillon.Ajoute(Partage.__vcUser.stCoordonnées)

// ─── Construction des champs de fusion standard ──────────────────────────
// TODO : adapter l'appel ci-dessous selon votre méthode de chargement des champs standard
tabBase est un tableau de structBase
taaChamps est un tableau associatif de structBaseChamp = Partage.CalculeTaaBaseChamp(tabBase, Faux, Faux)

// ─── Recomposer le wrapper complet à chaque ouverture ───────────────────
// Le wrapper (format, pages, fonds, champs, aperçu) est toujours recalculé
// depuis _stDocument. Seule la partie data est injectée depuis le JSON stocké.
jsWrapper = ChaîneVersUTF8(cpDesigner.ComposerJsonDesignerModele(_stDocument, Partage.__vcUser.IDClient, Partage.__vcUser.IDContact, _pIdModele, saiLibellé, taaChamps, tabEchantillon, _nLimTexte, _nLimImage, _nLimBarcode, _nLimQr))

si _stDocument.JsonDesignerData <> "" alors
	jsDesigner = ChaîneVersUTF8(_stDocument.JsonDesignerData)
	// Injecter la partie data stockée (zones texte, image, barcode, etc.)
	si jsDesigner.data..Existe alors
		jsWrapper.data = jsDesigner.data
	sinon
		jsWrapper.data = jsDesigner
	fin

	// Corriger cheminFond et urlFond dans les pages du data injecté
	// Le data stocké peut contenir d'anciens chemins (Upload, ancien répertoire)
	si jsWrapper.data.pages..Occurrence > 0 _et_ jsWrapper.Document.pages..Occurrence > 0 alors
		pour nInd = 1 _à_ Min(jsWrapper.data.pages..Occurrence, jsWrapper.Document.pages..Occurrence)
			jsWrapper.data.pages[nInd].cheminFond	= jsWrapper.Document.pages[nInd].cheminFond
			jsWrapper.data.pages[nInd].urlFond		= jsWrapper.Document.pages[nInd].urlFond
			Trace("btnModelePersonnaliser : page " + nInd + " cheminFond corrigé = " + jsWrapper.data.pages[nInd].cheminFond)
		fin
	fin
fin

gsJsonDesigner = JSONVersChaîne(jsWrapper)


// ═══════════════════════════════════════════════════════════════════════════
// Code navigateur retour Ajax :
// ═══════════════════════════════════════════════════════════════════════════

// Envoyer le JSON au Designer
si gsJsonDesigner <> "" alors
	celLtrModeleDocument.Plan	= 2
	EnvoyerMessageIframe(gsJsonDesigner)
	gsJsonDesigner					= ""
fin
