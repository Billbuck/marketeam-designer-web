// CheminModele — Procédure interne
// Retourne le chemin pour un type de répertoire selon le propriétaire du modèle
// ─────────────────────────────────────────────────────────────────────────────
	procédure interne CheminModele(pTypeRepertoire est une chaîne, pPhysique est un booléen = Vrai)
		selon Vrai
			cas ltr_modele.IdMarque > 0	: renvoyer CheminMarque(ltr_modele.IdMarque, ltr_modele.Type, pTypeRepertoire, pPhysique)
			cas ltr_modele.IdClient > 0	: renvoyer CheminClient(ltr_modele.IdClient, ltr_modele.Type, pTypeRepertoire, pPhysique)
			autre cas								: renvoyer CheminMarketeam(ltr_modele.Type, pTypeRepertoire, pPhysique)
		fin
	fin
