// ─────────────────────────────────────────────────────────────────────────────
// ServeurTraiterMessageDesignerModele — Page pgeLtrDocument (modèles)
// Réception du message export du Designer
//
// Flux simplifié pour les modèles :
//   1. Sauvegarder le data (zones, pages) + constraints dans JsonDesignerData
//   2. Corriger cheminFond (peut contenir d'anciens chemins Upload)
//   3. Générer un PSMD BAT temporaire (avec placeholders pour zones image vides)
//   4. Générer le BAT JPG depuis le PSMD BAT
//   5. Pas de PSMD production, pas de Printshop, pas d'images Designer
// ─────────────────────────────────────────────────────────────────────────────
procédure ServeurTraiterMessageDesignerModele()

sJsonData est une chaîne

si gsJsonRecu = "" alors renvoyer Faux

vMessage est un Variant = JSONVersVariant(gsJsonRecu)

si vMessage.action = Null alors renvoyer Faux

selon vMessage.action

	cas "validated"

		si vMessage.success = Vrai alors

			jDoc est un JSON = ChaîneVersUTF8(gsJsonRecu)
			sJsonData = JSONVersChaîne(jDoc.data)

			si sJsonData <> "" alors

				_stDocument.EstPersonnaliseDesigner	= Vrai
				_stDocument.DateHeureDesigner			= DateHeureSys()

				// ─── 1. Sauvegarder data + constraints ───────────────────
				// Les limites sont prioritairement lues depuis le JSON exporté
				// par le Designer (popup "Limites du modèle"). En l'absence,
				// on conserve les valeurs hardcodées des variables _nLimXxx.
				jStocke est un JSON = jDoc.data

				si jDoc.data.constraints.limites.texte..Existe alors
					_nLimTexte		= jDoc.data.constraints.limites.texte
					_nLimImage		= jDoc.data.constraints.limites.image
					_nLimQr			= jDoc.data.constraints.limites.qr
					_nLimBarcode	= jDoc.data.constraints.limites.barcode
				fin

				jStocke.constraints.limites.texte				= _nLimTexte
				jStocke.constraints.limites.image				= _nLimImage
				jStocke.constraints.limites.qr					= _nLimQr
				jStocke.constraints.limites.barcode				= _nLimBarcode

				// ─── 2. Corriger cheminFond ───────────────────────────────
				si jStocke.pages..Occurrence > 0 alors
					pour nInd = 1 _à_ jStocke.pages..Occurrence
						jStocke.pages[nInd].cheminFond = _stDocument.Repertoire + _stDocument.NomFichierTemp
					fin
				fin

				_stDocument.JsonDesignerData = JSONVersChaîne(jStocke)

				// ─── 3. Préparer JSON BAT avec placeholders ──────────────
				// Copie du data pour injection des placeholders dans les zones image vides
				// Le JsonDesignerData stocké reste propre (sans placeholders)
				jBat				est un JSON			= jStocke
				sRepTravail		est une chaîne		= _stDocument.RepertoireTravail
				sMontagne		est une chaîne		= cpProjet.__sRepRacine + "Designer\assets\montagne.png"
				sHachure			est une chaîne		= cpProjet.__sRepRacine + "Designer\assets\hachure.png"

				si jBat.zonesImage..Occurrence > 0 alors
					pour nInd = 1 _à_ jBat.zonesImage..Occurrence
						si jBat.zonesImage[nInd].source.nomFichier = "" _ou_ jBat.zonesImage[nInd].source.nomFichier = Null alors

							rLargeurZone			est un réel		= jBat.zonesImage[nInd].geometrie.largeurMm
							rHauteurZone			est un réel		= jBat.zonesImage[nInd].geometrie.hauteurMm
							sCheminPlaceholder	est une chaîne	= sRepTravail + "placeholder_" + nInd + ".png"

							sUrlPlaceholder		est une chaîne	= "http://localhost:5000/PyMuPdfExtract/generate_placeholder" +...
							"?width_mm=" + rLargeurZone +...
							"&height_mm=" + rHauteurZone +...
							"&output=" + URLEncode(sCheminPlaceholder) +...
							"&montagne=" + URLEncode(sMontagne) +...
							"&hachure=" + URLEncode(sHachure)

							hPlaceholder		est une httpRequête
							hRepPlaceholder	est un httpRéponse
							hPlaceholder.URL		= sUrlPlaceholder
							hPlaceholder.Méthode	= httpGet
							hRepPlaceholder		= hPlaceholder.Envoie()

							si hRepPlaceholder.CodeEtat = 200 alors
								jBat.zonesImage[nInd].source.nomFichier	= sCheminPlaceholder
								jBat.zonesImage[nInd].source.type			= "fichier"
							fin
						fin
					fin
				fin

				// ─── 4. Générer PSMD BAT temporaire ──────────────────────
				sNodeExe					est une chaîne		= INILit("NODE", "CHEMIN", "", NomFichierIni())
				sNodeCli					est une chaîne		= INILit("NODE", "CLI", "", NomFichierIni())
				sCodeTemp				est une chaîne		= _stDocument.CodeTemp
				sCheminJsonTemp		est une chaîne		= sRepTravail + "psmd_json_" + sCodeTemp + ".tmp"
				sCheminPsmdBat			est une chaîne		= sRepTravail + "psmd_bat_" + sCodeTemp + ".psmd"

				// Substitution nom d'affichage -> nom technique dans le \fonttbl (Lot 2B)
				//   Opère sur une copie chaîne (jBat non modifié en base).
				sJsonModeleTemp		est une chaîne		= JSONVersChaîne(jBat)
				nIdClientPolice		est un entier		= 0
				quand exception dans
					nIdClientPolice = Partage.__vcUser.IDClient
				faire
					nIdClientPolice = 0
				fin
				sJsonModeleTemp = cpDesigner.SubstitueNomTechniquePolices(sJsonModeleTemp, nIdClientPolice)

				si pas fSauveTexte(sCheminJsonTemp, ChaîneVersUTF8(sJsonModeleTemp)) alors
					renvoyer Faux
				fin
				// Commande Node avec chemins ENTRE GUILLEMETS (protège les espaces
				// « C:\Program Files\… », « D:\Mes Sites\… ») — schéma identique au
				// chemin document (GenererPsmdServeurDocument). --no-bleed hors guillemets.
				sCommandeBat est une chaîne = ChaîneConstruit(
				[
					"%1" "%2" "%3" "%4"
				],
				sNodeExe, sNodeCli, sCheminJsonTemp, sCheminPsmdBat) + " --no-bleed"
				LanceAppli(sCommandeBat, exeIconise, exeBloquant)

				// Nettoyage JSON temp
				si fFichierExiste(sCheminJsonTemp) alors fSupprime(sCheminJsonTemp)

				si fFichierExiste(sCheminPsmdBat) alors

					_stDocument.CheminPsmdBat = sCheminPsmdBat

					// Lot 2C : copie des polices utilisées dans le Fonts du répertoire MODÈLE
					//   (répertoire de sortie distinct de l'opération). Non bloquant.
					cpDesigner.CopiePolicesUtiliseesVersFonts(sJsonModeleTemp, sRepTravail)

					// ─── 5. Générer BAT JPG ──────────────────────────────
					GenererBatDocumentDepuisPsmd(_stDocument, __tabEchantillonBaseLigne)

					// ─── Blanc tournant sur le BAT uniquement (pas le fond) ─
					// Le fond a déjà son blanc tournant depuis la création (UploadFichier Phase 3)
					// Seul le BAT (dans le Temp) doit être traité à chaque validation
					si _stDocument.tabPagePDF..Occurrence > 0 alors
						si _stDocument.tabPagePDF[1].EstFondPerdu = Faux alors
							pour tout stVignette de _stDocument.tabImageBat
								cpDesigner.AppliqueBlancTournant(stVignette.CheminPhysique, _stDocument.LargeurOuvert, _stDocument.HauteurOuvert)
							fin
						fin
					fin

					// ─── Mettre à jour l'aperçu ─────────────────────────
					znrAperçuUpload.SupprimeTout()
					pour nPage = 1 _à_ _stDocument.NombrePage
						znrAperçuUpload.Ajoute()
						si _stDocument.tabImageBat..Occurrence >= nPage _et_ _stDocument.tabImageBat[nPage].CheminPhysique <> "" alors
							znrAperçuUpload[nPage].imgUpload = _stDocument.tabImageBat[nPage].CheminRelatif + "?" + DateHeureSys()
						sinon si _stDocument.tabImageFond..Occurrence >= nPage alors
							znrAperçuUpload[nPage].imgUpload = _stDocument.tabImageFond[nPage].CheminRelatif + "?" + DateHeureSys()
						fin
						znrAperçuUpload[nPage].imgUpload..Visible = Vrai
					fin

				sinon
					PopupAfficheMessage("Erreur lors de la génération du document" + RC + "Merci de réitérer l'opération", "Erreur", __OPERATION_COURRIER__, __TOAST_ANNULATION__)
				fin

			fin

		fin

	cas "cancelled"

	autre cas

fin

gsJsonRecu = ""

renvoyer Vrai
