procédure InitCboMois()

sTexteRequete est une chaîne
sdReqMois est une Source de Données
dDate est une Date
dDate = DateSys()

sTexteRequete =
[
	SELECT DISTINCT
		MONTH(dos_operation.DateEnvoiSouhaite) Mois
	FROM
		dos_operation
		JOIN clt_mere ON dos_operation.IdClient = clt_mere.IdClient
		JOIN clt_client ON dos_operation.IdClient = clt_client.IdClient
	WHERE
		Statut >= [%__Statut_Prise_En_Charge_En_Attente__%]
		AND YEAR(dos_operation.DateEnvoiSouhaite) = '[%cboAnnée.ValeurMémorisée%]'
		AND clt_mere.IdMere = [%Partage.__vcUser.IdClient%]
	ORDER BY 1 DESC
]

si pas HExécuteRequêteSQL(sdReqMois, ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

cboMois.SupprimeTout()

cboMois.Ajoute("< Mois >" + gLien(0))

pour tout sdReqMois
	cboMois.Ajoute(MoisEnLettre(sdReqMois.Mois) + gLien(sdReqMois.Mois))
fin

si cboMois..Occurrence = 0 alors
	cboMois.Ajoute(DateVersMoisEnLettre(dDate) + gLien(dDate.Mois))
fin

si cboAnnée.ValeurMémorisée = Gauche(DateSys(),4) alors
	cboMois.SelectPlus(2)
sinon
	cboMois.SelectPlus(1)
fin

fin:
HAnnuleDéclaration(sdReqMois)
