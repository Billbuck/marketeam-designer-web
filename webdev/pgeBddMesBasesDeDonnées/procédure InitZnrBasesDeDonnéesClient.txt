	]

		AND clt_base.Nom LIKE '%[%saiRecherche%]%'
	]

sTexteRequete =
[
	SELECT
	clt_base.*,
	(SELECT GROUP_CONCAT(clt_base_filtre.IdFiltre SEPARATOR '\t') FROM clt_base_filtre WHERE clt_base_filtre.IdBase = clt_base.IdBase ORDER BY IdFiltre) IdFiltre,
	(SELECT GROUP_CONCAT(clt_base_filtre.Nom SEPARATOR '\n') FROM clt_base_filtre WHERE clt_base_filtre.IdBase = clt_base.IdBase ORDER BY IdFiltre) NomFiltre
	FROM
	clt_base
	WHERE
	clt_base.EstActif = 1
	[%sWhere%]
	ORDER BY IdBase DESC
]
