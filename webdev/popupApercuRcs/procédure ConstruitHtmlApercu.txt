// Résumé : Construit le HTML complet de l'écran d'aperçu RCS
// Syntaxe :
//[ <Résultat> = ] ConstruitHtmlApercu (<pJson>, <pMode>)
//
// Paramètres :
//	pJson : JSON d'aperçu (Partage.ConstruitJsonApercuRcs), 100 % ASCII
//	pMode : android | iphone | sms
// Valeur de retour :
// 	chaîne : document HTML complet (CSS + JavaScript inline), destiné à
// 	l'attribut srcdoc de l'iframe posé par ChargeEcran (l'échappement
// 	pour l'attribut est fait par l'appelant)
//
// Procédure locale de popupApercuRcs.
//
// Le HTML/CSS/JS est repris de la maquette validée
// docs/maquettes/maquette-apercu-rcs.html, avec ces adaptations :
//	- cadre du téléphone et barre de contrôle retirés (fournis par la popup),
//	  l'écran occupe 100 % de l'iFrame ;
//	- données depuis le JSON injecté (%%DONNEES%%) et mode depuis %%MODE%% ;
//	- type CARTE géré (carte unique pleine largeur, sans pile ni carrousel) ;
//	- vraies images des cartes (placeholder gris si aucune) ;
//	- codes :xxx: convertis en emoji à l'affichage (table donnees.emojis) ;
//	- balises [Libellé] affichées telles quelles (pas de personnalisation) ;
//	- actions réelles : LIEN et sous-types à URL → volet + ouverture réelle ;
//	  LANDING → volet + landing de test ; REACTION (h50), APPLICATION, EMAIL,
//	  SMS (h75) → la vraie page Bouton.awp dans une iframe du volet ;
//	  APPEL, AGENDA, ITINERAIRE, CARTE, WHATSAPP → bulle d'information ;
//	- mode sms : From + message dans la bulle verte, lien Cliquez ici
//	  rendu cliquable comme dans popupApercuSms.
//
// Les caractères hors Windows-1252 (icônes) sont écrits en entités HTML
// (&#x2197;) ou en échappements JavaScript (\u2197) : le source reste ANSI.
// L'encodage de sortie est géré par WebDev au rendu du LibelléHTML ;
// la <meta charset> interne est ignorée par un iframe srcdoc (il hérite
// de l'encodage de la page) et conservée car inoffensive.
procédure ConstruitHtmlApercu(pJson, pMode)

<script>
const donnees = %%DONNEES%%;
const mode = "%%MODE%%";
const icones = { LIEN:"\u2197", LANDING:"\u2197", REACTION:"\u2713", APPEL:"\u260E",
  AGENDA:"\u25A6", ITINERAIRE:"\u27A4", CARTE:"\u27A4", APPLICATION:"\u25A3",
  EMAIL:"\u2709", SMS:"\u2709", WHATSAPP:"\u2197" };
let pileIndex = 0;

function esc(s) { return String(s == null ? "" : s).replace(/[&<>"]/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c])); }

// Codes :xxx: convertis en emoji (table fournie par le serveur)
function emo(s) {
  s = String(s == null ? "" : s);
  if (!donnees.emojis) return s;
  return s.replace(/:([0-9a-z_+-]+):/gi, (m, code) => donnees.emojis[code] !== undefined ? donnees.emojis[code] : m);
}

function texteHtml(s) { return emo(esc(s)); }

function icone(b) { return icones[b.sousType] || "\u2197"; }

function urlBouton(b) { return String(donnees.urls.bouton || "").replace("[IdRcsBouton]", b.id); }

function heureCourante() {
  const d = new Date();
  return String(d.getHours()).padStart(2, "0") + ":" + String(d.getMinutes()).padStart(2, "0");
}

function renduEntete() {
  const e = document.getElementById("entete");
  const a = donnees.agent || {};
  document.documentElement.style.setProperty("--agent", a.couleur || "#4a5568");
  let logo;
  if (a.logo) logo = `<div class="logo"><img src="${esc(a.logo)}" alt=""></div>`;
  else logo = `<div class="logo">${esc(String(a.nom || "?").charAt(0).toUpperCase())}</div>`;
  e.innerHTML = logo + `<div class="nomagent">${texteHtml(a.nom)} <span class="verif">&#x2713;</span></div>`;
}

function carteHtml(c, i) {
  let h = `<div class="carte" data-i="${i}">`;
  const ht = "h-" + String(c.mediaHauteur || "MEDIUM").toLowerCase();
  if (c.mediaUrl) h += `<div class="media ${ht}"><img src="${esc(c.mediaUrl)}" alt=""></div>`;
  else h += `<div class="media ${ht}">Image</div>`;
  h += `<div class="corps">`;
  if (c.titre) h += `<p class="titre">${texteHtml(c.titre)}</p>`;
  if (c.texte) h += `<p class="texte">${texteHtml(c.texte)}</p>`;
  h += `</div>`;
  const btns = c.boutons || [];
  if (mode === "android") {
    if (btns.length) h += `<div class="btns">` + btns.map((b, j) => `<div class="btn" onclick="action(${i},${j})"><span class="ic">${icone(b)}</span>${texteHtml(b.libelle)}</div>`).join("") + `</div>`;
  } else if (btns.length === 1) {
    h += `<div class="ios-options" onclick="action(${i},0)">${texteHtml(btns[0].libelle)}</div>`;
  } else if (btns.length > 1) {
    h += `<div class="ios-options" onclick="ouvreOptions(${i})">Options <span style="font-size:12px">&#x25BE;</span></div>`;
  }
  return h + `</div>`;
}

function rendu() {
  renduEntete();
  document.getElementById("heure").textContent = heureCourante();
  const conv = document.getElementById("conv");
  const cartes = donnees.cartes || [];
  const estCarrousel = donnees.type === "CARROUSEL" && cartes.length > 1;
  let h = `<p class="heure">Aujourd'hui ${heureCourante()}</p>`;
  if (!estCarrousel) {
    if (cartes.length) h += carteHtml(cartes[0], 0);
  } else if (mode === "android") {
    h += `<div class="carrousel-android">` + cartes.map(carteHtml).join("") + `</div>`;
  } else {
    h += `<div class="pile" id="pile">` + cartes.map(carteHtml).join("") + `</div>`;
    h += `<div class="pile-nav" id="pileNav">` + cartes.map((c, i) => `<span class="${i === pileIndex ? "on" : ""}"></span>`).join("") + `</div>`;
    h += `<p class="aide">Glissez la carte pour voir la suivante</p>`;
  }
  const communs = donnees.boutonsCommuns || [];
  if (communs.length) h += `<div class="chips">` + communs.map((b, j) => `<span class="chip" onclick="action(-1,${j})">${texteHtml(b.libelle)}</span>`).join("") + `</div>`;
  conv.innerHTML = h;
  if (estCarrousel) { if (mode === "iphone") { placePile(); initSwipe(); } else { initGlisseAndroid(); } }
}

function initGlisseAndroid() {
  const bande = document.querySelector(".carrousel-android");
  if (!bande) return;
  let x0 = null, scroll0 = 0, bouge = false;
  bande.addEventListener("pointerdown", e => { x0 = e.clientX; scroll0 = bande.scrollLeft; bouge = false; bande.style.scrollSnapType = "none"; });
  bande.addEventListener("pointermove", e => {
    if (x0 === null) return;
    const dx = e.clientX - x0;
    if (Math.abs(dx) > 6) bouge = true;
    bande.scrollLeft = scroll0 - dx;
  });
  const fin = () => {
    if (x0 === null) return;
    x0 = null;
    const largeur = 260;
    const cible = Math.round(bande.scrollLeft / largeur) * largeur;
    bande.scrollTo({ left: cible, behavior: "smooth" });
    setTimeout(() => { bande.style.scrollSnapType = "x mandatory"; }, 350);
  };
  bande.addEventListener("pointerup", fin);
  bande.addEventListener("pointerleave", fin);
  bande.addEventListener("click", e => { if (bouge) { e.stopPropagation(); e.preventDefault(); } }, true);
}

function placePile() {
  const cartes = document.querySelectorAll("#pile .carte");
  const n = cartes.length;
  cartes.forEach(c => {
    const i = +c.dataset.i;
    const rang = (i - pileIndex + n) % n;
    c.style.zIndex = n - rang;
    c.style.transform = `translateY(${-rang * 16}px) scale(${1 - rang * 0.05})`;
    c.style.opacity = rang > 3 ? 0 : 1;
    c.classList.toggle("devant", rang === 0);
  });
  document.querySelectorAll("#pileNav span").forEach((s, i) => s.classList.toggle("on", i === pileIndex));
}

function initSwipe() {
  const pile = document.getElementById("pile");
  let x0 = null;
  pile.addEventListener("pointerdown", e => { x0 = e.clientX; });
  pile.addEventListener("pointerup", e => {
    if (x0 === null) return;
    const dx = e.clientX - x0; x0 = null;
    if (Math.abs(dx) < 40) return;
    const top = pile.querySelector(`.carte[data-i="${pileIndex}"]`);
    top.style.transform = `translateX(${dx < 0 ? -420 : 420}px) rotate(${dx < 0 ? -8 : 8}deg)`;
    setTimeout(() => { pileIndex = (pileIndex + 1) % donnees.cartes.length; placePile(); }, 300);
  });
}

function ouvreOptions(i) {
  const c = donnees.cartes[i];
  const html = `<div class="poignee"></div><div class="listeopt">` +
    c.boutons.map((b, j) => `<div class="btn" onclick="action(${i},${j})"><span class="ic">${icone(b)}</span>${texteHtml(b.libelle)}</div>`).join("") +
    `<div class="btn" style="color:#6b7280" onclick="fermeVolet()">Annuler</div></div>`;
  ouvreVolet(html, "h50");
}

function ouvreVolet(html, taille) {
  const v = document.getElementById("volet");
  v.className = "volet " + taille;
  v.innerHTML = html;
  document.getElementById("voile").classList.add("on");
}
function fermeVolet() { document.getElementById("voile").classList.remove("on"); }

function toast(txt) {
  const t = document.getElementById("toast");
  t.textContent = txt; t.classList.add("on");
  clearTimeout(t._h); t._h = setTimeout(() => t.classList.remove("on"), 3200);
}

function voletLien(u, note, libelle) {
  return `<div class="poignee"></div><div class="adresse"><span>${esc(u)}</span><button class="fermer" onclick="fermeVolet()" aria-label="Fermer">&#x2715;</button></div>
    <div class="webpage"><p>${note}</p><p><a href="${esc(u)}" target="_blank" rel="noopener">${libelle} &#x2197;</a></p></div>`;
}

function voletIframe(u) {
  return `<div class="poignee"></div><div class="adresse"><span>${esc(u)}</span><button class="fermer" onclick="fermeVolet()" aria-label="Fermer">&#x2715;</button></div>
    <iframe src="${esc(u)}"></iframe>`;
}

function action(i, j) {
  const b = i < 0 ? donnees.boutonsCommuns[j] : donnees.cartes[i].boutons[j];
  fermeVolet();
  const surTel = mode === "iphone" ? "l'iPhone" : "l'Android";
  switch (b.sousType) {
    case "LIEN":
      ouvreVolet(voletLien(b.url || "", "Le navigateur du téléphone ouvre cette adresse.", "Ouvrir réellement dans un nouvel onglet"), "h100");
      break;
    case "LANDING":
      ouvreVolet(voletLien(donnees.urls.landing || "", "Landing page de test du destinataire (code @test).", "Ouvrir la landing page"), "h100");
      break;
    case "REACTION":
      ouvreVolet(voletIframe(urlBouton(b)), "h50");
      break;
    case "APPLICATION":
    case "EMAIL":
    case "SMS":
      ouvreVolet(voletIframe(urlBouton(b)), "h75");
      break;
    case "APPEL":
      toast(`Sur ${surTel} : le composeur s'ouvre avec le ${b.telephone || ""}.`);
      break;
    case "AGENDA": {
      const a = b.agenda || {};
      toast(`Sur ${surTel} : l'agenda s'ouvre avec l'événement « ${emo(a.titre || "")} — ${a.date || ""} de ${a.debut || ""} à ${a.fin || ""} ».`);
      break; }
    case "ITINERAIRE":
      toast(`Sur ${surTel} : Plans / Google Maps s'ouvre avec l'itinéraire vers ${emo(b.adresse || "")}.`);
      break;
    case "CARTE":
      toast(`Sur ${surTel} : la carte s'ouvre sur ${emo(b.lieu || ((b.latitude || "") + ", " + (b.longitude || "")))}.`);
      break;
    case "WHATSAPP":
      toast(`Sur ${surTel} : WhatsApp s'ouvre sur la conversation${b.url ? " (" + b.url + ")" : ""}.`);
      break;
    default:
      // Sous-type non cité : comportement LIEN si une URL est présente
      if (b.url) ouvreVolet(voletLien(b.url, "Le navigateur du téléphone ouvre cette adresse.", "Ouvrir réellement dans un nouvel onglet"), "h100");
      else toast(`Ce bouton (${b.sousType || "?"}) s'exécute sur le téléphone.`);
  }
}

// Mode sms : From + message dans la bulle verte, comme popupApercuSms
function renduSms() {
  const r = donnees.repli || {};
  const e = document.getElementById("entete");
  e.innerHTML = `<div class="logo sms-logo">${esc(String(r.from || "?").charAt(0).toUpperCase())}</div><div class="nomagent">${esc(r.from)}</div>`;
  document.getElementById("heure").textContent = heureCourante();
  let brut = String(r.message || "");
  const gab = (donnees.urls && donnees.urls.cliquezIci) || "";
  let lien = "";
  if (gab && brut.indexOf(gab) >= 0) {
    const affiche = esc(gab.replace("[Code]", "Test"));
    if (donnees.urls.landing) lien = `<a href="${esc(donnees.urls.landing)}" target="_blank" rel="noopener">${affiche}</a>`;
    else lien = `<span class="sms-lien">${affiche}</span>`;
    brut = brut.split(gab).join("\u0001");
  }
  let msg = esc(brut);
  if (lien) msg = msg.split("\u0001").join(lien);
  msg = emo(msg);
  document.getElementById("conv").innerHTML = `<p class="heure">Aujourd'hui ${heureCourante()}</p><div class="sms-bulle">${msg}</div>`;
}

if (mode === "sms") renduSms(); else rendu();
