// Résumé : Reconstruit l'écran d'aperçu RCS dans le mode demandé
// Syntaxe :
//ChargeEcran (<pMode>)
//
// Paramètres :
//	pMode : android | iphone | sms
//
// Construit le document HTML complet de l'écran (ConstruitHtmlApercu,
// procédure locale) puis le pose dans ztrEcran..LibelléHTML sous forme
// d'un iframe alimenté par l'attribut srcdoc : le JavaScript du document
// s'exécute dans l'iframe, sans page ni adresse. L'iframe prend la
// taille du champ (ztrEcran..Largeur / ..Hauteur) et reste transparent :
// le cadre du téléphone est dessiné en CSS par le document lui-même.
// En mode sms, le cadre reste celui du dernier mode Android / iPhone.
//
// Échappement pour l'attribut HTML : & EN PREMIER (sinon il doublerait
// les échappements suivants), puis " < >. Le navigateur dé-échappe une
// fois en lisant l'attribut : le document interne retrouve exactement
// le HTML d'origine, entités comprises.
procédure ChargeEcran(pMode)

sHtmlEcran		est une chaîne
sHtmlEchappe	est une chaîne
sModeCadre		est une chaîne	= pMode

si pMode = "sms" alors sModeCadre = __sModeEcran

sHtmlEcran					= ConstruitHtmlApercu(__sJsonApercu, pMode, sModeCadre)

sHtmlEchappe					= Remplace(sHtmlEcran, "&", "&amp;")
sHtmlEchappe					= Remplace(sHtmlEchappe, """", "&quot;")
sHtmlEchappe					= Remplace(sHtmlEchappe, "<", "&lt;")
sHtmlEchappe					= Remplace(sHtmlEchappe, ">", "&gt;")

ztrEcran..LibelléHTML		= "<iframe style=""width:" + ztrEcran..Largeur + "px;height:" + ztrEcran..Hauteur + "px;border:0;display:block;background:transparent"" allowtransparency=""true"" srcdoc=""" + sHtmlEchappe + """></iframe>"
