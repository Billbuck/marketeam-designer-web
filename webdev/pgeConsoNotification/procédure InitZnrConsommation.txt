procédure InitZnrConsommation()

nInd				est un entier

sTexteRequete		est une chaîne
sdReqConsommation	est une Source de Données
sWhere				est une chaîne
mPériodePayé		est un monétaire
mPériodeImpayé		est un monétaire
mPériodeTotal		est un monétaire

si cboMois.ValeurMémorisée > 0 alors
	sWhere = "AND MONTH(DateHeureEnvoi) = '" + cboMois.ValeurMémorisée + "'"
fin

si cboSociété.ValeurMémorisée > 0 alors
	sWhere += [RC] + "AND clt_client.IdClient = " + cboSociété.ValeurMémorisée
fin


selon cboTypeOperation.ValeurMémorisée

	cas __OPERATION_SMS_NOTIFICATION__
		sTexteRequete =
		[
			SELECT
			'[%__OPERATION_SMS_NOTIFICATION__%]' TypeProjet,
			ope_sms_notification.IdClient,
			Societe,
			ope_sms_notification.Portable Destinataire,
			IdSmsNotification id,
			ope_sms_notification.NbrSmsFacture Nombre,
			ope_sms_notification.MontantHT,
			ope_sms_notification.MontantTTC,
			ope_sms_notification.TauxTVA,
			ope_sms_notification.EstPaye,
			DateHeureEnvoi
			FROM
			ope_sms_notification
			JOIN clt_client ON clt_client.IdClient = ope_sms_notification.IdClient
			JOIN clt_mere ON clt_mere.IdClient = ope_sms_notification.IdClient
			WHERE
			clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(DateHeureEnvoi) = '[%cboAnnée.ValeurMémorisée%]'
			[%sWhere%]
			ORDER BY
			DateHeureEnvoi DESC
		]

	cas __OPERATION_VMS_UNITAIRE__

		sTexteRequete =
		[
			SELECT
			'[%__OPERATION_VMS_UNITAIRE__%]' TypeProjet,
			ope_vms_unitaire.IdClient,
			Societe,
			ope_vms_unitaire.Portable Destinataire,
			IdVmsUnitaire id,
			1 Nombre,
			ope_vms_unitaire.MontantHT,
			ope_vms_unitaire.MontantTTC,
			ope_vms_unitaire.TauxTVA,
			ope_vms_unitaire.EstPaye,
			DateHeureEnvoi
			FROM
			ope_vms_unitaire
			JOIN clt_client ON clt_client.IdClient = ope_vms_unitaire.IdClient
			JOIN clt_mere ON clt_mere.IdClient = ope_vms_unitaire.IdClient
			WHERE
			clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(DateHeureEnvoi) = '[%cboAnnée.ValeurMémorisée%]'
			[%sWhere%]
			ORDER BY
			DateHeureEnvoi DESC
		]


	cas __OPERATION_EMAIL_NOTIFICATION__

		sTexteRequete =
		[
			SELECT
			'[%__OPERATION_EMAIL_NOTIFICATION__%]' TypeProjet,
			ope_email_notification.IdClient,
			Societe,
			ope_email_notification.Email Destinataire,
			IdEmailNotification id,
			1 Nombre,
			ope_email_notification.MontantHT,
			ope_email_notification.MontantTTC,
			ope_email_notification.TauxTVA,
			ope_email_notification.EstPaye,
			DateHeureEnvoi
			FROM
			ope_email_notification
			JOIN clt_client ON clt_client.IdClient = ope_email_notification.IdClient
			JOIN clt_mere ON clt_mere.IdClient = ope_email_notification.IdClient
			WHERE
			clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(DateHeureEnvoi) = '[%cboAnnée.ValeurMémorisée%]'
			[%sWhere%]
			ORDER BY
			DateHeureEnvoi DESC
		]


	autre cas

		sTexteRequete =
		[
			SELECT
			z.TypeProjet,
			z.IdClient,
			z.Societe,
			z.Destinataire,
			z.id,
			z.Nombre,
			z.MontantHT,
			z.MontantTTC,
			z.TauxTVA,
			z.EstPaye,
			z.DateHeureEnvoi
			FROM (
			SELECT
			'[%__OPERATION_SMS_NOTIFICATION__%]' TypeProjet,
			ope_sms_notification.IdClient,
			Societe,
			ope_sms_notification.Portable Destinataire,
			IdSmsNotification id,
			ope_sms_notification.NbrSmsFacture Nombre,
			ope_sms_notification.MontantHT,
			ope_sms_notification.MontantTTC,
			ope_sms_notification.TauxTVA,
			ope_sms_notification.EstPaye,
			DateHeureEnvoi
			FROM
			ope_sms_notification
			JOIN clt_client ON clt_client.IdClient = ope_sms_notification.IdClient
			JOIN clt_mere ON clt_mere.IdClient = ope_sms_notification.IdClient
			WHERE
			clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(DateHeureEnvoi) = '[%cboAnnée.ValeurMémorisée%]'
			[%sWhere%]
			
			UNION ALL
			
			SELECT
			'[%__OPERATION_VMS_UNITAIRE__%]' TypeProjet,
			ope_vms_unitaire.IdClient,
			Societe,
			ope_vms_unitaire.Portable Destinataire,
			IdVmsUnitaire id,
			1 Nombre,
			ope_vms_unitaire.MontantHT,
			ope_vms_unitaire.MontantTTC,
			ope_vms_unitaire.TauxTVA,
			ope_vms_unitaire.EstPaye,
			DateHeureEnvoi
			FROM
			ope_vms_unitaire
			JOIN clt_client ON clt_client.IdClient = ope_vms_unitaire.IdClient
			JOIN clt_mere ON clt_mere.IdClient = ope_vms_unitaire.IdClient
			WHERE
			clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(DateHeureEnvoi) = '[%cboAnnée.ValeurMémorisée%]'
			[%sWhere%]
			
			UNION ALL
			
			SELECT
			'[%__OPERATION_EMAIL_NOTIFICATION__%]' TypeProjet,
			ope_email_notification.IdClient,
			Societe,
			ope_email_notification.Email Destinataire,
			IdEmailNotification,
			1 Nombre,
			ope_email_notification.MontantHT,
			ope_email_notification.MontantTTC,
			ope_email_notification.TauxTVA,
			ope_email_notification.EstPaye,
			DateHeureEnvoi
			FROM
			ope_email_notification
			JOIN clt_client ON clt_client.IdClient = ope_email_notification.IdClient
			JOIN clt_mere ON clt_mere.IdClient = ope_email_notification.IdClient
			WHERE
			clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(DateHeureEnvoi) = '[%cboAnnée.ValeurMémorisée%]'
			[%sWhere%]
			) AS z
			
			ORDER BY
			z.DateHeureEnvoi DESC
		]


fin



si pas sdReqConsommation.ExécuteRequêteSQL(ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

znrConsommation.SupprimeTout()

pour tout sdReqConsommation
	mPériodeTotal += sdReqConsommation.MontantTTC
	si sdReqConsommation.EstPaye = Faux alors
		mPériodeImpayé += sdReqConsommation.MontantTTC
	sinon
		mPériodePayé += sdReqConsommation.MontantTTC
	fin

	nInd												= znrConsommation.AjouteLigne()
	znrConsommation[nInd].attId							= sdReqConsommation.Id
	znrConsommation[nInd].attTypeOperation				= sdReqConsommation.TypeProjet
	znrConsommation[nInd].Icone							= cpProjet.IconeTypeProjet(sdReqConsommation.TypeProjet, Vrai)
	znrConsommation[nInd].libTypeOperation				= Partage.LibelleTypeProjet(sdReqConsommation.TypeProjet)
	znrConsommation[nInd].libDate						= DateHeureVersChaîne(sdReqConsommation.DateHeureEnvoi, "Jjj JJ mmm AA à HH:mm")
	selon sdReqConsommation.TypeProjet
		cas __OPERATION_SMS_NOTIFICATION__				: znrConsommation[nInd].libDestinataire	= cpFonction.Telephone(sdReqConsommation.Destinataire,1,".")
		cas __OPERATION_VMS_UNITAIRE__					: znrConsommation[nInd].libDestinataire	= cpFonction.Telephone(sdReqConsommation.Destinataire,1,".")
		cas __OPERATION_EMAIL_NOTIFICATION__			: znrConsommation[nInd].libDestinataire	= sdReqConsommation.Destinataire
		autres cas : STOP
	fin
	znrConsommation[nInd].libSociété					= sdReqConsommation.Societe

	znrConsommation[nInd].libNombre					= NumériqueVersChaîne(sdReqConsommation.Nombre, "dS")
	znrConsommation[nInd].libMontantHT					= NumériqueVersChaîne(sdReqConsommation.MontantHT, ",3fS") + " €"
	znrConsommation[nInd].libResteAPayer				= (sdReqConsommation.EstPaye = 0) ? NumériqueVersChaîne(sdReqConsommation.MontantTTC, ",3fS") + " €" sinon "0,000 €"

	znrConsommation[nInd].dispDestinataire[2].Visible	= (cboSociété.ValeurMémorisée = 0)
fin

dispTitre[2].Visible = (cboSociété.ValeurMémorisée = 0)

si znrConsommation.Occurrence > 0 alors
	znrConsommation.SelectPlus(1)
	znrConsommation[1].btnAperçu.Visible		= Vrai
	_nLignePrecedente							= 1
sinon
	_nLignePrecedente 							= 0
fin

celConsommation.Visible	= (znrConsommation.Occurrence > 0)

saiPériodePayé			= mPériodePayé
saiPériodeImpayé		= mPériodeImpayé
saiPériodeTotal			= mPériodeTotal

fin:
sdReqConsommation.AnnuleDéclaration()
