procédure InitCboMois()

sTexteRequete est une chaîne
sdReqMois est une Source de Données
dDate est une Date
dDate = DateSys()

sTexteRequete =
[
	SELECT DISTINCT
	    Mois
	FROM (
	    SELECT MONTH(ope_sms_notification.DateHeureEnvoi) AS Mois
	    FROM ope_sms_notification
	    JOIN clt_mere ON clt_mere.IdClient = ope_sms_notification.IdClient
	    WHERE clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(ope_sms_notification.DateHeureEnvoi) = [%cboAnnée.ValeurMémorisée%]
	    GROUP BY MONTH(ope_sms_notification.DateHeureEnvoi)
	    
	    UNION
	    
	    SELECT MONTH(ope_vms_unitaire.DateHeureEnvoi) AS Mois
	    FROM ope_vms_unitaire
	    JOIN clt_mere ON clt_mere.IdClient = ope_vms_unitaire.IdClient
	    WHERE clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(ope_vms_unitaire.DateHeureEnvoi) = [%cboAnnée.ValeurMémorisée%]
	    GROUP BY MONTH(ope_vms_unitaire.DateHeureEnvoi)
	    
	    UNION
	    
	    SELECT MONTH(ope_email_notification.DateHeureEnvoi) AS Mois
	    FROM ope_email_notification
	    JOIN clt_mere ON clt_mere.IdClient = ope_email_notification.IdClient
	    WHERE clt_mere.IdMere = [%Partage.__vcUser.IDClient%]
			AND YEAR(ope_email_notification.DateHeureEnvoi) = [%cboAnnée.ValeurMémorisée%]
	    GROUP BY MONTH(ope_email_notification.DateHeureEnvoi)
	) t
	ORDER BY Mois DESC;
]

si pas sdReqMois.ExécuteRequêteSQL(ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	Info("Erreur requête")
	retour
fin

cboMois.SupprimeTout()

pour tout sdReqMois
	cboMois.Ajoute(MoisEnLettre(sdReqMois.Mois) + gLien(sdReqMois.Mois))
fin

si cboMois.Occurrence > 1 alors
	cboMois.Insère("< Mois >" + gLien(0),1)
fin

si cboMois..Occurrence = 0 alors
	cboMois.Ajoute(NomPropre(DateVersMoisEnLettre(dDate)) + gLien(dDate.Mois))
fin

si cboAnnée.ValeurMémorisée = Gauche(DateSys(),4) alors
	cboMois = dDate.Mois
fin

si cboMois < 0 alors
	 cboMois.SelectPlus(1)
fin

fin:
sdReqMois.AnnuleDéclaration()
