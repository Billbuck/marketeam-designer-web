procédure CalculePrixUnSms(pMessage)

mPrixSms 		est un monétaire
sMessage 		est une chaîne
nTaille 		est un entier
nNbrCredit 		est un entier

si pMessage <> "" alors
	sMessage = pMessage
	sMessage = sMessage.Remplace(RC,Caract(10))
	
	nTaille = sMessage.Taille()
	si cpSms.TestUnicodeSms(sMessage) = Faux alors
		nNbrCredit = CalculeNbrCreditSms(nTaille,Faux)
	sinon
		nNbrCredit = CalculeNbrCreditSms(nTaille,Vrai)
	fin
	si nNbrCredit > 8 alors nNbrCredit = 8
	mPrixSms = Partage.__vcUser.taaPrestation[__PRESTATION_ROUTAGE_SMS__].PrixVenteUnitaire * nNbrCredit
fin

renvoyer mPrixSms
