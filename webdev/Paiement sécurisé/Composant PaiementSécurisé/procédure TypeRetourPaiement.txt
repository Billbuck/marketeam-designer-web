		cas URL_CANCEL
			renvoyer URL_CANCEL
			
		autre cas	
			//on continue : le marqueur peut être plus loin dans la liste des paramètres
			//(l'ancien code renvoyait ici gsDernierTypeRetour, un état de session périmé :
			// un abandon PayPal après un paiement carte renvoyait "validé" à tort)
	fin
fin

//aucun marqueur dans les paramètres reçus : repli sur le dernier type mémorisé (paiement direct)
si gbModeDirect alors
	renvoyer gsDernierTypeRetour
fin
