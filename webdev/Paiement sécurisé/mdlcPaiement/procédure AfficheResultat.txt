procédure AfficheResultat(pStatut, pMessage = "")

//si la page appelante n'a pas fourni de message, on complète le libellé du statut
//avec le détail mémorisé par le dernier chemin d'échec (ErreurInfo Paybox, sInformation de TransactionEtat...)
si pMessage = "" _et_ _sInformationEchec <> "" alors
	selon pStatut
		cas _paiement_annulé_	: pMessage = "Paiement annulé" + RC + _sInformationEchec
		cas _paiement_refusé_	: pMessage = "Paiement refusé" + RC + _sInformationEchec
		cas _paiement_erreur_	: pMessage = "Paiement en erreur" + RC + _sInformationEchec
		autre cas				: pMessage = _sInformationEchec
	fin
fin
_sInformationEchec = ""

selon pStatut
	cas _paiement_validé_
		pMessage = (pMessage = "") ? "Paiement effectué" sinon pMessage
		PopupAfficheMessage(pMessage,*,_stPaiementParametre.TypeOperation,__TOAST_VALIDATION__,10s)
		
	cas _paiement_annulé_
		pMessage = (pMessage = "") ? "Paiement annulé" sinon pMessage
		PopupAfficheMessage(pMessage,*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		
	cas _paiement_refusé_
		pMessage = (pMessage = "") ? "Paiement refusé" sinon pMessage
		PopupAfficheMessage(pMessage,*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		
	cas _paiement_erreur_
		pMessage = (pMessage = "") ? "Paiement en erreur" sinon pMessage
		PopupAfficheMessage(pMessage,*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		
	autre cas
		pMessage = "Statut non défini " + pStatut
		PopupAfficheMessage(pMessage,*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
fin
