procédure NavSaisieCarteOk()

dDate		est une Date
sDateMMAA	est une chaîne = mixDateExpirationMMAA()

si sDateMMAA <> "" alors
	dDate.Année	= "20" + Droite(sDateMMAA,2)
	dDate.Mois	= Gauche(sDateMMAA,2)
	dDate.Jour	= 1
	dDate		= DernierJourDuMois(dDate)
fin

si _bCarteBancaireMémorisée = Faux alors
	si SansEspace(saiCarteBancairePorteur) = "" alors 
		PopupAfficheMessage("Saisir le nom du porteur",*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		renvoyer Faux
	fin
	
	si CaractOccurrenceType(saiCarteBancaireNumero,ctNumérique) pas dans (15,16) alors 
		PopupAfficheMessage("Numéro de carte bancaire erroné",*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		renvoyer Faux
	fin
	
	si sDateMMAA = "" alors 
		PopupAfficheMessage("Veuillez saisir la date d'expiration (MM/AAAA)",*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		renvoyer Faux
	fin
	
	si dDate < DateSys() alors
		PopupAfficheMessage("Date d'expiration invalide",*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		renvoyer Faux
	fin
	
	si Taille(saiCarteBancaireCryptogramme) < 3 alors 
		PopupAfficheMessage("Code de cryptogramme erroné",*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		renvoyer Faux
	fin
	
	si mixVerificationCarteBancaire(saiCarteBancaireNumero) = Faux alors
		PopupAfficheMessage("Numéro de carte bancaire erroné",*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		renvoyer Faux
	fin
	
	si AJAXExécute(ajaxActualiseChamps+ajaxSynchroniseVariablesServeur,NomCarteBancaireOK) = Faux alors 
		PopupAfficheMessage("Une carte bancaire porte déjà ce nom (" + SansEspace(saiCarteBancaireLibelle) +")",*,_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		renvoyer Faux
	fin
fin

si _stPaiementParametre.EstReferenceObligatoire = Vrai alors
	si SansEspace(saiRéférenceClient) <> "" alors 
		si _stPaiementParametre.ReferenceExpressionReguliere <> "" _et_ pas RegexVérifie(SansEspace(saiRéférenceClient),_stPaiementParametre.ReferenceExpressionReguliere) alors
			PopupAfficheMessage(_stPaiementParametre.ReferenceLibelle + " incorrect" + [RC] + _stPaiementParametre.ReferenceExpressionReguliereAide,"Erreur",_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
			renvoyer Faux
		fin
	sinon
		PopupAfficheMessage(_stPaiementParametre.ReferenceLibelle + " obligatoire","Erreur",_stPaiementParametre.TypeOperation,__TOAST_ANNULATION__,10s)
		renvoyer Faux
	fin

	
fin

renvoyer Vrai
