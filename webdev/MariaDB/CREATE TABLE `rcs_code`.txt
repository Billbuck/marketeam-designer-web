CREATE TABLE `rcs_code` (
	`TypeCode` ENUM('STATUT','ERREUR') NOT NULL COMMENT 'STATUT = status Infobip, ERREUR = error Infobip' COLLATE 'utf8mb4_unicode_ci',
	`CodeId` SMALLINT(5) UNSIGNED NOT NULL,
	`Nom` VARCHAR(60) NOT NULL COMMENT 'name Infobip' COLLATE 'utf8mb4_unicode_ci',
	`GroupeId` TINYINT(3) UNSIGNED NULL DEFAULT NULL,
	`GroupeNom` VARCHAR(30) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`EstPermanent` TINYINT(1) NULL DEFAULT NULL COMMENT 'Erreurs : permanent Infobip',
	`Libelle` VARCHAR(120) NULL DEFAULT NULL COMMENT 'Libellé français affiché dans Marketeam - NULL = à renseigner' COLLATE 'utf8mb4_unicode_ci',
	`Action` VARCHAR(20) NULL DEFAULT NULL COMMENT 'Ex. SMS_SEUL pour la qualification' COLLATE 'utf8mb4_unicode_ci',
	`DateHeureCreation` DATETIME NOT NULL DEFAULT current_timestamp(),
	PRIMARY KEY (`TypeCode`, `CodeId`) USING BTREE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
;
