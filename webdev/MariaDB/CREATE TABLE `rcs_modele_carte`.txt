CREATE TABLE `rcs_modele_carte` (
	`IdRcsModeleCarte` INT(11) NOT NULL AUTO_INCREMENT,
	`IdRcsModele` INT(11) NOT NULL,
	`Ordre` TINYINT(3) UNSIGNED NOT NULL DEFAULT '1' COMMENT 'Position dans le carrousel de 1 a 10',
	`IdImage` INT(11) NULL DEFAULT NULL COMMENT 'Media choisi dans ici_image',
	`Titre` VARCHAR(400) NULL DEFAULT NULL COMMENT 'Gabarit - 200 car. apres fusion' COLLATE 'utf8mb4_unicode_ci',
	`Description` VARCHAR(3000) NULL DEFAULT NULL COMMENT 'Gabarit - 2000 car. apres fusion' COLLATE 'utf8mb4_unicode_ci',
	`MediaUrl` VARCHAR(2048) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`MediaContentType` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`MediaHauteur` ENUM('SHORT','MEDIUM','TALL') NOT NULL DEFAULT 'MEDIUM' COLLATE 'utf8mb4_unicode_ci',
	`MediaThumbnailUrl` VARCHAR(2048) NULL DEFAULT NULL COMMENT 'Video uniquement' COLLATE 'utf8mb4_unicode_ci',
	`MediaLargeurPx` SMALLINT(5) UNSIGNED NULL DEFAULT NULL,
	`MediaHauteurPx` SMALLINT(5) UNSIGNED NULL DEFAULT NULL,
	`DateHeureCreation` DATETIME NOT NULL,
	`DateHeureModification` DATETIME NULL DEFAULT NULL,
	`UserCreation` VARCHAR(35) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`UserModification` VARCHAR(35) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	PRIMARY KEY (`IdRcsModeleCarte`) USING BTREE,
	INDEX `IdRcsModele_Ordre` (`IdRcsModele`, `Ordre`) USING BTREE,
	INDEX `IdImage` (`IdImage`) USING BTREE,
	CONSTRAINT `FK_rcs_modele_carte_ici_image` FOREIGN KEY (`IdImage`) REFERENCES `ici_image` (`IdImage`) ON UPDATE RESTRICT ON DELETE SET NULL,
	CONSTRAINT `FK_rcs_modele_carte_rcs_modele` FOREIGN KEY (`IdRcsModele`) REFERENCES `rcs_modele` (`IdRcsModele`) ON UPDATE RESTRICT ON DELETE CASCADE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
ROW_FORMAT=DYNAMIC
AUTO_INCREMENT=51
;
