	`FenetreHeureDebut` TIME NULL DEFAULT NULL COMMENT 'Heure UTC et non heure de Paris',

	`EstRepliSms` TINYINT(1) NOT NULL DEFAULT '1',

	`RepliSmsMessage` VARCHAR(1200) NULL DEFAULT NULL COMMENT 'GSM-7, 1 à 3 segments' COLLATE 'utf8mb4_unicode_ci',
	`RepliCreditSms` VARCHAR(1000) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`DateFin` DATE NULL DEFAULT NULL,
	`HeureFin` TIME NULL DEFAULT NULL,
	`Cadence` MEDIUMINT(8) UNSIGNED NULL DEFAULT NULL COMMENT 'Calculé par rapport à la cadence',
	`Paquet` SMALLINT(5) UNSIGNED NULL DEFAULT NULL COMMENT 'Calculé par rapport à la cadence',

	INDEX `IdRcsAgent` (`IdRcsAgent`) USING BTREE,

	CONSTRAINT `FK_ope_rcs_rcs_agent` FOREIGN KEY (`IdRcsAgent`) REFERENCES `rcs_agent` (`IdRcsAgent`) ON UPDATE RESTRICT ON DELETE RESTRICT
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB

;
