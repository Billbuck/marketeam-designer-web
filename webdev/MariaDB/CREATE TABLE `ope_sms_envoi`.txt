	`IdBaseLigne` INT(11) NOT NULL,
	`IdSms` INT(11) NOT NULL,
	`Statut` CHAR(2) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',

	`StatutOperateur` CHAR(4) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`Operateur` CHAR(5) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`Routeur` CHAR(2) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`PrixUnitaireFacture` DECIMAL(6,4) NULL DEFAULT NULL,
	`NbrSmsFacture` TINYINT(1) NULL DEFAULT NULL,
	`SmsTaille` SMALLINT(6) NULL DEFAULT NULL,
