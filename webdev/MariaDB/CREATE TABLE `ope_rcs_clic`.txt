CREATE TABLE `ope_rcs_clic` (
	`IdRcsClic` BIGINT(20) NOT NULL AUTO_INCREMENT,
	`IdOperation` INT(11) NULL DEFAULT NULL,
	`IdBaseLigne` INT(11) NULL DEFAULT NULL COMMENT 'Clic d un destinataire d opération (ope_rcs_envoi) - NULL pour un test',
	`IdRcsUnitaire` INT(11) NULL DEFAULT NULL COMMENT 'Clic sur un RCS de test (ope_rcs_unitaire) - NULL pour une opération',
	`IdRcsCarte` INT(11) NULL DEFAULT NULL COMMENT 'Resolu depuis PostbackData. NULL = clic sur un bouton commun',
	`IdRcsBouton` INT(11) NULL DEFAULT NULL COMMENT 'Resolu depuis PostbackData. NULL si le bouton a ete supprime depuis',
	`PostbackData` VARCHAR(100) NULL DEFAULT NULL COMMENT 'Valeur brute reçue du webhook - IdRcsCarte-IdRcsBouton' COLLATE 'utf8mb4_unicode_ci',
	`LibelleBouton` VARCHAR(60) NULL DEFAULT NULL COMMENT 'Libellé au moment du clic - historisé' COLLATE 'utf8mb4_unicode_ci',
	`TypeBouton` ENUM('OPEN_URL','DIAL_PHONE','SHOW_LOCATION') NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`DateHeureClic` DATETIME NOT NULL COMMENT 'receivedAt du webhook',
	`MessageIdEvenement` VARCHAR(64) NULL DEFAULT NULL COMMENT 'Idempotence - évite les doublons de webhook' COLLATE 'utf8mb4_unicode_ci',
	PRIMARY KEY (`IdRcsClic`) USING BTREE,
	UNIQUE INDEX `MessageIdEvenement` (`MessageIdEvenement`) USING BTREE,
	INDEX `IdOperation_IdBaseLigne` (`IdOperation`, `IdBaseLigne`) USING BTREE,
	INDEX `IdOperation_DateHeureClic` (`IdOperation`, `DateHeureClic`) USING BTREE,
	INDEX `IdRcsUnitaire` (`IdRcsUnitaire`) USING BTREE,
	INDEX `IdRcsCarte` (`IdRcsCarte`) USING BTREE,
	INDEX `IdRcsBouton` (`IdRcsBouton`) USING BTREE,
	CONSTRAINT `FK_ope_rcs_clic_dos_operation` FOREIGN KEY (`IdOperation`) REFERENCES `dos_operation` (`IdOperation`) ON UPDATE NO ACTION ON DELETE CASCADE,
	CONSTRAINT `FK_ope_rcs_clic_ope_rcs_unitaire` FOREIGN KEY (`IdRcsUnitaire`) REFERENCES `ope_rcs_unitaire` (`IdRcsUnitaire`) ON UPDATE NO ACTION ON DELETE CASCADE,
	CONSTRAINT `FK_ope_rcs_clic_ope_rcs_carte` FOREIGN KEY (`IdRcsCarte`) REFERENCES `ope_rcs_carte` (`IdRcsCarte`) ON UPDATE NO ACTION ON DELETE SET NULL,
	CONSTRAINT `FK_ope_rcs_clic_ope_rcs_bouton` FOREIGN KEY (`IdRcsBouton`) REFERENCES `ope_rcs_bouton` (`IdRcsBouton`) ON UPDATE NO ACTION ON DELETE SET NULL,
	CONSTRAINT `CK_ope_rcs_clic_source` CHECK (`IdBaseLigne` is not null or `IdRcsUnitaire` is not null)
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
ROW_FORMAT=DYNAMIC
;
