CREATE TABLE `rcs_clic` (
	`IdRcsClic` BIGINT(20) NOT NULL AUTO_INCREMENT,
	`IdRcsEnvoi` INT(11) NOT NULL,
	`IdOperation` INT(11) NULL DEFAULT NULL,
	`IdRcsCarte` INT(11) NULL DEFAULT NULL COMMENT 'Resolu depuis PostbackData. NULL = clic sur un bouton commun',
	`IdRcsBouton` INT(11) NULL DEFAULT NULL COMMENT 'Resolu depuis PostbackData. NULL si le bouton a ete supprime depuis',
	`PostbackData` VARCHAR(100) NULL DEFAULT NULL COMMENT 'Valeur brute reçue du webhook' COLLATE 'utf8mb4_unicode_ci',
	`LibelleBouton` VARCHAR(60) NULL DEFAULT NULL COMMENT 'Libellé au moment du clic - historisé' COLLATE 'utf8mb4_unicode_ci',
	`TypeBouton` ENUM('OPEN_URL','DIAL_PHONE','SHOW_LOCATION') NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`DateHeureClic` DATETIME NOT NULL COMMENT 'receivedAt du webhook',
	`MessageIdEvenement` VARCHAR(64) NULL DEFAULT NULL COMMENT 'Idempotence - évite les doublons de webhook' COLLATE 'utf8mb4_unicode_ci',
	PRIMARY KEY (`IdRcsClic`) USING BTREE,
	UNIQUE INDEX `MessageIdEvenement` (`MessageIdEvenement`) USING BTREE,
	INDEX `IdRcsEnvoi` (`IdRcsEnvoi`) USING BTREE,
	INDEX `IdOperation_DateHeureClic` (`IdOperation`, `DateHeureClic`) USING BTREE,
	INDEX `IdRcsBouton` (`IdRcsBouton`) USING BTREE,
	INDEX `FK_rcs_clic_ope_rcs_carte` (`IdRcsCarte`) USING BTREE,
	CONSTRAINT `FK_rcs_clic_ope_rcs_carte` FOREIGN KEY (`IdRcsCarte`) REFERENCES `ope_rcs_carte` (`IdRcsCarte`) ON UPDATE NO ACTION ON DELETE SET NULL,
	CONSTRAINT `FK_rcs_clic_ope_rcs_bouton` FOREIGN KEY (`IdRcsBouton`) REFERENCES `ope_rcs_bouton` (`IdRcsBouton`) ON UPDATE NO ACTION ON DELETE SET NULL,
	CONSTRAINT `FK_rcs_clic_rcs_envoi` FOREIGN KEY (`IdRcsEnvoi`) REFERENCES `rcs_envoi` (`IdRcsEnvoi`) ON UPDATE NO ACTION ON DELETE CASCADE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
ROW_FORMAT=DYNAMIC
;
