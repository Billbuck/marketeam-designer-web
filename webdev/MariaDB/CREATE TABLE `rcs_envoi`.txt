CREATE TABLE `rcs_envoi` (
	`IdRcsEnvoi` INT(11) NOT NULL AUTO_INCREMENT,
	`IdClient` INT(11) NULL DEFAULT NULL,
	`IdContact` INT(11) NULL DEFAULT NULL,
	`IdOperation` INT(11) NULL DEFAULT NULL,
	`TypeOperation` CHAR(3) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`IdFacture` INT(11) NULL DEFAULT NULL,
	`TypeFacture` CHAR(3) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`IdAbonnement` INT(11) NULL DEFAULT NULL,
	`TypeAbonnement` CHAR(3) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`Variable` TEXT NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`CodeRcsType` VARCHAR(16) NOT NULL COLLATE 'utf8mb4_unicode_ci',
	`DateHeureDemande` DATETIME NOT NULL,
	`DateHeureEnvoi` DATETIME NULL DEFAULT NULL,
	`Portable` CHAR(20) NOT NULL COLLATE 'utf8mb4_unicode_ci',
	`CapaciteRcs` ENUM('INCONNU','ENABLED','UNREACHABLE') NOT NULL DEFAULT 'INCONNU' COLLATE 'utf8mb4_unicode_ci',
	`DateHeureCapacite` DATETIME NULL DEFAULT NULL,
	`CanalUtilise` ENUM('ATTENTE','RCS','SMS') NOT NULL DEFAULT 'ATTENTE' COMMENT 'Canal réellement facturé - renseigné par le DLR' COLLATE 'utf8mb4_unicode_ci',
	`BulkId` VARCHAR(64) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`MessageId` VARCHAR(64) NULL DEFAULT NULL COMMENT 'Clé de rapprochement des webhooks' COLLATE 'utf8mb4_unicode_ci',
	`CallbackData` VARCHAR(200) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`StatutGroupeId` TINYINT(3) UNSIGNED NULL DEFAULT NULL COMMENT '1 PENDING 2 UNDELIVERABLE 3 DELIVERED 4 EXPIRED 5 REJECTED',
	`StatutGroupeNom` VARCHAR(30) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`StatutId` SMALLINT(5) UNSIGNED NULL DEFAULT NULL,
	`StatutNom` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`DateHeureStatut` DATETIME NULL DEFAULT NULL COMMENT 'doneAt du DLR',
	`ErreurId` SMALLINT(5) UNSIGNED NULL DEFAULT NULL,
	`ErreurNom` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`ErreurPermanente` TINYINT(1) NULL DEFAULT NULL COMMENT 'Si 1 ne pas retenter - candidat liste noire',
	`DateHeureLu` DATETIME NULL DEFAULT NULL COMMENT 'seenAt - spécifique RCS',
	`TrafficType` VARCHAR(20) NULL DEFAULT NULL COMMENT 'BASIC SINGLE - contrôle de facturation' COLLATE 'utf8mb4_unicode_ci',
	`PrixUnitaire` DECIMAL(10,5) NULL DEFAULT NULL,
	`Devise` CHAR(3) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`MccMnc` VARCHAR(10) NULL DEFAULT NULL COMMENT 'Opérateur réel du destinataire' COLLATE 'utf8mb4_unicode_ci',
	`NbTentatives` TINYINT(3) UNSIGNED NOT NULL DEFAULT '0',
	PRIMARY KEY (`IdRcsEnvoi`) USING BTREE,
	UNIQUE INDEX `MessageId` (`MessageId`) USING BTREE,
	INDEX `CodeRcsType` (`CodeRcsType`) USING BTREE,
	INDEX `DateHeureDemande` (`DateHeureDemande`) USING BTREE,
	INDEX `DateHeureEnvoi` (`DateHeureEnvoi`) USING BTREE,
	INDEX `IdOperation` (`IdOperation`) USING BTREE,
	INDEX `IdOperation_CodeRcsType` (`IdOperation`, `CodeRcsType`) USING BTREE,
	INDEX `IdOperation_CanalUtilise` (`IdOperation`, `CanalUtilise`) USING BTREE,
	INDEX `IdOperation_StatutGroupeId` (`IdOperation`, `StatutGroupeId`) USING BTREE,
	INDEX `BulkId` (`BulkId`) USING BTREE,
	INDEX `Portable` (`Portable`) USING BTREE,
	INDEX `IdFacture` (`IdFacture`) USING BTREE,
	INDEX `FK_rcs_envoi_clt_contact` (`IdContact`) USING BTREE,
	INDEX `IdAbonnement` (`IdAbonnement`) USING BTREE,
	INDEX `IdClient` (`IdClient`) USING BTREE,
	CONSTRAINT `FK_rcs_envoi_clt_client` FOREIGN KEY (`IdClient`) REFERENCES `clt_client` (`IdClient`) ON UPDATE NO ACTION ON DELETE CASCADE,
	CONSTRAINT `FK_rcs_envoi_clt_contact` FOREIGN KEY (`IdContact`) REFERENCES `clt_contact` (`IdContact`) ON UPDATE NO ACTION ON DELETE CASCADE,
	CONSTRAINT `FK_rcs_envoi_clt_facture` FOREIGN KEY (`IdFacture`) REFERENCES `clt_facture` (`IdFacture`) ON UPDATE NO ACTION ON DELETE CASCADE,
	CONSTRAINT `FK_rcs_envoi_dos_operation` FOREIGN KEY (`IdOperation`) REFERENCES `dos_operation` (`IdOperation`) ON UPDATE NO ACTION ON DELETE CASCADE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
ROW_FORMAT=DYNAMIC
;
