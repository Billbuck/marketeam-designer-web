CREATE TABLE `clt_base_filtre` (
	`IdFiltre` INT(11) NOT NULL AUTO_INCREMENT,
	`IdClient` INT(11) NOT NULL,
	`IdContact` INT(11) NOT NULL,
	`IdBase` INT(11) NOT NULL,
	`Nom` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`Phrase` VARCHAR(200) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`NbrTotal` INT(10) UNSIGNED NULL DEFAULT NULL,
	`NbrSociete` INT(10) UNSIGNED NULL DEFAULT NULL,
	`NbrContact` INT(10) UNSIGNED NULL DEFAULT NULL,
	`NbrAdresse` INT(10) UNSIGNED NULL DEFAULT NULL,
	`NbrSms` INT(10) UNSIGNED NULL DEFAULT NULL,
	`NbrFixe` INT(10) UNSIGNED NULL DEFAULT NULL,
	`NbrTel` INT(10) UNSIGNED NULL DEFAULT NULL COMMENT 'Ensemble Fixe + Portable',
	`NbrFax` INT(10) UNSIGNED NULL DEFAULT NULL,
	`NbrEmail` INT(10) UNSIGNED NULL DEFAULT NULL,
	`DateHeureCreation` DATETIME NOT NULL DEFAULT current_timestamp(),
	`DateHeureModification` DATETIME NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
	`UserCreation` VARCHAR(35) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`UserModification` VARCHAR(35) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	PRIMARY KEY (`IdFiltre`) USING BTREE,
	INDEX `IdBase` (`IdBase`) USING BTREE,
	INDEX `FK_clt_base_filtre_clt_client` (`IdClient`) USING BTREE,
	INDEX `FK_clt_base_filtre_clt_contact` (`IdContact`) USING BTREE,
	INDEX `IdFiltre_IdClient_IdBase` (`IdFiltre`, `IdClient`, `IdBase`) USING BTREE,
	CONSTRAINT `FK_clt_base_filtre_clt_base` FOREIGN KEY (`IdBase`) REFERENCES `clt_base` (`IdBase`) ON UPDATE RESTRICT ON DELETE CASCADE,
	CONSTRAINT `FK_clt_base_filtre_clt_client` FOREIGN KEY (`IdClient`) REFERENCES `clt_client` (`IdClient`) ON UPDATE NO ACTION ON DELETE CASCADE,
	CONSTRAINT `FK_clt_base_filtre_clt_contact` FOREIGN KEY (`IdContact`) REFERENCES `clt_contact` (`IdContact`) ON UPDATE NO ACTION ON DELETE CASCADE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
AUTO_INCREMENT=870
;
