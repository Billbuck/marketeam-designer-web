CREATE TABLE `clt_base_rcs_lot` (
	`IdBaseRcsLot` INT(11) NOT NULL AUTO_INCREMENT,
	`IdBase` INT(11) NOT NULL,
	`BulkId` VARCHAR(64) NULL DEFAULT NULL COMMENT 'bulkId Infobip du capability check' COLLATE 'utf8mb4_unicode_ci',
	`NbrEnvoye` INT(10) UNSIGNED NOT NULL DEFAULT '0',
	`NbrRecu` INT(10) UNSIGNED NOT NULL DEFAULT '0',
	`Statut` ENUM('ENVOYE','TERMINE','ERREUR') NOT NULL DEFAULT 'ENVOYE' COLLATE 'utf8mb4_unicode_ci',
	`Erreur` VARCHAR(500) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`DateHeureEnvoi` DATETIME NOT NULL DEFAULT current_timestamp(),
	`DateHeureDernierRetour` DATETIME NULL DEFAULT NULL COMMENT 'Dernier resultat recu par pgeRcsCapacite',
	`DateHeureFin` DATETIME NULL DEFAULT NULL,
