CREATE TABLE `rcs_agent_historique` (
	`IdRcsAgentHistorique` BIGINT(20) NOT NULL AUTO_INCREMENT,
	`IdRcsAgent` INT(11) NOT NULL,
	`TypeEvenement` ENUM('CREATION_LOCALE','CREATION_INFOBIP','MAJ_FICHE','MAJ_STATUT','DEMANDE_LANCEMENT','MAJ_OPERATEUR','CORRECTION_DEMANDEE','TESTEUR_AJOUTE','TESTEUR_APPROUVE','TESTEUR_EXPIRE','TEST_ENVOYE','ACTIVATION','DESACTIVATION') NOT NULL COLLATE 'utf8mb4_unicode_ci',
	`StatutAvant` VARCHAR(30) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`StatutApres` VARCHAR(30) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`CodePays` CHAR(2) NULL DEFAULT NULL COMMENT 'Si evenement operateur' COLLATE 'utf8mb4_unicode_ci',
	`Operateur` VARCHAR(50) NULL DEFAULT NULL COMMENT 'Si evenement operateur' COLLATE 'utf8mb4_unicode_ci',
	`Detail` TEXT NULL DEFAULT NULL COMMENT 'Motif de rejet, numero concerne, message d erreur' COLLATE 'utf8mb4_unicode_ci',
	`Source` ENUM('MARKETEAM','WEBHOOK','SYNCHRO') NOT NULL DEFAULT 'MARKETEAM' COLLATE 'utf8mb4_unicode_ci',
	`DateHeureEvenement` DATETIME NOT NULL,
	`UserCreation` VARCHAR(35) NULL DEFAULT NULL COMMENT 'NULL si origine webhook' COLLATE 'utf8mb4_unicode_ci',
	PRIMARY KEY (`IdRcsAgentHistorique`) USING BTREE,
	INDEX `IdRcsAgent_DateHeureEvenement` (`IdRcsAgent`, `DateHeureEvenement`) USING BTREE,
	INDEX `TypeEvenement` (`TypeEvenement`) USING BTREE,
	CONSTRAINT `FK_rcs_agent_historique_rcs_agent` FOREIGN KEY (`IdRcsAgent`) REFERENCES `rcs_agent` (`IdRcsAgent`) ON UPDATE NO ACTION ON DELETE CASCADE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
ROW_FORMAT=DYNAMIC
;
