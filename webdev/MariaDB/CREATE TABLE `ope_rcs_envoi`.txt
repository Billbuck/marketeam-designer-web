CREATE TABLE `ope_rcs_envoi` (
	`IdRcs` INT(11) NOT NULL COMMENT 'IdOperation de dos_operation - voir ope_rcs',
	`IdBaseLigne` INT(11) NOT NULL COMMENT 'Destinataire - voir dos_base_ligne',
	`Statut` CHAR(2) NULL DEFAULT NULL COMMENT 'Codes Marketeam communs au SMS (__SMS_STATUT_*) : 10 enregistre 12 pris en charge 15 erreur envoi 20 envoye 42 stop pub 80 non delivre 90 delivre' COLLATE 'utf8mb4_unicode_ci',
	`NbTentatives` TINYINT(3) UNSIGNED NOT NULL DEFAULT '0' COMMENT 'Relance automatique des erreurs d envoi (statut 15) par batchRcs',
	`CanalUtilise` ENUM('ATTENTE','RCS','SMS') NOT NULL DEFAULT 'ATTENTE' COMMENT 'Canal réellement facturé - renseigné par le DLR' COLLATE 'utf8mb4_unicode_ci',
	`BulkId` VARCHAR(64) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`MessageId` VARCHAR(64) NULL DEFAULT NULL COMMENT 'Clé de rapprochement des webhooks' COLLATE 'utf8mb4_unicode_ci',
	`StatutGroupeId` TINYINT(3) UNSIGNED NULL DEFAULT NULL COMMENT '1 PENDING 2 UNDELIVERABLE 3 DELIVERED 4 EXPIRED 5 REJECTED',
	`StatutGroupeNom` VARCHAR(30) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`StatutId` SMALLINT(5) UNSIGNED NULL DEFAULT NULL,
	`StatutNom` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`ErreurId` SMALLINT(5) UNSIGNED NULL DEFAULT NULL,
	`ErreurNom` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`ErreurPermanente` TINYINT(1) NULL DEFAULT NULL COMMENT 'Si 1 ne pas retenter - candidat liste noire',
	`MccMnc` VARCHAR(10) NULL DEFAULT NULL COMMENT 'Opérateur réel du destinataire' COLLATE 'utf8mb4_unicode_ci',
	`PrixUnitaire` DECIMAL(10,5) NULL DEFAULT NULL COMMENT 'Coût Infobip - suivi de marge, pas de régularisation du devis',
	`Devise` CHAR(3) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`SmsNombre` TINYINT(4) NULL DEFAULT NULL COMMENT 'Nombre de segments si le repli SMS a été utilisé',
	`DateHeureEnvoi` DATETIME NULL DEFAULT NULL COMMENT 'Envoi par batchRcs',
	`DateHeurePriseEnCharge` DATETIME NULL DEFAULT NULL COMMENT 'Acceptation par Infobip',
	`DateHeureReception` DATETIME NULL DEFAULT NULL COMMENT 'doneAt du DLR',
	`DateHeureLu` DATETIME NULL DEFAULT NULL COMMENT 'seenAt - spécifique RCS',
	`ContenuJson` MEDIUMTEXT NULL DEFAULT NULL COMMENT 'JSON envoyé à Infobip - rempli par batchRcs seulement si prm_parametre Rcs_DebugJson = 1' COLLATE 'utf8mb4_unicode_ci',
	PRIMARY KEY (`IdRcs`, `IdBaseLigne`) USING BTREE,
	UNIQUE INDEX `MessageId` (`MessageId`) USING BTREE,
	INDEX `Statut` (`Statut`) USING BTREE,
	INDEX `IdBaseLigne` (`IdBaseLigne`) USING BTREE,
	CONSTRAINT `FK_ope_rcs_envoi_dos_operation` FOREIGN KEY (`IdRcs`) REFERENCES `dos_operation` (`IdOperation`) ON UPDATE NO ACTION ON DELETE CASCADE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
ROW_FORMAT=DYNAMIC
;
