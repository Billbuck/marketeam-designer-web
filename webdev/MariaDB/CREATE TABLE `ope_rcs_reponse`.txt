CREATE TABLE `ope_rcs_reponse` (
	`IdRcsReponse` BIGINT(20) NOT NULL AUTO_INCREMENT,
	`MessageIdEvenement` VARCHAR(64) NULL DEFAULT NULL COMMENT 'Id Infobip du message entrant - idempotence (rejeu des webhooks)' COLLATE 'utf8mb4_unicode_ci',
	`IdOperation` INT(11) NULL DEFAULT NULL,
	`IdBaseLigne` INT(11) NULL DEFAULT NULL COMMENT 'Destinataire d operation (ope_rcs_envoi) - NULL si unitaire ou non resolu',
	`IdRcsUnitaire` INT(11) NULL DEFAULT NULL COMMENT 'Envoi unitaire (ope_rcs_unitaire) - NULL si operation',
	`Portable` VARCHAR(20) NOT NULL COLLATE 'utf8mb4_unicode_ci',
	`TypeContenu` ENUM('TEXT','FILE','LOCATION') NOT NULL DEFAULT 'TEXT' COLLATE 'utf8mb4_unicode_ci',
	`Texte` VARCHAR(2048) NULL DEFAULT NULL COMMENT 'Texte de la reponse, URL du fichier ou latitude,longitude' COLLATE 'utf8mb4_unicode_ci',
	`DateHeureReception` DATETIME NOT NULL COMMENT 'receivedAt du webhook',
	PRIMARY KEY (`IdRcsReponse`) USING BTREE,
	UNIQUE INDEX `MessageIdEvenement` (`MessageIdEvenement`) USING BTREE,
	INDEX `IdOperation` (`IdOperation`) USING BTREE,
	INDEX `IdBaseLigne` (`IdBaseLigne`) USING BTREE,
	INDEX `IdRcsUnitaire` (`IdRcsUnitaire`) USING BTREE,
	INDEX `Portable` (`Portable`) USING BTREE,
	INDEX `DateHeureReception` (`DateHeureReception`) USING BTREE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB
ROW_FORMAT=DYNAMIC
;
