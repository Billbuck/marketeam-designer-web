	`StatutKannel` CHAR(2) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`StatutOperateur` CHAR(4) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
	`Operateur` CHAR(5) NULL DEFAULT NULL COLLATE 'utf8mb4_unicode_ci',
