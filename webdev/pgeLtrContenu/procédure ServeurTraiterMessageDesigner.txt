//!
//! @brief <indiquez ici ce que fait la procédure>
//!
//! @return (booléen)  <indiquez ici les valeurs possibles ainsi que leur interprétation>
//!
//!

procédure ServeurTraiterMessageDesigner()


sJsonData	est une chaîne


si gsJsonRecu = "" alors renvoyer Faux

vMessage est un Variant = JSONVersVariant(gsJsonRecu)

si vMessage.action = Null alors renvoyer Faux

selon vMessage.action

	cas "validated"

		// L'utilisateur a validé son document dans le Designer
		si vMessage.success = Vrai alors
			// vMessage.data contient le JSON complet exporté par exportToWebDev()
			Trace("Document validé reçu du Designer")
			jDoc est un JSON = ChaîneVersUTF8(gsJsonRecu)

			// ═════════════════════════════════════════════════════════════
			// L18/A34 — PARTIE 3 : PURGE DES COLONNES BASE AVANT STOCKAGE
			// L18/A35 — Test rendu STRICT et ROBUSTE sur attribut JSON absent.
			//
			//   Les champs portant `injecteParBase = "1"` (CHAÎNE) sont des
			//   colonnes de base injectées à l'exploitation par l'étape C
			//   (cf. SelectionModèle.txt / btnDocumentPersonnaliser.txt
			//   PARTIE 1). Ces colonnes sont VOLATILES — elles ne doivent
			//   JAMAIS être persistées dans le JsonDesignerData du document
			//   du tunnel.
			//
			//   Régression L18/A34 corrigée :
			//     Le test originel `injecteParBase = Vrai` (booléen WL)
			//     avait un comportement non fiable sur attribut JSON
			//     dynamique ABSENT. Pour un champ du modèle qui n'a JAMAIS
			//     reçu cet attribut, le test pouvait retourner Vrai à tort
			//     → purge de TOUS les champs côté validation aussi.
			//
			//   Solution L18/A35 :
			//     - Marqueur stocké comme CHAÎNE "1" (et non booléen Vrai)
			//     - Lecture défensive en chaîne : un attribut JSON absent
			//       lu en chaîne retourne "" → comparaison `= "1"` retourne
			//       Faux → champ CONSERVÉ. Aucun faux positif possible.
			//
			//   Justification fonctionnelle : le document stocké doit rester
			//   « propre » (générique), sans les colonnes base. Ainsi, quel
			//   que soit l'état futur de la base (suppression, changement,
			//   ajout), la prochaine ouverture du Designer repart d'une
			//   base saine et la PARTIE 2 + étape C recalculent les
			//   colonnes selon la base actuelle.
			// ═════════════════════════════════════════════════════════════
			si jDoc.data..Existe _et_ jDoc.data.champsFusion..Occurrence > 0 alors
				jChampsFusionPurge est un JSON = "[]"
				pour nIndPart3 = 1 _à_ jDoc.data.champsFusion..Occurrence
					// Lecture défensive en chaîne (attribut JSON absent → "")
					sInjectePB3 est une chaîne = jDoc.data.champsFusion[nIndPart3].injecteParBase
					sLibelleP3 est une chaîne = jDoc.data.champsFusion[nIndPart3].libelle

					si sInjectePB3 = "1" alors
						Trace("PARTIE 3 : champ " + sLibelleP3 + " injecteParBase=[" + sInjectePB3 + "] => PURGÉ (non persisté)")
					sinon
						Trace("PARTIE 3 : champ " + sLibelleP3 + " injecteParBase=[" + sInjectePB3 + "] => persisté")
						nIdxConservePart3 est un entier = jChampsFusionPurge..Occurrence + 1
						jChampsFusionPurge[nIdxConservePart3] = jDoc.data.champsFusion[nIndPart3]
					fin
				fin
				jDoc.data.champsFusion = jChampsFusionPurge
			fin

			sJsonData = JSONVersChaîne(jDoc.data)
			si sJsonData <> "" alors
				si _nIndDesignerEnCours = 0 alors
					// -----------------------------------//
					//              ENVELOPPE             //
					// -----------------------------------//
					__stOperation.stOperationLettre.stEnveloppe.JsonDesignerData = sJsonData
					si cpDesigner.GenererPsmdServeurEnveloppe(__stOperation.stOperationLettre.stEnveloppe, __stOperation.CodeTemp) alors
						GenererBatEnveloppeDepuisPsmd(__stOperation.stOperationLettre.stEnveloppe, __stOperation.CodeTemp)
						cpDesigner.TraiterManifestImagesDesigner(__stOperation.stOperationLettre.stEnveloppe.CheminPsmdEnveloppe, __stOperation.stOperationLettre.stEnveloppe.JsonDesignerData)
						intAdressageExpediteur.Etat		= Grisé
						intAdressageLogo.Etat				= Grisé
						intAdressagePromotion.Etat			= Grisé
						btnEnveloppeSupprimer..Visible	= Vrai
						btnEnveloppeAperçu.Visible			= Vrai
					fin


				sinon
					// -----------------------------------//
					//               DOCUMENT             //
					// -----------------------------------//
					__stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].JsonDesignerData = sJsonData
					si cpDesigner.GenererPsmdServeurDocument(__stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours]) alors
						cpDesigner.GenererBatDocumentDepuisPsmd(__stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours], __tabEchantillonBaseLigne)
						cpDesigner.TraiterManifestImagesDesigner(__stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].CheminPsmdTemp, __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].JsonDesignerData)
						si __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].tabPagePDF..Occurrence > 0 alors
							si __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].tabPagePDF[1].EstFondPerdu = Faux alors
								pour tout stVignette de __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].tabImageFond
									cpDesigner.AppliqueBlancTournant(stVignette.CheminPhysique, __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].LargeurOuvert, __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].HauteurOuvert)
								fin
								pour tout stVignette de __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].tabImageBat
									cpDesigner.AppliqueBlancTournant(stVignette.CheminPhysique, __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].LargeurOuvert, __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].HauteurOuvert)
								fin
							fin
						fin

						// ─── Marquer le document comme personnalisé ─────────────
						__stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].EstPersonnaliseDesigner	= Vrai
						__stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].DateHeureDesigner			= DateHeureSys()

						// ─── Détection QR Code Interactif (= opération "Cliquez Ici") ─
						// Flag par document + propagation au flag global de l'opération.
						__stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].EstCliquezIci				= cpDesigner.JsonContientQrInteractif(sJsonData)

						// Recalcul du flag global : Vrai si AU MOINS un document est EstCliquezIci
						__stOperation.EstCliquezIci																						= Faux
						__stOperation.TypeProjet																							= __OPERATION_COURRIER__
						pour tout stDoc de __stOperation.stOperationLettre.tabStDocument
							si stDoc.EstCliquezIci alors
								__stOperation.EstCliquezIci	= Vrai
								__stOperation.TypeProjet		= __OPERATION_COURRIER_INTERACTIF__

								sortir
							fin
						fin

						InitTitreOperation()
						InitProgressBar()

						// ─── Verrouiller les combos + afficher btnSupprimer + cacher btnImporter ─
						znrLtrContenu[_nIndDesignerEnCours].cboSource.Etat															= Inactif
						znrLtrContenu[_nIndDesignerEnCours].cboBibliothèque.Etat													= Inactif
						znrLtrContenu[_nIndDesignerEnCours].cboSelection.Etat														= Inactif
						znrLtrContenu[_nIndDesignerEnCours].btnDocumentSupprimer..Visible										= Vrai
						znrLtrContenu[_nIndDesignerEnCours].btnDocumentImporter..Visible										= Faux

					sinon
						PopupAfficheMessage(
						"Erreur lors de la génération du document" + RC + "Merci de réitérer l'opération", "Erreur", __OPERATION_COURRIER__, __TOAST_ANNULATION__)
					fin
				fin
			fin

		sinon
			Trace("Validation échouée : " + vMessage.error)
		fin

	cas "cancelled"
		// L'utilisateur a annulé le projet
		Trace("Projet annulé par l'utilisateur")
		// TODO : Retour à la page précédente du tunnel
		// PageAffiche(PAGE_TunnelCommande, "")

	autre cas
		Trace("Action non gérée côté serveur : " + vMessage.action)

fin

gsJsonRecu = ""

renvoyer Vrai
