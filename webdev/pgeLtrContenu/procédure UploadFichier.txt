		fin

		rFpHautNouveau		est un réel	= 0
		rFpBasNouveau		est un réel	= 0
		rFpGaucheNouveau	est un réel	= 0
		rFpDroiteNouveau	est un réel	= 0
		si stDocTest.tabPagePDF..Occurrence > 0 _et_ stDocTest.tabPagePDF[1].EstFondPerdu alors
			rFpHautNouveau		= stDocTest.tabPagePDF[1].FondPerduHautMm
			rFpBasNouveau		= stDocTest.tabPagePDF[1].FondPerduBasMm
			rFpGaucheNouveau	= stDocTest.tabPagePDF[1].FondPerduGaucheMm
			rFpDroiteNouveau	= stDocTest.tabPagePDF[1].FondPerduDroiteMm
		fin

		si Abs(rFpHautNouveau - rFpHautAttendu) > 0.5 _ou_ ...
			Abs(rFpBasNouveau - rFpBasAttendu) > 0.5 _ou_ ...
			Abs(rFpGaucheNouveau - rFpGaucheAttendu) > 0.5 _ou_ ...
			Abs(rFpDroiteNouveau - rFpDroiteAttendu) > 0.5 alors
			fSupprime(sRepUpload + sFichier)
			renvoyer "Le fond perdu du document importé (" + rFpHautNouveau + " mm) est différent" + RC + ...
			"de celui du document personnalisé (" + rFpHautAttendu + " mm)" + RC + ...
			"Les zones personnalisées seraient décalées"
		fin

		renvoyer ""
	fin
