jsDesigner	est un JSON
jsWrapper 	est un JSON

_nIndDesignerEnCours = znrLtrContenu

si __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].JsonDesignerData = "" alors
	retour
fin

// ─── Recomposer le wrapper du tunnel à chaque ouverture ─────────────────
// Le wrapper apporte : auth, bases, constraints, polices, format, pages,
// champsFusion, donneesApercu, et zones système (adresse destinataire si
// document porte-adresse d'une enveloppe à fenêtre)
// On y injecte ensuite la partie data du document (zones utilisateur ou modèle)
jsWrapper = ChaîneVersUTF8(cpDesigner.ComposerJsonDesignerCreation(__stOperation, __tabEchantillonBaseLigne, __stOperation.TypeOperation, Faux, _nIndDesignerEnCours, -1, -1, -1, (_nIndDesignerEnCours = 1)))

si __stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].JsonDesignerData <> "" alors
	jsDesigner = ChaîneVersUTF8(__stOperation.stOperationLettre.tabStDocument[_nIndDesignerEnCours].JsonDesignerData)

	// Source des zones utilisateur : data.xxx si imbriqué, sinon racine
	jsData est un JSON
	si jsDesigner.data..Existe alors
		jsData = jsDesigner.data
	sinon
		jsData = jsDesigner
	fin

	// ─── Fusion des zones utilisateur dans le wrapper ───────────────────
	// Les zones système (sys-adresse-*) déjà présentes dans le wrapper
	// sont préservées. On y ajoute les zones du modèle/utilisateur.
	si jsData.zonesTexte..Occurrence > 0 alors
		pour nInd = 1 _à_ jsData.zonesTexte..Occurrence
			jsWrapper.data.zonesTexte[jsWrapper.data.zonesTexte..Occurrence + 1] = jsData.zonesTexte[nInd]
		fin
	fin
	si jsData.zonesImage..Occurrence > 0 alors
		pour nInd = 1 _à_ jsData.zonesImage..Occurrence
			jsWrapper.data.zonesImage[jsWrapper.data.zonesImage..Occurrence + 1] = jsData.zonesImage[nInd]
		fin
	fin
	si jsData.zonesCodeBarres..Occurrence > 0 alors
		pour nInd = 1 _à_ jsData.zonesCodeBarres..Occurrence
			jsWrapper.data.zonesCodeBarres[jsWrapper.data.zonesCodeBarres..Occurrence + 1] = jsData.zonesCodeBarres[nInd]
		fin
	fin
	si jsData.zonesQR..Occurrence > 0 alors
		pour nInd = 1 _à_ jsData.zonesQR..Occurrence
			jsWrapper.data.zonesQR[jsWrapper.data.zonesQR..Occurrence + 1] = jsData.zonesQR[nInd]
		fin
	fin

	// Récupérer les contraintes globales du modèle si présentes
	si jsDesigner.constraints..Existe alors
		jsWrapper.constraints = jsDesigner.constraints
	fin

	// ─── Transfert du snapshot des zones initiales du modèle ───────────
	// Ce snapshot représente les zones placées à l'origine (par le créateur
	// du modèle ou lors de la 1ère ouverture du tunnel). Il sert de référence
	// pour la sémantique "zones SUPPLÉMENTAIRES" des limites côté Designer
	// (script.js : initialZoneCounts).
	// À la 1ère ouverture, ce snapshot n'existe pas encore : le Designer le
	// calcule depuis les zones reçues, puis le sauvegarde à la validation.
	// Aux ouvertures suivantes, on le transfert depuis le JSON stocké pour
	// que les limites continuent de fonctionner correctement.
	si jsDesigner.initialZoneCounts..Existe alors
		jsWrapper.data.initialZoneCounts = jsDesigner.initialZoneCounts
	fin
fin

gsJsonDesigner = JSONVersChaîne(jsWrapper)
