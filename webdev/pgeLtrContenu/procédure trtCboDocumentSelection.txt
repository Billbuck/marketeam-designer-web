// Résumé : <indiquez ici ce que fait la procédure>
// Syntaxe :
//trtCboDocumentSelection (<pNumDocument>)
//
// Paramètres :
// 	pNumDocument : <indiquez ici le rôle de pNumDocument>
// Valeur de retour :
// 	Aucune
//
// Exemple :
// Indiquez ici un exemple d'utilisation.
//
procédure trtCboDocumentSelection(pNumDocument)


nIdModele, nIdDocumentClient sont des entiers

sCboSourceGlien		est une chaîne
nCboSourceValeur		est un entier

sCboSourceGlien 	= znrLtrContenu[pNumDocument].attCboSouceGlien
nCboSourceValeur	= znrLtrContenu[pNumDocument].attCboSourceValeur

selon ExtraitChaîne(sCboSourceGlien, nCboSourceValeur)
	cas "F" // Fichier
		// Remplacement de fond : visible aussi quand EstPersonnaliseDesigner
		// (la confirmation est gérée par le clic de btnDocumentImporter)
		znrLtrContenu[pNumDocument].btnDocumentImporter..Visible 									= Vrai
		znrLtrContenu[pNumDocument].btnDocumentPersonnaliser..Visible								= Faux
		znrLtrContenu[pNumDocument].btnDocumentAperçu..Visible										= (__stOperation.stOperationLettre.tabStDocument[pNumDocument].EstDocumentEnAttente = Faux)
		__stOperation.stOperationLettre.tabStDocument[pNumDocument].ModeImpressionSelectionne		= __MODE_AUTOMATIQUE__


	cas "M" // Modèle
		nIdModele = ExtraitChaîne(attCboModeleGlien,attCboSelectionValeur)
		znrLtrContenu[pNumDocument].btnDocumentImporter..Visible 									= Faux
		znrLtrContenu[pNumDocument].btnDocumentPersonnaliser..Visible								= (nIdModele > 0)
		znrLtrContenu[pNumDocument].btnDocumentAperçu..Visible										= (nIdModele > 0)
		si pNumDocument = 1 _et_ __stOperation.stOperationLettre.EstEnveloppeFenetre = Vrai alors
			__stOperation.stOperationLettre.tabStDocument[pNumDocument].ModeImpressionSelectionne		= __MODE_LASER__
		sinon
			__stOperation.stOperationLettre.tabStDocument[pNumDocument].ModeImpressionSelectionne		= __MODE_AUTOMATIQUE__
		fin
		si nIdModele > 0 alors
			SelectionModèle(nIdModele,pNumDocument)
		fin

	cas "C" // Document client
		nIdDocumentClient = ExtraitChaîne(attCboDocumentGlien,attCboSelectionValeur)
		znrLtrContenu[pNumDocument].btnDocumentImporter..Visible 									= Faux
		znrLtrContenu[pNumDocument].btnDocumentPersonnaliser..Visible								= Faux
		znrLtrContenu[pNumDocument].btnDocumentAperçu..Visible										= (nIdDocumentClient > 0)
		__stOperation.stOperationLettre.tabStDocument[pNumDocument].ModeImpressionSelectionne		= __MODE_FOURNI__
		SelectionDocument(nIdDocumentClient)
fin

TunnelCalculeTarif()
TunnelAfficheInformationsBandeauDroit()
