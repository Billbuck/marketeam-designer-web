// ─────────────────────────────────────────────────────────────────────────────
// SelectionModèle — Page pgeLtrContenu (tunnel)
// Charge un modèle dans __stOperation.stOperationLettre.tabStDocument[pLigne]
//
// Flux :
//   1. Lire ltr_modele + ltr_support
//   2. Déterminer la bibliothèque source (Client / Marque / Marketeam)
//   3. Nettoyer les anciens fichiers du document (changement de modèle/PDF)
//   4. Initialiser les paramètres du document depuis le modèle
//   5. Copier les fichiers du modèle vers Upload\{CodeTempOp}\
//   6. Analyser le PDF + construire tabImageFond et tabImageBat
//   7. Charger JsonDesignerData du modèle
//
// Paramètres :
//   pIdModele : ID du modèle sélectionné
//   pLigne    : Indice de la ligne dans tabStDocument
// ─────────────────────────────────────────────────────────────────────────────
procédure SelectionModèle(pIdModele, pLigne)

sRepModele		est une chaîne
sRepFond			est une chaîne
sRepVignette	est une chaîne
sCodeTemp		est une chaîne
sRepUpload		est une chaîne
sFicPdfModele	est une chaîne
sFicPdfTemp		est une chaîne
sNomFond			est une chaîne
sNomBat			est une chaîne
sMessage			est une chaîne
nPage				est un entier

// Nettoyage centralisé des fichiers du document (anciens PDF/modèle/JSON)
NettoyerFichiersDocumentTemp(pLigne)

// Répertoire de travail dédié à l'opération en création
sRepUpload = cpProjet.__sRepRacine + cpProjet.__sRepUpload + ["\"] + __stOperation.CodeTemp + ["\"]
fRepCrée(sRepUpload)

// ─── Lecture du modèle et du support ─────────────────────────────────────
si pas ltr_modele.LitRecherchePremier(IdModele, pIdModele) alors
	Trace("SelectionModèle : modèle introuvable IdModele=" + pIdModele)
	retour
fin

si pas ltr_support.LitRecherchePremier(IdSupport, ltr_modele.IdSupport) alors
	Trace("SelectionModèle : support introuvable IdSupport=" + ltr_modele.IdSupport)
	retour
fin

// ─── Détermination de la bibliothèque source ─────────────────────────────
selon Vrai
	cas ltr_modele.IdMarque > 0
		sRepModele		= cpProjet.CheminMarque(ltr_modele.IdMarque, __OPERATION_COURRIER__, __REP_MODELE__, Vrai)
		sRepFond			= cpProjet.CheminMarque(ltr_modele.IdMarque, __OPERATION_COURRIER__, __REP_FOND__, Vrai)
		sRepVignette	= cpProjet.CheminMarque(ltr_modele.IdMarque, __OPERATION_COURRIER__, __REP_VIGNETTE__, Vrai)
	cas ltr_modele.IdClient > 0
		sRepModele		= cpProjet.CheminClient(ltr_modele.IdClient, __OPERATION_COURRIER__, __REP_MODELE__, Vrai)
		sRepFond			= cpProjet.CheminClient(ltr_modele.IdClient, __OPERATION_COURRIER__, __REP_FOND__, Vrai)
		sRepVignette	= cpProjet.CheminClient(ltr_modele.IdClient, __OPERATION_COURRIER__, __REP_VIGNETTE__, Vrai)
	autre cas
		sRepModele		= cpProjet.CheminMarketeam(__OPERATION_COURRIER__, __REP_MODELE__, Vrai)
		sRepFond			= cpProjet.CheminMarketeam(__OPERATION_COURRIER__, __REP_FOND__, Vrai)
		sRepVignette	= cpProjet.CheminMarketeam(__OPERATION_COURRIER__, __REP_VIGNETTE__, Vrai)
fin

sFicPdfModele = sRepModele + __DOC_LTR_MODELE__ + pIdModele + __PDF__

si pas fFichierExiste(sFicPdfModele) alors
	Trace("SelectionModèle : PDF du modèle introuvable : " + sFicPdfModele)
	retour
fin

// ─── Initialisation des paramètres du document ───────────────────────────
sCodeTemp																								= Partage.__vcUser.IDContact + "-" + DateHeureSys()

// EstUploadé = Vrai : indique à DéplaceFichierUpload qu'il y a des fichiers
// de travail (PDF, fonds, BAT dans Upload) à déplacer vers les répertoires
// définitifs de l'opération
__stOperation.stOperationLettre.tabStDocument[pLigne].EstUploadé						= Vrai
__stOperation.stOperationLettre.tabStDocument[pLigne].EstDocumentEnAttente			= Faux
__stOperation.stOperationLettre.tabStDocument[pLigne].EstPdfClient					= Faux
__stOperation.stOperationLettre.tabStDocument[pLigne].IdModele							= pIdModele
__stOperation.stOperationLettre.tabStDocument[pLigne].IdDocumentClient				= 0
__stOperation.stOperationLettre.tabStDocument[pLigne].IdSupport						= ltr_modele.IdSupport
__stOperation.stOperationLettre.tabStDocument[pLigne].EstFondPerdu					= ltr_modele.EstFondPerdu
__stOperation.stOperationLettre.tabStDocument[pLigne].EstRectoVerso					= ltr_modele.EstRectoVerso
__stOperation.stOperationLettre.tabStDocument[pLigne].EstPersonnalisable			= ltr_modele.EstPersonnalisable
__stOperation.stOperationLettre.tabStDocument[pLigne].PeutEtrePorteAdresse			= ltr_modele.EstPorteAdresse
__stOperation.stOperationLettre.tabStDocument[pLigne].LargeurOuvert					= ltr_support.LargeurOuvert
__stOperation.stOperationLettre.tabStDocument[pLigne].HauteurOuvert					= ltr_support.HauteurOuvert
__stOperation.stOperationLettre.tabStDocument[pLigne].LargeurFerme					= ltr_support.LargeurFerme
__stOperation.stOperationLettre.tabStDocument[pLigne].HauteurFerme					= ltr_support.HauteurFerme

__stOperation.stOperationLettre.tabStDocument[pLigne].CodeTemp							= sCodeTemp
__stOperation.stOperationLettre.tabStDocument[pLigne].NomFichierOrigine				= ltr_modele.NomFichierOrigine
__stOperation.stOperationLettre.tabStDocument[pLigne].NomFichierTemp					= __DOC_LTR__ + sCodeTemp + __PDF__
__stOperation.stOperationLettre.tabStDocument[pLigne].Repertoire						= sRepUpload
__stOperation.stOperationLettre.tabStDocument[pLigne].RepertoireTravail				= sRepUpload

// JsonDesignerData du modèle (data + constraints)
si pas ltr_modele.JsonDesignerData.Null _et_ ltr_modele.JsonDesignerData <> "" alors
	__stOperation.stOperationLettre.tabStDocument[pLigne].JsonDesignerData		= ltr_modele.JsonDesignerData
sinon
	__stOperation.stOperationLettre.tabStDocument[pLigne].JsonDesignerData		= ""
fin

// ─── Détection QR Code Interactif (= opération "Cliquez Ici") ────────────
// Flag par document + propagation au flag global de l'opération. Le modèle
// peut déjà contenir un QR Dynamique défini par son créateur ; on doit le
// détecter dès le chargement pour activer l'étape Landing Page du tunnel.
__stOperation.stOperationLettre.tabStDocument[pLigne].EstCliquezIci	= cpDesigner.JsonContientQrInteractif(__stOperation.stOperationLettre.tabStDocument[pLigne].JsonDesignerData)

// Recalcul du flag global : Vrai si AU MOINS un document est EstCliquezIci
__stOperation.EstCliquezIci														= Faux
pour tout stDoc de __stOperation.stOperationLettre.tabStDocument
	si stDoc.EstCliquezIci alors
		__stOperation.EstCliquezIci = Vrai
		sortir
	fin
fin

// ─── Nombre de pages attendu ─────────────────────────────────────────────
si ltr_support.NbPageObligatoire = 0 alors
	__stOperation.stOperationLettre.tabStDocument[pLigne].NombrePage = (ltr_modele.EstRectoVerso = Faux) ? 1 sinon 2
sinon
	__stOperation.stOperationLettre.tabStDocument[pLigne].NombrePage = ltr_support.NbPageObligatoire
fin

// ─── 1. Copie du PDF source du modèle vers Upload ────────────────────────
sFicPdfTemp = sRepUpload + __stOperation.stOperationLettre.tabStDocument[pLigne].NomFichierTemp
si fFichierExiste(sFicPdfTemp) alors fSupprime(sFicPdfTemp)
si pas fCopieFichier(sFicPdfModele, sFicPdfTemp) alors
	Trace("SelectionModèle : échec copie PDF " + sFicPdfModele + " --> " + sFicPdfTemp)
	retour
fin
Trace("SelectionModèle : PDF copié --> " + sFicPdfTemp)

// ─── 2. Analyse du PDF copié ─────────────────────────────────────────────
sMessage = cpDesigner.PdfAnalyseDocument(__stOperation.stOperationLettre.tabStDocument[pLigne])
si sMessage <> "" alors
	Trace("SelectionModèle : erreur PdfAnalyseDocument : " + sMessage)
	PopupAfficheMessage(sMessage, "Erreur", __OPERATION_COURRIER__, __TOAST_ANNULATION__)
	retour
fin

// URL relative du sous-répertoire d'upload
sRepUploadRel est une chaîne = cpProjet.__sHttpMarketeam + cpProjet.__sRepUpload + ["/"] + __stOperation.CodeTemp + ["/"]

// ─── 3. Copie des fonds JPG du modèle vers Upload (tabImageFond) ─────────
pour nPage = 1 _à_ __stOperation.stOperationLettre.tabStDocument[pLigne].NombrePage
	sNomFond = __DOC_LTR_MODELE__ + "Fond" + pIdModele + "p" + nPage + __JPG__
	sFicFondModele est une chaîne = sRepFond + sNomFond

	si fFichierExiste(sFicFondModele) alors
		sFicFondTemp est une chaîne = sRepUpload + __DOC_LTR_FOND__ + sCodeTemp + "p" + nPage + __JPG__
		si fFichierExiste(sFicFondTemp) alors fSupprime(sFicFondTemp)
		si fCopieFichier(sFicFondModele, sFicFondTemp) alors
			stImgFond est une structDocumentImage
			stImgFond.CheminPhysique	= sFicFondTemp
			stImgFond.CheminRelatif		= sRepUploadRel + __DOC_LTR_FOND__ + sCodeTemp + "p" + nPage + __JPG__
			__stOperation.stOperationLettre.tabStDocument[pLigne].tabImageFond.Ajoute(stImgFond)
			Trace("SelectionModèle : Fond p" + nPage + " copié --> " + sFicFondTemp)
		fin
	sinon
		Trace("SelectionModèle : Fond modèle introuvable : " + sFicFondModele)
	fin
fin

// ─── 4. Copie des vignettes BAT du modèle vers Upload (tabImageBat) ──────
pour nPage = 1 _à_ __stOperation.stOperationLettre.tabStDocument[pLigne].NombrePage
	sNomBat = __DOC_LTR_MODELE__ + pIdModele + "p" + nPage + __JPG__
	sFicBatModele est une chaîne = sRepVignette + sNomBat

	si fFichierExiste(sFicBatModele) alors
		sFicBatTemp est une chaîne = sRepUpload + __DOC_LTR_BAT__ + sCodeTemp + "p" + nPage + __JPG__
		si fFichierExiste(sFicBatTemp) alors fSupprime(sFicBatTemp)
		si fCopieFichier(sFicBatModele, sFicBatTemp) alors
			stImgBat est une structDocumentImage
			stImgBat.CheminPhysique		= sFicBatTemp
			stImgBat.CheminRelatif		= sRepUploadRel + __DOC_LTR_BAT__ + sCodeTemp + "p" + nPage + __JPG__
			__stOperation.stOperationLettre.tabStDocument[pLigne].tabImageBat.Ajoute(stImgBat)
			Trace("SelectionModèle : BAT p" + nPage + " copié --> " + sFicBatTemp)
		fin
	sinon
		Trace("SelectionModèle : Vignette modèle introuvable : " + sFicBatModele)
	fin
fin

// ─── 5. Si enveloppe à fenêtre : régénérer BAT avec zones système d'adresse ──
// Pour les documents d'une enveloppe à fenêtre, il faut régénérer le BAT via
// Designer/PSMD pour intégrer les zones système (adresse destinataire si pLigne=1).
// Le BAT pré-généré du modèle ne contient que les zones du modèle, pas les
// zones systèmes propres au tunnel.
si __stOperation.stOperationLettre.EstEnveloppeFenetre = Vrai alors

	nAvecAdresse		est un entier	= (pLigne = 1) ? 1 sinon 0

	// Sauvegarde du JsonDesignerData original (data only — zones du modèle)
	sJsonDataModele	est une chaîne	= __stOperation.stOperationLettre.tabStDocument[pLigne].JsonDesignerData

	// 5.1 — Composition du wrapper avec zones système (cf. btnDocumentPersonnaliser)
	jsWrapper			est un JSON		= ChaîneVersUTF8(cpDesigner.ComposerJsonDesignerCreation( ...
	__stOperation, __tabEchantillonBaseLigne, __OPERATION_COURRIER__, Faux, pLigne, -1, -1, -1, nAvecAdresse))

	// 5.2 — Fusion des zones du modèle dans le wrapper
	si sJsonDataModele <> "" alors
		jsDesigner		est un JSON		= ChaîneVersUTF8(sJsonDataModele)

		// Source des zones modèle : data.xxx si imbriqué, sinon racine
		jsData			est un JSON
		si jsDesigner.data..Existe alors
			jsData = jsDesigner.data
		sinon
			jsData = jsDesigner
		fin

		si jsData.zonesTexte..Occurrence > 0 alors
			pour nIndZone = 1 _à_ jsData.zonesTexte..Occurrence
				jsWrapper.data.zonesTexte[jsWrapper.data.zonesTexte..Occurrence + 1]	= jsData.zonesTexte[nIndZone]
			fin
		fin
		si jsData.zonesImage..Occurrence > 0 alors
			pour nIndZone = 1 _à_ jsData.zonesImage..Occurrence
				jsWrapper.data.zonesImage[jsWrapper.data.zonesImage..Occurrence + 1]	= jsData.zonesImage[nIndZone]
			fin
		fin
		si jsData.zonesCodeBarres..Occurrence > 0 alors
			pour nIndZone = 1 _à_ jsData.zonesCodeBarres..Occurrence
				jsWrapper.data.zonesCodeBarres[jsWrapper.data.zonesCodeBarres..Occurrence + 1]	= jsData.zonesCodeBarres[nIndZone]
			fin
		fin
		si jsData.zonesQR..Occurrence > 0 alors
			pour nIndZone = 1 _à_ jsData.zonesQR..Occurrence
				jsWrapper.data.zonesQR[jsWrapper.data.zonesQR..Occurrence + 1]	= jsData.zonesQR[nIndZone]
			fin
		fin

		// ─── Reconstruction champsFusion : doctrine "fidélité au modèle" ───
		// Avant ce patch : on AJOUTAIT les champs du modèle au wrapper, qui
		// contenait déjà ~20 champs standards par défaut issus de
		// RemplirDesignerChampsFusion(stOperation.taaBaseChamp). Résultat :
		// 22 champs exposés dans la popup -> utilisateur noyé, perte de la
		// promesse "ce que le créateur du modèle a choisi".
		//
		// Doctrine retenue : la popup contient EXACTEMENT :
		//   1. Les champs SYSTÈME du wrapper (type = "SYS") -> indispensables
		//      au pipeline PSMD (Sequentiel, Timbre/Affranchissement).
		//   2. Les champs définis dans le modèle (jsData.champsFusion),
		//      mappés ou non.
		// Les champs standards par défaut du wrapper (non-SYS) sont ÉCARTÉS.
		// L'utilisateur peut les rajouter à la demande via "+ Ajouter un
		// champ".
		//
		// Critère SYS : cf. docs/Structure Webdev Designer V3.md ligne 183
		//   « Le code SYS reste réservé aux zones système injectées par
		//     WebDev (adresse destinataire, séquentiel, datamatrix...) »
		jsChampsFusionFinal	est un JSON			= "[]"
		sNomsSysteme				est une chaîne		= ","	// pseudo-set ",NomA,NomB,"

		// ═════════════════════════════════════════════════════════════════
		// L18/A38 v2 — INJECTION DÉFENSIVE DES SYS, CONDITIONNÉE AU MODE MODÈLE
		//
		//   Compensation Option C : ComposerJsonDesignerCreation produit
		//   désormais un champsFusion VIDE (pas même les SYS). En mode
		//   modèle, on doit réinjecter les SYS pour préserver la doctrine
		//   "fidélité modèle" (champsFusion = SYS + champs modèle).
		//
		//   ⚠️ Garde-fou symétrique à btnDocumentPersonnaliser : on n'injecte
		//   les SYS QUE si jsData.champsFusion contient au moins 1 entrée
		//   (= modèle réel avec champs). Cas d'un modèle vraiment vide
		//   (sJsonDataModele = "{}" ou champsFusion absent) -> pas d'injection,
		//   le wrapper reste vide (Option C stricte).
		//
		//   Garde-fou interne : si le wrapper venait à contenir déjà les SYS
		//   (régression future de ComposerJsonDesignerCreation), pas de
		//   doublon grâce à bSeqPresent / bTimbrePresent.
		//
		//   Valeurs hardcodées identiques à RemplirDesignerChampsFusion.txt
		//   lignes 33-43 (source historique du contrat).
		// ═════════════════════════════════════════════════════════════════
		si jsData.champsFusion..Occurrence > 0 alors
			bSeqPresent		est un booléen	= Faux
			bTimbrePresent	est un booléen	= Faux
			pour nIndSysCheck = 1 _à_ jsWrapper.data.champsFusion..Occurrence
				si jsWrapper.data.champsFusion[nIndSysCheck].type = "SYS" alors
					si jsWrapper.data.champsFusion[nIndSysCheck].nom = "Sequentiel" alors
						bSeqPresent = Vrai
					sinon si jsWrapper.data.champsFusion[nIndSysCheck].nom = "Timbre" alors
						bTimbrePresent = Vrai
					fin
				fin
			fin

			si pas bSeqPresent alors
				nIdxAjoutSeq est un entier = jsWrapper.data.champsFusion..Occurrence + 1
				jsWrapper.data.champsFusion[nIdxAjoutSeq].nom			= "Sequentiel"
				jsWrapper.data.champsFusion[nIdxAjoutSeq].type		= "SYS"
				jsWrapper.data.champsFusion[nIdxAjoutSeq].libelle	= "Séquentiel"
				jsWrapper.data.champsFusion[nIdxAjoutSeq].ordre		= 998
			fin

			si pas bTimbrePresent alors
				nIdxAjoutTimbre est un entier = jsWrapper.data.champsFusion..Occurrence + 1
				jsWrapper.data.champsFusion[nIdxAjoutTimbre].nom			= "Timbre"
				jsWrapper.data.champsFusion[nIdxAjoutTimbre].type		= "SYS"
				jsWrapper.data.champsFusion[nIdxAjoutTimbre].libelle	= "Affranchissement"
				jsWrapper.data.champsFusion[nIdxAjoutTimbre].ordre		= 999
			fin
		fin

		// (1) Conserver les champs SYSTÈME du wrapper
		si jsWrapper.data.champsFusion..Occurrence > 0 alors
			pour nIndW = 1 _à_ jsWrapper.data.champsFusion..Occurrence
				si jsWrapper.data.champsFusion[nIndW].type = "SYS" alors
					jsChampsFusionFinal[jsChampsFusionFinal..Occurrence + 1] = ...
					jsWrapper.data.champsFusion[nIndW]
					si jsWrapper.data.champsFusion[nIndW].nom <> "" alors
						sNomsSysteme += jsWrapper.data.champsFusion[nIndW].nom + ","
					fin
				fin
			fin
		fin

		// (2) Ajouter les champs du modèle (mappés ou non).
		//     Garde-fou : si un champ du modèle a le même nom non-vide qu'un
		//     SYS déjà retenu, on l'ignore (priorité au wrapper pour préserver
		//     la référence système intacte). Cas théoriquement impossible —
		//     un modèle ne devrait pas contenir Sequentiel/Timbre.
		si jsData.champsFusion..Occurrence > 0 alors
			pour nIndM = 1 _à_ jsData.champsFusion..Occurrence
				bDoublonSys est un booléen = Faux
				si jsData.champsFusion[nIndM].nom <> "" alors
					si Position(sNomsSysteme, "," + jsData.champsFusion[nIndM].nom + ",") > 0 alors
						bDoublonSys = Vrai
					fin
				fin

				si pas bDoublonSys alors
					jsChampsFusionFinal[jsChampsFusionFinal..Occurrence + 1] = ...
					jsData.champsFusion[nIndM]
				sinon
					Trace("SelectionModèle : champ modèle ignoré (doublon SYS) : " ...
					+ jsData.champsFusion[nIndM].nom)
				fin
			fin
		fin

		jsWrapper.data.champsFusion = jsChampsFusionFinal

		// ═════════════════════════════════════════════════════════════════
		// L18/A34 — PARTIE 2 : PURGE DES COLONNES BASE RÉSIDUELLES
		// L18/A35 — Test rendu STRICT et ROBUSTE sur attribut JSON absent.
		//
		//   Les champs portant `injecteParBase = "1"` (CHAÎNE) sont des
		//   colonnes de base injectées lors d'une exploitation précédente
		//   (étape C, marquées par PARTIE 1). Elles sont VOLATILES :
		//   recalculées à chaque ouverture du Designer selon la base
		//   actuelle.
		//
		//   On les purge AVANT tout traitement (avant la purge des LOCAL_xxx
		//   résiduels, avant le mapping étape A, et SURTOUT avant la
		//   réinjection par l'étape C si une base est présente).
		//
		//   Régression L18/A34 corrigée :
		//     Le test originel `injecteParBase = Vrai` (booléen WL) avait un
		//     comportement non fiable sur attribut JSON dynamique ABSENT. Pour
		//     un champ du modèle qui n'a JAMAIS reçu cet attribut, le test
		//     pouvait retourner Vrai à tort -> purge de TOUS les champs.
		//
		//   Solution L18/A35 :
		//     - Marqueur stocké comme CHAÎNE "1" (et non booléen Vrai)
		//     - Lecture défensive en chaîne : un attribut JSON absent lu
		//       en chaîne retourne "" -> comparaison `= "1"` retourne Faux
		//       -> champ CONSERVÉ. Aucun faux positif possible.
		//
		//   Résultat :
		//   - Avec base : purge des anciennes colonnes + réinjection par
		//     l'étape C des colonnes de la base actuelle.
		//   - Sans base : purge des anciennes colonnes, l'étape C
		//     n'injecte rien (régime A) -> retour aux champs du modèle
		//     uniquement.
		// ═════════════════════════════════════════════════════════════════
		si jsWrapper.data.champsFusion..Occurrence > 0 alors
			jsChampsFusionSansBase est un JSON = "[]"
			pour nIndPurgeBase = 1 _à_ jsWrapper.data.champsFusion..Occurrence
				// Lecture défensive en chaîne (attribut JSON absent -> "")
				sInjectePB2	est une chaîne	= jsWrapper.data.champsFusion[nIndPurgeBase].injecteParBase
				sLibelleP2	est une chaîne	= jsWrapper.data.champsFusion[nIndPurgeBase].libelle

				si sInjectePB2 = "1" alors
					// Champ injecté par la base à l'exploitation précédente — à purger
					Trace("PARTIE 2 : champ " + sLibelleP2 + " injecteParBase=[" + sInjectePB2 + "] => PURGÉ")
				sinon
					// Champ du modèle (ou SYS) — préservé
					Trace("PARTIE 2 : champ " + sLibelleP2 + " injecteParBase=[" + sInjectePB2 + "] => conservé")
					nIdxConserve est un entier = jsChampsFusionSansBase..Occurrence + 1
					jsChampsFusionSansBase[nIdxConserve] = jsWrapper.data.champsFusion[nIndPurgeBase]
				fin
			fin
			jsWrapper.data.champsFusion = jsChampsFusionSansBase
		fin

		// L18/A29 - Le bloc `presenteEnBase` historique (V2.5 par nom
		//   technique) est DÉPLACÉ ci-dessous (après l'application des
		//   valeurs du modèle) et REFONDU en 4 étapes A/B/C/D qui exploitent
		//   le matching par LIBELLÉ via taaBaseChamp. Cf. plus bas.

		// ─── Application des valeurs du modèle sur donneesApercu ────────
		// L15/A25 — Le bloc précédent de FILTRAGE de donneesApercu a été
		//   RETIRÉ. Justification :
		//
		//   Le filtrage écartait des donneesApercu toutes les valeurs dont
		//   le nom ne figurait pas dans les champsFusion retenus (SYS +
		//   modèle). C'était cohérent du point de vue popup (l'utilisateur
		//   ne voit que les champs du modèle), MAIS cela cassait l'affichage
		//   du PAVÉ ADRESSE SYSTÈME du BAT pré-version : les zones
		//   sys-adresse-* ont besoin des valeurs Adresse1, Adresse2,
		//   Adresse3, Adresse4, CodePostal, Ville, Pays… pour s'afficher,

			+ "] avecBase=" + bAvecCorrespBase + " echantDefaut=[" + jsWrapper.data.champsFusion[nIndB].echantillonDefaut + "]")

			si jsWrapper.data.donneesApercu..Occurrence = 0 alors
				continuer  // pas d'enregistrement à remplir
			fin

			// Boucler sur TOUS les enregistrements
			pour nIndEB = 1 _à_ jsWrapper.data.donneesApercu..Occurrence
				sValeurB est une chaîne = ""

				si bAvecCorrespBase alors
					// CAS AVEC BASE + CORRESPONDANCE : valeur de la colonne base
					//   dans cet enregistrement (inchangé).
					pour nIndVB = 1 _à_ jsWrapper.data.donneesApercu[nIndEB].enregistrement..Occurrence
						si jsWrapper.data.donneesApercu[nIndEB].enregistrement[nIndVB].nom = sNomTechB alors
							sValeurB = jsWrapper.data.donneesApercu[nIndEB].enregistrement[nIndVB].valeur
							sortir
						fin
					fin
				sinon si bAvecBaseLocal alors
					// Cahier des charges §3 (Point 2, Sous-étape A) — CAS AVEC BASE
					//   + champ ABSENT (pas de correspondance dans la base courante) :
					//   valeur VIDE. La base fait foi -> un champ non mappé en
					//   base n'a pas de valeur ; il s'affichera rouge et son
					//   aperçu sera vide. Ne PAS retomber sur echantillonDefaut
					//   ici (ce serait afficher une valeur figée pour un champ
					//   que la base ne supporte pas — cf. cahier §3 « valeur
					//   en aperçu : VIDE »).
					sValeurB = ""
				sinon
					// CAS SANS BASE (régime A) : echantillonDefaut du modèle
					//   (inchangé, cf. cahier §2 « SANS base + spécifique ->
					//   échantillon obligatoire »).
					sValeurB = jsWrapper.data.champsFusion[nIndB].echantillonDefaut
				fin

				// TRACE TEMPORAIRE (Point 2, Sous-étape A) — à retirer après validation.
				//   Loggue uniquement pour le 1ᵉʳ enregistrement pour limiter le bruit
				//   en cas de multi-records (la décision dépend du champ, pas du record).
				si nIndEB = 1 alors
					Trace("[Point2-A] " + sLibB + " (" + sNomTechB + ") regimeBase=" ...
					+ (bAvecBaseLocal ? "V" sinon "F") ...
					+ " corresp=" + (bAvecCorrespBase ? "V" sinon "F") ...
					+ " -> valeur='" + sValeurB + "'")
				fin

				// Injecter ou mettre à jour la clé sCleLocaleB dans cet enregistrement
				// Pattern d'assignation directe (pas de variable JSON intermédiaire)
				bDejaB est un booléen = Faux
				pour nIndCkB = 1 _à_ jsWrapper.data.donneesApercu[nIndEB].enregistrement..Occurrence
					si jsWrapper.data.donneesApercu[nIndEB].enregistrement[nIndCkB].nom = sCleLocaleB alors
						jsWrapper.data.donneesApercu[nIndEB].enregistrement[nIndCkB].valeur	= sValeurB
						bDejaB																						= Vrai
						sortir
					fin
				fin
				si pas bDejaB alors
					nIdxNouveauB est un entier = jsWrapper.data.donneesApercu[nIndEB].enregistrement..Occurrence + 1
					jsWrapper.data.donneesApercu[nIndEB].enregistrement[nIdxNouveauB].nom			= sCleLocaleB
					jsWrapper.data.donneesApercu[nIndEB].enregistrement[nIdxNouveauB].valeur		= sValeurB
				fin
			fin
		fin

		si jsDesigner.constraints..Existe alors
			jsWrapper.constraints = jsDesigner.constraints
		fin
	fin

	// 5.3 — Injection des placeholders pour zones image vides (pour BAT lisible)
	// (cf. ServeurTraiterMessageDesignerModele page pgeLtrDocument)
	// Le PSMD production temporaire contiendra ces placeholders, mais sera
	// régénéré à la sauvegarde par AjoutModificationOperationCourrier qui
	// utilisera le JsonDesignerData propre (restauré en 5.7).
	si jsWrapper.data.zonesImage..Occurrence > 0 alors
		sMontagne	est une chaîne	= cpProjet.__sRepRacine + "Designer\assets\montagne.png"
		sHachure		est une chaîne	= cpProjet.__sRepRacine + "Designer\assets\hachure.png"

		pour nIndImg = 1 _à_ jsWrapper.data.zonesImage..Occurrence
			si jsWrapper.data.zonesImage[nIndImg].source.nomFichier = "" _ou_ ...
				jsWrapper.data.zonesImage[nIndImg].source.nomFichier = Null alors

				rLargeurZone				est un réel		= jsWrapper.data.zonesImage[nIndImg].geometrie.largeurMm
				rHauteurZone				est un réel		= jsWrapper.data.zonesImage[nIndImg].geometrie.hauteurMm
				sCheminPlaceholder		est une chaîne	= sRepUpload + "placeholder_" + pLigne + "_" + nIndImg + ".png"

				sUrlPlaceholder			est une chaîne	= "http://localhost:5000/PyMuPdfExtract/generate_placeholder" +...
				"?width_mm=" + rLargeurZone +...
				"&height_mm=" + rHauteurZone +...
				"&output=" + URLEncode(sCheminPlaceholder) +...
				"&montagne=" + URLEncode(sMontagne) +...
				"&hachure=" + URLEncode(sHachure)

				hPlaceholder		est une httpRequête
				hRepPlaceholder	est un httpRéponse
				hPlaceholder.URL		= sUrlPlaceholder
				hPlaceholder.Méthode	= httpGet
				hRepPlaceholder		= hPlaceholder.Envoie()

				si hRepPlaceholder.CodeEtat = 200 alors
					jsWrapper.data.zonesImage[nIndImg].source.nomFichier	= sCheminPlaceholder
					jsWrapper.data.zonesImage[nIndImg].source.type			= "fichier"
					Trace("SelectionModèle : placeholder zone image " + nIndImg + " " + rLargeurZone + "x" + rHauteurZone + "mm --> " + sCheminPlaceholder)
				sinon
					Trace("SelectionModèle : échec placeholder zone image " + nIndImg)
				fin
			fin
		fin
	fin

	// 5.4 — Affecter temporairement le wrapper complet pour génération PSMD
	__stOperation.stOperationLettre.tabStDocument[pLigne].JsonDesignerData = JSONVersChaîne(jsWrapper)

	// 5.5 — Vider tabImageBat et supprimer les BAT du modèle déjà copiés
	__stOperation.stOperationLettre.tabStDocument[pLigne].tabImageBat.SupprimeTout()
	pour nPage = 1 _à_ __stOperation.stOperationLettre.tabStDocument[pLigne].NombrePage
		sFicBatExist est une chaîne = sRepUpload + __DOC_LTR_BAT__ + sCodeTemp + "p" + nPage + __JPG__
		si fFichierExiste(sFicBatExist) alors fSupprime(sFicBatExist)
	fin

	// 5.6 — Générer PSMD + BAT via PrintShop
	si cpDesigner.GenererPsmdServeurDocument(__stOperation.stOperationLettre.tabStDocument[pLigne]) alors
		cpDesigner.GenererBatDocumentDepuisPsmd(__stOperation.stOperationLettre.tabStDocument[pLigne], __tabEchantillonBaseLigne)

		// Blanc tournant si non fond perdu
		si __stOperation.stOperationLettre.tabStDocument[pLigne].tabPagePDF..Occurrence > 0 _et_ ...
			__stOperation.stOperationLettre.tabStDocument[pLigne].tabPagePDF[1].EstFondPerdu = Faux alors
			pour tout stVignette de __stOperation.stOperationLettre.tabStDocument[pLigne].tabImageBat
				cpDesigner.AppliqueBlancTournant(stVignette.CheminPhysique, ...
				__stOperation.stOperationLettre.tabStDocument[pLigne].LargeurOuvert, ...
				__stOperation.stOperationLettre.tabStDocument[pLigne].HauteurOuvert)
			fin
		fin
		Trace("SelectionModèle : BAT régénéré via Designer/PSMD (porte-adresse=" + nAvecAdresse + ") pour CodeTemp=" + sCodeTemp)
	sinon
		Trace("SelectionModèle : échec GenererPsmdServeurDocument (porte-adresse) pour CodeTemp=" + sCodeTemp)
	fin

	// 5.7 — Restaurer JsonDesignerData en data only original (sans placeholders)
	// Le PSMD production sera régénéré proprement à la sauvegarde dans
	// AjoutModificationOperationCourrier à partir de cette data propre.
	__stOperation.stOperationLettre.tabStDocument[pLigne].JsonDesignerData = sJsonDataModele

fin

Trace("SelectionModèle : modèle " + pIdModele + " chargé pour ligne " + pLigne + " (CodeTemp=" + sCodeTemp + ")")
