// =========================================================================
// Génère un JPG BAT depuis un fichier PSMD via l'API PSM PHP.
//
// Principe :
//   1. Construit les données de fusion (pTabDonnees ou fallback __vcUser)
//   2. Envoie les données en POST (paramètre "data") au format JSON [{Champ,Valeur}...]
//      via Sérialise() -> L'API PHP crée le CSV temporaire et le supprime après
//   3. Décode le JPG retourné en base64 et le sauve sur disque
//
// Paramètres :
//	pCheminPsmd (chaîne ANSI) : Chemin physique complet du fichier .psmd
//	pCheminJpgSortie (chaîne ANSI) : Chemin physique complet du JPG à produire
//	pNumPage (entier - valeur par défaut=1) : Numéro de layout/page (défaut = 1)
//	ptabPrintshopData (tableau (stPrintshopData)) : <indiquez ici le rôle de ptabPrintshopData>

procédure GenerationBatDepuisPsmd(pCheminPsmd est une chaîne, pCheminJpgSortie est une chaîne, pNumPage est un entier = 1, ptabPrintshopData est un tableau de stPrintshopData = [])

stImageBat			est une structDocumentImage
sServeurPsm			est une chaîne
sUrlPsm				est une chaîne
sJsonData			est une chaîne ANSI
hPsm					est une httpRequête
hReponse				est un httpRéponse
bufPrintshop		est un Buffer
imgPrintshop		est une Image

// ── Validation du PSMD ───────────────────────────────────────────────────
si pCheminPsmd = "" ou pas fFichierExiste(pCheminPsmd) alors
	renvoyer stImageBat
fin

// ── Sérialisation JSON et appel API PSM PHP en POST ──────────────────────
Sérialise(ptabPrintshopData, sJsonData, psdJSON)

sServeurPsm				= cpFonction.ValeurParametre("PSM_ServeurUrl", prm_parametre.ValChaine..Nom)
sUrlPsm					= "http://" + sServeurPsm + "psm_marketeam_jpg.php?p=" + URLEncode(pCheminPsmd) + "&l=" + pNumPage


hPsm.URL					= sUrlPsm
hPsm.Méthode			= httpPost
hPsm.ContentType		= "application/x-www-form-urlencoded"
hPsm.Contenu			= "data=" + sJsonData.URLEncode()

hReponse					= hPsm.Envoie()

si ErreurDétectée alors
	renvoyer stImageBat
fin

si hReponse.CodeEtat <> 200 alors
	renvoyer stImageBat
fin

si fFichierExiste(pCheminJpgSortie) alors
	si pas fSupprime(pCheminJpgSortie) alors
		renvoyer stImageBat
	fin
fin


bufPrintshop	= hReponse.Contenu.Décode(encodeBASE64)
imgPrintshop	= bufPrintshop

si imgPrintshop.SauveJPEG(pCheminJpgSortie, 100) alors
	stImageBat.CheminPhysique	= pCheminJpgSortie
	// URL web : on retire la MÊME racine physique que celle ayant construit le
	// fichier (__sRepRacine), puis on normalise les \ en /. Retirer une autre
	// racine (__sMarketeamRepRacine) échouait en prod (racine physique locale
	// ≠ racine réseau), laissant fuir le chemin physique vers le navigateur.
	stImageBat.CheminRelatif	= pCheminJpgSortie.Remplace(cpProjet.__sRepRacine, cpProjet.__sHttpMarketeam)
	stImageBat.CheminRelatif	= stImageBat.CheminRelatif.Remplace("\", "/")
fin

renvoyer stImageBat
