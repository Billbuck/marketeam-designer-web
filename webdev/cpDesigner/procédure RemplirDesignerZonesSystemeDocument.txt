// cpDesigner.RemplirDesignerZonesSystemeDocument()
// Rôle : Générer la zone système adresse destinataire sur le document
//        porte-adresse, positionnée en face de la fenêtre de l'enveloppe.
//        Coordonnées lues depuis ltr_enveloppe (ZoneFenetreDoc*)
// Paramètres :
//	stDesignerDocument (stDesignerDocument) : ID de l'enveloppe dans ltr_enveloppe
//	pIdEnveloppe : <indiquez ici le rôle de pIdEnveloppe>

// 	pDocLargeur : <indiquez ici le rôle de pDocLargeur>
// 	pDocHauteur : <indiquez ici le rôle de pDocHauteur>
procédure RemplirDesignerZonesSystemeDocument(stDesignerDocument est une structDesignerDocument,  pIdEnveloppe, pTaaBaseChamp est un tableau associatif de structBaseChamp, pBAvecBase est un booléen)

stZoneSystemeTexte		est une structDesignerZoneSystemeTexte
stZoneSystemeDatamatrix	est une structDesignerZoneSystemeDatamatrix

// =========================================================================
// LECTURE DES COORDONNÉES EN BDD
// =========================================================================
si pas ltr_enveloppe.LitRecherchePremier(IdEnveloppe, pIdEnveloppe) alors
	retour
fin

// Pas de fenêtre = pas de zone adresse sur le document
si ltr_enveloppe.NbrFenetre = 0 alors
	retour
fin

// Coordonnées non renseignées
si ltr_enveloppe.ZoneFenetreLargeur <= 0 _ou_ ltr_enveloppe.ZoneFenetreHauteur <= 0 alors
	retour
fin
