// ─────────────────────────────────────────────────────────────────────────────
// ComposerJsonDesignerModele
// Compose le JSON complet pour ouvrir le Designer en mode "template"
// (création/modification de modèle)
//
// Paramètres :
//	pStDocument (structDocument) : Document avec PDF, dimensions, fond, etc.
//	pIdClient (entier) : ID client
//	pIdContact (entier) : ID contact
//	pIdModele (entier) : ID modèle (0 si nouveau)
//	pLibelle (chaîne Unicode) : Libellé du modèle
//	pTaaBaseChamp : Tableau associatif des champs de fusion standard
//	pTabEchantillon (tableau (StructBaseLigne)) : Échantillon d'aperçu (coordonnées client)
//	pLimTexte (entier - valeur par défaut=-1) : Limite zones texte pour l'utilisateur du tunnel
//	pLimImage (entier - valeur par défaut=-1) : Limite zones image pour l'utilisateur du tunnel
//	pLimBarcode (entier - valeur par défaut=-1) : Limite zones code-barres pour l'utilisateur du tunnel
//	pLimQr (entier - valeur par défaut=0) : Limite zones QR pour l'utilisateur du tunnel

// Note : les limites sont celles destinées à l'UTILISATEUR FINAL dans le tunnel.
procédure ComposerJsonDesignerModele(pStDocument est une structDocument, pIdClient est un entier, pIdContact est un entier, pIdModele est un entier, pLibelle est une chaîne, pTaaBaseChamp, pTabEchantillon est un tableau de StructBaseLigne, pLimTexte est un entier = -1, pLimImage est un entier = -1, pLimBarcode est un entier = -1, pLimQr est un entier = 0)

stLoad			est une structDesignerLoad
sJsonDocument	est une chaîne ANSI

// =========================================================================
// BLOCS D'ENVIRONNEMENT
// =========================================================================
stLoad.action						= "load"
stLoad.mode							= "template"
stLoad.auth							= RemplirDesignerAuth(pIdClient, pIdContact)
stLoad.constraints				= RemplirDesignerConstraints(pLimTexte, pLimImage, pLimQr, pLimBarcode)
stLoad.limites						= RemplirDesignerLimites()
stLoad.policesDisponibles		= RemplirDesignerPolices(pIdClient)
stLoad.policesSysteme			= RemplirDesignerPolicesSysteme()		// §3.10 : polices système (hors dsn_police)
// V3.4 - Listes de référence pour la modale "Ajouter un champ" du Designer
//        (cf. docs/cahier_des_charges_creation_champs_fusion.md V2.2 §5).
//        TOUJOURS envoyées en mode template — le créateur du modèle peut
//        ajouter des champs dans tous les cas (sur base existante OU sans base).
stLoad.champsStandard			= RemplirDesignerChampsStandard(pIdClient)
stLoad.typesDisponibles			= RemplirDesignerTypesDisponibles()
// V3.5 - Verrouillage global de la gestion des champs : TOUJOURS Vrai en mode
//        template (le créateur du modèle a toujours le droit d'ajouter,
//        modifier, supprimer des champs).
//        Cf. cahier des charges V2.3 §5.2.
stLoad.autoriserGestionChamps	= Vrai
stLoad.Theme						= "LTR"

// =========================================================================
// DOCUMENT (data)
// =========================================================================

// Identification
sIdDocument est une chaîne = "modele-" + (pIdModele > 0 ? pIdModele sinon "nouveau")
stLoad.Document.identification = RemplirDesignerIdentification(sIdDocument, "modele", 1, pLibelle)

// Format : fond perdu depuis les dimensions détaillées de structPagePdf
si pStDocument.EstFondPerdu _et_ pStDocument.tabPagePDF..Occurrence >= 1 alors

	nFormatPapierLargeur		est un réel		= 0
	nFormatPapierHauteur		est un réel		= 0
	nIdFormatImpression		est un entier	= pStDocument.tabPagePDF[1].IdFormatImpression

	si ltr_format_impression.LitRecherchePremier(IdFormatImpression, nIdFormatImpression) alors
		si ltr_format_impression.IdFormatPapier > 0 alors
			si ltr_format_impression.LitRecherchePremier(IdFormatImpression, ltr_format_impression.IdFormatPapier) alors
				nFormatPapierLargeur	= ltr_format_impression.Largeur
				nFormatPapierHauteur	= ltr_format_impression.Hauteur
			fin
		fin
	fin

	stLoad.Document.formatDocument = RemplirDesignerFormat(
	pStDocument.LargeurOuvert, pStDocument.HauteurOuvert,
	Vrai,
	pStDocument.tabPagePDF[1].FondPerduHautMm,
	pStDocument.tabPagePDF[1].FondPerduBasMm,
	pStDocument.tabPagePDF[1].FondPerduGaucheMm,
	pStDocument.tabPagePDF[1].FondPerduDroiteMm,
	nFormatPapierLargeur,
	nFormatPapierHauteur)

sinon
	stLoad.Document.formatDocument = RemplirDesignerFormat(pStDocument.LargeurOuvert, pStDocument.HauteurOuvert, Faux, 0, 0, 0, 0)
fin

// Pages : recto seul ou recto/verso, images de fond
stLoad.Document.pages = RemplirDesignerPages(pStDocument.EstRectoVerso, pStDocument.tabImageFond, pStDocument.Repertoire + pStDocument.NomFichierTemp)

Trace("ComposerJsonDesignerModele : cheminFond page 1 = [" + stLoad.Document.pages[1].cheminFond + "]")
Trace("ComposerJsonDesignerModele : Nombre de pages = " + stLoad.Document.pages..Occurrence)

// =========================================================================
// BLOCS COMMUNS
// =========================================================================

// V3.4 - Mode template : champsFusion et donneesApercu vides au load.
//        L'utilisateur peuple lui-même la liste via la modale "Ajouter un
//        champ" (onglet Standard puisant dans champsStandard envoyés en
//        racine, ou onglet Spécifique pour des champs non mappés gérés via
//        localId). Les paramètres pTaaBaseChamp et pTabEchantillon ne sont
//        donc PAS exploités pour produire le load (mais on les conserve dans
//        la signature pour ne pas casser les appels existants et pour pouvoir
//        revenir à l'option B - peuplement initial depuis une base "défaut"
//        client - si la décision Q4 venait à évoluer).
//        Décision donneur d'ordre du 11/05/2026 (V2.1 §4 -> V2.2 §6 / §9
//        du cahier des charges).
tabChampsFusionVide		est un tableau de structDesignerChampFusion
tabApercuVide				est un tableau de structDesignerEnregistrement
stLoad.Document.champsFusion	= tabChampsFusionVide
stLoad.Document.donneesApercu	= tabApercuVide

// =========================================================================
// SÉRIALISATION EN JSON
// =========================================================================
Sérialise(stLoad, sJsonDocument, psdJSON + psdMiseEnForme)

sJsonDocument = sJsonDocument.Remplace("""##Null##""","null")

renvoyer sJsonDocument
