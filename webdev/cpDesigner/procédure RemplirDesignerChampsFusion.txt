// -----------------------------------------------------------------------------
// cpDesigner.RemplirDesignerChampsFusion()
// Rôle : Remplir la liste des champs de fusion disponibles pour le Designer
//        Parcourt le tableau associatif des champs communs aux bases de l'opération
//        La clé du tableau = libellé d'origine du champ client
//        Champ = nom de la colonne MariaDB
//        Type : transmission du TYPE FIN (TXT/ENT/DEC/MON/DAT/TIM/EML/TEL/
//        SMS/CDP/URL/IMG/ALG) — cf. docs/cahier-des-charges-formatage-
//        champs-fusion.md §3.1 (Lot 1). Les constantes __CHAMP_TYPE_*__
//        contiennent directement les codes Designer. Fallback "TXT" si vide.
// Paramètres :
//   pTaaBaseChamp (tableau associatif de structBaseChamp) : champs communs aux bases
// Retour : tableau de stDesignerChampFusion rempli
// -----------------------------------------------------------------------------
procédure RemplirDesignerChampsFusion(pTaaBaseChamp)

tabChamps	est un tableau de structDesignerChampFusion
stUnChamp	est une structDesignerChampFusion
nOrdre		est un entier


pour tout stChamp, sLibellé de pTaaBaseChamp
	stUnChamp.nom = stChamp.Champ
	// Lot 1 formatage — type FIN transmis tel quel (les constantes
	// __CHAMP_TYPE_*__ = codes Designer : TXT/ENT/.../IMG/ALG).
	// L'aplatissement historique "IMG sinon TXT" est levé (cf. CDC §3.1).
	si stChamp.Type <> "" alors
		stUnChamp.type = stChamp.Type
	sinon
		stUnChamp.type = "TXT"
	fin
	stUnChamp.libelle	= sLibellé
	stUnChamp.ordre		= stChamp.Ordre
	nOrdre = stUnChamp.ordre
	tabChamps.Ajoute(stUnChamp)
fin

stUnChamp.nom 		= "Sequentiel"
stUnChamp.type 		= "SYS"
stUnChamp.libelle	= "Séquentiel"
stUnChamp.ordre		= ++nOrdre
tabChamps.Ajoute(stUnChamp)

stUnChamp.nom		= "Timbre"
stUnChamp.type		= "SYS"
stUnChamp.libelle	= "Affranchissement"
stUnChamp.ordre		= ++nOrdre
tabChamps.Ajoute(stUnChamp)


renvoyer tabChamps
