// ─────────────────────────────────────────────────────────────────────────────
// cpDesigner.ImporteGoogleFontsGenerales
// Rôle : Peupler dsn_police (portée GÉNÉRALE) à partir de l'API Google Fonts,
//        pour une liste de 21 familles. PROCÉDURE PONCTUELLE, REJOUABLE
//        (UPSERT idempotent). NE JAMAIS appeler au runtime du Designer.
//
//        Lot 0 (B). Source de vérité : docs/cahier-des-charges-polices-designer.md
//        §3.1 (1 graisse = 1 entrée), §3.3 (Nom ≠ NomTechnique), §3.6 (portée).
//
// Périmètre Lot 0 (B) :
//   - On STOCKE les URLs Google (forcées en https) dans Url/Bold/Italic/BoldItalicUrl.
//   - PAS de téléchargement de fichiers ni de sous-dossier Fonts (print = Lot 2).
//   - Convention BoldUrl/BoldItalicUrl = OPTION A (remplis uniquement pour la
//     graisse Regular 400 → 700/700italic ; vides pour les autres graisses).
//
// Réutilise le pattern existant (landing pages) : HTTPRequête → HTTPDonneRésultat.
// La réponse est traitée en JSON DYNAMIQUE car le map "files" a des clés
// dynamiques ("100","regular","300italic"…).
//
// Retour : chaîne récapitulative (familles importées / non trouvées / nb créées-MAJ).
//          Également tracée (Trace).
// ─────────────────────────────────────────────────────────────────────────────
procédure ImporteGoogleFontsGenerales() : chaîne

sCle					est une chaîne
sUrlBase				est une chaîne
sUrl					est une chaîne
ResLancement		est un booléen
sResCode				est une chaîne
jReponse				est un JSON
bJsonOk				est un booléen	= Vrai
sFamily				est une chaîne
sRecap				est une chaîne
sNonTrouvees		est une chaîne	= ""

nNbFamilles			est un entier	= 0
nNbCreees			est un entier	= 0
nNbFichiers			est un entier	= 0		// Lot 2A-2 : fichiers TTF téléchargés (regular + variantes)
nNbNomTech			est un entier	= 0		// Lot 2A-2 : NomTechnique corrigés depuis le nameID1 réel
nNbPurgees			est un entier	= 0		// Voie B : entrées Google générales purgées avant reconstruction
nNbFichSuppr		est un entier	= 0		// Voie B : fichiers cache supprimés lors de la purge
sEchecsTtf			est une chaîne	= ""		// Lot 2A-2 : polices dont le NomTechnique n'a PU être corrigé
sRepCache			est une chaîne	= ""		// Lot 2A-2 : dossier cache global des polices

taaCible				est un tableau associatif de booléens		// familles cibles (clé = family)
taaTrouvee			est un tableau associatif de booléens		// familles cibles trouvées dans l'API
taaLibelleGraisse	est un tableau associatif de chaînes		// Code(entier) -> libellé principal (ici_police_weight)
tabAjoutFam			est un tableau de structDesignerPolice		// entrées créées pour une famille (retour de TraiteFamilleGooglePolice)

// ─── 1. Clé API (jamais en dur) ──────────────────────────────────────────
sCle		= cpFonction.ValeurParametre("GoogleFonts_Api_Key", prm_parametre.ValChaine..Nom)
sUrlBase	= cpFonction.ValeurParametre("GoogleFonts_Api_Url", prm_parametre.ValChaine..Nom)
si sUrlBase = "" alors
	sUrlBase = "https://www.googleapis.com/webfonts/v1/webfonts"
fin

si sCle = "" alors
	renvoyer "Import annulé : paramètre GoogleFonts_Api_Key absent dans prm_parametre."
fin

// ─── 2. Familles cibles (correspondance EXACTE sur family) ────────────────
//   Attention : « Source Sans 3 » = ex « Source Sans Pro ».
taaCible["Roboto"]				= Vrai
taaCible["Open Sans"]			= Vrai
taaCible["Lato"]					= Vrai
taaCible["Montserrat"]			= Vrai
taaCible["Poppins"]				= Vrai
taaCible["Inter"]					= Vrai
taaCible["Source Sans 3"]		= Vrai
taaCible["Nunito"]				= Vrai
taaCible["Raleway"]				= Vrai
taaCible["Work Sans"]			= Vrai
taaCible["Oswald"]				= Vrai
taaCible["Roboto Condensed"]	= Vrai
taaCible["Bebas Neue"]			= Vrai
taaCible["Anton"]					= Vrai
taaCible["Merriweather"]		= Vrai
taaCible["Playfair Display"]	= Vrai
taaCible["Lora"]					= Vrai
taaCible["Roboto Slab"]			= Vrai
taaCible["PT Serif"]				= Vrai
taaCible["Roboto Mono"]			= Vrai
taaCible["Source Code Pro"]	= Vrai

// ─── 3. Libellés de graisses « principales » (ici_police_weight) ──────────
taaLibelleGraisse = RemplirLibellesGraisses()

// ─── 3 bis. Cache global des fichiers de police (Lot 2A-2, stratégie A) ────
//   Si indisponible : l'import DB continue, mais sans téléchargement ni
//   correction de NomTechnique (tracé).
sRepCache = RepCachePolices()
si sRepCache = "" alors
	Trace("ImporteGoogleFonts : cache polices indisponible -> téléchargement/NomTechnique sautés")
fin

// ─── 3 ter. PURGE des entrées Google générales (Voie B : import idempotent) ──
//   À chaque import, on supprime TOUTES les entrées Google de portée générale
//   (Origine='google' AND IdMarque IS NULL AND IdClient IS NULL) + leurs fichiers
//   de cache, PUIS on reconstruit un jeu unique. Idempotent par construction,
//   indépendant de tout matching par Nom (la recherche d'existant cassée est
//   supprimée — cf. AjoutePolice, INSERT direct).
//   PRÉSERVE strictement : les polices ajoutées à la main (Origine <> 'google')
//   et toutes les portées marque/client.
PurgePolicesGoogleGenerales()

// ─── 4. Appel API (liste COMPLÈTE, jamais ?family=) ───────────────────────
sUrl = sUrlBase + "?key=" + sCle + "&sort=popularity"
ResLancement = HTTPRequête(sUrl)
si pas ResLancement alors
	renvoyer "Import annulé : échec HTTPRequête vers l'API Google Fonts."
fin
sResCode = HTTPDonneRésultat()

// ─── 5. Parsing JSON dynamique ────────────────────────────────────────────
QUAND EXCEPTION DANS
	jReponse = sResCode
FAIRE
	bJsonOk = Faux
FIN
si bJsonOk = Faux _ou_ jReponse.items..Existe = Faux alors
	renvoyer "Import annulé : réponse API illisible ou sans 'items'."
fin

// ─── 6. Parcours + filtre + mapping ───────────────────────────────────────
POUR nI = 1 _À_ jReponse.items..Occurrence
	sFamily = ""
	QUAND EXCEPTION DANS
		si jReponse.items[nI].family..Existe alors
			sFamily = jReponse.items[nI].family
		fin
	FAIRE
	FIN

	si sFamily <> "" _et_ taaCible[sFamily]..Existe alors
		taaTrouvee[sFamily]	= Vrai
		nNbFamilles++
		// Version Google au niveau famille (pour la colonne dsn_police.Version)
		sVersionFam est une chaîne = ""
		QUAND EXCEPTION DANS
			si jReponse.items[nI].version..Existe alors
				sVersionFam = jReponse.items[nI].version
			fin
		FAIRE
		FIN
		// Traitement par famille délégué à la procédure partagée (Lot A1).
		// Compteurs accumulés PAR RÉFÉRENCE ; nNbCreees = somme des entrées créées.
		tabAjoutFam = TraiteFamilleGooglePolice(jReponse.items[nI], sFamily, taaLibelleGraisse, sRepCache, sVersionFam, nNbFichiers, nNbNomTech, sEchecsTtf)
		nNbCreees += tabAjoutFam..Occurrence
	fin
FIN

// ─── 7. Familles cibles non trouvées (trace) ──────────────────────────────
POUR TOUT bCible, sCibleNom DE taaCible
	si taaTrouvee[sCibleNom]..Existe = Faux alors
		si sNonTrouvees <> "" alors sNonTrouvees += ", "
		sNonTrouvees += sCibleNom
	fin
FIN

// ─── 8. Récap ─────────────────────────────────────────────────────────────
sRecap = "Import Google Fonts terminé — familles importées : " + nNbFamilles ...
	+ " / 21 ; entrées Google générales purgées : " + nNbPurgees ...
	+ " ; fichiers cache supprimés : " + nNbFichSuppr ...
	+ " ; entrées créées : " + nNbCreees ...
	+ " ; fichiers TTF téléchargés : " + nNbFichiers ...
	+ " ; NomTechnique corrigés : " + nNbNomTech
si sNonTrouvees <> "" alors
	sRecap += " ; familles NON trouvées : " + sNonTrouvees
fin
si sEchecsTtf <> "" alors
	sRecap += " ; polices en échec (NomTechnique non corrigé) : " + sEchecsTtf
fin
Trace(sRecap)

renvoyer sRecap


// ═════════════════════════════════════════════════════════════════════════
// PROCÉDURES INTERNES
// ═════════════════════════════════════════════════════════════════════════

	// ─── Purge des entrées Google GÉNÉRALES + leurs fichiers cache (Voie B) ──
	//   Supprime TOUTES les entrées Origine='google' de portée générale
	//   (IdMarque IS NULL AND IdClient IS NULL) ainsi que leurs fichiers cache.
	//   PRÉSERVE les polices manuelles (Origine <> 'google') et les portées
	//   marque/client. On collecte d'abord les IdPolice dans un tableau pour ne
	//   pas modifier dsn_police pendant l'itération du curseur de requête.
	PROCÉDURE INTERNE PurgePolicesGoogleGenerales()
		tabIds		est un tableau de entier
		sReqDel		est une chaîne ANSI
		sdReqDel	est une Source de Données
		sReqDel = [
			SELECT IdPolice FROM dsn_police
			WHERE Origine = 'google' AND IdMarque IS NULL AND IdClient IS NULL
		]
		si sdReqDel.ExécuteRequêteSQL(dsn_police.Connexion, hRequêteSansCorrection, sReqDel) alors
			POUR TOUT sdReqDel
				tabIds.Ajoute(sdReqDel.IdPolice)
			FIN
		fin
		sdReqDel.AnnuleDéclaration()

		POUR TOUT nIdSuppr DE tabIds
			SupprimeFichiersCachePolice(nIdSuppr)
			si dsn_police.LitRecherchePremier(IdPolice, nIdSuppr) alors
				si dsn_police.Supprime() alors
					nNbPurgees++
				fin
			fin
		FIN
	FIN

	// ─── Supprime les 4 fichiers de cache d'une police (si présents) ────────
	PROCÉDURE INTERNE SupprimeFichiersCachePolice(LOCAL pId est un entier)
		si sRepCache = "" alors
			RETOUR
		fin
		SupprimeSiExiste(sRepCache + "police_" + pId + "_regular.ttf")
		SupprimeSiExiste(sRepCache + "police_" + pId + "_bold.ttf")
		SupprimeSiExiste(sRepCache + "police_" + pId + "_italic.ttf")
		SupprimeSiExiste(sRepCache + "police_" + pId + "_bolditalic.ttf")
	FIN

	PROCÉDURE INTERNE SupprimeSiExiste(LOCAL pChemin est une chaîne)
		si fFichierExiste(pChemin) alors
			si fSupprime(pChemin) alors
				nNbFichSuppr++
			fin
		fin
	FIN

	// ─── Traitement par famille : délégué à la procédure PARTAGÉE ───────────
	//   cpDesigner.TraiteFamilleGooglePolice (extraite au Lot A1) — réutilisée
	//   par l'ajout unitaire. UrlVariante / TelechargeVariante / l'INSERT y sont
	//   désormais internes. La purge globale ci-dessus reste propre à l'import.
