// =========================================================================
// Partage.GenererBatDocumentDepuisPsmd()
// Rôle : Générer les BAT (aperçus JPEG) d'un document à partir du PSMD
//        via l'API PHP PrintShop Mail (psm_marketeam_jpg.php)
//
//        Utilise en priorité CheminPsmdBat (format fini sans fond perdu)
//        pour éviter un bug de décalage des codes-barres dans la preview JPEG.
//        Si CheminPsmdBat est vide, fallback sur CheminPsmdTemp (production).
//
//        Après génération, le PSMD BAT temporaire est supprimé.
//
// PROCÉDURE PARTAGÉE (Partage) — utilisable depuis pgeLtrContenu et pgeLtrDocument
//
// Paramètres :
//   stDocument          (structDocument)                : Document en cours
//   pTabEchantillon     (tableau de StructBaseLigne)    : Données échantillon pour le publipostage
// =========================================================================

procédure GenererBatDocumentDepuisPsmd(stDocument est une structDocument, pTabEchantillon est un tableau de StructBaseLigne)

sRepTravail				est une chaîne			= stDocument.RepertoireTravail
sCheminPsmd				est une chaîne
tabPrintshopData		est un tableau de stPrintshopData
stPrintshopData		est un stPrintshopData
stImageBat				est une structDocumentImage
sCheminJpg				est une chaîne
nPage						est un entier

// ─── Choix du PSMD source pour le BAT ────────────────────────────────────
sCheminPsmd = stDocument.CheminPsmdBat
si sCheminPsmd = "" alors sCheminPsmd = stDocument.CheminPsmdTemp


// ─── Validation : le PSMD doit exister sur disque ────────────────────────
si sCheminPsmd = "" _ou_ pas fFichierExiste(sCheminPsmd) alors
	retour
fin


// ─── Construction du tableau de fusion ───────────────────────────────────
// L16/A26 - REFONTE : la source des valeurs de fusion est désormais
//   `donneesApercu[1].enregistrement[*]` du JSON Designer, et NON plus
//   `champsFusion` + indirection WLanguage sur StructBaseLigne.
//
//   Pourquoi ce changement :
//
//   1. champsFusion est désormais FILTRÉ par la doctrine "fidélité au
//      modèle" (cf. SelectionModèle.txt + btnDocumentPersonnaliser.txt) :
//      il ne contient que les champs SYS + les champs du modèle. Les
//      champs d'adresse système (Societe, Adresse1-4, CodePostal, Ville,
//      Pays…) utilisés par le pavé adresse système (zones sys-adresse-*)
//      n'y figurent PLUS quand un modèle est chargé. Conséquence avec
//      l'ancienne logique : pavé adresse incomplet dans le BAT
//      pré-version (« Monsieur ATTALI Michel » sans CHRONODIRECT/rue/CP/
//      Ville). Bug remonté tour précédent par le donneur d'ordre.
//
//   2. donneesApercu reste COMPLET (= toutes les valeurs du wrapper +
//      écrasement par les valeurs du modèle, cf. L15/A25). C'est la
//      source EXHAUSTIVE des paires {nom, valeur} prêtes à l'emploi
//      pour PrintShop Mail.
//
//   3. Plus besoin d'indirection WLanguage sur StructBaseLigne — la
//      SaaS (RemplirDesignerApercu) a déjà résolu les valeurs au
//      moment de la composition du wrapper. Code plus simple, moins
//      de cas particuliers.
//
//   4. Cas Sequentiel/Timbre : les valeurs simulées (« 000001 »,
//      « Affranchissement 001 ») sont désormais transmises par la SaaS
//      via donneesApercu, vérifié sur les traces. Ces hardcodes
//      historiques sont retirés au profit de la source unique.
//      Si en production une valeur Sequentiel/Timbre est manquante,
//      corriger à la source (RemplirDesignerApercu) plutôt que de
//      réintroduire un fallback hardcode ici.
//
//   5. Couvre AUTOMATIQUEMENT les 3 cas de l'ancienne logique :
//      - Champs mappés (nom non vide)              → entrées par nom
//      - Champs spécifiques non mappés (LOCAL_xxx)  → entrées par LOCAL_xxx
//      - Champs SYS Sequentiel/Timbre               → entrées par nom SYS
//      Tous présents dans donneesApercu sous le même format.
//
// pTabEchantillon reste en paramètre pour préserver la signature
// publique, mais N'EST PLUS UTILISÉ dans ce flux. Le fallback historique
// (__vcUser.stCoordonnées) est retiré : si donneesApercu est vide ou
// absent, PrintShop Mail reçoit un tableau vide → marqueurs substitués
// par chaîne vide (comportement sain en mode dégradé).

si stDocument.JsonDesignerData <> "" alors
	jDoc						est un JSON									= stDocument.JsonDesignerData.VersUTF8()

	// ─── CONTRAT NOMMAGE (docs/nommage-variables-psm.md §4) ──────────────
	// Le PSMD émet désormais les champs spécifiques sous leur LIBELLÉ
	// SANITISÉ (@Vehicule@) — cf. GenererPsmdServeurDocument. Le BAT doit
	// donc fournir les paires {libellé sanitisé, valeur} EN PLUS des
	// paires existantes ChampN / LOCAL_xxx (conservées : aperçu écran et
	// rétro-compatibilité du canal). MÊME SOURCE de mapping que le PSMD :
	// Partage.ConstruireMappingPsmLibelle — pas deux logiques.
	taaMappingLibelleBat	est un tableau associatif de chaîne	= Partage.ConstruireMappingPsmLibelle(stDocument.JsonDesignerData)

	// Le JSON peut avoir 2 formats selon le contexte d'appel :
	//   - Wrapper complet {action, mode, auth, bases, data:{...}}
	//     (cas BAT initial : SelectionModèle, UploadFichier porte-adresse, ...)
	//     → donneesApercu est sous jDoc.data.donneesApercu
	//   - Data only (cas après validation Designer)
	//     → donneesApercu est à la racine jDoc.donneesApercu
	// On teste l'existence de jDoc.data ET de donneesApercu sous data
	// pour décider, pour rester aligné sur l'ancienne logique.
	jDocChamps				est un JSON
	si jDoc.data..Existe _et_ jDoc.data.donneesApercu..Existe alors
		jDocChamps = jDoc.data
	sinon
		jDocChamps = jDoc
	fin

	si jDocChamps.donneesApercu..Occurrence > 0 _et_ ...
		jDocChamps.donneesApercu[1].enregistrement..Occurrence > 0 alors

		pour nIEnr = 1 _à_ jDocChamps.donneesApercu[1].enregistrement..Occurrence
			sNomEnr	est une chaîne	= jDocChamps.donneesApercu[1].enregistrement[nIEnr].nom
			sValEnr	est une chaîne	= jDocChamps.donneesApercu[1].enregistrement[nIEnr].valeur

			// Garde-fou : un nom vide ne doit pas être transmis à PrintShop
			// Mail (clé invalide). Ne devrait jamais arriver mais protection
			// défensive.
			si sNomEnr <> "" alors
				AjoutePrintshopData(sNomEnr, sValEnr)

				// CONTRAT NOMMAGE : paire supplémentaire {libellé sanitisé,
				// valeur} pour les spécifiques mappés — c'est elle que le
				// PSMD renommé (@Vehicule@) consomme au BAT.
				si sNomEnr.Gauche(6) = "LOCAL_" _et_ taaMappingLibelleBat[sNomEnr]..Existe alors
					AjoutePrintshopData(taaMappingLibelleBat[sNomEnr], sValEnr)
				fin
			fin
		fin
	fin
fin

// ─── Génération BAT JPG page par page ────────────────────────────────────
stDocument.tabImageBat.SupprimeTout()

pour nPage = 1 _à_ stDocument.NombrePage

	sCheminJpg	= sRepTravail + __DOC_LTR_BAT__ + stDocument.CodeTemp + "p" + nPage + __JPG__
	stImageBat	= cpDesigner.GenerationBatDepuisPsmd(sCheminPsmd, sCheminJpg, nPage, tabPrintshopData)

	si stImageBat.CheminPhysique <> "" alors
		stDocument.tabImageBat.Ajoute(stImageBat)
	fin
fin

// ─── Nettoyage du PSMD BAT temporaire ────────────────────────────────────
si stDocument.CheminPsmdBat <> "" _et_ stDocument.CheminPsmdBat <> stDocument.CheminPsmdTemp alors
	si fFichierExiste(stDocument.CheminPsmdBat) alors
		fSupprime(stDocument.CheminPsmdBat)
	fin
	stDocument.CheminPsmdBat = ""
fin

	procédure interne AjoutePrintshopData(pChamp, pValeur)
		stPrintshopData.Champ	= pChamp
		stPrintshopData.Valeur	= pValeur
		tabPrintshopData.Ajoute(stPrintshopData)
	fin
