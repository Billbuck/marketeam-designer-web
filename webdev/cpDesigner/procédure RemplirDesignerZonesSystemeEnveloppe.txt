// cpDesigner.RemplirDesignerZonesSystemeEnveloppe()
// Rôle : Générer les zones système sur l'enveloppe pour la pré-version BAT
//        Même logique que RemplirDesignerZonesSystemeDocument
// Paramètres :
//	stDesignerDocument (stDesignerDocument) : document Designer à remplir
//	pIdEnveloppe (entier) : ID de l'enveloppe dans ltr_enveloppe

procédure RemplirDesignerZonesSystemeEnveloppe(stDesignerDocument est une structDesignerDocument, pIdEnveloppe est un entier)

stZoneSystemeTexte		est une structDesignerZoneSystemeTexte
stZoneSystemeDatamatrix	est une structDesignerZoneSystemeDatamatrix

// =========================================================================
// LECTURE ENVELOPPE EN BDD
// =========================================================================
si pas ltr_enveloppe.LitRecherchePremier(IdEnveloppe, pIdEnveloppe) alors
	retour
fin


si ltr_enveloppe.NbrFenetre = 0 alors
	// =========================================================================
	// ZONE 1 : CONTOUR FENÊTRE (sys-adresse-contour)
	// =========================================================================
	VariableRAZ(stZoneSystemeTexte)
	stZoneSystemeTexte.Id							= "sys-adresse-contour"
	stZoneSystemeTexte.Page							= 1
	stZoneSystemeTexte.Nom							= ""
	stZoneSystemeTexte.PosX							= ltr_enveloppe.ZoneFenetreX
	stZoneSystemeTexte.PosY							= ltr_enveloppe.ZoneFenetreY
	stZoneSystemeTexte.Largeur						= ltr_enveloppe.ZoneFenetreLargeur
	stZoneSystemeTexte.Hauteur						= ltr_enveloppe.ZoneFenetreHauteur
	stZoneSystemeTexte.FondTransparent				= Faux
	stZoneSystemeTexte.BordureEpaisseur				= 1
	stZoneSystemeTexte.BordureStyle					= "dashed"
	stZoneSystemeTexte.Imprimable					= Vrai
	stDesignerDocument.zonesTexte.Ajoute(RemplirDesignerZoneTexteSysteme(stZoneSystemeTexte))
fin
