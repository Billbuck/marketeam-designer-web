// ─────────────────────────────────────────────────────────────────────────────
// cpDesigner.TraiteFamilleGooglePolice
// Rôle : Traiter UNE famille Google (item d'API) : créer les entrées dsn_police
//        (portée GÉNÉRALE), télécharger les TTF dans le cache, renseigner le
//        NomTechnique (nameID1), selon la RÈGLE PAIRE DE BASE :
//          - regular -> 400 = entrée ancre portant BoldUrl(700)/BoldItalicUrl(700italic) ;
//          - « italic », « 700 », « 700italic » NE créent PAS d'entrée ;
//          - autres poids (100/200/300/500/600/800/900) = entrées propres + ItalicUrl.
//
//        Procédure PARTAGÉE (extraite de ImporteGoogleFontsGenerales au Lot A1,
//        logique INCHANGÉE) réutilisée par l'import global ET l'ajout unitaire.
//        Ne fait PAS de purge : l'appelant gère la purge (globale ou ciblée).
//
// Paramètres :
//   jItem             (JSON, local)   : item de l'API webfonts pour la famille
//   pFamily           (chaîne, local) : nom de famille (persisté dans dsn_police.Famille)
//   taaLibelleGraisse (tableau assoc, local) : Code -> libellé (RemplirLibellesGraisses)
//   pRepCache         (chaîne, local) : dossier cache global (RepCachePolices)
//   nNbFichiers       (entier, PAR RÉFÉRENCE) : compteur TTF téléchargés (accumulé)
//   nNbNomTech        (entier, PAR RÉFÉRENCE) : compteur NomTechnique corrigés (accumulé)
//   sEchecs           (chaîne, PAR RÉFÉRENCE) : liste des échecs (accumulée)
//
// Retour : tableau de structDesignerPolice (entrées créées, mappées comme
//          RemplirDesignerPolices : id/nom/nomTechnique/famille/graisse/weight/
//          style/url/boldUrl/italicUrl/boldItalicUrl).
// ─────────────────────────────────────────────────────────────────────────────
procédure TraiteFamilleGooglePolice(local jItem est un JSON, local pFamily est une chaîne, local taaLibelleGraisse est un tableau associatif de chaînes, local pRepCache est une chaîne, local pVersion est une chaîne, nNbFichiers est un entier, nNbNomTech est un entier, sEchecs est une chaîne) : tableau de structDesignerPolice

tabPolices		est un tableau de structDesignerPolice
taaWeights		est un tableau associatif de booléens		// clé chaîne = poids présent
sVar				est une chaîne
nW					est un entier
sNom				est une chaîne
sLibGraisse		est une chaîne
sUrlReg			est une chaîne
sUrlIta			est une chaîne
sUrlBold			est une chaîne
sUrlBoldIta		est une chaîne
nId				est un entier
sCheminReg		est une chaîne
sNameId1			est une chaîne
sNomTechFinal	est une chaîne
stP				est une structDesignerPolice

// ─── Recensement des graisses NON-italiques présentes (regular -> 400) ──────
quand exception dans
	si jItem.variants..Existe alors
		pour nV = 1 _à_ jItem.variants..Occurrence
			sVar = jItem.variants[nV]
			si sVar.Position("italic") > 0 alors
				// variante italique : pas d'entrée
			sinon si sVar = "regular" alors
				taaWeights["400"] = Vrai
			sinon si sVar = "700" alors
				// Bold de l'ancre 400 : pas d'entrée propre
			sinon si EstNumérique(sVar) alors
				taaWeights[sVar] = Vrai
			fin
		fin
	fin
faire
fin


// ─── Une entrée par graisse présente ────────────────────────────────────────
pour tout bW, sWCle de taaWeights
	nW = Val(sWCle)

	// URLs (test présence + https forcé)
	si nW = 400 alors
		sUrlReg		= UrlVariante(jItem, "regular")
		sUrlIta		= UrlVariante(jItem, "italic")
		sUrlBold		= UrlVariante(jItem, "700")
		sUrlBoldIta	= UrlVariante(jItem, "700italic")
	sinon
		sUrlReg		= UrlVariante(jItem, sWCle)
		sUrlIta		= UrlVariante(jItem, sWCle + "italic")
		sUrlBold		= ""
		sUrlBoldIta	= ""
	fin

	// Libellé (Nom) : Regular -> Famille seule ; sinon Famille + libellé graisse
	si nW = 400 alors
		sNom = pFamily
	sinon
		si taaLibelleGraisse[nW]..Existe alors
			sLibGraisse = taaLibelleGraisse[nW]
		sinon
			sLibGraisse = sWCle
		fin
		sNom = pFamily + " " + sLibGraisse
	fin

	// ─── UPSERT par (Famille, Weight) en portée générale google ─────────────
	//   Si la ligne existe -> UPDATE en GARDANT l'IdPolice (références préservées,
	//   rien n'est détruit) ; sinon -> INSERT. Ne touche PAS à EstActif.
	//   Après purge globale (import), TrouveIdPolice ne trouve rien -> INSERT
	//   -> résultat identique à l'ancien comportement.
	nId = TrouveIdPolice(pFamily, nW)
	si nId > 0 alors
		// ─── UPDATE en place (IdPolice inchangé) ───
		si dsn_police.LitRecherchePremier(IdPolice, nId) alors
			dsn_police.Nom							= sNom
			dsn_police.NomTechnique				= sNom
			dsn_police.Famille					= pFamily
			dsn_police.Url							= sUrlReg
			dsn_police.BoldUrl					= sUrlBold
			dsn_police.ItalicUrl					= sUrlIta
			dsn_police.BoldItalicUrl			= sUrlBoldIta
			dsn_police.Weight						= nW
			dsn_police.Style						= "normal"
			dsn_police.Origine					= "google"
			dsn_police.Version					= pVersion
			dsn_police.DateHeureModification	= DateHeureSys()
			si pas dsn_police.Modifie() alors
				nId = 0
			fin
		sinon
			nId = 0
		fin
	sinon
		// ─── INSERT (portée générale : IdMarque/IdClient NULL) ───
		dsn_police.RAZ()
		dsn_police.Nom						= sNom
		dsn_police.NomTechnique			= sNom
		dsn_police.Famille				= pFamily
		dsn_police.Url						= sUrlReg
		dsn_police.BoldUrl				= sUrlBold
		dsn_police.ItalicUrl				= sUrlIta
		dsn_police.BoldItalicUrl		= sUrlBoldIta
		dsn_police.Weight					= nW
		dsn_police.Style					= "normal"
		dsn_police.IdMarque..Null		= Vrai
		dsn_police.IdClient..Null		= Vrai
		dsn_police.EstActif				= 1
		dsn_police.Origine				= "google"
		dsn_police.Version				= pVersion
		dsn_police.DateHeureCreation	= DateHeureSys()
		si dsn_police.Ajoute() alors
			nId = dsn_police.IdPolice
		fin
	fin

	si nId > 0 alors
		sNomTechFinal = sNom

		// ─── Téléchargement des variantes existantes + NomTechnique réel ────
		si pRepCache <> "" alors
			sCheminReg = TelechargeVariante(nId, "regular", sUrlReg, pRepCache)
			si sCheminReg <> "" alors nNbFichiers++
			si TelechargeVariante(nId, "bold", sUrlBold, pRepCache) <> "" alors nNbFichiers++
			si TelechargeVariante(nId, "italic", sUrlIta, pRepCache) <> "" alors nNbFichiers++
			si TelechargeVariante(nId, "bolditalic", sUrlBoldIta, pRepCache) <> "" alors nNbFichiers++

			si sCheminReg <> "" alors
				sNameId1 = LitNomInterneTtf(sCheminReg)
				si sNameId1 <> "" alors
					si dsn_police.LitRecherchePremier(IdPolice, nId) alors
						dsn_police.NomTechnique				= sNameId1
						dsn_police.DateHeureModification	= DateHeureSys()
						si dsn_police.Modifie() alors
							nNbNomTech++
							sNomTechFinal = sNameId1
						fin
					fin
				sinon
					si sEchecs <> "" alors sEchecs += ", "
					sEchecs += sNom + " (nameID1 illisible)"
				fin
			sinon
				si sEchecs <> "" alors sEchecs += ", "
				sEchecs += sNom + " (TTF régulier non téléchargé)"
			fin
		fin

		// ─── Mapping structDesignerPolice (comme RemplirDesignerPolices) ────
		stP.id				= nId
		stP.nom				= sNom
		stP.nomTechnique	= sNomTechFinal
		stP.famille			= pFamily
		si taaLibelleGraisse[nW]..Existe alors
			stP.graisse = taaLibelleGraisse[nW]
		sinon
			stP.graisse = ""
		fin
		stP.weight			= nW
		stP.style			= "normal"
		stP.url				= sUrlReg
		stP.boldUrl			= sUrlBold
		stP.italicUrl		= sUrlIta
		stP.boldItalicUrl	= sUrlBoldIta
		tabPolices.Ajoute(stP)
	fin
fin

renvoyer tabPolices

// PROCÉDURES INTERNES
// ═════════════════════════════════════════════════════════════════════════

// ─── Recherche d'une ligne existante par (Famille, Weight) générale google ──
	//   Paramètre chaîne QUOTÉ. Renvoie l'IdPolice existant, 0 si absent.
	procédure interne TrouveIdPolice(local pFamilleL est une chaîne, local pWeightL est un entier) : entier
		nIdT		est un entier	= 0
		sReqT		est une chaîne ANSI
		sdReqT	est une Source de Données
		sReqT = [
			SELECT IdPolice FROM dsn_police
			WHERE Origine = 'google' AND IdMarque IS NULL AND IdClient IS NULL
			AND Famille = '[%pFamilleL%]' AND Weight = [%pWeightL%]
		]
		si sdReqT.ExécuteRequêteSQL(dsn_police.Connexion, hRequêteSansCorrection, sReqT) alors
			pour tout sdReqT
				nIdT = sdReqT.IdPolice
				sortir
			fin
		fin
		sdReqT.AnnuleDéclaration()
		renvoyer nIdT
	fin

	// ─── URL d'une variante (clé dynamique dans files) + https forcé ────────
	procédure interne UrlVariante(local jItemL est un JSON, local pVariant est une chaîne) : chaîne
		sU est une chaîne = ""
		quand exception dans
			si jItemL.files[pVariant]..Existe alors
				sU = jItemL.files[pVariant]
			fin
		faire
		fin
		si sU <> "" _et_ sU.Gauche(7) = "http://" alors
			sU = "https://" + sU.Milieu(8)
		fin
		renvoyer sU
	fin

// ─── Télécharge une variante dans le cache (nommage déterministe) ───────
//   <cache>\police_<IdPolice>_<suffixe>.ttf ; URL vide -> rien.
	//   Renvoie le chemin local si OK, "" sinon (tracé).
	procédure interne TelechargeVariante(local pIdPolice est un entier, local pSuffixe est une chaîne, local pUrl est une chaîne, local pRepCacheL est une chaîne) : chaîne
		sDest est une chaîne
		si pUrl = "" _ou_ pRepCacheL = "" alors
			renvoyer ""
		fin
		sDest = pRepCacheL + "police_" + pIdPolice + "_" + pSuffixe + ".ttf"
		si TelechargeFichierBinaire(pUrl, sDest) alors
			renvoyer sDest
		fin
		renvoyer ""
	fin
