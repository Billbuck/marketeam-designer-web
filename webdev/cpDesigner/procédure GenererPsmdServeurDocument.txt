// =========================================================================
// Partage.GenererPsmdServeur()
// Rôle : Générer un fichier .psmd à partir du JSON Designer
//        en appelant psmd_cli.js via Node.js côté serveur.
//        Écrit le JSON dans un fichier temp, appelle Node, vérifie le résultat,
//        met à jour CheminPsmdTemp et NomFichierPsmd dans tabStDocument.
//        Génère également un PSMD BAT (sans fond perdu) pour éviter un bug
//        de rendu des codes-barres dans CreateJPEGPreviewOfLayout.
//
// Paramètres :
//	stDocument (structDocument) : <indiquez ici le rôle de pIndexDocument>

procédure GenererPsmdServeurDocument(stDocument est une structDocument)

sNodeExe					est une chaîne	= INILit("NODE", "CHEMIN", "", NomFichierIni())
sNodeCli					est une chaîne	= INILit("NODE", "CLI", "", NomFichierIni())
sRepTravail				est une chaîne
sCodeTemp				est une chaîne
sCheminJsonTemp		est une chaîne
sCheminPsmd				est une chaîne
sCheminPsmdBat			est une chaîne
sNomFichierPsmd		est une chaîne
sCommande				est une chaîne
sCommandeBat			est une chaîne

// ─── Validation du JSON en entrée ────────────────────────────────────────
si stDocument.JsonDesignerData = "" alors
	renvoyer Faux
fin

// ─── Validation de la configuration INI ──────────────────────────────────
si sNodeExe = "" ou sNodeCli = "" alors
	renvoyer Faux
fin

// ─── Construction des chemins — UTILISE LE REPERTOIRE DU DOCUMENT ────────
sRepTravail				= stDocument.RepertoireTravail
sCodeTemp				= stDocument.CodeTemp
sNomFichierPsmd		= __DOC_LTR_PSMD__ + sCodeTemp + ".psmd"
sCheminPsmd				= sRepTravail + sNomFichierPsmd
sCheminJsonTemp		= sRepTravail + "psmd_json_" + sCodeTemp + ".tmp"

// ═════════════════════════════════════════════════════════════════════
// L18/A44 v5 + CONTRAT NOMMAGE (10/06/2026) —
//   CONVERSION @LOCAL_<localId>@ > @<libellé sanitisé>@ POUR PSMD
//   (approche SUBSTITUTION STRING DIRECTE)
//
//   Le JsonDesignerData stocké en BDD utilise des marqueurs stables
//   @LOCAL_<localId>@ pour les champs spécifiques (doctrine "survie aux
//   changements de base" — cf. L18/A39+A41). Un marqueur @LOCAL_xxx@
//   dans le PSMD ne trouverait jamais de colonne correspondante dans
//   les données de fusion > champ vide.
//
//   CONTRAT NOMMAGE (docs/nommage-variables-psm.md) : le nom cible est
//   le LIBELLÉ SANITISÉ du champ (@Vehicule@, @Date_de_livraison@) —
//   plus jamais la colonne physique (@Champ3@) — pour vérification
//   visuelle à l'atelier. Le module d'export CSV de production
//   (PAS ENCORE CONSTRUIT, dernière étape du projet) DEVRA émettre ses
//   en-têtes via Partage.SanitiseNomPsm : noms et casse identiques
//   PSMD/CSV par construction. Le BAT ajoute les paires
//   {libellé sanitisé, valeur} dans tabPrintshopData
//   (cf. GenererBatDocumentDepuisPsmd).
//
//   Le mapping LOCAL_<localId> > libellé sanitisé est construit par
//   Partage.ConstruireMappingPsmLibelle (SOURCE UNIQUE, partagée avec
//   le BAT et le futur export production — pas deux logiques). Il est
//   utilisé uniquement pour piloter les substitutions STRING sur la
//   chaîne JSON entière (approche v5 : évite les problèmes d'alias/
//   copie WLangage observés en v3-v4).
//
//   Motifs de substitution par entrée du mapping (couverture sans
//   risque de faux positif) :
//     a. "@LOCAL_xxx@" > "@Vehicule@"
//        Cible : contenu/contenuRtf des zonesTexte, qrConfig.fields[*]
//        tokens, et tout marqueur RTF/texte dans le JSON.
//     a2. "@LOCAL_xxx__SUF@" > "@Vehicule__SUF@" pour les suffixes de
//        formatage __MAJ/__MIN/__PRO (casse, Lot 1) et __DATE1 à
//        __DATE6 (dates, Lot 2) — cf. CDC formatage §2.3.2, §3.3.
//        Sans ambiguïté : un nom sanitisé ne contient jamais "__".
//     b. "key":"LOCAL_xxx" > "key":"Vehicule"
//        Cible : quillDelta.ops[*] des embeds merge-tag.
//     c. "champFusion":"LOCAL_xxx" > "champFusion":"Vehicule"
//        Cible : zonesImage[*].source.champFusion + zonesCodeBarres[*]
//        .champFusion + zonesQR[*].champFusion (valeur sans @).
//
//   donneesApercu[*].enregistrement[*].nom = "LOCAL_xxx" n'est PAS
//   substitué (motif "nom":"LOCAL_xxx" non ciblé) — intentionnel :
//   le canal parallèle BAT (tabPrintshopData) doit conserver les
//   paires LOCAL_xxx pour rétro-compatibilité.
//
//   Le JsonDesignerData STOCKÉ en BDD n'est PAS modifié — on travaille
//   sur la copie chaîne locale sJsonPsmdTemp, écrite dans le fichier
//   temp passé à psmd_cli.js et jamais ré-écrite en BDD.
//
//   Couverture des chemins d'appel : ServeurTraiterMessageDesigner,
//   BTN_TraiterMessageModele, AjoutModificationOperationCourrier (CAS 1
//   ajout base sans réouverture Designer), SelectionModèle, UploadFichier,
//   SupprimerDocument — tous accèdent à __stOperation.taaBaseChamp.
// ═════════════════════════════════════════════════════════════════════

// 1+2. Construire le mapping LOCAL_<localId> > libellé sanitisé via la
//      procédure partagée (contrat docs/nommage-variables-psm.md §4).
//      Garde-fou : si __stOperation/taaBaseChamp inaccessibles (hors
//      tunnel courrier), mapping vide > aucune substitution > PSMD
//      écrit avec @LOCAL_xxx@ comme avant.
taaMappingPsmd		est un tableau associatif de chaîne	= Partage.ConstruireMappingPsmLibelle(stDocument.JsonDesignerData)

// 3. SUBSTITUTION TEXTUELLE DIRECTE sur la chaîne JSON entière.
//    sJsonPsmdTemp est la copie chaîne destinée au fichier temp.
//    Pour chaque entrée du mapping, 3 motifs de substitution.
sJsonPsmdTemp		est une chaîne								= stDocument.JsonDesignerData
nNbSubst				est un entier								= 0

si taaMappingPsmd..Occurrence > 0 alors
	pour tout sNomPsmSubst, sCleMappingSubst de taaMappingPsmd
		// sCleMappingSubst contient "LOCAL_xxx", sNomPsmSubst le libellé
		// sanitisé (cf. Partage.ConstruireMappingPsmLibelle)

		// a. Marqueurs @KEY@ dans contenu / contenuRtf / qrConfig.fields tokens
		sMarqA	est une chaîne	= "@" + sCleMappingSubst + "@"
		sMarqB	est une chaîne	= "@" + sNomPsmSubst + "@"
		nPosA		est un entier	= sJsonPsmdTemp.Position(sMarqA)
		si nPosA > 0 alors
			sJsonPsmdTemp = sJsonPsmdTemp.Remplace(sMarqA, sMarqB)
			nNbSubst++
		fin

		// a2. Lots 1 & 2 formatage — marqueurs SUFFIXÉS @LOCAL_xxx__MAJ@ /
		//     __MIN / __PRO (casse) et __DATE1 à __DATE6 (dates,
		//     cf. CDC formatage §2.3.2, §3.3). Le motif a. ne les couvre
		//     pas : le @ fermant vient APRÈS le suffixe. Le suffixe est
		//     conservé dans le nom PSM (@Vehicule__MAJ@) pour que
		//     psmd-generator émette la variable alias correspondante.
		pour nIndSufFmt = 1 _à_ 9
			sSufFmt		est une chaîne	= ExtraitChaîne("__MAJ" + TAB + "__MIN" + TAB + "__PRO" + TAB + "__DATE1" + TAB + "__DATE2" + TAB + "__DATE3" + TAB + "__DATE4" + TAB + "__DATE5" + TAB + "__DATE6", nIndSufFmt, TAB)
			sMarqA2		est une chaîne	= "@" + sCleMappingSubst + sSufFmt + "@"
			sMarqB2		est une chaîne	= "@" + sNomPsmSubst + sSufFmt + "@"
			si sJsonPsmdTemp.Position(sMarqA2) > 0 alors
				sJsonPsmdTemp = sJsonPsmdTemp.Remplace(sMarqA2, sMarqB2)
				nNbSubst++
			fin
		fin

		// b. quillDelta ops merge-tag : "key":"LOCAL_xxx"
		// Guillemets WLangage : "" = un guillemet littéral.
		sKeyA	est une chaîne	= """key"":""" + sCleMappingSubst + """"
		sKeyB	est une chaîne	= """key"":""" + sNomPsmSubst + """"
		nPosB	est un entier	= sJsonPsmdTemp.Position(sKeyA)
		si nPosB > 0 alors
			sJsonPsmdTemp = sJsonPsmdTemp.Remplace(sKeyA, sKeyB)
			nNbSubst++
		fin

		// c. source.champFusion / champFusion : "champFusion":"LOCAL_xxx"
		sCFA	est une chaîne	= """champFusion"":""" + sCleMappingSubst + """"
		sCFB	est une chaîne	= """champFusion"":""" + sNomPsmSubst + """"
		nPosC	est un entier	= sJsonPsmdTemp.Position(sCFA)
		si nPosC > 0 alors
			sJsonPsmdTemp = sJsonPsmdTemp.Remplace(sCFA, sCFB)
			nNbSubst++
		fin
	fin
fin

// ─── Substitution nom d'affichage -> nom technique dans le \fonttbl (Lot 2B) ──
//   Couvre PROD (--imposition) ET BAT (--no-bleed) : même fichier temp.
//   Opère sur la copie chaîne sJsonPsmdTemp (le JsonDesignerData en base reste intact).
nIdClientPolice est un entier = 0
quand exception dans
	nIdClientPolice = __stOperation.IdClient
faire
	nIdClientPolice = 0
fin
sJsonPsmdTemp = SubstitueNomTechniquePolices(sJsonPsmdTemp, nIdClientPolice)

// ─── Injection de l'URL générique dans les QR interactifs (zonesQR) ──────────
//   Couvre PROD (--imposition) ET BAT (--no-bleed) : même fichier temp.
//   Tout QR interactif = toute entrée de zonesQR (zone système sys-qr-cliquezici
//   OU QR ajouté manuellement "zone-N", créé sans contenu). Sans contenu, le
//   générateur encode une valeur par défaut (scan -> recherche web) : on impose
//   donc l'URL générique ICI, à la génération. Ciblage par la COLLECTION zonesQR
//   (pas par id), parsing JSON (jamais de substitution textuelle aveugle). Le
//   JsonDesignerData stocké en base reste intact : on n'opère que sur la copie
//   chaîne sJsonPsmdTemp. L'URL variable par destinataire relève du module d'export.
sUrlQrInteractif est une chaîne = cpFonction.ValeurParametre("Url_Courrier_Interactif_Generique", prm_parametre.ValChaine..Nom)
si sUrlQrInteractif = "" alors
	sUrlQrInteractif = "https://www.marketeam.com"
fin

quand exception dans
	jPsmdQr		est un JSON			= sJsonPsmdTemp.VersUTF8()
	bModifieQr	est un booléen		= Faux

	// Deux formes possibles, comme JsonContientQrInteractif : data.zonesQR ou racine zonesQR
	si jPsmdQr.data..Existe alors
		si jPsmdQr.data.zonesQR..Existe alors
			pour nIndQrInj = 1 _à_ jPsmdQr.data.zonesQR..Occurrence
				jPsmdQr.data.zonesQR[nIndQrInj].contenu = sUrlQrInteractif
				// Sécurité : si un qrConfig typé existait (anormal pour un QR Marketeam),
				// le générateur le lit en priorité -> y écrire aussi l'URL.
				si jPsmdQr.data.zonesQR[nIndQrInj].qrConfig..Existe _et_ jPsmdQr.data.zonesQR[nIndQrInj].qrConfig.type..Existe alors
					jPsmdQr.data.zonesQR[nIndQrInj].qrConfig.fields.url = sUrlQrInteractif
				fin
				bModifieQr = Vrai
			fin
		fin
	sinon
		si jPsmdQr.zonesQR..Existe alors
			pour nIndQrInj = 1 _à_ jPsmdQr.zonesQR..Occurrence
				jPsmdQr.zonesQR[nIndQrInj].contenu = sUrlQrInteractif
				si jPsmdQr.zonesQR[nIndQrInj].qrConfig..Existe _et_ jPsmdQr.zonesQR[nIndQrInj].qrConfig.type..Existe alors
					jPsmdQr.zonesQR[nIndQrInj].qrConfig.fields.url = sUrlQrInteractif
				fin
				bModifieQr = Vrai
			fin
		fin
	fin

	si bModifieQr alors
		sJsonPsmdTemp = jPsmdQr.VersChaîne()
	fin
faire
//	Trace("GenererPsmdServeur : injection QR interactif ignorée (JSON non parsable / structure inattendue) : " + ExceptionInfo(errMessage))
fin

// ─── Écriture du JSON dans le fichier temporaire ─────────────────────────
si pas fSauveTexte(sCheminJsonTemp, sJsonPsmdTemp.VersUTF8()) alors
	renvoyer Faux
fin

si fFichierExiste(sCheminPsmd) alors
	si pas fSupprime(sCheminPsmd) alors
		renvoyer Faux
	fin
fin

// ─── Construction et exécution de la commande (PSMD production) ──────────
sCommande = ChaîneConstruit(
[
	"%1" "%2" "%3" "%4" --imposition
],
sNodeExe, sNodeCli, sCheminJsonTemp, sCheminPsmd)

si pas LanceAppli(sCommande, exeIconise, exeBloquant) alors
	renvoyer Faux
fin

// ─── Génération du PSMD BAT (sans fond perdu) ───────────────────────────
// Corrige un bug de rendu des codes-barres dans CreateJPEGPreviewOfLayout
// quand le fond perdu est en mode "Ajouter à la tâche"
sCheminPsmdBat	= sRepTravail + "psmd_bat_" + sCodeTemp + ".psmd"
sCommandeBat	= ChaîneConstruit(
[
	"%1" "%2" "%3" "%4"
],
sNodeExe, sNodeCli, sCheminJsonTemp, sCheminPsmdBat) + " --no-bleed"

LanceAppli(sCommandeBat, exeIconise, exeBloquant)

// ─── Nettoyage du fichier JSON temporaire ────────────────────────────────
si fFichierExiste(sCheminJsonTemp) alors
	fSupprime(sCheminJsonTemp)
fin

// ─── Vérification de la présence du .psmd produit ────────────────────────
si pas fFichierExiste(sCheminPsmd) alors
	renvoyer Faux
fin

// ─── Mise à jour de la structure document ────────────────────────────────
stDocument.CheminPsmdTemp	= sCheminPsmd
stDocument.NomFichierPsmd	= sNomFichierPsmd

// PSMD BAT : utilisé par GenerationBatDepuisPsmd à la place du PSMD production
si fFichierExiste(sCheminPsmdBat) alors
	stDocument.CheminPsmdBat	= sCheminPsmdBat
sinon
	stDocument.CheminPsmdBat	= sCheminPsmd
fin

// ─── Lot 2C : copie des polices utilisées dans Fonts (avant rendu BAT) ───────
//   Garantit les TTF du document dans …\Fonts\ pour sa preview BAT et la prod ;
//   l'union projet se construit au fil des documents (même répertoire). Non bloquant.
CopiePolicesUtiliseesVersFonts(stDocument.JsonDesignerData, stDocument.RepertoireTravail)

renvoyer Vrai
