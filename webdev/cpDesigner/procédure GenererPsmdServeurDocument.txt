// =========================================================================
// Partage.GenererPsmdServeur()
// Rôle : Générer un fichier .psmd à partir du JSON Designer
//        en appelant psmd_cli.js via Node.js côté serveur.
//        Écrit le JSON dans un fichier temp, appelle Node, vérifie le résultat,
//        met à jour CheminPsmdTemp et NomFichierPsmd dans tabStDocument.
//        Génère également un PSMD BAT (sans fond perdu) pour éviter un bug
//        de rendu des codes-barres dans CreateJPEGPreviewOfLayout.
//
// Paramètres :
//	stDocument (structDocument) : <indiquez ici le rôle de pIndexDocument>

procédure GenererPsmdServeurDocument(stDocument est une structDocument)

sNodeExe					est une chaîne	= INILit("NODE", "CHEMIN", "", NomFichierIni())
sNodeCli					est une chaîne	= INILit("NODE", "CLI", "", NomFichierIni())
sRepTravail				est une chaîne
sCodeTemp				est une chaîne
sCheminJsonTemp		est une chaîne
sCheminPsmd				est une chaîne
sCheminPsmdBat			est une chaîne
sNomFichierPsmd		est une chaîne
sCommande				est une chaîne
sCommandeBat			est une chaîne

// ─── Validation du JSON en entrée ────────────────────────────────────────
si stDocument.JsonDesignerData = "" alors
	Trace("GenererPsmdServeur : pJsonDesigner vide")
	renvoyer Faux
fin

// ─── Validation de la configuration INI ──────────────────────────────────
si sNodeExe = "" ou sNodeCli = "" alors
	Trace("GenererPsmdServeur : section [NODE] manquante dans le fichier INI (CHEMIN / CLI)")
	renvoyer Faux
fin

// ─── Construction des chemins — UTILISE LE REPERTOIRE DU DOCUMENT ────────
sRepTravail				= stDocument.RepertoireTravail
sCodeTemp				= stDocument.CodeTemp
sNomFichierPsmd		= __DOC_LTR_PSMD__ + sCodeTemp + ".psmd"
sCheminPsmd				= sRepTravail + sNomFichierPsmd
sCheminJsonTemp		= sRepTravail + "psmd_json_" + sCodeTemp + ".tmp"

// ═════════════════════════════════════════════════════════════════════
// L18/A44 v5 — CONVERSION @LOCAL_<localId>@ > @<nom technique>@ POUR PSMD
//   (approche SUBSTITUTION STRING DIRECTE)
//
//   Le JsonDesignerData stocké en BDD utilise des marqueurs stables
//   @LOCAL_<localId>@ pour les champs spécifiques (doctrine "survie aux
//   changements de base" — cf. L18/A39+A41). Mais PrintShop Mail en
//   PRODUCTION charge une source CSV externe dont les colonnes
//   s'appellent par leurs noms TECHNIQUES (Civilite, Champ3, Champ4…).
//   Un marqueur @LOCAL_xxx@ dans le PSMD ne trouverait jamais de
//   colonne correspondante > champ vide en production.
//   (Le BAT, lui, masquait le bug via un canal parallèle qui injecte
//   tabPrintshopData {nom:LOCAL_xxx, valeur:...}.)
//
//   SOLUTION v5 : on opère par SUBSTITUTION TEXTUELLE DIRECTE sur la
//   chaîne JSON entière (stDocument.JsonDesignerData), pour éviter
//   les problèmes d'alias/copie WLangage observés en v3-v4 (l'API
//   structurée jDocChampsConv = jDocConverti créait une copie
//   indépendante, les modifs ne se propageaient pas à la sérialisation
//   finale > PSMD conservait @LOCAL_xxx@).
//
//   Pour construire le mapping LOCAL_<localId> > <nom technique>, on
//   parse une COPIE en JSON pour lire champsFusion[*].libelle et le
//   matcher avec __stOperation.taaBaseChamp[libelle].Champ. Ce mapping
//   est utilisé ensuite uniquement pour piloter les substitutions
//   STRING sur la chaîne JSON entière.
//
//   3 motifs de substitution par entrée du mapping (couverture sans
//   risque de faux positif) :
//     a. "@LOCAL_xxx@" > "@Champ3@"
//        Cible : contenu/contenuRtf des zonesTexte, qrConfig.fields[*]
//        tokens, et tout marqueur RTF/texte dans le JSON.
//     b. "key":"LOCAL_xxx" > "key":"Champ3"
//        Cible : quillDelta.ops[*] des embeds merge-tag.
//     c. "champFusion":"LOCAL_xxx" > "champFusion":"Champ3"
//        Cible : zonesImage[*].source.champFusion + zonesCodeBarres[*]
//        .champFusion + zonesQR[*].champFusion (valeur sans @).
//
//   donneesApercu[*].enregistrement[*].nom = "LOCAL_xxx" n'est PAS
//   substitué (motif "nom":"LOCAL_xxx" non ciblé) — intentionnel :
//   le canal parallèle BAT (tabPrintshopData) doit conserver les
//   paires LOCAL_xxx pour rétro-compatibilité.
//
//   Le JsonDesignerData STOCKÉ en BDD n'est PAS modifié — on travaille
//   sur la copie chaîne locale sJsonPsmdTemp, écrite dans le fichier
//   temp passé à psmd_cli.js et jamais ré-écrite en BDD.
//
//   Couverture des chemins d'appel : ServeurTraiterMessageDesigner,
//   BTN_TraiterMessageModele, AjoutModificationOperationCourrier (CAS 1
//   ajout base sans réouverture Designer), SelectionModèle, UploadFichier,
//   SupprimerDocument — tous accèdent à __stOperation.taaBaseChamp.
// ═════════════════════════════════════════════════════════════════════

// 1. Parser le JsonDesignerData en JSON pour pouvoir lire champsFusion.
//    UNIQUEMENT pour construire le mapping — les substitutions se font
//    sur la chaîne JSON entière (étape 3 ci-dessous).
jDocLec				est un JSON		= stDocument.JsonDesignerData
jDocChampsLec	est un JSON
si jDocLec.data..Existe _et_ jDocLec.data.champsFusion..Existe alors
	jDocChampsLec = jDocLec.data
sinon
	jDocChampsLec = jDocLec
fin

// 2. Construire le mapping LOCAL_<localId> > <nom technique> par LIBELLÉ.
//    Garde-fou : si __stOperation ou taaBaseChamp non accessibles
//    (cas hors tunnel courrier), mappingPsmd reste vide > aucune
//    substitution > PSMD écrit avec @LOCAL_xxx@ comme avant.
taaMappingPsmd		est un tableau associatif de chaîne

si jDocChampsLec.champsFusion..Occurrence > 0 alors
	pour nIndCF = 1 _à_ jDocChampsLec.champsFusion..Occurrence
		sLocalIdCF	est une chaîne	= jDocChampsLec.champsFusion[nIndCF].localId
		sLibelleCF	est une chaîne	= jDocChampsLec.champsFusion[nIndCF].libelle

		// Critères : besoin d'un localId réel ET d'un libellé qui matche
		// une colonne base. Les standards (Civilite/Nom) sans localId réel
		// ne passent pas le filtre — ils sont déjà en nom technique stable
		// dans le contenu (cf. L18/A43 Correction A).
		si sLocalIdCF <> "" _et_ sLocalIdCF <> "0" _et_ sLibelleCF <> "" alors
			bExisteEnBase est un booléen = Faux
			QUAND EXCEPTION DANS
				bExisteEnBase = __stOperation.taaBaseChamp[sLibelleCF]..Existe
			FAIRE
				bExisteEnBase = Faux
			FIN
			si bExisteEnBase alors
				sNomTechCF est une chaîne = __stOperation.taaBaseChamp[sLibelleCF].Champ
				si sNomTechCF <> "" alors
					taaMappingPsmd["LOCAL_" + sLocalIdCF] = sNomTechCF
				fin
			fin
		fin
	fin
fin

// 3. SUBSTITUTION TEXTUELLE DIRECTE sur la chaîne JSON entière.
//    sJsonPsmdTemp est la copie chaîne destinée au fichier temp.
//    Pour chaque entrée du mapping, 3 motifs de substitution.
sJsonPsmdTemp	est une chaîne	= stDocument.JsonDesignerData
nNbSubst			est un entier	= 0

si taaMappingPsmd..Occurrence > 0 alors
	pour tout sNomTechSubst, sCleMappingSubst de taaMappingPsmd
		// sCleMappingSubst contient déjà "LOCAL_xxx" (cf. construction
		// du mapping ligne `taaMappingPsmd["LOCAL_" + sLocalIdCF] = ...`)

		// a. Marqueurs @KEY@ dans contenu / contenuRtf / qrConfig.fields tokens
		sMarqA	est une chaîne	= "@" + sCleMappingSubst + "@"
		sMarqB	est une chaîne	= "@" + sNomTechSubst + "@"
		nPosA		est un entier	= Position(sJsonPsmdTemp, sMarqA)
		si nPosA > 0 alors
			sJsonPsmdTemp = Remplace(sJsonPsmdTemp, sMarqA, sMarqB)
			nNbSubst++
		fin

		// b. quillDelta ops merge-tag : "key":"LOCAL_xxx"
		// Guillemets WLangage : "" = un guillemet littéral.
		sKeyA	est une chaîne	= """key"":""" + sCleMappingSubst + """"
		sKeyB	est une chaîne	= """key"":""" + sNomTechSubst + """"
		nPosB	est un entier	= Position(sJsonPsmdTemp, sKeyA)
		si nPosB > 0 alors
			sJsonPsmdTemp = Remplace(sJsonPsmdTemp, sKeyA, sKeyB)
			nNbSubst++
		fin

		// c. source.champFusion / champFusion : "champFusion":"LOCAL_xxx"
		sCFA	est une chaîne	= """champFusion"":""" + sCleMappingSubst + """"
		sCFB	est une chaîne	= """champFusion"":""" + sNomTechSubst + """"
		nPosC	est un entier	= Position(sJsonPsmdTemp, sCFA)
		si nPosC > 0 alors
			sJsonPsmdTemp = Remplace(sJsonPsmdTemp, sCFA, sCFB)
			nNbSubst++
		fin
	fin
	Trace("L18/A44 v5 : " + nNbSubst + " substitution(s) effectuée(s) sur la chaîne JSON.")
sinon
	Trace("L18/A44 v5 : aucun mapping LOCAL_xxx > nom (régime A ou pas de spécifique mappable). Substitution PSMD sautée.")
fin

// ─── Substitution nom d'affichage -> nom technique dans le \fonttbl (Lot 2B) ──
//   Couvre PROD (--imposition) ET BAT (--no-bleed) : même fichier temp.
//   Opère sur la copie chaîne sJsonPsmdTemp (le JsonDesignerData en base reste intact).
nIdClientPolice est un entier = 0
QUAND EXCEPTION DANS
	nIdClientPolice = __stOperation.IdClient
FAIRE
	nIdClientPolice = 0
FIN
sJsonPsmdTemp = SubstitueNomTechniquePolices(sJsonPsmdTemp, nIdClientPolice)

// ─── Écriture du JSON dans le fichier temporaire ─────────────────────────
si pas fSauveTexte(sCheminJsonTemp, ChaîneVersUTF8(sJsonPsmdTemp)) alors
	Trace("GenererPsmdServeur : échec fSauveTexte JSON temp : " + sCheminJsonTemp)
	renvoyer Faux
fin

si fFichierExiste(sCheminPsmd) alors
	si pas fSupprime(sCheminPsmd) alors
		renvoyer Faux
	fin
fin

// ─── Construction et exécution de la commande (PSMD production) ──────────
sCommande = ChaîneConstruit(
[
	"%1" "%2" "%3" "%4" --imposition
],
sNodeExe, sNodeCli, sCheminJsonTemp, sCheminPsmd)

si pas LanceAppli(sCommande, exeIconise, exeBloquant) alors
	Trace("GenererPsmdServeur : LanceAppli a échoué — " + ErreurInfo(errComplet))
	renvoyer Faux
fin

// ─── Génération du PSMD BAT (sans fond perdu) ───────────────────────────
// Corrige un bug de rendu des codes-barres dans CreateJPEGPreviewOfLayout
// quand le fond perdu est en mode "Ajouter à la tâche"
sCheminPsmdBat	= sRepTravail + "psmd_bat_" + sCodeTemp + ".psmd"
sCommandeBat	= ChaîneConstruit(
[
	"%1" "%2" "%3" "%4"
],
sNodeExe, sNodeCli, sCheminJsonTemp, sCheminPsmdBat) + " --no-bleed"

LanceAppli(sCommandeBat, exeIconise, exeBloquant)

// ─── Nettoyage du fichier JSON temporaire ────────────────────────────────
si fFichierExiste(sCheminJsonTemp) alors
	fSupprime(sCheminJsonTemp)
fin

// ─── Vérification de la présence du .psmd produit ────────────────────────
si pas fFichierExiste(sCheminPsmd) alors
	Trace("GenererPsmdServeur : .psmd introuvable après génération : " + sCheminPsmd)
	renvoyer Faux
fin

// ─── Mise à jour de la structure document ────────────────────────────────
stDocument.CheminPsmdTemp	= sCheminPsmd
stDocument.NomFichierPsmd	= sNomFichierPsmd

// PSMD BAT : utilisé par GenerationBatDepuisPsmd à la place du PSMD production
si fFichierExiste(sCheminPsmdBat) alors
	stDocument.CheminPsmdBat	= sCheminPsmdBat
sinon
	stDocument.CheminPsmdBat	= sCheminPsmd
	Trace("GenererPsmdServeur : PSMD BAT absent, fallback sur PSMD production")
fin

// ─── Lot 2C : copie des polices utilisées dans Fonts (avant rendu BAT) ───────
//   Garantit les TTF du document dans …\Fonts\ pour sa preview BAT et la prod ;
//   l'union projet se construit au fil des documents (même répertoire). Non bloquant.
CopiePolicesUtiliseesVersFonts(stDocument.JsonDesignerData, stDocument.RepertoireTravail)

renvoyer Vrai
