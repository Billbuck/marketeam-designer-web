selon _stSynthèse.TypeOperation
	cas __OPERATION_COURRIER__, __OPERATION_SMARTEVA__, __OPERATION_TRACTEVA__

procédure interne ChargeTaaDate()
		sTexteRequete	est une chaîne
		sdReqDate		est une Source de Données

		sTexteRequete =
		[
			SELECT 
			z.statut,
			MAX(z.DateHeureStatut) DateHeureStatut
			FROM 
				(
					SELECT DISTINCT s.Statut, s.DateHeureStatut, 'o' AS Origine FROM dos_operation_statut s WHERE s.IdOperation = [%_pIdOperation%]
					UNION
					SELECT o.Statut, o.DateHeureStatut, 'n' AS Origine FROM dos_operation o WHERE o.IdOperation = [%_pIdOperation%]
					UNION
					SELECT '00' AS Statut, o.DateHeureCreation, 'c' AS Origine FROM dos_operation o WHERE o.IdOperation = [%_pIdOperation%]
				) AS z
			WHERE
				z.Statut <= '[%_stSynthèse.Statut%]'
			GROUP BY 
				z.Statut
		]

		si pas sdReqDate.ExécuteRequêteSQL(ltr_support..Connexion, hRequêteSansCorrection, sTexteRequete) alors
			Info("Erreur requête")
			retour
		fin

		sdReqDate.LitPremier(hSansRafraîchir)
		tantque pas sdReqDate.EnDehors()
			taaDate[sdReqDate.Statut] = sdReqDate.DateHeureStatut
			sdReqDate.LitSuivant()
		fin


	fin:
	sdReqDate.AnnuleDéclaration()
fin
