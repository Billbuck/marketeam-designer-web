procédure navSmsDatePossible(pDate = _stCalendrier.DateEnvoiSouhaite)

selon Vrai
	cas pas DateValide(pDate) 						: renvoyer Faux
	cas pDate < DateSys() 							: renvoyer Faux
	cas cpFonction.__taaJoursFeriés[pDate] = Vrai 	: renvoyer Faux
	autres cas 										: renvoyer Vrai
fin
