	selon vReponse.notificationType

			sFichier = ComplèteRep(fRepWeb()) + "Erreur TYPE - Tracking Email - " + DateHeureSys() + ".txt"
			fSauveTexte(sFichier,sParamBuffer)
	fin
sinon
	sFichier = ComplèteRep(fRepWeb()) + "ErreurJSON - Tracking Email - " + DateHeureSys() + ".txt"
	fSauveTexte(sFichier,sParamBuffer)
fin
