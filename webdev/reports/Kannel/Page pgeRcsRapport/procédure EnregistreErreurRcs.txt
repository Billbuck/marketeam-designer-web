// Une seule écriture fichier par erreur (le fichier n'est ouvert qu'une fois).
procédure EnregistreErreurRcs(pTitre, pRequete)

sLigne est une chaîne

sLigne = RC + DateHeureVersChaîne(DateHeureSys(), "JJ/MM/AAAA HH:mm:SS")
sLigne += RC + pTitre

sLigne += RC + ErreurInfo(errComplet)
