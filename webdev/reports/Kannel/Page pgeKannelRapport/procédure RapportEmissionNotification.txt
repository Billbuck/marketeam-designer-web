procédure RapportEmissionNotification(stRapportKannel est une structRapportKannel)

sStatut						est une chaîne
sTexteRequete				est une chaîne
sdReqRapportNotification	est une Source de Données

sStatut = MatchingStatus(stRapportKannel.Statut, stRapportKannel.Detail, stRapportKannel.Routeur)

sTexteRequete =
[
	UPDATE LOW_PRIORITY
		ope_sms_notification
	SET
		ope_sms_notification.Statut = '[%sStatut%]',
		ope_sms_notification.StatutKannel = '[%stRapportKannel.Statut%]',
		ope_sms_notification.StatutOperateur = '[%stRapportKannel.Detail%]',
		ope_sms_notification.Operateur = '[%stRapportKannel.Operateur%]',
		ope_sms_notification.PrixUnitaireFacture = [%stRapportKannel.Price%],
		ope_sms_notification.DateHeureReception = '[%stRapportKannel.DateHeure%]'
	WHERE
		ope_sms_notification.IdSmsNotification = [%stRapportKannel.stNotification.IdSmsNotification%];
	]


si pas sdReqRapportNotification.ExécuteRequêteSQL(cpKannel.__cnxKannel, hRequêteSansCorrection, sTexteRequete) alors
	EnregistreErreurRequete("RapportEmissionNotification UPDATE", sTexteRequete)
	retour
fin

si stRapportKannel.Detail dans (__IB_STATUT_BLACKLISTED_DESTINATIONADDRESS__, __CM_STATUT_BLACKLISTED_PERMANENTLY__) alors
	sTexteRequete =
	[
		INSERT IGNORE dos_base_ligne_erreur (TypeErreur, Portable, DateHeureErreur)
		VALUES('[%__ERREUR_STOP_PUB_OPERATEUR__%]', '[%stRapportKannel.Portable%]', NOW()) 
	]
fin

si pas sdReqRapportNotification.ExécuteRequêteSQL(cpKannel.__cnxKannel, hRequêteSansCorrection, sTexteRequete) alors
	EnregistreErreurRequete("RapportEmissionNotification INSERT", sTexteRequete)
	retour
fin



fin:
sdReqRapportNotification.AnnuleDéclaration()
