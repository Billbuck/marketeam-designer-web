// Résumé : Vide puis remplit ope_rcs_envoi à l'enregistrement d'une opération RCS,
// selon la qualification des lignes (dos_base_ligne.EstRcs) :
//	EstRcs = 1 (RCS)      → une ligne CanalPrevu = RCS par portable distinct ;
//	EstRcs = 0 (SMS seul) → une ligne CanalPrevu = SMS par portable distinct,
//	                        SEULEMENT si le repli SMS est activé (envoi SMS
//	                        direct par batchRcs au lot 4) ;
//	EstRcs = 2 (invalide) ou NULL (non qualifié) → jamais de ligne.
// Calquée sur EnregistrementProgrammationSms, sans calcul de taille ni de
// nombre de SMS. L'envoi est fait par batchRcs (statut enregistrée).
procédure EnregistrementProgrammationRcs(stOperation)


sTexteRequete			est une chaîne
sdReqRcs				est une Source de Données
sListeIdBase			est une chaîne

pour tout stBase de stOperation.tabBase
	sListeIdBase += [","] + stBase.IdBase
fin

sTexteRequete = "DELETE FROM ope_rcs_envoi WHERE IdRcs = " + stOperation.IdOperation

si pas sdReqRcs.ExécuteRequêteSQL(ope_rcs_envoi..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	<compile si TypeConfiguration <> Webservice>
		Info("Erreur requête EnregistrementProgrammationRcs 1 : " + HErreurInfo(hErrComplet))
	<fin>
	retour
fin

// Destinataires RCS : un seul enregistrement par Portable, lignes qualifiées RCS
sTexteRequete	=
[
INSERT INTO ope_rcs_envoi (IdRcs, IdBaseLigne, Statut, CanalPrevu)
SELECT [%stOperation.IdOperation%] 
, z.IdBaseLigne
, '[%__SMS_STATUT_ENREGISTREE__%]' Statut
, '[%__CANAL_PREVU_RCS__%]' CanalPrevu
FROM
(SELECT * FROM dos_base_ligne WHERE IdBase IN ([%sListeIdBase%]) AND Portable IS NOT NULL AND EstRcs = [%__LIGNE_RCS_OUI__%] GROUP BY Portable) AS z
]

si pas sdReqRcs.ExécuteRequêteSQL(ope_rcs_envoi..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	<compile si TypeConfiguration <> Webservice>
		Info("Erreur requête EnregistrementProgrammationRcs 2 : " + HErreurInfo(hErrComplet))
	<fin>
	retour
fin

// Destinataires SMS seul : contactés uniquement si le repli SMS est activé
si stOperation.stOperationRcs.EstRepliSms alors
	sTexteRequete	=
	[
	INSERT INTO ope_rcs_envoi (IdRcs, IdBaseLigne, Statut, CanalPrevu)
	SELECT [%stOperation.IdOperation%] 
	, z.IdBaseLigne
	, '[%__SMS_STATUT_ENREGISTREE__%]' Statut
	, '[%__CANAL_PREVU_SMS__%]' CanalPrevu
	FROM
	(SELECT * FROM dos_base_ligne WHERE IdBase IN ([%sListeIdBase%]) AND Portable IS NOT NULL AND EstRcs = [%__LIGNE_RCS_SMS__%] GROUP BY Portable) AS z
	]

	si pas sdReqRcs.ExécuteRequêteSQL(ope_rcs_envoi..Connexion, hRequêteSansCorrection, sTexteRequete) alors
		<compile si TypeConfiguration <> Webservice>
			Info("Erreur requête EnregistrementProgrammationRcs 3 : " + HErreurInfo(hErrComplet))
		<fin>
		retour
	fin
fin


fin:
sdReqRcs.AnnuleDéclaration()
