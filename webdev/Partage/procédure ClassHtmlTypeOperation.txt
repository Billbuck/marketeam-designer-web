procédure ClassHtmlTypeOperation(pTypeChamp, pTypeOperation, pOption = "")


sClassHtml est une chaîne
selon pTypeChamp
	cas "lnk"	: sClassHtml = "lnk-"
	cas "btn"	: sClassHtml = "bouton-"
	cas "cbo"	: sClassHtml = "combo-"
	cas "sai"	: sClassHtml = "saisie-"
	cas "lib"	: sClassHtml = "libelle-"
	cas "znr"	: sClassHtml = "znr-"
	cas "int"	: sClassHtml = "interrupteur-"
	cas "sel"	: sClassHtml = "selecteur-"
	cas "jge"	: sClassHtml = "jauge-"
	cas "cel"	: sClassHtml = "cellule-"
	cas "lgo"	: sClassHtml = "logo-"
fin

selon pTypeOperation
	cas __OPERATION_COURRIER__								: sClassHtml += "ltr"
	cas __OPERATION_COURRIER_INTERACTIF__				: sClassHtml += "ltr"
	cas __OPERATION_TRACTEVA__								: sClassHtml += "ttb"
	cas __OPERATION_SMARTEVA__								: sClassHtml += "ttb"
	cas __OPERATION_COURRIER_GESTION__					: sClassHtml += "ltr"
	cas __OPERATION_SMS__									: sClassHtml += "sms"
	cas __OPERATION_SMS_INTERACTIF__						: sClassHtml += "sms"
	cas __OPERATION_RCS__									: sClassHtml += "sms"
	cas __OPERATION_RCS_INTERACTIF__						: sClassHtml += "sms"
	cas __OPERATION_VMS__									: sClassHtml += "vms"
	cas __OPERATION_SMS_NOTIFICATION__					: sClassHtml += "sms"
	cas __OPERATION_EMAIL__									: sClassHtml += "eml"
	cas __OPERATION_WEB__									: sClassHtml += "mkt"
	cas __OPERATION_TEL__									: sClassHtml += "tel"
	cas __OPERATION_BASE_DE_DONNEES__					: sClassHtml += "bdd"
	cas __OPERATION_RNVP__									: sClassHtml += "ltr"
	cas __OPERATION_BLOCTEL__								: sClassHtml += "tel"
	cas __OPERATION_MARKETEAM__							: sClassHtml += "mkt"
	autres cas
fin

si pOption <> "" alors
	sClassHtml += "-" + pOption
fin

renvoyer sClassHtml
