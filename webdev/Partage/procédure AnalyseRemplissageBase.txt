// =============================================================================
// Partage.AnalyseRemplissageBase
// Rôle : Moteur du DIAGNOSTIC DE REMPLISSAGE (chantier 2, Lot 1).
//        Pour TOUS les champs réellement insérés dans l'ENSEMBLE des documents
//        de l'opération, calcule le taux de remplissage + min/max par type, sur
//        TOUTE la base (clt et/ou dos selon l'origine), en UNE seule requête.
//
//        N'AFFICHE rien, NE BLOQUE rien : produit une structRemplissageBase.
//
// Spec : docs/cahier-des-charges-diagnostic-remplissage-base.md (§2, §3, §6).
//
// Réutilise :
//   - cpDesigner.CollecteChampsInseresDocument (clés insérées par document) ;
//   - le WHERE de Partage.ChargeEchantillonTabBaseLigne (clt/dos selon EstAjoute,
//     filtres GenereRequeteFiltreClientBase) — MAIS SANS LIMIT.
//
// Paramètre : stOperation (structOperation)
// Retour    : structRemplissageBase
// =============================================================================
procédure AnalyseRemplissageBase(stOperation est une structOperation) : structRemplissageBase

stRemplissage			est une structRemplissageBase
taaColonnesBase		est un tableau associatif de structBaseChamp	= stOperation.taaBaseChamp

// Locals de résolution (partagés avec les procédures internes)
taaChampVu				est un tableau associatif de structRemplissageChamp	// clé = .Champ (dédup colonne)
tabChampsUtilises		est un tableau de structRemplissageChamp
taaLocalLibelle		est un tableau associatif de chaînes						// "LOCAL_xxx" -> libellé (par document)

// Locals SQL
sWhereBase				est une chaîne
sListeIdBase			est une chaîne
sFiltre					est une chaîne
sConditionType			est une chaîne
sColsClt					est une chaîne
sColsDos					est une chaîne
sAgregats				est une chaîne
sInner					est une chaîne
sRequete					est une chaîne
nIdx						est un entier
bAvecBase				est un booléen											= Faux
sdReqDiag				est une Source de Données


// ─── a) Une base est-elle présente ? ─────────────────────────────────────
pour tout stBase de stOperation.tabBase
	si stBase.EstSupprime = Faux alors
		bAvecBase = Vrai
		sortir
	fin
fin
stRemplissage.BaseChargee = bAvecBase
si pas bAvecBase alors
	renvoyer stRemplissage		// structure vide
fin


// ─── b) Liste UNIQUE des champs utilisés (tous documents) ────────────────
pour tout stDocument de stOperation.stOperationLettre.tabStDocument
	si stDocument.JsonDesignerData <> "" alors
		ChargeChampsFusionLocal(stDocument.JsonDesignerData)	// remplit taaLocalLibelle (par doc)
		tabClesDoc est un tableau de chaînes = cpDesigner.CollecteChampsInseresDocument(stDocument.JsonDesignerData)
		pour tout sCleDoc de tabClesDoc
			ResoutCle(sCleDoc)	// résout + dédup -> tabChampsUtilises
		fin

		// Pavé adresse DESTINATAIRE (zones sys-adresse-*, champ "contenu") :
		//   non capté par CollecteChampsInseresDocument (zones sys- exclues +
		//   parsing par embeds). Mêmes noms standard -> résolus par ResoutCle ;
		//   la dédup gère le recouvrement avec les champs déjà insérés.
		tabClesPave est un tableau de chaînes = cpDesigner.CollecteChampsPaveAdresse(stDocument.JsonDesignerData)
		pour tout sClePave de tabClesPave
			ResoutCle(sClePave)
		fin
	fin
fin

// Pavé adresse DESTINATAIRE de l'ENVELOPPE (cas enveloppe sans porte-adresse
// document) : mêmes champs standard ; la dédup gère le recouvrement.
si stOperation.stOperationLettre.stEnveloppe.JsonDesignerData <> "" alors
	tabClesPaveEnv est un tableau de chaînes = cpDesigner.CollecteChampsPaveAdresse(stOperation.stOperationLettre.stEnveloppe.JsonDesignerData)
	pour tout sClePaveEnv de tabClesPaveEnv
		ResoutCle(sClePaveEnv)
	fin
fin


// ─── c) Construction de la requête de diagnostic (SANS LIMIT) ────────────

// Condition par type d'opération (miroir de ChargeEchantillonTabBaseLigne)
selon stOperation.TypeOperation
	cas __OPERATION_SMS__			: sConditionType = "Portable IS NOT NULL"
	cas __OPERATION_RCS__ 			: sConditionType = "Portable IS NOT NULL"
	cas __OPERATION_VMS__ 			: sConditionType = "Portable IS NOT NULL"
	cas __OPERATION_EMAIL__ 		: sConditionType = "Email IS NOT NULL"
	cas __OPERATION_TEL__ 			: sConditionType = "Tel IS NOT NULL OR Portable IS NOT NULL"
	cas __OPERATION_COURRIER__		: sConditionType = "(Adresse1 IS NOT NULL OR Adresse2 IS NOT NULL OR Adresse3 IS NOT NULL OR Adresse4 IS NOT NULL) AND (CodePostal IS NOT NULL AND Ville IS NOT NULL)"
	autre cas
		sConditionType = "1=1"
fin

// WHERE clt / liste dos selon EstAjoute (miroir de ChargeEchantillonTabBaseLigne)
pour tout stBase de stOperation.tabBase
	si stBase.EstSupprime = Faux alors
		si stBase.EstAjoute = Vrai alors
			si stBase.IdFiltre = 0 alors
				sWhereBase += [" OR "] + "(clt_base_ligne.IdBase = " + stBase.IdBaseClient + ")"
			sinon
				sFiltre = GenereRequeteFiltreClientBase(stBase.IdBaseClient, stBase.IdFiltre)
				sWhereBase += [" OR "] + "(" + sFiltre + ")"
			fin
		sinon
			sListeIdBase += [","] + stBase.IdBase
		fin
	fin
fin

// Colonnes sélectionnées (par table) + agrégats (par colonne utilisée)
nIdx = 0
pour tout stCh de tabChampsUtilises
	nIdx++
	sColBt est une chaîne = "`" + stCh.Champ + "`"
	sColsClt += [", "] + "clt_base_ligne." + sColBt + " AS " + sColBt
	sColsDos += [", "] + "dos_base_ligne." + sColBt + " AS " + sColBt
	sAgregats += ", " + ConstruitAgregats(sColBt, stCh.Type, nIdx)
fin

// Aucun champ utilisé : on compte quand même les enregistrements
si tabChampsUtilises..Occurrence = 0 alors
	sColsClt	= "clt_base_ligne.IdBase AS k"
	sColsDos	= "dos_base_ligne.IdBase AS k"
fin

// Sous-requête interne : clt, dos, ou UNION ALL des deux (sans LIMIT)
si sWhereBase <> "" alors
	sInner = "SELECT " + sColsClt + " FROM clt_base_ligne JOIN clt_base ON clt_base.IdBase = clt_base_ligne.IdBase WHERE (" + sWhereBase + ") AND (" + sConditionType + ")"
fin
si sListeIdBase <> "" alors
	sDosPart est une chaîne = "SELECT " + sColsDos + " FROM dos_base_ligne JOIN dos_base ON dos_base.IdBase = dos_base_ligne.IdBase WHERE dos_base.IdBase IN (" + sListeIdBase + ") AND (" + sConditionType + ")"
	si sInner <> "" alors
		sInner += " UNION ALL " + sDosPart
	sinon
		sInner = sDosPart
	fin
fin

si sInner = "" alors
	renvoyer stRemplissage		// aucune base exploitable (sécurité)
fin

sRequete = "SELECT COUNT(*) AS nbtotal" + sAgregats + " FROM (" + sInner + ") t"


// ─── d) Exécution + remplissage de la structure ──────────────────────────
si pas sdReqDiag.ExécuteRequêteSQL(ltr_support..Connexion, hRequêteSansCorrection, sRequete) alors
	renvoyer stRemplissage		// BaseChargee = Vrai mais pas de données
fin

sdReqDiag.LitPremier(hSansRafraîchir)
si pas sdReqDiag.EnDehors() alors
	stRemplissage.NbEnregistrements	= sdReqDiag.nbtotal

	nIdx										= 0
	pour tout stCh de tabChampsUtilises
		nIdx++
		stCh.NbTotal		= stRemplissage.NbEnregistrements
		// Lecture des alias dynamiques par indirection (idiome RemplirDesignerApercu:64)
		stCh.NbRenseignes	= {"sdReqDiag.rempli" + nIdx, indVariable}
		si stCh.NbTotal > 0 alors
			stCh.Taux = (stCh.NbRenseignes * 100.0) / stCh.NbTotal
		fin

		stCh.MinLongueur	= {"sdReqDiag.minlen" + nIdx, indVariable}
		stCh.MaxLongueur	= {"sdReqDiag.maxlen" + nIdx, indVariable}

		sMinV	est une chaîne	= {"sdReqDiag.minval" + nIdx, indVariable}
		sMaxV	est une chaîne	= {"sdReqDiag.maxval" + nIdx, indVariable}

		// Numériques : le CAST SQL renvoie un point décimal -> on remet la
		// virgule (cohérent avec le format base et l'entrée de FormateNombre).
		// Coordonnée : on garde le point. Texte/date : inchangé.
		selon stCh.Type
			cas __CHAMP_TYPE_ENTIER__, __CHAMP_TYPE_DECIMAL__, __CHAMP_TYPE_MONETAIRE__
				sMinV	= Remplace(sMinV, ".", ",")
				sMaxV	= Remplace(sMaxV, ".", ",")
			autre cas
		fin

		stCh.MinValeur	= sMinV
		stCh.MaxValeur	= sMaxV
		stRemplissage.tabChamps.Ajoute(stCh)
	fin
fin

renvoyer stRemplissage


// ═════════════════════════════════════════════════════════════════════════
// PROCÉDURES INTERNES
// ═════════════════════════════════════════════════════════════════════════

// ─── champsFusion d'un document : localId -> libellé (réplique minimale
	//     de ChargeChampsFusion de VerifieConcordanceBaseDesigner) ──────────
	procédure interne ChargeChampsFusionLocal(local pJson est une chaîne)
		jDoc			est un JSON
		jData			est un JSON
		bJsonOk		est un booléen	= Vrai
		sLocalId		est une chaîne
		sLibelle		est une chaîne

		taaLocalLibelle.SupprimeTout()

		si pJson <> "" alors
			quand exception dans
				jDoc = pJson
			faire
				bJsonOk = Faux
			fin

			si bJsonOk alors
				si jDoc.data..Existe alors
					jData = jDoc.data
				sinon
					jData = jDoc
				fin

				si jData.champsFusion..Existe alors
					pour nC = 1 _à_ jData.champsFusion..Occurrence
						sLocalId	= ""
						sLibelle	= ""
						quand exception dans
							si jData.champsFusion[nC].localId..Existe alors sLocalId = jData.champsFusion[nC].localId
							si jData.champsFusion[nC].libelle..Existe alors sLibelle = jData.champsFusion[nC].libelle
						faire
						fin
						si sLocalId <> "" _et_ sLocalId <> "0" alors
							taaLocalLibelle["LOCAL_" + sLocalId] = sLibelle
						fin
					fin
				fin
			fin
		fin
	fin

	// ─── Champ SYSTÈME ? (exclu du diagnostic) ───────────────────────────────
	procédure interne EstChampSystemeLocal(local pCle est une chaîne) : booléen
		bSys est un booléen = Faux
		si pCle = "Sequentiel" _ou_ pCle = "Timbre" _ou_ pCle = "Rapprochement" alors
			bSys = Vrai
		fin
		quand exception dans
			si pCle = __CHAMP_CLIQUEZ_ICI__ alors bSys = Vrai
		faire
		fin
		renvoyer bSys
	fin

// ─── Résout une clé insérée -> colonne base (Champ/Libelle/Type), dédup ──
//   LOCAL_xxx : libellé via taaLocalLibelle -> colonne via taaColonnesBase
//   nom technique : recherche par .Champ dans taaColonnesBase
	//   Non résolvable à une colonne base -> ignoré (relève de la concordance).
	procédure interne ResoutCle(local pCle est une chaîne)
		sChamp		est une chaîne	= ""
		sLibelle		est une chaîne	= ""
		sType			est une chaîne	= ""

		si EstChampSystemeLocal(pCle) alors retour

		si Gauche(pCle, 6) = "LOCAL_" alors
			// ─── SPÉCIFIQUE ───
			si taaLocalLibelle[pCle]..Existe = Faux alors retour
			sLibelle = taaLocalLibelle[pCle]
			si sLibelle = "" alors retour
			si taaColonnesBase[sLibelle]..Existe = Faux alors retour
			si taaColonnesBase[sLibelle].EstSupprimé alors retour
			sChamp	= taaColonnesBase[sLibelle].Champ
			sType		= taaColonnesBase[sLibelle].Type
		sinon
			// ─── STANDARD / nom technique : recherche par colonne ───
			bTrouve est un booléen = Faux
			pour tout stCol, sLib de taaColonnesBase
				si stCol.EstSupprimé = Faux _et_ stCol.Champ = pCle alors
					sChamp		= stCol.Champ
					sLibelle		= sLib
					sType			= stCol.Type
					bTrouve		= Vrai
					sortir
				fin
			fin
			si pas bTrouve alors retour
		fin

		si sChamp = "" alors retour
		si taaChampVu[sChamp]..Existe alors retour		// dédup par colonne

		stCh est une structRemplissageChamp
		stCh.Libelle			= sLibelle
		stCh.Champ				= sChamp
		stCh.Type				= sType
		taaChampVu[sChamp]	= stCh
		tabChampsUtilises.Ajoute(stCh)
	fin

// ─── Fragment SQL des agrégats d'une colonne ─────────────────────────────
//   Émet TOUJOURS 5 alias : rempli<i>, minlen<i>, maxlen<i>, minval<i>,
//   maxval<i>. Tout en UNE passe (aucune sous-requête / scan additionnel).
//
//   - NUM/DAT/COO : minval/maxval = valeur min/max (CAST AS CHAR) ;
//     minlen/maxlen = 0 (non pertinent).
//   - TEXTE : minlen/maxlen = longueur mini/maxi ; minval/maxval = la VALEUR
//     à cette longueur, via le trick argmin/argmax
//     SUBSTRING(MIN/MAX(CONCAT(LPAD(longueur,10,'0'), valeur)), 11) :
//     le préfixe longueur zero-paddé fait trier d'abord par longueur, sans
	//     GROUP_CONCAT (donc sans limite de taille).
	procédure interne ConstruitAgregats(local pColBt est une chaîne, local pType est une chaîne, local pIdx est un entier) : chaîne
		sCondVide	est une chaîne	= pColBt + " IS NULL OR " + pColBt + " = ''"
		sRempli		est une chaîne	= "SUM(CASE WHEN " + sCondVide + " THEN 0 ELSE 1 END) AS rempli" + pIdx
		sMinLen		est une chaîne
		sMaxLen		est une chaîne
		sMinVal		est une chaîne
		sMaxVal		est une chaîne
		sValeur		est une chaîne

		selon pType
			cas __CHAMP_TYPE_ENTIER__, __CHAMP_TYPE_DECIMAL__, __CHAMP_TYPE_MONETAIRE__
				// virgule décimale en base -> tri texte FAUX -> CAST obligatoire
				sValeur	= "CAST(REPLACE(" + pColBt + ",',','.') AS DECIMAL(20,4))"
				sMinLen	= "0 AS minlen" + pIdx
				sMaxLen	= "0 AS maxlen" + pIdx
				sMinVal	= "CAST(MIN(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sValeur + " END) AS CHAR) AS minval" + pIdx
				sMaxVal	= "CAST(MAX(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sValeur + " END) AS CHAR) AS maxval" + pIdx

			cas __CHAMP_TYPE_COORDONNEE__
				// point décimal déjà présent (CleanCoordonnee)
				sValeur	= "CAST(" + pColBt + " AS DECIMAL(18,8))"
				sMinLen	= "0 AS minlen" + pIdx
				sMaxLen	= "0 AS maxlen" + pIdx
				sMinVal	= "CAST(MIN(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sValeur + " END) AS CHAR) AS minval" + pIdx
				sMaxVal	= "CAST(MAX(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sValeur + " END) AS CHAR) AS maxval" + pIdx

			cas __CHAMP_TYPE_DATE__
				// AAAAMMJJ : tri texte déjà correct
				sValeur	= pColBt
				sMinLen	= "0 AS minlen" + pIdx
				sMaxLen	= "0 AS maxlen" + pIdx
				sMinVal	= "CAST(MIN(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sValeur + " END) AS CHAR) AS minval" + pIdx
				sMaxVal	= "CAST(MAX(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sValeur + " END) AS CHAR) AS maxval" + pIdx

			autre cas
				// TEXTE (TXT/EML/URL/CDP/ALG/TEL/SMS/TIM/IMG/inconnu)
				sLenExpr	est une chaîne	= "CHAR_LENGTH(" + pColBt + ")"
				sArg		est une chaîne	= "CONCAT(LPAD(" + sLenExpr + ",10,'0'), " + pColBt + ")"
				sMinLen	= "MIN(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sLenExpr + " END) AS minlen" + pIdx
				sMaxLen	= "MAX(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sLenExpr + " END) AS maxlen" + pIdx
				sMinVal	= "SUBSTRING(MIN(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sArg + " END), 11) AS minval" + pIdx
				sMaxVal	= "SUBSTRING(MAX(CASE WHEN " + sCondVide + " THEN NULL ELSE " + sArg + " END), 11) AS maxval" + pIdx
		fin

		renvoyer sRempli + ", " + sMinLen + ", " + sMaxLen + ", " + sMinVal + ", " + sMaxVal
	fin
