//	pIdBase : <indiquez ici le rôle de pIdBase>
//	pIdFiltre : <indiquez ici le rôle de pIdFiltre>

// 	Aucune
//
// Exemple :
// <Indiquez ici un exemple d'utilisation>

procédure CalculeTotauxClientBaseFiltre(pIdBase, pIdFiltre)



sTexteRequete		est une chaîne
sTexteRequeteTotal	est une chaîne
sdReqBddTotal		est une Source de Données
sFiltre				est une chaîne
nTotal				est un entier
nTotalSociete		est un entier
nTotalContact		est un entier
nTotalAdresse		est un entier
nTotalSms			est un entier
nTotalTel			est un entier
nTotalFixe			est un entier
nTotalEmail			est un entier


sTexteRequeteTotal =
[
	SELECT COUNT(*) Total FROM clt_base_ligne WHERE IdBase = %1 %2
	%3
]

sFiltre = " AND (" + GenereRequeteFiltreClientBase(pIdBase,pIdFiltre) + ")"

nTotal			= CalculSurFiltreBddClient(pIdBase,"")
nTotalSociete	= CalculSurFiltreBddClient(pIdBase,"AND Societe IS NOT NULL")
nTotalContact	= CalculSurFiltreBddClient(pIdBase,"AND Contact IS NOT NULL")
nTotalAdresse	= CalculSurFiltreBddClient(pIdBase,"AND (Adresse1 IS NOT NULL OR Adresse2 IS NOT NULL) AND CodePostal IS NOT NULL AND Ville IS NOT NULL")
nTotalSms		= CalculSurFiltreBddClient(pIdBase,"AND Portable IS NOT NULL")
nTotalTel		= CalculSurFiltreBddClient(pIdBase,"AND (Tel IS NOT NULL OR Portable IS NOT NULL) ")
nTotalFixe		= CalculSurFiltreBddClient(pIdBase,"AND Tel IS NOT NULL")
nTotalEmail		= CalculSurFiltreBddClient(pIdBase,"AND Email IS NOT NULL")

si clt_base_filtre.LitRecherchePremier(IdFiltre,pIdFiltre) alors
	clt_base_filtre.NbrTotal 		= nTotal
	clt_base_filtre.NbrSociete		= nTotalSociete
	clt_base_filtre.NbrContact		= nTotalContact
	clt_base_filtre.NbrAdresse		= nTotalAdresse
	clt_base_filtre.NbrSms			= nTotalSms
	clt_base_filtre.NbrTel			= nTotalTel
	clt_base_filtre.NbrFixe			= nTotalFixe
	clt_base_filtre.NbrEmail		= nTotalEmail
	clt_base_filtre.Modifie()
fin

procédure interne CalculSurFiltreBddClient(piBddClient,pColonne)

	sTexteRequete = ChaîneConstruit(sTexteRequeteTotal,piBddClient, pColonne, sFiltre)

	si pas sdReqBddTotal.ExécuteRequêteSQL(clt_base_ligne..Connexion, hRequêteSansCorrection, sTexteRequete) alors
			<compile si TypeConfiguration <> Webservice>
				Info("Erreur requête CalculSurFiltreBddClient")
			<fin>
		renvoyer ""
	fin


	si sdReqBddTotal.LitPremier(hSansRafraîchir) _et_ Val(sdReqBddTotal.Total) > 0 alors
		renvoyer sdReqBddTotal.Total
	sinon
		renvoyer 0
	fin

fin

fin:
sdReqBddTotal.AnnuleDéclaration()
