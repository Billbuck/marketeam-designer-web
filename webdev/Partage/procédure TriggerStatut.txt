procédure TriggerStatut()

bEstPaiementReceptionFacture	est un booléen
bEstPaiementSurReserve 			est un booléen
bEstPlafondEncoursDepasse		est un booléen
bEstRéférenceObligatoire		est un booléen
sNouveauStatut					est une chaîne
dhLimitePriseEnCharge			est une DateHeure	= ""
sTexteRequete 					est une chaîne
sdReqTriggerStatut 				est une Source de Données
sStatutPrecedent 				est une chaîne

sTexteRequete = [
	SELECT Statut FROM dos_operation where IdOperation = [%dos_operation.IdOperation%]
]

si H.FonctionTrigger = "HModifie" alors
	sdReqTriggerStatut.ExécuteRequêteSQL(dos_operation..Connexion,hRequêteSansCorrection,sTexteRequete)
	sdReqTriggerStatut.LitPremier()
	sStatutPrecedent = sdReqTriggerStatut.Statut
fin

__bChangementStatut = (dos_operation.Statut <> sStatutPrecedent)

si dos_operation.Statut dans (__Statut_Annulée__,__Statut_Terminée__) alors
	retour
fin

si dos_operation.EstTerminee = Faux _et_ dos_operation.Statut dans (__Statut_Prise_En_Charge__,__Statut_Production_En_Cours__,__Statut_Distribution_En_Attente__,__Statut_Distribution_En_Cours__,__Statut_Envoyée__) alors
	retour
fin


si DateHeureValide(dos_operation.DateHeureRefus) alors
	sNouveauStatut = __Statut_Validation_Refusée__
	retour
fin

si dos_operation.EstBaseEnAttente = Vrai _ou_ dos_operation.EstDocumentEnAttente = Vrai _ou_ dos_operation.EstInjectionEnAttente = Vrai _ou_ dos_operation.EstLandingPageEnAttente = Vrai _ou_ dos_operation.EstMessageEnAttente = Vrai alors
	sNouveauStatut = __Statut_Element_En_Attente__
	si dos_operation.DateHeureLimiteValidation < DateHeureSys() alors
		sNouveauStatut = __Statut_Element_Hors_Delais__
	fin
	retour
fin

// En brouillon si l'opération est complète
si dos_operation.EstBrouillon = Vrai alors
	sNouveauStatut = __Statut_Enregistrée__
	retour
fin

// Statut Audit Remplissage
si dos_operation.TypeOperation = __OPERATION_COURRIER__ alors
	si pas DateHeureValide(dos_operation.DateHeureCheckRemplissage) alors
		si Partage.EstEnveloppeAFenetre(dos_operation.IdOperation) alors
			sNouveauStatut = __Statut_Audit_Remplissage_En_Attente__
			si dos_operation.DateHeureLimiteValidation < DateHeureSys() alors
				sNouveauStatut = __Statut_Audit_Remplissage_Hors_Delais__
			fin
			retour
		fin
	fin
fin



bEstPaiementSurReserve			= clt_paiement.LitRecherchePremier(IdOperation, dos_operation.IdOperation) _et_ clt_paiement.EstReserve = Vrai
bEstPaiementReceptionFacture 	= bEstPaiementSurReserve = Faux _et_ EstClientEnCompte(dos_operation.IdClientFacturation)
bEstPlafondEncoursDepasse 		= (bEstPaiementSurReserve = Faux _et_ bEstPaiementReceptionFacture = Vrai _et_ PlafondEncoursDepassé(dos_operation.IdClientFacturation, dos_operation.ResteAPayer, dos_operation.IdOperation))
bEstRéférenceObligatoire 		= (bEstPaiementSurReserve = Faux _et_ bEstPaiementReceptionFacture = Vrai _et_ RéférenceObligatoire(dos_operation.IdAdresseFacturation))

si bEstRéférenceObligatoire = Vrai _et_ dos_operation.NumCommandeClient = "" alors
	sNouveauStatut = __Statut_Référence_En_Attente__
	si dos_operation.DateHeureLimiteValidation < DateHeureSys() alors
		sNouveauStatut = __Statut_Référence_Hors_Delais__
	fin
	retour
fin

si dos_operation.Statut < __Statut_Prise_En_Charge_En_Attente__ _et_ dos_operation.ResteAPayer > 0 _et_ (bEstPaiementReceptionFacture = Faux _ou_ bEstPlafondEncoursDepasse = Vrai) alors
	// On attend le règlement
	selon StatutRéglementOpération(dos_operation.IdOperation)
		cas __REGLEMENT_STATUT_INEXISTANT__		: sNouveauStatut = __Statut_Règlement_En_Attente__ // Pas de règlement
		cas __REGLEMENT_STATUT_EN_ATTENTE__		: sNouveauStatut = __Statut_Règlement_En_Attente__
		cas __REGLEMENT_STATUT_EN_COURS__		: sNouveauStatut = __Statut_Règlement_En_Cours__
		cas __REGLEMENT_STATUT_ANNULE__			: sNouveauStatut = __Statut_Règlement_Annulé__
		cas __REGLEMENT_STATUT_REFUSE__			: sNouveauStatut = __Statut_Règlement_Rejeté__

		cas __REGLEMENT_STATUT_VALIDE__			: sNouveauStatut = __Statut_Règlement_Erreur__ 		; Erreur("Le statut de règlement est validé alors que Reste à payer > 0") ; STOP // Impossible
		cas __REGLEMENT_STATUT_ERREUR__ 			: sNouveauStatut = __Statut_Règlement_Erreur__ 		; Erreur("Erreur lecture Statut Règlement  Opération") ; STOP //  Erreur
		autre cas								: sNouveauStatut = __Statut_Règlement_Erreur__ 		; Erreur("Statut de règlement non défini :'" + StatutRéglementOpération(dos_operation.IdOperation) +"'"); STOP // A définir
	fin

	dos_operation.DateHeureValidation.Null	= Vrai // Si le client a validé en ligne mais qu'il doit payer alors on validera l’opération après le règlement
	si dos_operation.DateHeureLimiteValidation < DateHeureSys() alors
		sNouveauStatut = __Statut_Règlement_Hors_Delais__
	fin
	retour
fin


// Statut de la commande
si pas DateHeureValide(dos_operation.DateHeureValidation) alors
	sNouveauStatut = __Statut_Validation_En_Attente__
	si dos_operation.DateHeureLimiteValidation < DateHeureSys() alors
		sNouveauStatut = __Statut_Validation_Hors_Delais__
	fin
	retour
fin



// Statut de la prise en charge
si pas DateHeureValide(dos_operation.DateHeurePriseEnCharge) alors

	si DateHeureSys() < dos_operation.DateHeureLimitePriseEnCharge alors
		sNouveauStatut = __Statut_Prise_En_Charge_En_Attente__
	sinon
		dos_operation.DateHeureValidation..Null = Vrai
		sNouveauStatut = __Statut_Validation_Hors_Delais__
	fin

fin


// Statut de la facturation
si dos_operation.EstTerminee = Vrai alors

	si bEstPaiementSurReserve = Vrai alors
		// Si l'opération a été payé sur la réserve c'est terminé
		sNouveauStatut = __Statut_Terminée__
		retour
	sinon
		sNouveauStatut = __Statut_Facturation_En_Attente__
		// Si la facture n'a pas été créé on s’arrête là
		si dos_operation.IdFactureLigne = 0 alors
			retour
		fin
	fin
sinon
	retour
fin


si dos_operation.ResteAPayer > 0 alors

	selon StatutRéglementOpération(dos_operation.IdOperation)
		cas __REGLEMENT_STATUT_INEXISTANT__	: sNouveauStatut = __Statut_Solde_En_Attente__ // Pas de règlement
		cas __REGLEMENT_STATUT_EN_ATTENTE__	: sNouveauStatut = __Statut_Solde_En_Attente__
		cas __REGLEMENT_STATUT_EN_COURS__	: sNouveauStatut = __Statut_Solde_En_Cours__
		cas __REGLEMENT_STATUT_ANNULE__		: sNouveauStatut = __Statut_Solde_Annulé__
		cas __REGLEMENT_STATUT_REFUSE__		: sNouveauStatut = __Statut_Solde_Rejeté__
		cas __REGLEMENT_STATUT_VALIDE__		: sNouveauStatut = __Statut_Solde_Erreur__ ; Erreur("Le statut de règlement est validé alors que Reste à payer > 0") ; STOP // Impossible
		cas __REGLEMENT_STATUT_ERREUR__ 		: sNouveauStatut = __Statut_Solde_Erreur__ ; Erreur("Erreur lecture Statut Règlement  Opération") ; STOP //  Erreur
		autre cas							: sNouveauStatut = __Statut_Solde_Erreur__ ; Erreur("Statut de règlement non défini :'" + StatutRéglementOpération(dos_operation.IdOperation) +"'"); STOP // A définir
	fin
sinon
	sNouveauStatut = __Statut_Terminée__
fin


fin:

si sNouveauStatut <> "" _et_ dos_operation.Statut <> sNouveauStatut alors

	__bChangementStatut = Vrai

	si H.FonctionTrigger = "HModifie" alors
		dos_operation.DateHeureModification		= DateHeureSys()
		dos_operation.UserModification 			= "TriggerStatut : " + Partage.LibelleStatutOperation(sNouveauStatut)
	fin

	dos_operation.Statut = sNouveauStatut

	si DateHeureValide(dhLimitePriseEnCharge) alors
		dos_operation.DateHeureLimitePriseEnCharge = dhLimitePriseEnCharge[[à 14]]
	fin

fin
