// Syntaxe :
//[ <Résultat> = ] CalculeTaaBaseChamp (<tabBase> est tableau (StructBase), <bConserverChampsSupprimés> est booléen, <bAutoriserColonnesInversées> est booléen)
//
// Paramètres :
//	tabBase (tableau (StructBase)) : <indiquez ici le rôle de pStOperation>
//	bConserverChampsSupprimés (booléen) : <indiquez ici le rôle de pBConserverChampsSupprimés>
//	bAutoriserColonnesInversées (booléen) : <indiquez ici le rôle de pBAutoriserColonnesInversées>

// Valeur de retour :
// 	tableau associatif (structBaseChamp) : <indiquez ici le rôle de la valeur de retour>
//
// Exemple :
// Indiquez ici un exemple d'utilisation.
//
// 	pEstCliquezIci : <indiquez ici le rôle de pEstCliquezIci>
procédure CalculeTaaBaseChamp(tabBase est un tableau de StructBase, bConserverChampsSupprimés est un booléen, bAutoriserColonnesInversées est un booléen)

// ICI ON CALCULE LES COLONNES COMMUNES A TOUTES LES BASES DE DONNEES


stBase				est une StructBase
stBaseChamp			est une structBaseChamp
taaBaseLibelle		est un tableau associatif de structBaseChamp
taaBaseChamp		est un tableau associatif de chaînes
taaChampPrincipaux	est un tableau associatif de booléens
nInd				est un entier

taaChampPrincipaux[__CHAMP_REFERENCE__]		= Vrai
taaChampPrincipaux[__CHAMP_SOCIETE__]		= Vrai
taaChampPrincipaux[__CHAMP_ENSEIGNE__]		= Vrai
taaChampPrincipaux[__CHAMP_CONTACT__]		= Vrai
taaChampPrincipaux[__CHAMP_CIVILITE__]		= Vrai
taaChampPrincipaux[__CHAMP_NOM__]			= Vrai
taaChampPrincipaux[__CHAMP_PRENOM__]		= Vrai
taaChampPrincipaux[__CHAMP_ADRESSE1__]		= Vrai
taaChampPrincipaux[__CHAMP_ADRESSE2__]		= Vrai
taaChampPrincipaux[__CHAMP_ADRESSE3__]		= Vrai
taaChampPrincipaux[__CHAMP_ADRESSE4__]		= Vrai
taaChampPrincipaux[__CHAMP_CODE_POSTAL__]	= Vrai
taaChampPrincipaux[__CHAMP_VILLE__]			= Vrai
taaChampPrincipaux[__CHAMP_PAYS__]			= Vrai
taaChampPrincipaux[__CHAMP_TEL__]			= Vrai
taaChampPrincipaux[__CHAMP_PORTABLE__]		= Vrai
taaChampPrincipaux[__CHAMP_EMAIL__]			= Vrai
taaChampPrincipaux[__CHAMP_ALLIAGE__]		= Vrai


pour tout stBase de tabBase
	si stBase.EstSupprime = Faux alors
		nInd++
		pour tout stBaseChamp, sLibelle de stBase.taaBaseChamp
			si taaBaseLibelle[sLibelle]..Vide alors
				stBaseChamp.Pointer = 1
				taaBaseLibelle[sLibelle] = stBaseChamp
				taaBaseChamp[stBaseChamp.Champ] = sLibelle
			sinon
				si bAutoriserColonnesInversées alors
					taaBaseLibelle[sLibelle].Pointer++
				sinon
					si stBaseChamp.Champ = taaBaseLibelle[sLibelle].Champ alors
						taaBaseLibelle[sLibelle].Pointer++
					sinon
						stBaseChamp.ColonnesInversées				+= [" - "] + taaBaseChamp[taaBaseLibelle[sLibelle].Champ]
						taaBaseLibelle[sLibelle].ColonnesInversées	+= [" - "] + taaBaseChamp[stBaseChamp.Champ]
					fin
				fin
			fin
		fin
	fin
fin

pour tout stBaseChamp, sLibelle de taaBaseLibelle
	selon Vrai
		cas taaChampPrincipaux[stBaseChamp.Champ] = Vrai

		cas stBaseChamp.Pointer <> nInd
			si bConserverChampsSupprimés = Vrai alors
				stBaseChamp.EstSupprimé = Vrai
			sinon
				TableauSupprime(taaBaseLibelle,sLibelle)
			fin

		autre cas

	fin
fin



renvoyer taaBaseLibelle
