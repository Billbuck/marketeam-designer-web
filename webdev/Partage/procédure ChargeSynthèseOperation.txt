	,clt_client_facturation.Societe FacturationSociete

	,clt_contact_valideur.Civilite ValideurCivilite

	,clt_contact_valideur.Prenom ValideurPrenom
	,clt_paiement.IdPaiement

	[%sSelect%]
	FROM
	dos_operation
	JOIN clt_client ON clt_client.IdClient = dos_operation.IdClientFacturation
	JOIN clt_contact ON clt_contact.IdContact = dos_operation.IdContact
	JOIN clt_client clt_client_facturation ON clt_client_facturation.IdClient = dos_operation.IdClientFacturation
	JOIN clt_contact clt_contact_valideur ON clt_contact_valideur.IdContact = dos_operation.IdValideur
	[%sFrom%]
	LEFT JOIN clt_paiement ON clt_paiement.IdOperation = dos_operation.IdOperation
	LEFT JOIN clt_reglement ON clt_reglement.IdReglement = clt_paiement.IdReglement
	LEFT JOIN clt_facture_ligne ON clt_facture_ligne.IdOperation = dos_operation.IdOperation
	LEFT JOIN prm_mode_reglement ON prm_mode_reglement.CodeModeReglement = clt_reglement.CodeModeReglement
	WHERE
	dos_operation.IdOperation = [%pIdOperation%]
