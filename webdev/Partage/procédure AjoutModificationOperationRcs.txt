procédure AjoutModificationOperationRcs(stOperation est une structOperation)

sCreditSms			est une chaîne
sTexteRequete		est une chaîne
sdReqRcs				est une Source de Données
bEstModeCreation	est un booléen	= (stOperation.IdOperation = 0)		// On fait le test maintenant car après l'appel de AjoutModificationOperation l'IdOperation sera renseigné


AjoutModificationOperation(stOperation)  // On ajoute / met à jour les données dans la table dos_operation

si bEstModeCreation alors
	ope_rcs.RAZ()
	ope_rcs.IdRcs					= stOperation.IdOperation
	ope_rcs.DateHeureCreation	= DateHeureSys()
	ope_rcs.UserCreation			= Partage.__vcUser.NomUser

sinon
	si pas ope_rcs.LitRecherchePremier(IdRcs, stOperation.IdOperation) alors
		<compile si TypeConfiguration <> Webservice>
			Info("Erreur positionnement ope_rcs")
		<fin>
		renvoyer Faux
	fin

fin

MémoireVersFichier(stOperation.stOperationRcs, ope_rcs)

Sérialise(stOperation.stOperationRcs.taaRepliCreditSms, sCreditSms, psdJSON)
ope_rcs.RepliCreditSms						= sCreditSms
ope_rcs.RepliCreditSms.Null				= (sCreditSms = "")

ope_rcs.RepliSmsMessage						= Remplace(ope_rcs.RepliSmsMessage, RC, Caract(10))
si ope_rcs.TypeContenu = "" alors ope_rcs.TypeContenu = "CARD"
si ope_rcs.Orientation = "" alors ope_rcs.Orientation = "VERTICAL"
si ope_rcs.Alignement = "" alors ope_rcs.Alignement = "LEFT"
si ope_rcs.LargeurCarte = "" alors ope_rcs.LargeurCarte = "MEDIUM"
ope_rcs.IdRcsModele..Null					= (ope_rcs.IdRcsModele = 0)
ope_rcs.IdRcsAgent..Null					= (ope_rcs.IdRcsAgent = 0)
ope_rcs.ModeleVariable..Null				= (ope_rcs.ModeleVariable = "")
ope_rcs.CampaignReferenceId..Null		= (ope_rcs.CampaignReferenceId = "")
ope_rcs.ValiditeDuree..Null				= (ope_rcs.ValiditeDuree = 0)
ope_rcs.DateFin..Null						= (stOperation.stOperationRcs.DateFin = "")
ope_rcs.HeureFin..Null						= (stOperation.stOperationRcs.HeureFin = "")
ope_rcs.FenetreHeureDebut..Null			= (pas HeureValide(stOperation.stOperationRcs.FenetreHeureDebut))
ope_rcs.FenetreHeureFin..Null				= (pas HeureValide(stOperation.stOperationRcs.FenetreHeureFin))
ope_rcs.DateHeureQualification..Null	= (pas DateHeureValide(stOperation.stOperationRcs.DateHeureQualification))


si bEstModeCreation alors
	si pas ope_rcs.Ajoute() alors
		<compile si TypeConfiguration <> Webservice>
			Info("Erreur :", ErreurInfo(errComplet))
		<fin>
		renvoyer Faux
	fin
sinon
	ope_rcs.DateHeureModification	= DateHeureSys()
	ope_rcs.UserModification		= Partage.__vcUser.NomUser
	si pas ope_rcs.Modifie() alors
		<compile si TypeConfiguration <> Webservice>
			Info("Erreur :", ErreurInfo(errComplet))
		<fin>
		renvoyer Faux
	fin
fin


// On remplace tout : suppression puis réécriture dans l'ordre
sTexteRequete = ChaîneConstruit("DELETE FROM ope_rcs_bouton WHERE IdRcs = %1", stOperation.IdOperation)
si pas sdReqRcs.ExécuteRequêteSQL(ope_rcs_bouton..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	<compile si TypeConfiguration <> Webservice>
		Info("Erreur requête AjoutModificationOperationRcs boutons")
	<fin>
	renvoyer Faux
fin
sTexteRequete = ChaîneConstruit("DELETE FROM ope_rcs_carte WHERE IdRcs = %1", stOperation.IdOperation)
si pas sdReqRcs.ExécuteRequêteSQL(ope_rcs_carte..Connexion, hRequêteSansCorrection, sTexteRequete) alors
	<compile si TypeConfiguration <> Webservice>
		Info("Erreur requête AjoutModificationOperationRcs cartes")
	<fin>
	renvoyer Faux
fin

pour tout stCarte de stOperation.stOperationRcs.tabCarte sur Ordre
	ope_rcs_carte.RAZ()
	ope_rcs_carte.DateHeureCreation		= DateHeureSys()
	ope_rcs_carte.UserCreation				= Partage.__vcUser.NomUser
	MémoireVersFichier(stCarte, ope_rcs_carte)
	ope_rcs_carte.IdRcs						= stOperation.IdOperation
	ope_rcs_carte.IdRcsCarte				= 0
	si ope_rcs_carte.MediaHauteur = "" alors ope_rcs_carte.MediaHauteur = "MEDIUM"
	ope_rcs_carte.IdImage..Null					= (ope_rcs_carte.IdImage = 0)
	ope_rcs_carte.MediaThumbnailUrl..Null		= (ope_rcs_carte.MediaThumbnailUrl = "")

	si pas ope_rcs_carte.Ajoute() alors
		<compile si TypeConfiguration <> Webservice>
			Info("Erreur :", ErreurInfo(errComplet))
		<fin>
		renvoyer Faux
	fin
	stCarte.IdRcsCarte = ope_rcs_carte.IdRcsCarte

	pour tout stBouton de stCarte.tabBouton sur Ordre
		si pas EnregistreUnBoutonRcs(stBouton, stCarte.IdRcsCarte, stOperation.IdOperation) alors
			renvoyer Faux
		fin
	fin
fin

pour tout stBouton de stOperation.stOperationRcs.tabBoutonCommun sur Ordre
	si pas EnregistreUnBoutonRcs(stBouton, 0, stOperation.IdOperation) alors
		renvoyer Faux
	fin
fin

renvoyer Vrai

	procédure interne EnregistreUnBoutonRcs(pStBoutonRcs est une structOperationRcsBouton, pIdCarteRattachement est un entier, pIdOperation est un entier)
		ope_rcs_bouton.RAZ()
		ope_rcs_bouton.DateHeureCreation		= DateHeureSys()
		ope_rcs_bouton.UserCreation			= Partage.__vcUser.NomUser
		MémoireVersFichier(pStBoutonRcs, ope_rcs_bouton)
		ope_rcs_bouton.IdRcs					= pIdOperation
		ope_rcs_bouton.IdRcsBouton			= 0
		si pIdCarteRattachement = 0 alors
			ope_rcs_bouton.IdRcsCarte..Null = Vrai
		sinon
			ope_rcs_bouton.IdRcsCarte			= pIdCarteRattachement
			ope_rcs_bouton.IdRcsCarte..Null	= Faux
		fin
		si ope_rcs_bouton.TypeBouton = "" alors ope_rcs_bouton.TypeBouton = "OPEN_URL"
		ope_rcs_bouton.UrlApplication..Null	= (ope_rcs_bouton.UrlApplication = "")
		ope_rcs_bouton.UrlWebviewMode..Null	= (ope_rcs_bouton.UrlWebviewMode = "")
		ope_rcs_bouton.SousType..Null			= (ope_rcs_bouton.SousType = "")
		ope_rcs_bouton.Parametres..Null		= (ope_rcs_bouton.Parametres = "")

		si pas ope_rcs_bouton.Ajoute() alors
			<compile si TypeConfiguration <> Webservice>
				Info("Erreur :", ErreurInfo(errComplet))
			<fin>
			renvoyer Faux
		fin
		pStBoutonRcs.IdRcsBouton = ope_rcs_bouton.IdRcsBouton
		renvoyer Vrai
	fin

fin:
sdReqRcs.AnnuleDéclaration()
