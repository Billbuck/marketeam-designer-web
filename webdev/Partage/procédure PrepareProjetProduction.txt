// =========================================================================
// Partage.PrepareProjetProduction()
// Rôle : Geste « préparer le projet de production » d'une opération Courrier :
//          1. exporter le CSV de la base dans le répertoire Printshop
//             (Partage.ExporteBaseOperationCsv) ;
//          2. finaliser TOUS les PSMD du répertoire pour qu'ils lisent ce CSV
//             (Partage.FinalisePsmdOperation).
//
// Paramètre :
//   pIdOperation : identifiant de l'opération.
// Retour :
//   booléen : Vrai si l'export ET la finalisation des PSMD ont réussi.
// =========================================================================
procédure PrepareProjetProduction(pIdOperation) : booléen

sNomCsv est une chaîne = Partage.ExporteBaseOperationCsv(pIdOperation)

si sNomCsv = "" alors
	Trace("PrepareProjetProduction : export CSV échoué (IdOperation=" + pIdOperation + ")")
	renvoyer Faux
fin

renvoyer Partage.FinalisePsmdOperation(pIdOperation, sNomCsv)
