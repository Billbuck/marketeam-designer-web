// ─────────────────────────────────────────────────────────────────────────────
// Bouton JETABLE : btnTestPolicePrimitives  (Lot 2A-1 — preuve des primitives)
// Libellé conseillé : « TEST primitives police (download + nameID1) »
//
// Rôle : prouver, sur UNE police, les 2 capacités neuves :
//        1) téléchargement BINAIRE d'un .ttf https vers le cache global ;
//        2) lecture du nameID1 (Family name) par parse WLangage.
//
// Cas piège visé : une graisse à 2 mots (SemiBold/ExtraBold) dont le nameID1
//        est COLLÉ sans espace interne (ex. « Montserrat SemiBold »).
//
// EMPLACEMENT : page d'admin INTERNE uniquement (garde EstChronodirect).
// JETABLE : à retirer après validation. Ne touche ni à l'import, ni au print,
//           ni au Designer (lecture seule sur dsn_police).
// ─────────────────────────────────────────────────────────────────────────────

sUrl			est une chaîne	= ""
sNomBase		est une chaîne	= ""
sNomTechBase	est une chaîne	= ""
nWeightBase		est un entier	= 0
sReq			est une chaîne ANSI
sdReq			est une Source de Données
sCache			est une chaîne
sDest			est une chaîne
nTaille			est un entier
sNameId1		est une chaîne

// ─── Garde de sécurité : personnel interne uniquement ───────────────────────
SI Partage.__vcUser.EstChronodirect = Faux ALORS
	Erreur("Action réservée au personnel interne Marketeam.")
	RETOUR
FIN

// ─── 1. Récupérer une URL de variante à 2 mots (SemiBold 600, sinon ExtraBold 800) ──
//   Lecture SEULE de dsn_police (portée générale, Origine google).
sReq = [
	SELECT Nom, NomTechnique, Url, Weight
	FROM dsn_police
	WHERE Weight IN (600, 800)
	  AND Origine = 'google'
	  AND IdMarque IS NULL AND IdClient IS NULL
	  AND Url <> ''
	ORDER BY Weight
]
si sdReq.ExécuteRequêteSQL(dsn_police.Connexion, hRequêteSansCorrection, sReq) alors
	POUR TOUT sdReq
		sUrl			= sdReq.Url
		sNomBase		= sdReq.Nom
		sNomTechBase	= sdReq.NomTechnique
		nWeightBase		= sdReq.Weight
		SORTIR
	FIN
fin
sdReq.AnnuleDéclaration()

si sUrl = "" alors
	Erreur("Aucune entrée SemiBold/ExtraBold trouvée dans dsn_police." + RC ...
		+ "Lance d'abord l'import Google Fonts (Lot 0 B), puis relance ce test.")
	RETOUR
fin

// ─── 2. Télécharger le TTF dans le cache global ─────────────────────────────
sCache = cpDesigner.RepCachePolices()
si sCache = "" alors
	Erreur("Impossible de créer/atteindre le dossier de cache des polices.")
	RETOUR
fin
sDest = sCache + "test_nameid1.ttf"

si pas cpDesigner.TelechargeFichierBinaire(sUrl, sDest) alors
	Erreur("Échec du téléchargement binaire du TTF." + RC + "URL : " + sUrl)
	RETOUR
fin

nTaille = fTaille(sDest)

// ─── 3. Lire le nameID1 (Family name) ───────────────────────────────────────
sNameId1 = cpDesigner.LitNomInterneTtf(sDest)

// ─── 4. Affichage du résultat ───────────────────────────────────────────────
Info("TEST primitives police (Lot 2A-1)" + RC + RC ...
	+ "Entrée base   : " + sNomBase + "  (Weight " + nWeightBase + ")" + RC ...
	+ "NomTechnique  : " + sNomTechBase + RC ...
	+ "URL           : " + sUrl + RC ...
	+ "Fichier       : " + sDest + RC ...
	+ "Taille        : " + nTaille + " octets" + RC ...
	+ "nameID1 (TTF) : [" + sNameId1 + "]" + RC + RC ...
	+ "Comparer NomTechnique vs nameID1 (cas piège : espace vs collé).")

Trace("TEST 2A-1 : Nom=" + sNomBase + " | NomTechnique=" + sNomTechBase ...
	+ " | nameID1=[" + sNameId1 + "] | taille=" + nTaille + " | url=" + sUrl)
