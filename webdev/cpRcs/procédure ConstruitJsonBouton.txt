procédure ConstruitJsonBouton(stBouton est une structOperationRcsBouton, pIdRcsCarte, taaValeurChamp est un tableau associatif de chaînes, pCodeLanding, bEstTest)

sJson			est une chaîne
sLibelle		est une chaîne
sPostback		est une chaîne
sUrl			est une chaîne
sTelephone		est une chaîne
sLabel			est une chaîne
sTypeJson		est une chaîne
sApplication	est une chaîne
sWebview		est une chaîne
jParam			est un JSON
sTitre			est une chaîne
sDescription	est une chaîne
sDate			est une chaîne
sHeureDebut	est une chaîne
sHeureFin		est une chaîne
dAgenda		est une Date
hDebut			est une Heure
hFin			est une Heure
sDebutIso		est une chaîne
sFinIso		est une chaîne
sLat			est une chaîne
sLon			est une chaîne
bJsonOk		est un booléen


sLibelle	= FusionneTexte(stBouton.Libelle, taaValeurChamp)
si SansEspace(sLibelle) = "" alors
	Trace("cpRcs : bouton " + stBouton.IdRcsBouton + " ignoré (libellé vide)")
	renvoyer ""
fin

sPostback = pIdRcsCarte + "-" + stBouton.IdRcsBouton

selon stBouton.SousType
	cas __RCS_ACTION_AGENDA__
		sTypeJson = "CREATE_CALENDAR_EVENT"
		bJsonOk = Vrai
		quand exception dans
			jParam = stBouton.Parametres
		faire
			bJsonOk = Faux
		fin
		si bJsonOk = Faux _ou_ stBouton.Parametres = "" alors
			Trace("cpRcs : bouton agenda " + stBouton.IdRcsBouton + " ignoré (Parametres)")
			renvoyer ""
		fin
		sDate = jParam.AgendaDate
		si sDate = "" _ou_ sDate = "0000-00-00" alors
			Trace("cpRcs : bouton agenda " + stBouton.IdRcsBouton + " ignoré (date)")
			renvoyer ""
		fin
		dAgenda = ChaîneVersDate(sDate, "AAAA-MM-JJ")
		si pas DateValide(dAgenda) alors
			Trace("cpRcs : bouton agenda " + stBouton.IdRcsBouton + " ignoré (date)")
			renvoyer ""
		fin
		sHeureDebut	= Gauche(jParam.AgendaHeureDebut, 8)
		sHeureFin		= Gauche(jParam.AgendaHeureFin, 8)
		hDebut			= ChaîneVersHeure(sHeureDebut, "HH:MM:SS")
		hFin			= ChaîneVersHeure(sHeureFin, "HH:MM:SS")
		sDebutIso		= DateHeureAgendaIso(dAgenda, hDebut)
		sFinIso		= DateHeureAgendaIso(dAgenda, hFin)
		si sDebutIso = "" _ou_ sFinIso = "" alors
			Trace("cpRcs : bouton agenda " + stBouton.IdRcsBouton + " ignoré (horaire)")
			renvoyer ""
		fin
		sTitre			= FusionneTexte(jParam.AgendaTitre, taaValeurChamp)
		sDescription	= FusionneTexte(jParam.AgendaDescription, taaValeurChamp)
		si jParam.AgendaLieu <> "" alors
			si sDescription <> "" alors
				sDescription += " — " + FusionneTexte(jParam.AgendaLieu, taaValeurChamp)
			sinon
				sDescription = FusionneTexte(jParam.AgendaLieu, taaValeurChamp)
			fin
		fin
		sJson = "{"
		sJson += """text"":""" + EncodeAsciiJson(sLibelle) + """"
		sJson += ",""postbackData"":""" + EncodeAsciiJson(sPostback) + """"
		sJson += ",""type"":""CREATE_CALENDAR_EVENT"""
		sJson += ",""startTime"":""" + EncodeAsciiJson(sDebutIso) + """"
		sJson += ",""endTime"":""" + EncodeAsciiJson(sFinIso) + """"
		sJson += ",""title"":""" + EncodeAsciiJson(sTitre) + """"
		sJson += ",""description"":""" + EncodeAsciiJson(sDescription) + """"
		sJson += "}"
		renvoyer sJson

	cas __RCS_ACTION_LANDING__
		sTypeJson		= __RCS_BOUTON_OPEN_URL__
		sUrl			= UrlLanding(pCodeLanding, bEstTest)
		sApplication	= __RCS_URL_BROWSER__
		sWebview		= ""

	cas __RCS_ACTION_REACTION__
		sTypeJson		= __RCS_BOUTON_OPEN_URL__
		sUrl			= UrlBouton(stBouton.IdRcsBouton, pCodeLanding)
		sApplication	= __RCS_URL_WEBVIEW__
		sWebview		= __RCS_WEBVIEW_HALF__

	cas __RCS_ACTION_APPLICATION__, __RCS_ACTION_EMAIL__, __RCS_ACTION_SMS__
		sTypeJson		= __RCS_BOUTON_OPEN_URL__
		sUrl			= UrlBouton(stBouton.IdRcsBouton, pCodeLanding)
		sApplication	= __RCS_URL_BROWSER__
		sWebview		= ""

	cas __RCS_ACTION_APPEL__
		sTypeJson	= __RCS_BOUTON_DIAL_PHONE__
		sTelephone	= FusionneTexte(stBouton.Telephone, taaValeurChamp)
		sTelephone	= Remplace(sTelephone, " ", "")
		sTelephone	= Remplace(sTelephone, "+", "")
		si SansEspace(sTelephone) = "" alors
			Trace("cpRcs : bouton appel " + stBouton.IdRcsBouton + " ignoré (téléphone vide)")
			renvoyer ""
		fin
		sJson = "{"
		sJson += """text"":""" + EncodeAsciiJson(sLibelle) + """"
		sJson += ",""postbackData"":""" + EncodeAsciiJson(sPostback) + """"
		sJson += ",""type"":""DIAL_PHONE"""
		sJson += ",""phoneNumber"":""" + EncodeAsciiJson(sTelephone) + """"
		sJson += "}"
		renvoyer sJson

	cas __RCS_ACTION_CARTE__
		sTypeJson	= __RCS_BOUTON_SHOW_LOCATION__
		sLabel		= FusionneTexte(stBouton.LibelleLieu, taaValeurChamp)
		sLat		= Remplace(NumériqueVersChaîne(stBouton.Latitude, "6.6f"), ",", ".")
		sLon		= Remplace(NumériqueVersChaîne(stBouton.Longitude, "6.6f"), ",", ".")
		sJson = "{"
		sJson += """text"":""" + EncodeAsciiJson(sLibelle) + """"
		sJson += ",""postbackData"":""" + EncodeAsciiJson(sPostback) + """"
		sJson += ",""type"":""SHOW_LOCATION"""
		sJson += ",""latitude"":" + sLat
		sJson += ",""longitude"":" + sLon
		si SansEspace(sLabel) <> "" alors
			sJson += ",""label"":""" + EncodeAsciiJson(sLabel) + """"
		fin
		sJson += "}"
		renvoyer sJson

	autres cas
		sTypeJson = stBouton.TypeBouton
		si sTypeJson = "" alors sTypeJson = __RCS_BOUTON_OPEN_URL__

		si sTypeJson = __RCS_BOUTON_DIAL_PHONE__ alors
			sTelephone	= FusionneTexte(stBouton.Telephone, taaValeurChamp)
			sTelephone	= Remplace(sTelephone, " ", "")
			sTelephone	= Remplace(sTelephone, "+", "")
			si SansEspace(sTelephone) = "" alors renvoyer ""
			sJson = "{"
			sJson += """text"":""" + EncodeAsciiJson(sLibelle) + """"
			sJson += ",""postbackData"":""" + EncodeAsciiJson(sPostback) + """"
			sJson += ",""type"":""DIAL_PHONE"""
			sJson += ",""phoneNumber"":""" + EncodeAsciiJson(sTelephone) + """"
			sJson += "}"
			renvoyer sJson
		fin

		si sTypeJson = __RCS_BOUTON_SHOW_LOCATION__ alors
			sLabel	= FusionneTexte(stBouton.LibelleLieu, taaValeurChamp)
			sLat	= Remplace(NumériqueVersChaîne(stBouton.Latitude, "6.6f"), ",", ".")
			sLon	= Remplace(NumériqueVersChaîne(stBouton.Longitude, "6.6f"), ",", ".")
			sJson = "{"
			sJson += """text"":""" + EncodeAsciiJson(sLibelle) + """"
			sJson += ",""postbackData"":""" + EncodeAsciiJson(sPostback) + """"
			sJson += ",""type"":""SHOW_LOCATION"""
			sJson += ",""latitude"":" + sLat
			sJson += ",""longitude"":" + sLon
			si SansEspace(sLabel) <> "" alors
				sJson += ",""label"":""" + EncodeAsciiJson(sLabel) + """"
			fin
			sJson += "}"
			renvoyer sJson
		fin

		sUrl			= FusionneTexte(stBouton.Url, taaValeurChamp)
		sApplication	= stBouton.UrlApplication
		sWebview		= stBouton.UrlWebviewMode
		si sApplication = "" alors sApplication = __RCS_URL_BROWSER__
fin

si sTypeJson <> __RCS_BOUTON_OPEN_URL__ alors renvoyer ""

si pas EstUrlHttpHttps(sUrl) alors
	Trace("cpRcs : bouton " + stBouton.IdRcsBouton + " ignoré (URL non http/https)")
	renvoyer ""
fin

sJson = "{"
sJson += """text"":""" + EncodeAsciiJson(sLibelle) + """"
sJson += ",""postbackData"":""" + EncodeAsciiJson(sPostback) + """"
sJson += ",""type"":""OPEN_URL"""
sJson += ",""url"":""" + EncodeAsciiJson(sUrl) + """"
sJson += ",""application"":""" + EncodeAsciiJson(sApplication) + """"
si sApplication = __RCS_URL_WEBVIEW__ _et_ sWebview <> "" alors
	sJson += ",""webviewViewMode"":""" + EncodeAsciiJson(sWebview) + """"
fin
sJson += "}"
renvoyer sJson


	procédure interne EstUrlHttpHttps(pUrlCdb)
		si Position(pUrlCdb, "https://", SansCasse) = 1 alors renvoyer Vrai
		si Position(pUrlCdb, "http://", SansCasse) = 1 alors renvoyer Vrai
		renvoyer Faux
	fin
