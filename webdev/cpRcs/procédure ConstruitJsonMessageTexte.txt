// Résumé : Construit le JSON Infobip d'UN message RCS texte simple (confirmation STOP, notification, test).
// Même enveloppe que ConstruitJsonMessage (sender, destinations, options, webhooks),
// mais content = { "type": "TEXT", "text": ... }. Repli SMS optionnel (pTexteRepli), sans expéditeur :
// Infobip applique l'expéditeur par défaut du compte.
// Syntaxe :
// sJson = ConstruitJsonMessageTexte(sSenderName, sPortable, sMessageId, sTexte [, sTexteRepli])
procédure ConstruitJsonMessageTexte(pSenderName, pPortable, pMessageId, pTexte, pTexteRepli = "")

sJson			est une chaîne
sPortable	est une chaîne


sPortable		= Remplace(Remplace(SansEspace(pPortable), "+", ""), " ", "")

sJson			= "{"
sJson += """sender"":""" + EncodeAsciiJson(pSenderName) + """"
sJson += ",""destinations"":[{""to"":""" + EncodeAsciiJson(sPortable) + """,""messageId"":""" + EncodeAsciiJson(pMessageId) + """}]"

sJson += ",""content"":{""type"":""TEXT"",""text"":""" + EncodeAsciiJson(pTexte) + """}"

sJson += ",""options"":{""validityPeriod"":{""amount"":" + __nValiditeHeures + ",""timeUnit"":""HOURS""}"
si SansEspace(pTexteRepli) <> "" alors
	sJson += ",""smsFailover"":{""text"":""" + EncodeAsciiJson(pTexteRepli) + """"
	sJson += ",""validityPeriod"":{""amount"":" + __nValiditeHeures + ",""timeUnit"":""HOURS""}}"
fin
sJson += "}"

si __sUrlWebhook <> "" alors
	sJson += ",""webhooks"":{"
	sJson += """delivery"":{""url"":""" + EncodeAsciiJson(__sUrlWebhook) + """}"
	sJson += ",""seen"":{""url"":""" + EncodeAsciiJson(__sUrlWebhook) + """}"
	sJson += "}"
fin

sJson += "}"
renvoyer sJson
