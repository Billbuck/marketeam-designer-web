// Résumé : Capability check SYNCHRONE Infobip (qualification RCS, mode hybride)
// Syntaxe :
//[ <Résultat> = ] VerifieCapacite (<pSenderName>, <tabNumeros>, <taaCodeParNumero>, <sErreur>)
//
// Paramètres :
//	pSenderName : agent RCS (rcs_agent.SenderName)
//	tabNumeros : numéros au format international sans + (33670299592), 500 max
//	taaCodeParNumero : en sortie, tableau associatif numéro → code Infobip
//		(ENABLED, UNREACHABLE, REJECTED_NETWORK, UNKNOWN_CODE,
//		INVALID_DESTINATION_ADDRESS...). Un numéro absent de la réponse
//		n'a pas d'entrée : c'est à l'appelant de le classer (invalide).
//	sErreur : en sortie, message d'erreur si échec (vide si OK)
// Valeur de retour :
// 	booléen : Vrai si HTTP 200/201 et réponse exploitable
//
// POST {Rcs_Infobip_Url}/rcs/2/capability-check/query — corps
// { "sender": ..., "phoneNumbers": [...] }, réponse immédiate
// { "capabilityCheckResults": [ { "messageId", "phoneNumber", "code" } ] }.
// Mesuré le 23/09/2026 : 500 numéros par appel acceptés (1,23 s, 49 Ko),
// 4 appels en parallèle sans code 429 (§ 6 du cahier des charges).
procédure VerifieCapacite(pSenderName, tabNumeros est un tableau de chaînes, taaCodeParNumero est un tableau associatif de chaînes, sErreur)

requete	est un httpRequête
reponse	est un httpRéponse
vRep		est un Variant
vRes		est un Variant
sJson		est une chaîne
nInd		est un entier
bJsonOk	est un booléen


taaCodeParNumero.SupprimeTout()
sErreur = ""

si __sUrlInfobip = "" _ou_ __sApiKey = "" alors
	sErreur = "URL ou clé API manquante (InitCpRcs non appelé ?)"
	Trace("cpRcs VerifieCapacite : " + sErreur)
	renvoyer Faux
fin

si tabNumeros..Occurrence = 0 alors
	sErreur = "Aucun numéro à vérifier"
	renvoyer Faux
fin

sJson = "{""sender"":""" + EncodeAsciiJson(pSenderName) + """"
sJson += ",""phoneNumbers"":["
pour nInd = 1 _à_ tabNumeros..Occurrence
	si nInd > 1 alors sJson += ","
	sJson += """" + EncodeAsciiJson(tabNumeros[nInd]) + """"
fin
sJson += "]}"

requete.URL								= __sUrlInfobip + "/rcs/2/capability-check/query"
requete.Méthode						= httpPost
requete.ContentType					= typeMimeJSON
requete.Entête["Authorization"]	= "App " + __sApiKey
requete.Timeout						= 60s
requete.Contenu						= sJson

reponse									= HTTPEnvoie(requete)

si ErreurDétectée alors
	sErreur = ErreurInfo()
	Trace("cpRcs VerifieCapacite : " + sErreur)
	renvoyer Faux
fin

si reponse.CodeEtat <> 200 _et_ reponse.CodeEtat <> 201 alors
	sErreur = "HTTP " + reponse.CodeEtat + " " + reponse.Contenu
	Trace("cpRcs VerifieCapacite : " + sErreur)
	renvoyer Faux
fin

bJsonOk = Vrai
quand exception dans
	vRep = JSONVersVariant(reponse.Contenu)
faire
	bJsonOk = Faux
fin
si bJsonOk = Faux _ou_ pas vRep.capabilityCheckResults..Existe alors
	sErreur = "Réponse illisible ou capabilityCheckResults absent : " + reponse.Contenu
	Trace("cpRcs VerifieCapacite : " + sErreur)
	renvoyer Faux
fin

pour nInd = 1 _à_ vRep.capabilityCheckResults..Occurrence
	vRes = vRep.capabilityCheckResults[nInd]
	si vRes.phoneNumber..Existe _et_ vRes.code..Existe alors
		taaCodeParNumero[vRes.phoneNumber] = vRes.code
	fin
fin

renvoyer Vrai
